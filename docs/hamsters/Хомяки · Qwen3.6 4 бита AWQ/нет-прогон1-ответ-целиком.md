Here is a complete, single-file browser project. You can save this code as `index.html` and open it in any modern web browser (Chrome, Firefox, Edge, Safari).

It uses Three.js via a CDN, generates all 3D geometry programmatically (no external model downloads required), and features autonomous hamster AI.

```html
<!DOCTYPE html>

<head>

        }

            pointer-events: none;
        }

<body>

    <div id="loading">Building Cage...</div>
    <div class="instructions">
        <strong>Controls:</strong><br>
        Left Click + Drag: Rotate Camera<br>
        Right Click + Drag: Pan<br>

    <!-- Import Map for Three.js -->
    <script type="importmap">
        {
            "imports": {
                "three": "https://unpkg.com/three@0.160.0/build/three.module.js",
                "three/addons/": "https://unpkg.com/three@0.160.0/examples/jsm/"
            }
        }
    </script>

    <script type="module">
        import * as THREE from 'three';
        import { OrbitControls } from 'three/addons/controls/OrbitControls.js';

        // --- Global Variables ---
        let scene, camera, renderer, controls, clock;
        let hamsters = [];
        let wheelGroup;
        const FLOOR_SIZE = 12;
        
        init();
        animate();

        function init() {
            // 1. Setup Scene
            scene = new THREE.Scene();
            scene.background = new THREE.Color(0x87CEEB); // Sky blue
            scene.fog = new THREE.Fog(0x87CEEB, 20, 50);

            // 2. Setup Camera
            camera = new THREE.PerspectiveCamera(45, window.innerWidth / window.innerHeight, 0.1, 100);
            camera.position.set(0, 15, 20);
            camera.lookAt(0, 0, 0);

            // 3. Setup Renderer
            renderer = new THREE.WebGLRenderer({ antialias: true });
            renderer.setSize(window.innerWidth, window.innerHeight);
            renderer.shadowMap.enabled = true;
            renderer.shadowMap.type = THREE.PCFSoftShadowMap;
            document.body.appendChild(renderer.domElement);

            // 4. Controls
            controls = new OrbitControls(camera, renderer.domElement);
            controls.enableDamping = true;
            controls.dampingFactor = 0.05;
            controls.maxPolarAngle = Math.PI / 2 - 0.1; // Prevent going below ground

            // 5. Lighting
            setupLighting();

            // 6. Build World
            buildCage();
            buildWheel();
            
            // 7. Create Hamsters
            createHamster(0x8B4513, 4, 0, 4, true);   // Brown
            createHamster(0xFFFFFF, -4, 0, -4, false); // White
            createHamster(0xFFD700, 0, 0, -5, false);  // Golden

            // Remove loading text
            document.getElementById('loading').style.display = 'none';

            // Handle Resize
            window.addEventListener('resize', onWindowResize);
            
            clock = new THREE.Clock();
        }

        function setupLighting() {
            const ambientLight = new THREE.AmbientLight(0xffffff, 0.6);
            scene.add(ambientLight);

            const dirLight = new THREE.DirectionalLight(0xffffff, 1);
            dirLight.position.set(10, 20, 10);
            dirLight.castShadow = true;
            dirLight.shadow.mapSize.width = 2048;
            dirLight.shadow.mapSize.height = 2048;
            dirLight.shadow.camera.near = 0.5;
            dirLight.shadow.camera.far = 50;
            dirLight.shadow.camera.left = -15;
            dirLight.shadow.camera.right = 15;
            dirLight.shadow.camera.top = 15;
            dirLight.shadow.camera.bottom = -15;
            scene.add(dirLight);
        }

        // --- Object Builders ---

        function buildCage() {
            // Materials
            const woodMat = new THREE.MeshStandardMaterial({ color: 0xDEB887, flatShading: true }); // Burlywood
            const floorMat = new THREE.MeshStandardMaterial({ color: 0xE8C39E, flatShading: true });
            const glassMat = new THREE.MeshPhysicalMaterial({ 
                color: 0xffffff, 
                transparent: true, 
                opacity: 0.2, 
                roughness: 0, 
                metalness: 0,
                transmission: 0.5
            });

            // Floor Tray
            const floorGeo = new THREE.BoxGeometry(FLOOR_SIZE, 1, FLOOR_SIZE);
            const floor = new THREE.Mesh(floorGeo, floorMat);
            floor.position.y = -0.5;
            floor.receiveShadow = true;
            scene.add(floor);

            // Bedding (small cubes scattered)
            const beddingGeo = new THREE.BoxGeometry(0.5, 0.1, 0.5);
            const beddingMat = new THREE.MeshStandardMaterial({ color: 0xFFE4C4, flatShading: true });
            for(let i=0; i<40; i++) {
                const bedding = new THREE.Mesh(beddingGeo, beddingMat);
                bedding.position.set(
                    (Math.random() - 0.5) * (FLOOR_SIZE - 2),
                    0.05,
                    (Math.random() - 0.5) * (FLOOR_SIZE - 2)
                );
                bedding.rotation.y = Math.random();
                bedding.receiveShadow = true;
                scene.add(bedding);
            }

            // Glass Walls
            const wallHeight = 5;
            const wallThickness = 0.2;
            
            // Back Wall
            const backWall = new THREE.Mesh(new THREE.BoxGeometry(FLOOR_SIZE, wallHeight, wallThickness), glassMat);
            backWall.position.set(0, wallHeight/2, -FLOOR_SIZE/2);
            scene.add(backWall);

            // Front Wall
            const frontWall = new THREE.Mesh(new THREE.BoxGeometry(FLOOR_SIZE, wallHeight, wallThickness), glassMat);
            frontWall.position.set(0, wallHeight/2, FLOOR_SIZE/2);
            scene.add(frontWall);

            // Side Walls
            const sideWallGeo = new THREE.BoxGeometry(wallThickness, wallHeight, FLOOR_SIZE);
            
            const leftWall = new THREE.Mesh(sideWallGeo, glassMat);
            leftWall.position.set(-FLOOR_SIZE/2, wallHeight/2, 0);
            scene.add(leftWall);

            const rightWall = new THREE.Mesh(sideWallGeo, glassMat);
            rightWall.position.set(FLOOR_SIZE/2, wallHeight/2, 0);
            scene.add(rightWall);

            // Wood Frame (Top and Bottom beams)
            const frameGeoLong = new THREE.BoxGeometry(FLOOR_SIZE + 1, 0.5, 0.5);
            const frameGeoShort = new THREE.BoxGeometry(0.5, 0.5, FLOOR_SIZE + 1);

            // Bottom Frame
            addFrame(frameGeoLong, 0, 0, -FLOOR_SIZE/2);
            addFrame(frameGeoLong, 0, 0, FLOOR_SIZE/2);
            addFrame(frameGeoShort, -FLOOR_SIZE/2, 0, 0);
            addFrame(frameGeoShort, FLOOR_SIZE/2, 0, 0);

            // Top Frame
            addFrame(frameGeoLong, 0, wallHeight, -FLOOR_SIZE/2);
            addFrame(frameGeoLong, 0, wallHeight, FLOOR_SIZE/2);
            addFrame(frameGeoShort, -FLOOR_SIZE/2, wallHeight, 0);
            addFrame(frameGeoShort, FLOOR_SIZE/2, wallHeight, 0);

            // Vertical Corners
            const cornerGeo = new THREE.BoxGeometry(0.5, wallHeight, 0.5);
            addFrame(cornerGeo, -FLOOR_SIZE/2, wallHeight/2, -FLOOR_SIZE/2);
            addFrame(cornerGeo, FLOOR_SIZE/2, wallHeight/2, -FLOOR_SIZE/2);
            addFrame(cornerGeo, -FLOOR_SIZE/2, wallHeight/2, FLOOR_SIZE/2);
            addFrame(cornerGeo, FLOOR_SIZE/2, wallHeight/2, FLOOR_SIZE/2);
        }

        function addFrame(geo, x, y, z) {
            const mesh = new THREE.Mesh(geo, new THREE.MeshStandardMaterial({ color: 0x8B4513, flatShading: true }));
            mesh.position.set(x, y, z);
            mesh.castShadow = true;
            scene.add(mesh);
        }

        function buildWheel() {
            wheelGroup = new THREE.Group();
            
            const rimColor = 0xFF69B4; // Hot pink
            const mat = new THREE.MeshStandardMaterial({ color: rimColor, flatShading: true });
            const innerMat = new THREE.MeshStandardMaterial({ color: 0xE0E0E0, flatShading: true });

            // Rim (Torus)
            const radius = 2;
            const tube = 0.3;
            const segments = 12; // Low poly look
            const rimGeo = new THREE.TorusGeometry(radius, tube, 8, segments);
            const rim = new THREE.Mesh(rimGeo, mat);
            rim.castShadow = true;
            wheelGroup.add(rim);

            // Inner Cylinder (The running surface)
            const cylGeo = new THREE.CylinderGeometry(radius - 0.1, radius - 0.1, 1.5, 12, 1, true);
            // Rotate cylinder to face sideways
            const cylinder = new THREE.Mesh(cylGeo, innerMat);
            cylinder.rotation.x = Math.PI / 2;
            cylinder.castShadow = true;
            wheelGroup.add(cylinder);

            // Axle
            const axleGeo = new THREE.CylinderGeometry(0.2, 0.2, 2.5, 8);
            const axle = new THREE.Mesh(axleGeo, new THREE.MeshStandardMaterial({color: 0x555555}));
            axle.rotation.x = Math.PI / 2;
            wheelGroup.add(axle);

            // Stand
            const standGeo = new THREE.BoxGeometry(1, 3, 1);
            const standMat = new THREE.MeshStandardMaterial({ color: 0x8B4513 });
            const stand = new THREE.Mesh(standGeo, standMat);
            stand.position.set(0, -1.5, 0);
            stand.castShadow = true;
            wheelGroup.add(stand);

            // Position Wheel
            wheelGroup.position.set(0, 0, -4);
            wheelGroup.rotation.x = 0; // Wheel spins on local X axis
            
            scene.add(wheelGroup);
        }

        // --- Hamster Logic ---

        class Hamster {
            constructor(color, x, z, isRunner) {
                this.group = new THREE.Group();
                this.isRunner = isRunner; // If true, tries to go to the wheel
                
                // State Machine
                this.state = 'IDLE'; // IDLE, WALK, TURN, RUN_WHEEL
                this.timer = 0;
                this.speed = 0;
                this.targetPos = new THREE.Vector3(x, 0, z);
                
                this.createMeshes(color);
                this.group.position.set(x, 0, z);
                
                // Random rotation initially
                this.group.rotation.y = Math.random() * Math.PI * 2;
                
                scene.add(this.group);
            }

            createMeshes(color) {
                const bodyMat = new THREE.MeshStandardMaterial({ color: color, flatShading: true });
                const darkMat = new THREE.MeshStandardMaterial({ color: 0x222222, flatShading: true }); // Eyes/Nose

                // Body (Capsule-ish: Sphere + Cylinder)
                const bodyGeo = new THREE.SphereGeometry(0.8, 8, 6);
                this.body = new THREE.Mesh(bodyGeo, bodyMat);
                this.body.position.y = 0.8;
                this.body.scale.set(1, 0.8, 1.2); // Make it slightly elongated
                this.body.castShadow = true;
                this.group.add(this.body);

                // Head
                const headGeo = new THREE.SphereGeometry(0.6, 8, 6);
                this.head = new THREE.Mesh(headGeo, bodyMat);
                this.head.position.set(0, 1.2, 0.8);
                this.head.castShadow = true;
                this.group.add(this.head);

                // Ears
                const earGeo = new THREE.ConeGeometry(0.2, 0.3, 4);
                const earL = new THREE.Mesh(earGeo, bodyMat);
                earL.position.set(-0.4, 1.6, 0.8);
                earL.rotation.x = -0.5;
                this.group.add(earL);
                
                const earR = new THREE.Mesh(earGeo, bodyMat);
                earR.position.set(0.4, 1.6, 0.8);
                earR.rotation.x = -0.5;
                this.group.add(earR);

                // Eyes
                const eyeGeo = new THREE.SphereGeometry(0.1, 4, 4);
                const eyeL = new THREE.Mesh(eyeGeo, darkMat);
                eyeL.position.set(-0.25, 1.3, 1.3);
                this.group.add(eyeL);
                
                const eyeR = new THREE.Mesh(eyeGeo, darkMat);
                eyeR.position.set(0.25, 1.3, 1.3);
                this.group.add(eyeR);

                // Cheeks (puffed out look)
                const cheekGeo = new THREE.SphereGeometry(0.25, 6, 6);
                const cheekL = new THREE.Mesh(cheekGeo, bodyMat);
                cheekL.position.set(-0.45, 1.1, 1.1);
                this.group.add(cheekL);
                
                const cheekR = new THREE.Mesh(cheekGeo, bodyMat);
                cheekR.position.set(0.45, 1.1, 1.1);
                this.group.add(cheekR);

                // Legs (Simple boxes)
                const legGeo = new THREE.BoxGeometry(0.3, 0.3, 0.5);
                const legMat = bodyMat.clone();
                
                this.legFL = new THREE.Mesh(legGeo, legMat);
                this.legFL.position.set(-0.4, 0.15, 0.5);
                this.group.add(this.legFL);

                this.legFR = new THREE.Mesh(legGeo, legMat);
                this.legFR.position.set(0.4, 0.15, 0.5);
                this.group.add(this.legFR);

                this.legBL = new THREE.Mesh(legGeo, legMat);
                this.legBL.position.set(-0.4, 0.15, -0.5);
                this.group.add(this.legBL);

                this.legBR = new THREE.Mesh(legGeo, legMat);
                this.legBR.position.set(0.4, 0.15, -0.5);
                this.group.add(this.legBR);
            }

            update(dt) {
                this.timer -= dt;

                // State Machine Transitions
                if (this.timer <= 0) {
                    this.pickNextState();
                }

                // Execute State
                switch (this.state) {
                    case 'IDLE':
                        this.speed = 0;
                        // Breathing animation
                        const breathe = Math.sin(Date.now() * 0.005) * 0.05;
                        this.body.scale.y = 0.8 + breathe;
                        break;
                    
                    case 'WALK':
                        this.speed = 2.0;
                        this.animateWalk(dt);
                        break;

                    case 'TURN':
                        this.speed = 0;
                        this.group.rotation.y += dt * 2; // Spin
                        break;

                    case 'RUN_WHEEL':
                        // Logic handled in global loop for wheel interaction
                        this.speed = 0; // Don't walk while running
                        this.animateRun(dt);
                        break;
                }

                // Movement Logic
                if (this.state === 'WALK') {
                    const direction = new THREE.Vector3(0, 0, 1);
                    direction.applyAxisAngle(new THREE.Vector3(0, 1, 0), this.group.rotation.y);
                    
                    const nextPos = this.group.position.clone().add(direction.multiplyScalar(this.speed * dt));

                    // Boundary Check
                    const limit = FLOOR_SIZE / 2 - 1;
                    if (nextPos.x > limit) { this.state = 'TURN'; this.timer = 0.5; return; }
                    if (nextPos.x < -limit) { this.state = 'TURN'; this.timer = 0.5; return; }
                    if (nextPos.z > limit) { this.state = 'TURN'; this.timer = 0.5; return; }
                    if (nextPos.z < -limit) { this.state = 'TURN'; this.timer = 0.5; return; }

                    this.group.position.copy(nextPos);
                }
            }

            pickNextState() {
                if (this.isRunner && this.state !== 'RUN_WHEEL') {
                    // Runner logic: Go to wheel
                    const distToWheel = this.group.position.distanceTo(new THREE.Vector3(0,0,-4));
                    if (distToWheel < 1.5) {
                        this.state = 'RUN_WHEEL';
                        this.timer = 5; // Run for 5 seconds
                        // Face the wheel center
                        this.group.lookAt(0, 0, -4);
                        return;
                    } else {
                        // Move towards wheel
                        this.targetPos.set(0, 0, -4);
                        this.group.lookAt(this.targetPos);
                        this.state = 'WALK';
                        this.timer = 2;
                        return;
                    }
                }

                // Normal random behavior
                const rand = Math.random();
                if (rand < 0.4) {
                    this.state = 'IDLE';
                    this.timer = 1 + Math.random() * 2;
                } else if (rand < 0.8) {
                    this.state = 'WALK';
                    this.timer = 2 + Math.random() * 2;
                } else {
                    this.state = 'TURN';
                    this.timer = 0.5 + Math.random() * 0.5;
                }
            }

            animateWalk(dt) {
                // Simple bobbing
                const bounce = Math.sin(Date.now() * 0.02) * 0.1;
                this.group.position.y = bounce;
                
                // Leg swing
                const angle = Date.now() * 0.015;
                this.legFL.rotation.x = Math.sin(angle);
                this.legBR.rotation.x = Math.sin(angle);
                this.legFR.rotation.x = Math.sin(angle + Math.PI);
                this.legBL.rotation.x = Math.sin(angle + Math.PI);
            }

            animateRun(dt) {
                // Faster bobbing
                const bounce = Math.sin(Date.now() * 0.04) * 0.05;
                this.body.position.y = 0.8 + bounce;
            }
        }

        function createHamster(color, x, z, isRunner) {
            hamsters.push(new Hamster(color, x, z, isRunner));
        }

        function onWindowResize() {
            camera.aspect = window.innerWidth / window.innerHeight;
            camera.updateProjectionMatrix();
            renderer.setSize(window.innerWidth, window.innerHeight);
        }

        function animate() {
            requestAnimationFrame(animate);

            const dt = clock.getDelta();

            // Update Hamsters
            hamsters.forEach(h => h.update(dt));

            // Wheel Interaction Logic
            hamsters.forEach(h => {
                if (h.state === 'RUN_WHEEL') {
                    // Spin the wheel
                    wheelGroup.rotation.x -= 0.15; 
                    
                    // Vibrate hamster slightly
                    h.body.rotation.z = Math.sin(Date.now() * 0.05) * 0.1;
                    h.head.rotation.z = Math.cos(Date.now() * 0.05) * 0.1;
                } else {
                    // Reset pose
                    h.body.rotation.z = 0;
                    h.head.rotation.z = 0;
                }
            });

            controls.update();
            renderer.render(scene, camera);
        }
    </script>
</body>
</html>
```
