javascript function generateChunk(cx, cz) { const data = new Uint8Array(CHUNK_SIZE * CHUNK_SIZE * CHUNK_HEIGHT); const setLocal = (x, y, z, id) => { data[x + z*16 + y*256] = id; }; const getLocal = (x, y, z) => { if (x<0||x>=16||z<0||z>=16||y<0||y>=80) return -1; // sentinel for out-of-chunk return data[x + z*16 + y*256]; }; for (let z=0; z<16; z++) { for (let x=0; x<16; x++) { const wx = cx*16 + x; const wz = cz*16 + z; // terrain height const m = fractal2(wx*0.004, wz*0.004, 4); const h = fractal2(wx*0.02, wz*0.02, 4); const H = Math.floor(5 + m*m*58 + h*10); // fill for (let y=0; y= 37) setLocal(x,y,z, STONE); else setLocal(x,y,z, DIRT); } } // surface if (H >= 46) setLocal(x,H,SNOW); else if (H >= 37) setLocal(x,H,STONE); else if (H <= 16) setLocal(x,H,SAND); else setLocal(x,H,GRASS); // caves for (let y=3; y<=H-2; y++) { if (noise3(wx*0.09, y*0.09, wz*0.09) > 0.67) { setLocal(x,y,z, AIR); } } // trees if (getSurfaceType(...) === grass && treeHash(wx,wz) < 0.02) { // place tree if trunk fits in chunk // trunk at (x,z), need leaves within chunk: check x,z in [2,13]? // spec says trunk fits inside chunk, so just 0<=x<16, 0<=z<16 // place leaves only if within chunk } } } chunks.set(chunkKey(cx,cz), { data: data, mesh: null }); }
