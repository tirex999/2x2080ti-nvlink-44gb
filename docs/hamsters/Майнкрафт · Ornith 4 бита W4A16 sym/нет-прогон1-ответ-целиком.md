	// ---------- Value noise ----------
	function hash2(x, y) {
		x |= 0; y |= 0;
		var h = (x * 374761393 + y * 668265263) | 0;
		h = (h ^ (h >>> 13)) * 1274126177 | 0;
		h = (h ^ (h >>> 16)) >>> 0;
		return h / 4294967296;
	}
	function hash3(x, y, z) {
		var h = (x * 374761393 + y * 668265263 + z * 2147483647) | 0;
		h = (h ^ (h >>> 13)) * 1274126177 | 0;
		h = (h ^ (h >>> 16)) >>> 0;
		return h / 4294967296;
	}
	function smoothstep(t) { return t * t * (3 - 2 * t); }
	function lerp(a, b, t) { return a + (b - a) * t; }

	function valueNoise2(x, y) {
		var x0 = Math.floor(x), y0 = Math.floor(y);
		var fx = x - x0, fy = y - y0;
		var v00 = hash2(x0, y0), v10 = hash2(x0 + 1, y0);
		var v01 = hash2(x0, y0 + 1), v11 = hash2(x0 + 1, y0 + 1);
		var u = smoothstep(fx), v = smoothstep(fy);
		return lerp(lerp(v00, v10, u), lerp(v01, v11, u), v);
	}
	function valueNoise3(x, y, z) {
		var x0 = Math.floor(x), y0 = Math.floor(y), z0 = Math.floor(z);
		var fx = x - x0, fy = y - y0, fz = z - z0;
		function h(dx, dy, dz) { return hash3(x0 + dx, y0 + dy, z0 + dz); }
		var c000 = h(0,0,0), c100 = h(1,0,0), c010 = h(0,1,0), c110 = h(1,1,0);
		var c001 = h(0,0,1), c101 = h(1,0,1), c011 = h(0,1,1), c111 = h(1,1,1);
		var u = smoothstep(fx), v = smoothstep(fy), w = smoothstep(fz);
		var a = lerp(c000, c100, u), b = lerp(c010, c110, u);
		var c = lerp(c001, c101, u), d = lerp(c011, c111, u);
		return lerp(lerp(a, b, v), lerp(c, d, v), w);
	}
	function fbm2(x, y) {
		var amp = 1, freq = 1, sum = 0, norm = 0;
		for (var i = 0; i < 4; i++) {
			sum += amp * valueNoise2(x * freq, y * freq);
			norm += amp;
			amp *= 0.5; freq *= 2;
		}
		return sum / norm;
	}

	// ---------- Chunk data helpers ----------
	function chunkKey(cx, cz) { return cx + "," + cz; }

	function getBlock(x, y, z) {
		if (y < 0 || y >= CHUNK_H) return 0;
		var cx = Math.floor(x / CHUNK), lz = x - cx * CHUNK;
		var cz = Math.floor(z / CHUNK), lz_ = z - cz * CHUNK;
		var c = chunks[chunkKey(cx, cz)];
		if (!c) return 0;
		return c.data[lz + lz_ * CHUNK + y * CHUNK * CHUNK];
	}

	function setBlock(x, y, z, v) {
		if (y < 0 || y >= CHUNK_H) return;
		var cx = Math.floor(x / CHUNK), lz = x - cx * CHUNK;
		var cz = Math.floor(z / CHUNK), lz_ = z - cz * CHUNK;
		var key = chunkKey(cx, cz);
		var c = chunks[key];
		if (!c) { c = { data: new Uint8Array(CHUNK * CHUNK * CHUNK_H), mesh: null }; chunks[key] = c; }
		c.data[lz + lz_ * CHUNK + y * CHUNK * CHUNK] = v;
		buildChunkMesh(cx, cz);
		if (lz === 0) buildChunkMesh(cx - 1, cz);
		if (lz === CHUNK - 1) buildChunkMesh(cx + 1, cz);
		if (lz_ === 0) buildChunkMesh(cx, cz - 1);
		if (lz_ === CHUNK - 1) buildChunkMesh(cx, cz + 1);
	}

	function makeChunkData(cx, cz) {
		var data = new Uint8Array(CHUNK * CHUNK * CHUNK_H);

		function heightAt(hx, hz) {
			var wx = cx * CHUNK + hx, wz = cz * CHUNK + hz;
			var m = fbm2(wx * 0.004, wz * 0.004);
			var h = fbm2(wx * 0.02, wz * 0.02);
			return Math.floor(5 + m * m * 58 + h * 10);
		}

		for (var hx = 0; hx < CHUNK; hx++) {
			for (var hz = 0; hz < CHUNK; hz++) {
				var wx = cx * CHUNK + hx, wz = cz * CHUNK + hz;
				var H = heightAt(hx, hz);
				var isSand = H <= 16;
				var isStoneH = H >= 37;
				var isSnow = H >= 46;

				data[hx + hz * CHUNK + 0 * CHUNK * CHUNK] = 3;
				for (var y = 1; y < H; y++) {
					if (isSand && y === H - 1) {
						data[hx + hz * CHUNK + y * CHUNK * CHUNK] = 4;
					} else if (isStoneH && y >= H - 3) {
						data[hx + hz * CHUNK + y * CHUNK * CHUNK] = 3;
					} else {
						data[hx + hz * CHUNK + y * CHUNK * CHUNK] = 2;
					}
				}
				if (isSnow) {
					data[hx + hz * CHUNK + (H - 1) * CHUNK * CHUNK] = 7;
				} else if (isStoneH) {
					data[hx + hz * CHUNK + (H - 1) * CHUNK * CHUNK] = 3;
				} else if (isSand) {
					data[hx + hz * CHUNK + (H - 1) * CHUNK * CHUNK] = 4;
				} else {
					data[hx + hz * CHUNK + (H - 1) * CHUNK * CHUNK] = 1;
				}

				// Caves
				for (var cy = 3; cy < H - 2; cy++) {
					if (valueNoise3(wx * 0.09, cy * 0.09, wz * 0.09) > 0.67) {
						data[hx + hz * CHUNK + cy * CHUNK * CHUNK] = 0;
					}
				}

				// Trees
				if (data[hx + hz * CHUNK + (H - 1) * CHUNK * CHUNK] === 1 &&
					hash2(wx * 7 + 1, wz * 13 + 5) < 0.02 && hx >= 2 && hx <= 13 && hz >= 2 && hz <= 13 && H + 6 < CHUNK_H) {
					for (var t = 0; t < 4; t++) {
						data[hx + hz * CHUNK + (H + t) * CHUNK * CHUNK] = 5;
					}
					var top = H + 4;
					for (var lx = -2; lx <= 2; lx++) {
						for (var lz = -2; lz <= 2; lz++) {
							for (var ly = 0; ly < 2; ly++) {
								var tx = hx + lx, tz = hz + lz, ty = top + ly;
								if (tx >= 0 && tx < CHUNK && tz >= 0 && tz < CHUNK && ty < CHUNK_H) {
									if (data[tx + tz * CHUNK + ty * CHUNK * CHUNK] === 0)
										data[tx + tz * CHUNK + ty * CHUNK * CHUNK] = 6;
								}
							}
						}
					}
					for (var rx = -1; rx <= 1; rx++) {
						for (var rz = -1; rz <= 1; rz++) {
							var bx = hx + rx, bz = hz + rz, by = top + 2;
							if (bx >= 0 && bx < CHUNK && bz >= 0 && bz < CHUNK && by < CHUNK_H) {
								if (data[bx + bz * CHUNK + by * CHUNK * CHUNK] === 0)
									data[bx + bz * CHUNK + by * CHUNK * CHUNK] = 6;
							}
						}
					}
					if (H + 3 < CHUNK_H && data[hx + hz * CHUNK + (top + 3) * CHUNK * CHUNK] === 0) {
						data[hx + hz * CHUNK + (top + 3) * CHUNK * CHUNK] = 6;
					}
				}
			}
		}
		return data;
	}

	// ---------- Meshing ----------
	function buildChunkMesh(cx, cz) {
		var key = chunkKey(cx, cz);
		var c = chunks[key];
		if (!c) return;
		var pos = [], nrm = [], col = [], idx = [];
		var vc = 0;

		for (var y = 0; y < CHUNK_H; y++) {
			for (var z = 0; z < CHUNK; z++) {
				for (var x = 0; x < CHUNK; x++) {
					var id = c.data[x + z * CHUNK + y * CHUNK * CHUNK];
					if (id === 0) continue;
					var blk = BLOCKS[id];
					var bc = blk.color;
					var br = (bc >> 16) & 255, bg = (bc >> 8) & 255, bb = bc & 255;
					var wx = cx * CHUNK + x, wy = y, wz = cz * CHUNK + z;

					for (var f = 0; f < 6; f++) {
						var d = FACE_DIRS[f];
						var nx = wx + d[0], ny = wy + d[1], nz = wz + d[2];
						if (getBlock(nx, ny, nz) === 0) {
							var verts = FACE_VERTS[f];
							var shade = FACE_SHADE[f];
							for (var vi = 0; vi < 4; vi++) {
								var vv = verts[vi];
								pos.push(wx + vv[0], wy + vv[1], wz + vv[2]);
								nrm.push(d[0], d[1], d[2]);
								col.push(br / 255 * shade, bg / 255 * shade, bb / 255 * shade);
							}
			idx.push(vc, vc + 1, vc + 2, vc, vc + 2, vc + 3);
							vc += 4;
						}
					}
				}
			}
		}

		var mesh = c.mesh;
		if (mesh) {
			chunkList.splice(chunkList.indexOf(mesh), 1);
			var old = mesh.geometry;
			mesh.geometry.dispose();
			mesh.parent.remove(mesh);
		}
		if (pos.length === 0) { c.mesh = null; return; }

		var geo = new THREE.BufferGeometry();
		geo.setAttribute("position", new THREE.Float32BufferAttribute(pos, 3));
		geo.setAttribute("normal", new THREE.Float32BufferAttribute(nrm, 3));
		geo.setAttribute("color", new THREE.Float32BufferAttribute(col, 3));
		geo.setIndex(idx);

		var m = new THREE.Mesh(geo, sharedMaterial);
		m.position.set(cx * CHUNK, 0, cz * CHUNK);
		scene.add(m);
		c.mesh = m;
		chunkList.push(m);
	}

	// ---------- Collision ----------
	function collidesAt(bx, by, bz) {
		var minX = Math.floor(bx - PW), maxX = Math.floor(bx + PW);
		var minY = Math.floor(by),     maxY = Math.floor(by + PH);
		var minZ = Math.floor(bz - PW), maxZ = Math.floor(bz + PW);
		for (var y = minY; y <= maxY; y++)
			for (var z = minZ; z <= maxZ; z++)
				for (var x = minX; x <= maxX; x++)
					if (getBlock(x, y, z) !== 0) return true;
		return false;
	}
	function playerOverlaps(bx, by, bz) {
		var minX = Math.floor(playerBlock.x - PW), maxX = Math.floor(playerBlock.x + PW);
		var minY = Math.floor(playerBlock.y),     maxY = Math.floor(playerBlock.y + PH);
		var minZ = Math.floor(playerBlock.z - PW), maxZ = Math.floor(playerBlock.z + PW);
		return (bx >= minX && bx <= maxX &&
				by >= minY && by <= maxY &&
				bz >= minZ && bz <= maxZ);
	}

	function updatePlayer(dt) {
		var forward = new THREE.Vector3(-Math.sin(yaw), 0, -Math.cos(yaw));
		var right   = new THREE.Vector3(Math.cos(yaw), 0, -Math.sin(yaw));

		var wish = new THREE.Vector3();
		if (wantForward) wish.add(forward);
		if (wantBack)    wish.sub(forward);
		if (wantRight)   wish.add(right);
		if (wantLeft)    wish.sub(right);
		if (wish.lengthSq() > 0) wish.normalize().multiplyScalar(WALK * dt);

		camPos.x += wish.x;
		if (collidesAt(camPos.x, camPos.y, camPos.z)) camPos.x -= wish.x;
		camPos.z += wish.z;
		if (collidesAt(camPos.x, camPos.y, camPos.z)) camPos.z -= wish.z;

		if (!hasVel) { vel.set(0, -GRAVITY, 0); hasVel = true; }
		vel.y -= GRAVITY * dt;
		var movedY = vel.y * dt;
		var prevGround = canJump;
		canJump = false;

		camPos.y += movedY;
		if (collidesAt(camPos.x, camPos.y, camPos.z)) {
			if (vel.y < 0) canJump = true;
			camPos.y -= movedY;
		}
		hasVel = camPos.y !== prevGround || movedY !== 0;

		if (camPos.y < -20) {
			camPos.set(8, 0, 8); hasVel = false;
		}

		if (canJump) {
			var groundH = terrainHeightAt(Math.floor(camPos.x), Math.floor(camPos.z));
			if (groundH > camPos.y) camPos.y = groundH;
		}

		playerBlock.copy(camPos);
	}

	function terrainHeightAt(bx, bz) {
		var h = -1;
		for (var y = CHUNK_H - 1; y >= 0; y--) {
			if (getBlock(bx, y, bz) !== 0) { h = y + 1; break; }
		}
		return h < 0 ? 0 : h;
	}

	// ---------- Targeting ----------
	var n = new THREE.Vector3();
	function updateTarget() {
		camDir.set(0, 0, -1).applyEuler(camera.rotation).normalize();
		raycaster.set(camPos, camDir);
		raycaster.far = REACH;
		var hits = raycaster.intersectObjects(chunkList, false);
		if (hits.length > 0) {
			var hit = hits[0];
			n.copy(hit.face.normal);
			targetBlock.set(
				Math.floor(hit.point.x - n.x * 0.5),
				Math.floor(hit.point.y - n.y * 0.5),
				Math.floor(hit.point.z - n.z * 0.5)
			);
			haveTarget = true;
			outline.position.set(targetBlock.x + 0.5, targetBlock.y + 0.5, targetBlock.z + 0.5);
			outline.visible = true;
		} else {
			haveTarget = false;
			outline.visible = false;
		}
	}

	// ---------- World streaming ----------
	var genQueue = [], buildQueue = [];
	function streamWorld() {
		genQueue.length = 0; buildQueue.length = 0;
		var pcx = Math.floor(camPos.x / CHUNK), pcz = Math.floor(camPos.z / CHUNK);
		for (var dz = -5; dz <= 5; dz++) {
			for (var dx = -5; dx <= 5; dx++) {
				var cx = pcx + dx, cz = pcz + dz;
				var key = chunkKey(cx, cz);
				if (!chunks[key]) genQueue.push([cx, cz]);
			}
		}
		genQueue.sort(function (a, b) {
			var da = (a[0] - pcx) * (a[0] - pcx) + (a[1] - pcz) * (a[1] - pcz);
			var db = (b[0] - pcx) * (b[0] - pcx) + (b[1] - pcz) * (b[1] - pcz);
			return db - da;
		});

		var genCount = 0;
		while (genQueue.length > 0 && genCount < 4) {
			var g = genQueue.shift();
			var gc = chunks[chunkKey(g[0], g[1])] ||
				(chunks[chunkKey(g[0], g[1])] = { data: null, mesh: null });
			gc.data = makeChunkData(g[0], g[1]);
			genCount++;
		}

		for (var k = -4; k <= 4; k++) {
			for (var j = -4; j <= 4; j++) {
				var bcx = pcx + k, bcz = pcz + j;
				var bkey = chunkKey(bcx, bcz);
				var cc = chunks[bkey];
				if (cc && cc.data && !cc.mesh) {
					var haveN = true;
					if (!chunks[chunkKey(bcx - 1, bcz)] || !chunks[chunkKey(bcx - 1, bcz)].data) haveN = false;
					if (!chunks[chunkKey(bcx + 1, bcz)] || !chunks[chunkKey(bcx + 1, bcz)].data) haveN = false;
					if (!chunks[chunkKey(bcx, bcz - 1)] || !chunks[chunkKey(bcx, bcz - 1)].data) haveN = false;
					if (!chunks[chunkKey(bcx, bcz + 1)] || !chunks[chunkKey(bcx, bcz + 1)].data) haveN = false;
					if (haveN) buildQueue.push([bcx, bcz]);
				}
			}
		}
		buildQueue.sort(function (a, b) {
			var da = (a[0] - pcx) * (a[0] - pcx) + (a[1] - pcz) * (a[1] - pcz);
			var db = (b[0] - pcx) * (b[0] - pcx) + (b[1] - pcz) * (b[1] - pcz);
			return db - da;
		});

		var buildCount = 0;
		while (buildQueue.length > 0 && buildCount < 2) {
			var b = buildQueue.shift();
			buildChunkMesh(b[0], b[1]);
			buildCount++;
		}

		for (var kk = -7; kk <= 7; kk++) {
			for (var jj = -7; jj <= 7; jj++) {
				var dcx = pcx + kk, dcz = pcz + jj;
				var dkey = chunkKey(dcx, dcz);
				var dd = chunks[dkey];
				if (dd && dd.mesh && (Math.abs(kk) > 4 || Math.abs(jj) > 4)) {
					chunkList.splice(chunkList.indexOf(dd.mesh), 1);
					dd.mesh.geometry.dispose();
					scene.remove(dd.mesh);
					dd.mesh = null;
				}
			}
		}
	}

	// ---------- Clouds & water follow ----------
	function updateClouds() {
		for (var i = 0; i < clouds.length; i++) {
			var c = clouds[i];
			var cx = ((camPos.x + (c.userData.offset * 600)) % 600 + 600) % 600 - 300 + camPos.x;
			var cz = (((1 - c.userData.offset) * 600) % 600 + 600) % 600 - 300 + camPos.z;
			c.position.set(cx, 90, cz);
		}
		water.position.x = camPos.x;
		water.position.z = camPos.z;
	}

	// ---------- Init ----------
	function init() {
		camPos.set(8, 0, 8);
		playerBlock.copy(camPos);
		for (var i = 0; i < 200; i++) streamWorld();
		while (terrainHeightAt(8, 8) < CHUNK_H - 1) {
			var gh = terrainHeightAt(8, 8);
			camPos.y = gh + 1;
			playerBlock.copy(camPos);
			if (gh > 40) break;
		}
	}

	// ---------- Main loop ----------
	var lastT = performance.now();
	var fpsTimer = 0, frames = 0;
	var fpsEl = document.getElementById("fps");

	function animate() {
		requestAnimationFrame(animate);
		var now = performance.now();
		var dt = Math.min((now - lastT) / 1000, 0.05);
		lastT = now;

		updatePlayer(dt);

		camera.position.copy(camPos);
		camera.rotation.set(pitch, yaw, 0);

		if (locked) updateTarget();
		streamWorld();
		updateClouds();

		renderer.render(scene, camera);

		frames++; fpsTimer += dt;
		if (fpsTimer >= 1) { fpsEl.textContent = frames + " fps"; frames = 0; fpsTimer = 0; }
	}

	init();
	animate();
})();
