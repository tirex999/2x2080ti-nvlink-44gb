Причина: скрипт падал ещё до запуска цикла отрисовки `animate()`, поэтому WebGL‑канвас не получал ни одного кадра и оставался прозрачным — на экране виден был только кремовый фон страницы и HUD. Конкретно это делала фатальная ошибка в коде: обращение к несуществующей переменной `headParts` в `makeHamster` (`basePosition(headParts, face)` → `ReferenceError`), а также несостыковка метки цикла `let_owned:` с `break owned` (это вообще `SyntaxError`, который ронял весь скрипт на этапе парсинга). Из‑за этого ни один кадр не рендерился.

Исправил обе ошибки (убрал вызов с `headParts` и переписал цикл поиска объекта при клике), остальное сохранил.

function spawnHamster(fur, belly, pos) {
  const g = makeHamster(fur, belly);
  g.position.copy(pos);
  scene.add(g);

  const h = {
    group: g,
    mode: 'pause',

    job: null,                      // 'wheel' | 'bowl'
    jobT: 0,                        // how long to stay at the job
    bounceT: 0                      // click-hop countdown
  };
  g.userData.hamster = h;
  hamsters.push(h);
}

function lerpAngle(a, b, t) {
  let d = (b - a + Math.PI) % (Math.PI * 2) - Math.PI;
  return a + d * t;
}

function updateHamster(h, dt, worldTime) {
  h.timer -= dt;
  const p = h.group.position;

  // Free-choice pick-up-work logic
  if (h.mode === 'pause' && h.timer <= 0) {
    const roll = Math.random();
    if (roll < 0.18) { h.job = 'wheel'; h.target = WHEEL_POINT.clone(); h.mode = 'towork'; }
    else if (roll < 0.32) { h.job = 'bowl'; h.target = BOWL_POINT.clone(); h.mode = 'towork'; }
    else { h.target = randomWanderTarget(); h.mode = 'wander'; }
  }

  switch (h.mode) {
    case 'wander':
    case 'towork': {
      const dir = h.target.clone().sub(p); dir.y = 0;
      if (dir.length() < 0.25) {
        if (h.mode === 'towork') {
          h.mode = 'work';
          h.jobT = 2.5 + Math.random() * 2.5;
        } else {
          h.mode = 'pause';
          h.timer = 0.6 + Math.random() * 2;
        }
      } else {
        dir.normalize().multiplyScalar(h.speed * dt);
        p.add(dir);
        h.group.rotation.y = lerpAngle(h.group.rotation.y, Math.atan2(dir.x, dir.z), dt * 8);
      }
      break;
    }
    case 'work': {
      h.jobT -= dt;
      if (h.job === 'bowl') {
        // Face the bowl directly (slight offset so they don't stack exactly)
        const seat = h.target.clone();
        seat.x += (p.x > h.target.x ? 0.35 : -0.35);
        p.lerp(seat, dt * 3);
        h.group.rotation.y = lerpAngle(h.group.rotation.y, Math.atan2(h.target.x - p.x, h.target.z - p.z), dt * 4);
      }
      if (h.jobT <= 0) { h.mode = 'pause'; h.timer = 0.6 + Math.random() * 1.5; h.job = null; }
      break;
    }
    case 'pause':
      // little idle sway handled via bobbing below
      break;
  }

  // Bouncy locomotion
  const working = (h.mode === 'work' && h.job === 'wheel');
  const nibbling = (h.mode === 'work' && h.job === 'bowl');
  let bobFreq = 6, bobAmp = 0.05;
  if (h.mode === 'wander' || h.mode === 'towork') { bobFreq = 10; bobAmp = 0.08; }
  if (working)   { bobFreq = 16; bobAmp = 0.1; }
  if (nibbling)  { bobFreq = 8;  bobAmp = 0.04; }
  p.y = 0.55 + Math.sin(worldTime * bobFreq + h.bobPhase) * bobAmp;

  // Squash-and-stretch makes walks cute
  const squash = 1 + Math.sin(worldTime * bobFreq + h.bobPhase) * bobAmp * 1.5;
  h.group.scale.y = squash * (h.bounceT > 0 ? 1.15 : 1);

  // Click-hopping
  if (h.bounceT > 0) {
    h.bounceT -= dt;
    const t = 0.55 - h.bounceT;
    p.y += Math.sin((t / 0.55) * Math.PI) * 0.55;
  }
}

renderer.domElement.addEventListener('click', e => {
  pointer.set((e.clientX / innerWidth) * 2 - 1, -(e.clientY / innerHeight) * 2 + 1);
  raycaster.setFromCamera(pointer, camera);

  const targets = [tire, stand, bowl, ...hamsters.map(h => h.group)];
  const hits = raycaster.intersectObjects(targets, true);
  if (hits.length === 0) return;

  const hitObj = hits[0].object;

  // Walk up the parent chain to figure out which logical object was hit
  let ham = null, isWheel = false;
  let node = hitObj;
  while (node) {
    if (node.userData && node.userData.hamster) { ham = node.userData.hamster; break; }
    if (node === wheelPivot) { isWheel = true; break; }
    node = node.parent;
  }
  if (!isWheel && hitObj === stand) isWheel = true;

  if (ham) {
    ham.bounceT = 0.55;   // happy hop!
  }
  if (isWheel) {
    wheelVel += 14;       // give the wheel a boost
    // and send someone to celebrate in it
    const lazy = hamsters[Math.floor(Math.random() * hamsters.length)];
    lazy.job = 'wheel'; lazy.target = WHEEL_POINT.clone();
    lazy.mode = 'towork'; lazy.timer = 10;
  }
  if (hitObj === bowl) {
    const hungry = hamsters[Math.floor(Math.random() * hamsters.length)];
    hungry.job = 'bowl'; hungry.target = BOWL_POINT.clone();
    hungry.mode = 'towork'; hungry.timer = 10;

});

// ---------- Main loop ----------
const clock = new THREE.Clock();
function animate() {
  requestAnimationFrame(animate);
  const dt = Math.min(clock.getDelta(), 0.05);
  const t = clock.elapsedTime;

  hamsters.forEach(h => updateHamster(h, dt, t));

  // Wheel slows down smoothly unless a hamster is riding it
  const targetSpeed = hamsters.some(h => h.mode === 'work' && h.job === 'wheel') ? 9 : 0;
  wheelVel += (targetSpeed - wheelVel) * Math.min(dt * 1.5, 1);
  wheelPivot.rotation.y += wheelVel * dt;

  updateCamera();
  renderer.render(scene, camera);
}
