function randomSpot(){

function arrive(h){
  switch(h.arrive){
    case 'chew':
      h.state='chew'; h.t=R(2.2,4.0);
      break;
    case 'spin':
      h.state='spin'; h.t=R(3.5,6.0); h.rotY = lerpAngle(h.rotY, -Math.PI/2, 1);
      break;
    case 'tunnel':
      h.state='tuck'; h.t=.45; h.hFtarget=.3; h.rotY = Math.PI/2;
      break;
    default:
      decide(h);
  }
  h.arrive='idle';
}

/* ---------------- per-hamster frame update ---------------- */
function stepHam(h, dt, tAll){
  h.blinkT -= dt;
  if(h.blinkT < 0) h.blinkT = R(1.6,4.5);
  h.hF = damp3(h.hF, h.hFtarget, 7, dt);
  if(h.twitch > 0) h.twitch -= dt;

  switch(h.state){
    case 'idle':
      h.t -= dt;
      if(h.t <= 0) act(h);
      break;

    case 'walk': {
      const to = new THREE.Vector3(h.target.x - h.root.position.x, 0, h.target.z - h.root.position.z);
      const dist = to.length();
      if(dist < .05 || dist <= h.speed*dt){
        h.root.position.x = h.target.x; h.root.position.z = h.target.z;
        arrive(h); break;
      }
      to.normalize();
      h.root.position.addScaledVector(to, h.speed*dt);
      h.rotY = lerpAngle(h.rotY, Math.atan2(to.x, to.z), 1-Math.exp(-6.5*dt));
      h.ph += dt*10;
      break;
    }

    case 'chew': {
      h.t -= dt;
      h.rotY = lerpAngle(h.rotY, Math.atan2(BOWL.x-h.root.position.x, BOWL.z-h.root.position.z), 1-Math.exp(-6*dt));
      h.chew = (h.chew||0) + dt;
      if(Math.random() < dt*.7)
        spawnFx(h.root.position.clone().add(new THREE.Vector3(0,.9,.2)), FX_SETS.nibble, 1);
      if(h.t <= 0){
        if(h.chair){ h.chair.owner = null; h.chair = null; }
        decide(h);
      }
      break;
    }

    case 'tuck':
      h.t -= dt;
      if(h.t <= 0){
        h.root.position.copy(TUN.in);
        h.state='hiding'; h.t=R(.7,1.2);
      }
      break;

    case 'hiding':
      h.t -= dt;
      if(h.t <= 0) tunnelFinish(h);
      break;

    case 'pop':
      h.t -= dt;
      if(h.t <= 0) decide(h);
      break;

    case 'spin': {
      h.t -= dt;
      h.starAcc += dt;
      if(h.starAcc > .8){ h.starAcc = 0;
        spawnFx(h.root.position.clone().add(new THREE.Vector3(-.6,.9,0)), FX_SETS.spin, 1); }
      if(h.t <= 0){
        if(wheelHam === h) wheelHam = null;
        h.rotY = lerpAngle(h.rotY, Math.atan2(1,0), 1.0);
        decide(h);
      }
      break;
    }
  }

  // gentle separation between hamsters
  for(const o of hams){
    if(o===h) continue;
    if(h.hF < .55) continue;
    const dx = h.root.position.x-o.root.position.x, dz = h.root.position.z-o.root.position.z;
    const d = Math.hypot(dx,dz);
    if(d > .001 && d < .8){
      const push = (.8-d)/2, nx = dx/d, nz = dz/d;
      if(h.state !== 'tuck' && h.state !== 'hiding'){
        h.root.position.x += nx*push; h.root.position.z += nz*push;
      }
      if(o.state !== 'tuck' && o.state !== 'hiding'){
        o.root.position.x -= nx*push; o.root.position.z -= nz*push;
      }
    }
  }
  h.root.position.x = THREE.MathUtils.clamp(h.root.position.x, -TRAY_HW+.4, TRAY_HW-.4);
  h.root.position.z = THREE.MathUtils.clamp(h.root.position.z, -TRAY_HZ+.4, TRAY_HZ-.4);

  /* --- pose targets --- */
  const moving = h.state==='walk';
  const squat  = (h.state==='chew' || h.state==='spin' || h.state==='idle');
  let sX = 1, sY = 1, sZ = 1, yBob = 0;
  if(moving){
    sX = 1.04; sZ = .97;
    sY = .98 + Math.sin(h.ph)*.05;
    yBob = Math.abs(Math.sin(h.ph))*0.07;
  } else if(squat){
    sX = 1.07; sZ = 1.07; sY = .9;
  }
  if(h.state==='hiding'){ sX=sY=sZ = 1; }

  h.tilt = (moving ? Math.sin(h.ph)*.04 : 0)
         + (h.state==='chew' ? Math.sin((h.chew||0)*16)*.1 : 0)
         + (h.state==='spin' ? Math.abs(Math.sin(tAll*9))*-.05 : 0)
         + (h.twitch>0 ? Math.sin(h.twitch*25)*.3*h.twitch*3 : 0);

  h.sX = damp3(h.sX, sX, 8, dt);
  h.sY = damp3(h.sY, sY, 8, dt);
  h.sZ = damp3(h.sZ, sZ, 8, dt);
  h.cY = damp3(h.cY, yBob, 10, dt);
  h.root.scale.set( h.sX*h.hF, h.sY*h.hF, h.sZ*h.hF );
  h.root.position.y = h.cY;
  h.root.rotation.set(0, h.rotY, h.tilt);

  /* breathing */
  const br = 1 + Math.sin(tAll*2.2 + h.id)*.016;
  h.body.scale.set(1.3*br, 1.05/br, 1.15/br);

  /* blink */
  const eyeY = (h.blinkT < .12) ? .15 : 1;
  for(const e of h.eyes) e.scale.y = eyeY;

  /* paws shuffling */
  if(moving) h.pawsFeet.forEach((f,i)=>{
    f.m.position.y = f.by - .05 + Math.abs(Math.sin(h.ph + i*1.6))*.07;
  });
}

  for(const hit of hits){
    let o = hit.object;
    while(o && !hams.includes(o)) o = o.parent;
    const h = hams.find(x=>x.root===o);

      h.twitch = .6;

/* ---------------- main loop ---------------- */
const clock = new THREE.Clock();
function tick(){
  requestAnimationFrame(tick);
  const dt = Math.min(clock.getDelta(), .05);
  const tAll = clock.elapsedTime;

  for(const h of hams) stepHam(h, dt, tAll);

  // wheel animation
  const targetV = wheelHam ? 7.5 : 0;
  wheelVel += (targetV - wheelVel) * (1 - Math.exp(-(wheelHam?3:1.6)*dt));
  if(Math.abs(wheelVel) > .01) wheelSpin.rotation.x -= wheelVel*dt;

  // emoji particles
  for(let i=fxList.length-1; i>=0; i--){
    const f = fxList[i];
    f.age += dt;
    f.s.position.y += f.vy*dt;
    f.s.position.x += f.dx*dt;
    f.s.position.z += f.dz*dt;
    f.s.material.opacity = 1 - f.age/f.dur;
    f.s.scale.setScalar(.35 + f.age*.5);
    if(f.age >= f.dur){ scene.remove(f.s); f.s.material.dispose(); fxList.splice(i,1); }
  }

  controls.update();
  renderer.render(scene, camera);
}
tick();
