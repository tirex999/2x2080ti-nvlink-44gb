import * as THREE from 'three';
import { OrbitControls } from 'three/addons/controls/OrbitControls.js';

// ---------- Renderer / Scene / Camera ----------
const renderer = new THREE.WebGLRenderer({ antialias: true });
renderer.setPixelRatio(Math.min(devicePixelRatio, 2));
renderer.setSize(innerWidth, innerHeight);
renderer.shadowMap.enabled = true;
renderer.shadowMap.type = THREE.PCFSoftShadowMap;
document.body.appendChild(renderer.domElement);

const scene = new THREE.Scene();
scene.background = new THREE.Color(0xffe9f0);
scene.fog = new THREE.Fog(0xffe9f0, 18, 34);

const camera = new THREE.PerspectiveCamera(50, innerWidth / innerHeight, 0.1, 100);
camera.position.set(7.5, 6, 8.5);

const controls = new OrbitControls(camera, renderer.domElement);
controls.target.set(0, 1.2, 0);
controls.enableDamping = true;
controls.dampingFactor = 0.08;
controls.minDistance = 5;
controls.maxDistance = 16;
controls.maxPolarAngle = Math.PI * 0.48;

// ---------- Lights ----------
scene.add(new THREE.HemisphereLight(0xfff5ff, 0xffd9b0, 0.9));
const sun = new THREE.DirectionalLight(0xfff2df, 1.4);
sun.position.set(6, 10, 4);
sun.castShadow = true;
sun.shadow.mapSize.set(2048, 2048);
sun.shadow.camera.left = -8; sun.shadow.camera.right = 8;
sun.shadow.camera.top = 8;  sun.shadow.camera.bottom = -8;
scene.add(sun);

// ---------- Helpers ----------
const mat = (color, opts = {}) => new THREE.MeshStandardMaterial({ color, flatShading: true, roughness: 0.9, ...opts });
function mesh(geo, material, x = 0, y = 0, z = 0) {
  const m = new THREE.Mesh(geo, material);
  m.position.set(x, y, z);
  m.castShadow = true; m.receiveShadow = true;
  return m;
}

// ---------- Ground ----------
const ground = new THREE.Mesh(new THREE.CircleGeometry(14, 8), mat(0xf8cfe0));
ground.rotation.x = -Math.PI / 2;
ground.position.y = -0.3;
ground.receiveShadow = true;
scene.add(ground);

// ---------- Cage (tray + wire bars) ----------
const CAGE_R = 4, CAGE_H = 3;
{
  // tray
  const tray = mesh(new THREE.BoxGeometry(CAGE_R * 2 + 0.6, 0.35, CAGE_R * 2 + 0.6), mat(0xc98bc7), 0, -0.13, 0);
  scene.add(tray);
  const sand = mesh(new THREE.BoxGeometry(CAGE_R * 2 - 0.1, 0.12, CAGE_R * 2 - 0.1), mat(0xf0d5a0), 0, 0.12, 0);
  scene.add(sand);

  // wire bars
  const barMat = mat(0xbfd3dd, { roughness: 0.5, metalness: 0.4 });
  const bars = 18;
  for (let i = 0; i < bars; i++) {
    const a = (i / bars) * Math.PI * 2;
    const x = Math.cos(a) * CAGE_R, z = Math.sin(a) * CAGE_R;
    scene.add(mesh(new THREE.CylinderGeometry(0.04, 0.04, CAGE_H, 5), barMat, x, CAGE_H / 2 + 0.05, z));
  }
  const ringGeo = new THREE.TorusGeometry(CAGE_R, 0.08, 5, 24);
  const topRing = mesh(ringGeo, mat(0xa85fc0, { roughness: 0.6 }), 0, CAGE_H + 0.05, 0);
  topRing.rotation.x = Math.PI / 2;
  const botRing = mesh(ringGeo, mat(0xa85fc0, { roughness: 0.6 }), 0, 0.2, 0);
  botRing.rotation.x = Math.PI / 2;
  scene.add(topRing, botRing);
}

// ---------- Exercise wheel ----------
const wheelPos = new THREE.Vector3(2.7, 0, -1.1);
const wheel = new THREE.Group();
{
  const frameMat = mat(0x8ed0e8, { roughness: 0.6 });
  wheel.add(mesh(new THREE.BoxGeometry(0.7, 0.15, 0.7), frameMat, 0, 0.07, 0));      // base
  wheel.add(mesh(new THREE.CylinderGeometry(0.06, 0.08, 1.15, 6), frameMat, 0, 0.62, 0)); // post
  const hubH = 1.35;
  wheel.add(mesh(new THREE.CylinderGeometry(0.07, 0.07, 0.5, 6), frameMat, 0, hubH, 0));

  const rim = mesh(new THREE.TorusGeometry(0.95, 0.09, 6, 20), mat(0xff9f68));
  rim.rotation.y = Math.PI / 2;
  rim.position.y = hubH;
  const spokes = new THREE.Group();
  spokes.position.y = hubH;
  for (let i = 0; i < 5; i++) {
    const s = mesh(new THREE.CylinderGeometry(0.035, 0.035, 1.8, 5), mat(0xffc49e));
    s.rotation.z = (i / 5) * Math.PI;
    spokes.add(s);
  }
  wheel.add(rim, spokes);
  wheel.userData.spokes = spokes;
  wheel.userData.rim = rim;
}
wheel.position.copy(wheelPos);
wheel.rotation.y = -Math.atan2(wheelPos.y, -wheelPos.x) ; // face wheel toward center
scene.add(wheel);
let wheelSpeed = 0;

// ---------- Food bowl ----------
const bowlPos = new THREE.Vector3(-2.6, 0.18, 1.6);
{
  scene.add(mesh(new THREE.CylinderGeometry(0.55, 0.35, 0.28, 8), mat(0x9bd48b), bowlPos.x, 0.14, bowlPos.z));
  for (let i = 0; i < 4; i++) {
    const a = i * 1.7;
    scene.add(mesh(new THREE.SphereGeometry(0.09, 5, 4), mat(0x8a5a3b),
      bowlPos.x + Math.cos(a) * 0.2, 0.32, bowlPos.z + Math.sin(a) * 0.2));
  }
}

// ---------- Tunnel (decorative) ----------
{
  const tunnel = mesh(
    new THREE.CylinderGeometry(0.7, 0.7, 1.5, 8, 1, false, 0, Math.PI),
    mat(0xffb3d9), -1.4, 0.18, -2.5);
  tunnel.rotation.set(Math.PI / 2, 0, Math.PI / 2);
  tunnel.material.side = THREE.DoubleSide;
  scene.add(tunnel);
}

// ---------- Hamsters ----------
function makeHamster(bodyColor) {
  const g = new THREE.Group();
  const fur = mat(bodyColor);
  const cream = mat(0xfff3dc);

  const body = mesh(new THREE.SphereGeometry(0.45, 7, 5), fur, 0, 0.5, 0);
  body.scale.set(1, 0.85, 1.15);
  const belly = mesh(new THREE.SphereGeometry(0.34, 6, 4), cream, 0, 0.4, 0.18);
  belly.scale.set(0.9, 0.8, 1);
  const head = mesh(new THREE.SphereGeometry(0.3, 6, 5), fur, 0, 0.55, 0.5);
  const cheekL = mesh(new THREE.SphereGeometry(0.12, 5, 4), cream, -0.17, 0.47, 0.62);
  const cheekR = cheekL.clone(); cheekR.position.x = 0.17;
  const earL = mesh(new THREE.SphereGeometry(0.11, 5, 4), fur, -0.16, 0.78, 0.38);
  const earR = earL.clone(); earR.position.x = 0.16;
  earL.scale.set(1, 0.6, 1); earR.scale.set(1, 0.6, 1);
  const nose = mesh(new THREE.SphereGeometry(0.05, 5, 4), mat(0xf48fb1), 0, 0.52, 0.8);
  const eyeGeo = new THREE.SphereGeometry(0.05, 5, 4);
  const eyeMat = mat(0x2b2b2b);
  const eyeL = new THREE.Mesh(eyeGeo, eyeMat); eyeL.position.set(-0.14, 0.62, 0.74);
  const eyeR = eyeL.clone(); eyeR.position.x = 0.14;
  const footGeo = new THREE.SphereGeometry(0.08, 5, 4);
  const feet = [];
  [[-0.22, -0.28], [0.22, -0.28], [-0.22, 0.35], [0.22, 0.35]].forEach(([fx, fz]) => {
    const f = new THREE.Mesh(footGeo, eyeMat);
    f.position.set(fx, 0.08, fz);
    f.scale.set(1, 0.5, 1.4);
    feet.push(f);
  });
  const tail = mesh(new THREE.SphereGeometry(0.07, 5, 4), fur, 0, 0.42, -0.5);
  tail.visible = false;

  g.add(body, belly, head, cheekL, cheekR, earL, earR, nose, eyeL, eyeR, tail, ...feet);
  g.userData = { body, head, eyeL, eyeR, feet };
  return g;
}

const HAMSTER_COLORS = [0xf2b06a, 0xf7ead9, 0xc9c3d9, 0xe88a6a];
const hamsters = HAMSTER_COLORS.map((c, i) => {
  const a = i / HAMSTER_COLORS.length * Math.PI * 2;
  const h = {
    obj: makeHamster(c),
    pos: new THREE.Vector3(Math.cos(a) * 1.8, 0, Math.sin(a) * 1.8),
    heading: Math.random() * Math.PI * 2,
    state: 'pause',
    timer: 1 + Math.random() * 2,
    target: new THREE.Vector3(),
    interacting: false,
    blinkTimer: 2 + Math.random() * 4,
    seed: Math.random() * 10
  };
  h.obj.position.copy(h.pos);
  scene.add(h.obj);
  return h;
});

const rand = (a, b) => a + Math.random() * (b - a);

function pickTarget(h) {
  const a = Math.random() * Math.PI * 2, r = Math.sqrt(Math.random()) * 3.1;
  h.target.set(Math.cos(a) * r, 0, Math.sin(a) * r);
}

function chooseState(h) {
  const r = Math.random();
  if (r < 0.45) { h.state = 'pause'; h.timer = rand(0.8, 2.5); }
  else if (r < 0.75) { h.state = 'wander'; h.timer = rand(2, 5); pickTarget(h); }
  else if (r < 0.92) { h.state = 'wheel'; h.timer = rand(4, 7); }
  else { h.state = 'eat'; h.timer = rand(3, 5); }
}

// wheel stand spot (beside wheel, facing it)
const wheelSpot = new THREE.Vector3(wheelPos.x - 1.1, 0, wheelPos.z + 0.45);
const eatSpot = new THREE.Vector3(bowlPos.x + 0.4, 0, bowlPos.z - 0.9);

function updateHamster(h, dt, t) {
  h.timer -= dt;
  const { obj, pos } = h;
  const ud = obj.userData;
  let speed = 0;

  if (h.state === 'pause') {
    if (h.timer <= 0) chooseState(h);
    // sleepy squash
    ud.body.scale.y = 0.85 + Math.sin(t * 3 + h.seed) * 0.03;
  }
  else if (h.state === 'wander') {
    const dx = h.target.x - pos.x, dz = h.target.z - pos.z;
    if (dx * dx + dz * dz < 0.12 || h.timer <= 0) chooseState(h);
    else {
      const want = Math.atan2(dx, dz);
      h.heading = lerpAngle(h.heading, want, dt * 4);
      speed = 1.4;
    }
  }
  else if (h.state === 'wheel') {
    const done = pos.distanceTo(wheelSpot) < 0.35 &&
                 Math.abs(lerpAngle(h.heading, Math.atan2(wheelPos.x - pos.x, wheelPos.z - pos.z), 0.5)) < 0.7;
    if (done) {
      h.interacting = true;
      // frantic wiggling on the wheel
      ud.body.rotation.z = Math.sin(t * 18 + h.seed) * 0.12;
      obj.position.y = Math.abs(Math.sin(t * 14 + h.seed)) * 0.15;
      if (h.timer <= 0) { h.interacting = false; obj.position.y = 0; ud.body.rotation.z = 0; h.state = 'pause'; h.timer = rand(1, 2); }
    } else {
      const dx = wheelSpot.x - pos.x, dz = wheelSpot.z - pos.z;
      h.heading = lerpAngle(h.heading, Math.atan2(dx, dz), dt * 4);
      speed = 2.0; // sprint!
    }
  }
  else if (h.state === 'eat') {
    if (pos.distanceTo(eatSpot) < 0.4) {
      h.interacting = true;
      ud.head.position.y = 0.55 + Math.sin(t * 10 + h.seed) * 0.05 - 0.06; // nibbling
      ud.head.rotation.x = 0.25 + Math.sin(t * 10 + h.seed) * 0.1;
      if (h.timer <= 0) {
        h.interacting = false;
        ud.head.position.y = 0.55; ud.head.rotation.x = 0;
        h.state = 'pause'; h.timer = rand(1, 2);
      }
    } else {
      const dx = eatSpot.x - pos.x, dz = eatSpot.z - pos.z;
      h.heading = lerpAngle(h.heading, Math.atan2(dx, dz), dt * 4);
      speed = 1.6;
    }
  }

  if (speed > 0) {
    pos.x += Math.sin(h.heading) * speed * dt;
    pos.z += Math.cos(h.heading) * speed * dt;
    // keep inside cage
    const r = Math.hypot(pos.x, pos.z);
    if (r > 3.3) { pos.x *= 3.3 / r; pos.z *= 3.3 / r; }
    // walk bob + leg scurry
    const step = Math.sin(t * (8 + speed * 4) + h.seed) * 0.04;
    obj.position.y = Math.abs(step);
    obj.rotation.z = Math.sin(t * 10 + h.seed) * 0.06;
    ud.feet.forEach((f, i) => { f.position.y = 0.08 + Math.sin(t * 16 + i * Math.PI) * 0.05 * (speed > 0 ? 1 : 0); });
    ud.body.rotation.z = Math.sin(t * 10 + h.seed) * 0.04;
  }
  obj.position.x = pos.x;
  obj.position.z = pos.z;
  obj.rotation.y = h.heading;

  // blinking
  h.blinkTimer -= dt;
  if (h.blinkTimer < 0) h.blinkTimer = rand(2, 5);
  const squint = h.blinkTimer < 0.12 ? 0.1 : 1;
  ud.eyeL.scale.y = ud.eyeR.scale.y = squint;
}

function lerpAngle(a, b, k) {
  let d = (b - a) % (Math.PI * 2);
  if (d > Math.PI) d -= Math.PI * 2;
  if (d < -Math.PI) d += Math.PI * 2;
  return a + d * Math.min(1, k);
}

// gentle separation so they don't stack
function separate() {
  for (let i = 0; i < hamsters.length; i++) {
    for (let j = i + 1; j < hamsters.length; j++) {
      const A = hamsters[i].pos, B = hamsters[j].pos;
      const dx = B.x - A.x, dz = B.z - A.z;
      const d = Math.hypot(dx, dz);
      if (d < 0.85 && d > 0.001) {
        const push = (0.85 - d) / 2;
        A.x -= (dx / d) * push; A.z -= (dz / d) * push;
        B.x += (dx / d) * push; B.z += (dz / d) * push;
      }
    }
  }
}

// ---------- Main loop ----------
const clock = new THREE.Clock();
function animate() {
  requestAnimationFrame(animate);
  const dt = Math.min(clock.getDelta(), 0.05);
  const t = clock.elapsedTime;

  hamsters.forEach(h => updateHamster(h, dt, t));
  separate();

  // wheel physics: any hamster in there makes it spin, friction slows it
  const anyoneRunning = hamsters.some(h => h.interacting && h.state === 'wheel');
  wheelSpeed += ((anyoneRunning ? 9 : 0) - wheelSpeed) * Math.min(1, dt * 2.5);
  wheelSpeed *= (anyoneRunning ? 1 : 0.995 - dt);
  wheel.userData.rim.rotation.x += wheelSpeed * dt;
  wheel.userData.spokes.rotation.x = wheel.userData.rim.rotation.x;

  controls.update();
  renderer.render(scene, camera);
}
animate();

addEventListener('resize', () => {
  camera.aspect = innerWidth / innerHeight;
  camera.updateProjectionMatrix();
  renderer.setSize(innerWidth, innerHeight);
});
</script>
</body>
</html>

## What you get

- **🏠 Cage** — wire-bar round cage with pink rims sitting on a lavender tray filled with sand
- **🐹 4 unique hamsters** — cream, ginger, gray, and salmon; each with low-poly bodies, blush cheeks, round ears, blinking eyes, and scurrying feet
- **🎡 Interactive wheel** — hamsters randomly sprint over to it, wiggle frantically on it, and the rim/spokes actually spin up with momentum and coast to a stop when they leave
- **🥣 Food bowl** — hamsters occasionally waddle over and bobble their heads, nibbling
- **🛝 Decorations** — a pink half-tube tunnel, kibble in the bowl, fog-softened pastel ground
- **🎮 Controls** — drag to orbit, scroll to zoom, with smooth damping and sensible limits

**Behavior details:** each hamster runs a simple state machine (`wander` / `pause` / `wheel` / `eat`) with randomized timers, steers smoothly toward targets, squashes its body while idling, bobs while walking, and pairs gently repel each other so they never overlap. Everything runs on delta-time so it stays stable regardless of frame rate. No build step needed — just open the file.
