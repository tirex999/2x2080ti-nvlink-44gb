(function(){

// ---------- Globals ----------
const W = 36, H = 24, D = 20;                 // tank size
const BOUNDS = new THREE.Vector3(W/2 - 2, H/2 - 2, D/2 - 2); // swim bounds
const FLOOR_Y = -H/2 + 0.6;                      // sand surface
const WATER_TOP = H/2;                             // water surface
const clock = new THREE.Clock();

let scene, camera, renderer, controls, raycaster;
let dirLight, pointLightA, pointLightB;
let fishArray = [], bubbles = [], foods = [];
let seaweeds = [];
let lightOn = true;

// shared materials / geometries
const matBody = new THREE.MeshStandardMaterial({ color: 0xffffff, roughness: 0.55, metalness: 0.1 });
const matEye = new THREE.MeshStandardMaterial({ color: 0xf5f5f5, roughness: 0.25, metalness: 0.1 });
const matPupil = new THREE.MeshStandardMaterial({ color: 0x101018, roughness: 0.3 });
const matFin = new THREE.MeshStandardMaterial({ color: 0xffffff, roughness: 0.5, transparent: true, opacity: 0.85 });

const geoBody = new THREE.SphereGeometry(1, 20, 14);
const geoEye = new THREE.SphereGeometry(0.16, 12, 10);
const geoPupil = new THREE.SphereGeometry(0.08, 10, 8);
const geoTail = new THREE.ConeGeometry(0.55, 1.0, 10);
const geoFin = new THREE.ConeGeometry(0.35, 0.7, 8);
const geoFood = new THREE.SphereGeometry(0.28, 10, 8);
const geoBubble = new THREE.SphereGeometry(1, 12, 10);
const matFood = new THREE.MeshStandardMaterial({ color: 0xffcf5a, roughness: 0.5, emissive: 0x553300 });

});

// 8 color schemes (body, fin)

// ---------- Tank (glass) ----------
function buildTank(){
  const geo = new THREE.BoxGeometry(W, H, D);
  const glassMat = new THREE.MeshPhysicalMaterial({
    color: 0xaad4ff, roughness: 0.05, metalness: 0,
    transmission: 0.95, transparent: true, opacity: 0.25,
    clearcoat: 1.0, clearcoatRoughness: 0.1, side: THREE.DoubleSide
  });
  const box = new THREE.Mesh(geo, glassMat);
  box.castShadow = false; box.receiveShadow = false;
  scene.add(box);

  // wireframe frame
  const edges = new THREE.EdgesGeometry(geo);
  const line = new THREE.LineSegments(edges, new THREE.LineBasicMaterial({ color: 0x6fb0ff, transparent: true, opacity: 0.6 }));
  scene.add(line);
}

// ---------- Sand floor ----------
function buildSand(){
  const seg = 32;
  const geo = new THREE.PlaneGeometry(W - 0.4, D - 0.4, seg, seg);
  geo.rotateX(-Math.PI/2);
  const pos = geo.attributes.position;
  for(let i=0;i<pos.count;i++){
    const x = pos.getX(i), z = pos.getZ(i);
    pos.setY(i, FLOOR_Y + Math.sin(x*0.5)*Math.cos(z*0.5)*0.35 + Math.random()*0.15);
  }
  geo.computeVertexNormals();
  const mat = new THREE.MeshStandardMaterial({ color: 0xd8bd86, roughness: 0.95, metalness: 0 });
  const sand = new THREE.Mesh(geo, mat);
  sand.receiveShadow = true;
  scene.add(sand);
}

// ---------- Rocks ----------
function buildRocks(){
  for(let i=0;i<8;i++){
    const r = 0.6 + Math.random()*1.2;
    const g = new THREE.DodecahedronGeometry(r, 0);
    // deform
    const p = g.attributes.position;
    for(let j=0;j<p.count;j++){
      p.setXYZ(j,
        p.getX(j)*(0.8+Math.random()*0.5),
        p.getY(j)*(0.7+Math.random()*0.5),
        p.getZ(j)*(0.8+Math.random()*0.5));
    }
    g.computeVertexNormals();
    const m = new THREE.Mesh(g, new THREE.MeshStandardMaterial({ color: 0x6a6f78, roughness: 0.9 }));
    m.position.set((Math.random()-0.5)*(W-6), FLOOR_Y + r*0.4, (Math.random()-0.5)*(D-6));
    m.rotation.set(Math.random()*Math.PI, Math.random()*Math.PI, Math.random()*Math.PI);
    m.castShadow = true; m.receiveShadow = true;
    scene.add(m);
  }
}

// ---------- Seaweed ----------
function buildSeaweed(){
  for(let i=0;i<12;i++){
    const h = 3 + Math.random()*4;
    const base = new THREE.Vector3((Math.random()-0.5)*(W-6), FLOOR_Y, (Math.random()-0.5)*(D-6));
    const pts = [];
    const lean = (Math.random()-0.5)*0.6;
    for(let s=0;s<=6;s++){
      const t = s/6;
      pts.push(new THREE.Vector3(
        base.x + Math.sin(t*Math.PI*2 + i)*0.3*t + lean*h*t,
        base.y + h*t,
        base.z + Math.cos(t*Math.PI*2 + i)*0.3*t
      ));
    }
    const curve = new THREE.CatmullRomCurve3(pts);
    const tube = new THREE.TubeGeometry(curve, 20, 0.18, 6, false);
    const hue = 0.3 + Math.random()*0.1;
    const mat = new THREE.MeshStandardMaterial({ color: new THREE.Color().setHSL(hue, 0.6, 0.35), roughness: 0.7 });
    const mesh = new THREE.Mesh(tube, mat);
    mesh.castShadow = true;
    scene.add(mesh);
    seaweeds.push({ mesh, phase: Math.random()*Math.PI*2, amp: 0.05 + Math.random()*0.05 });
  }
}

// ---------- Fish ----------
function createFishMesh(scheme){
  const g = new THREE.Group();

  // body (scaled sphere)
  const body = new THREE.Mesh(geoBody, matBody.clone());
  body.scale.set(1.0, 0.55, 0.42);
  body.material.color.setHex(scheme.body);
  body.castShadow = true;
  g.add(body);

  // eyes + pupils
  [-1,1].forEach(side=>{
    const eye = new THREE.Mesh(geoEye, matEye);
    eye.position.set(0.55, 0.05, side*0.28);
    g.add(eye);
    const pup = new THREE.Mesh(geoPupil, matPupil);
    pup.position.set(0.62, 0.05, side*0.30);
    g.add(pup);
  });

  // tail (cone) - pivot at back
  const tail = new THREE.Mesh(geoTail, matFin.clone());
  tail.material.color.setHex(scheme.fin);
  tail.material.side = THREE.DoubleSide;
  tail.rotation.z = Math.PI/2;          // point backwards along -x
  tail.position.x = -0.9;
  tail.castShadow = true;
  g.add(tail);

  // dorsal fin (top)
  const topFin = new THREE.Mesh(geoFin, matFin.clone());
  topFin.material.color.setHex(scheme.fin);
  topFin.material.side = THREE.DoubleSide;
  topFin.position.set(-0.1, 0.35, 0);
  g.add(topFin);

  // side fins
  const lf = new THREE.Mesh(geoFin, matFin.clone());
  lf.material.color.setHex(scheme.fin);
  lf.material.side = THREE.DoubleSide;
  lf.position.set(0.1, -0.15, 0.3);
  lf.rotation.x = Math.PI/2;
  g.add(lf);
  const rf = lf.clone();
  rf.position.z = -0.3;
  g.add(rf);

  return { group: g, tail, topFin, leftFin: lf, rightFin: rf };
}

function spawnFish(n){
  for(let i=0;i<n;i++){
    const scheme = SCHEMES[Math.floor(Math.random()*SCHEMES.length)];
    const built = createFishMesh(scheme);
    const scale = 0.6 + Math.random()*0.6;
    built.group.scale.setScalar(scale);
    built.group.position.set(
      (Math.random()-0.5)*(W-6),
      (Math.random()-0.5)*(H-6),
      (Math.random()-0.5)*(D-6)
    );
    scene.add(built.group);

    const v = new THREE.Vector3((Math.random()-0.5), (Math.random()-0.5), (Math.random()-0.5)).normalize();
    fishArray.push({
      mesh: built.group,
      tail: built.tail,
      topFin: built.topFin,
      leftFin: built.leftFin,
      rightFin: built.rightFin,
      velocity: v,
      speed: 3 + Math.random()*3,
      tailSpeed: 4 + Math.random()*3,
      phase: Math.random()*Math.PI*2,
      targetFood: null,
      avoidanceRadius: 3.2,
      wanderTimer: Math.random()*4
    });
  }
  updateCounts();
}

// ---------- Bubbles ----------
function spawnBubbles(n){
  for(let i=0;i<n;i++){
    const s = 0.15 + Math.random()*0.4;
    const m = new THREE.Mesh(geoBubble, matBubble);
    m.scale.setScalar(s);
    m.position.set(
      (Math.random()-0.5)*(W-4),
      FLOOR_Y + Math.random()*(H-4),
      (Math.random()-0.5)*(D-4)
    );
    scene.add(m);
    bubbles.push({
      mesh: m,
      vel: 0.6 + Math.random()*1.4,
      sway: Math.random()*Math.PI*2,
      swayAmt: 0.3 + Math.random()*0.5
    });
  }
  updateCounts();
}

// ---------- Food ----------
function addFood(pos){
  const m = new THREE.Mesh(geoFood, matFood);
  m.position.copy(pos);
  m.castShadow = true;
  scene.add(m);
  foods.push({ mesh: m, vel: new THREE.Vector3(0, 0.5, 0), age: 0 });
}

// ---------- Update loops ----------
const _dir = new THREE.Vector3();
const _to = new THREE.Vector3();

function updateFish(dt, t){
  for(const f of fishArray){
    // --- food seeking ---
    if(!f.targetFood || !foods.includes(f.targetFood)){
      f.targetFood = null;
      for(const fd of foods){
        if(fd.mesh.position.distanceTo(f.mesh.position) < 15){
          f.targetFood = fd; break;
        }
      }
    }
    if(f.targetFood){
      _to.copy(f.targetFood.mesh.position).sub(f.mesh.position);
      const d = _to.length();
      if(d < 0.7){
        // eat
        f.mesh.scale.multiplyScalar(1.05);
        scene.remove(f.targetFood.mesh);
        foods.splice(foods.indexOf(f.targetFood), 1);
        f.targetFood = null;
      } else {
        _to.normalize();
        f.velocity.lerp(_to, 0.05).normalize();
      }
    } else {
      // --- random wandering ---
      f.wanderTimer -= dt;
      if(f.wanderTimer <= 0){
        f.wanderTimer = 2 + Math.random()*4;
        f.velocity.add(
          new THREE.Vector3((Math.random()-0.5), (Math.random()-0.5), (Math.random()-0.5))
        ).normalize();
      }
    }

    // --- collision avoidance ---
    for(const o of fishArray){
      if(o === f) continue;
      _to.copy(f.mesh.position).sub(o.mesh.position);
      const d = _to.length();
      if(d < f.avoidanceRadius && d > 0.0001){
        _to.normalize();
        f.velocity.addScaledVector(_to, (f.avoidanceRadius - d)/f.avoidanceRadius * 0.5);
      }
    }

    // --- wall reflection ---
    const p = f.mesh.position;
    if(p.x >  BOUNDS.x){ f.velocity.x = -Math.abs(f.velocity.x); p.x = BOUNDS.x; }
    if(p.x < -BOUNDS.x){ f.velocity.x =  Math.abs(f.velocity.x); p.x = -BOUNDS.x; }
    if(p.y >  BOUNDS.y){ f.velocity.y = -Math.abs(f.velocity.y); p.y = BOUNDS.y; }
    if(p.y < -BOUNDS.y){ f.velocity.y =  Math.abs(f.velocity.y); p.y = -BOUNDS.y; }
    if(p.z >  BOUNDS.z){ f.velocity.z = -Math.abs(f.velocity.z); p.z = BOUNDS.z; }
    if(p.z < -BOUNDS.z){ f.velocity.z =  Math.abs(f.velocity.z); p.z = -BOUNDS.z; }

    // normalize & move
    f.velocity.normalize();
    p.addScaledVector(f.velocity, f.speed * dt);

    // orient toward movement (Y axis)
    _dir.copy(f.velocity);
    f.mesh.quaternion.setFromUnitVectors(new THREE.Vector3(0,1,0), _dir.normalize());

    // --- tail animation ---
    f.phase += dt * f.tailSpeed;
    f.tail.rotation.y = Math.sin(f.phase) * 0.7;
    // fins flutter
    const fl = Math.sin(f.phase * 1.5) * 0.3;
    f.topFin.rotation.z = fl;
    f.leftFin.rotation.y = fl;
    f.rightFin.rotation.y = -fl;
  }
}

function updateBubbles(dt, t){
  for(const b of bubbles){
    b.sway += dt;
    b.mesh.position.y += b.vel * dt;
    b.mesh.position.x += Math.sin(b.sway + b.sway) * 0.01;
    b.mesh.position.z += Math.cos(b.sway) * 0.01;
    // reset at surface
    if(b.mesh.position.y > WATER_TOP - 0.5){
      b.mesh.position.set(
        (Math.random()-0.5)*(W-4),
        FLOOR_Y,
        (Math.random()-0.5)*(D-4)
      );
    }
  }
}

function updateFood(dt){
  for(let i=foods.length-1;i>=0;i--){
    const fd = foods[i];
    fd.age += dt;
    // gravity
    fd.vel.y -= 6 * dt;
    fd.mesh.position.addScaledVector(fd.vel, dt);
    // remove when hits floor or too old
    if(fd.mesh.position.y < FLOOR_Y + 0.3 || fd.age > 25){
      scene.remove(fd.mesh);
      foods.splice(i,1);
    }
  }
}

function updateSeaweed(dt, t){
  for(const s of seaweeds){
    s.mesh.rotation.x = Math.sin(t*1.2 + s.phase) * s.amp;
    s.mesh.rotation.z = Math.cos(t*1.0 + s.phase) * s.amp;
  }
}

// ---------- Click feeding ----------
function onClick(e){
  const rect = renderer.domElement.getBoundingClientRect();
  const mouse = new THREE.Vector2(
    ((e.clientX - rect.left) / rect.width) * 2 - 1,
    -((e.clientY - rect.top) / rect.height) * 2 + 1
  );
  raycaster.setFromCamera(mouse, camera);
  // drop food a bit in front of camera, inside tank
  const pt = new THREE.Vector3();
  raycaster.ray.at(25, pt);
  pt.x = THREE.MathUtils.clamp(pt.x, -BOUNDS.x, BOUNDS.x);
  pt.z = THREE.MathUtils.clamp(pt.z, -BOUNDS.z, BOUNDS.z);
  pt.y = Math.min(pt.y, WATER_TOP - 1);
  addFood(pt);
}

// ---------- UI ----------
function bindUI(){
  document.getElementById('addFish').onclick = ()=> spawnFish(1);
  document.getElementById('addBubbles').onclick = ()=> spawnBubbles(10);
  document.getElementById('toggleLight').onclick = ()=>{
    lightOn = !lightOn;
    dirLight.intensity = lightOn ? 1.1 : 0.15;
    pointLightA.intensity = lightOn ? 1.2 : 0.4;
    pointLightB.intensity = lightOn ? 1.2 : 0.4;
  };
}

function updateCounts(){
  document.getElementById('fishCount').textContent = fishArray.length;
  document.getElementById('bubbleCount').textContent = bubbles.length;
}

// ---------- Resize ----------
function onResize(){
  camera.aspect = window.innerWidth/window.innerHeight;
  camera.updateProjectionMatrix();
  renderer.setSize(window.innerWidth, window.innerHeight);
}

// ---------- FPS ----------
let frames = 0, fpsTime = 0;
function tickFPS(dt){
  frames++;
  fpsTime += dt;
  if(fpsTime >= 0.5){
    document.getElementById('fps').textContent = Math.round(frames/fpsTime);
    frames = 0; fpsTime = 0;
  }
}

// ---------- Main loop ----------
function animate(){
  requestAnimationFrame(animate);
  const dt = Math.min(clock.getDelta(), 0.05);
  const t = clock.elapsedTime;

  updateFish(dt, t);
  updateBubbles(dt, t);
  updateFood(dt);
  updateSeaweed(dt, t);

  controls.update();
  renderer.render(scene, camera);
  tickFPS(dt);
}

// go
init();

})();

### Ключевые особенности реализации

**Рыбки (15 шт.)** — каждая собирается из `THREE.Group`: вытянутое тело (масштабированная сфера), глаза с зрачками, анимируемый хвост (конус, вращение), спинной и боковые плавники. 8 цветовых схем, случайный масштаб 0.6–1.2.

**ИИ** — избегание столкновений (отталкивание в радиусе), отражение от стенок, периодическое блуждание, преследование корма (радиус 15), рост на 5% после еды. У каждой рыбки свои скорость, фаза и частота маха.

**Окружение** — стеклянный бак (`transmission: 0.95`) с wireframe-рамкой, песчаное дно с процедурными неровностями, 8 деформированных камней, 12 водорослей на `TubeGeometry` + `CatmullRomCurve3` с покачиванием.

**Пузыри и корм** — 30 пузырей со стеклом, плавают вверх с покачиванием и сбрасываются на дно; клик создаёт корм с гравитацией, который рыбки подхватывают.

**Свет и камера** — ambient + directional (тени 2048, PCFSoft) + 2 голубых point-light, кнопка переключает основной свет. OrbitControls с damping, зумом 10–60 и ограничением угла.

**UI** — glass-панели с blur: слева заголовок с градиентом, подсказки и кнопки «+ Рыбка», «+ Пузыри», «💡 Свет»; справа счётчики рыбок/пузырей и живой FPS. Туман `FogExp2` и градиентный фон имитируют воду.
