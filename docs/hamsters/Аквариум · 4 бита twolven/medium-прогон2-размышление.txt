function createFish(scheme, pos, scale){
  const group = new THREE.Group();
  const mat = new THREE.MeshStandardMaterial({ color:scheme.color, roughness:0.45, metalness:0.15 });
  const finMat = new THREE.MeshStandardMaterial({ color:scheme.color, roughness:0.5, metalness:0.1, side:THREE.DoubleSide, transparent:true, opacity:0.85 });

  // Тело
  const body = new THREE.Mesh(new THREE.SphereGeometry(1, 18, 14), mat);
  body.scale.set(0.75, 1.05, 1.9);
  body.castShadow = true;
  group.add(body);

  // Глаза + зрачки
  [-1,1].forEach(side=>{
    const eye = new THREE.Mesh(new THREE.SphereGeometry(0.17, 12, 10),
      new THREE.MeshStandardMaterial({ color:0xffffff, roughness:0.25 }));
    eye.position.set(side*0.5, 0.22, 1.25);
    group.add(eye);
    const pupil = new THREE.Mesh(new THREE.SphereGeometry(0.085, 10, 8),
      new THREE.MeshStandardMaterial({ color:0x101010, roughness:0.2 }));
    pupil.position.set(side*0.5, 0.22, 1.37);
    group.add(pupil);
  });

  // Хвост (пивот на задней части тела)
  const tailPivot = new THREE.Group();

  tail.scale.set(0.9, 1.5, 1);

function updateFood(dt){

    const fd = foodArray[i];
    if (!fd.alive){
      scene.remove(fd.mesh);
      foodArray.splice(i,1);
      continue;
    }
    fd.life += dt;

    fd.mesh.position.addScaledVector(fd.velocity, dt);
