```js
function hash2i(x,z){
  let h=(x*374761393+z*668265263)|0;
  h=(h^(h>>>13))*1274126177|0;
  h^=h>>>16;
  return (h>>>0)/4294967295;
}
function sm(t){return t*t*(3-2*t);}
function noise2(x,z){
  const xi=Math.floor(x),zi=Math.floor(z);
  const fx=sm(x-xi),fz=sm(z-zi);
  const a=hash2i(xi,zi),b=hash2i(xi+1,zi),c=hash2i(xi,zi+1),d=hash2i(xi+1,zi+1);
  return a+(b-a)*fx+(c-a)*fz+(a-b-c+d)*fx*fz;
}
function fractal2(x,z){
  let v=0,amp=0.5,f=1;
  for(let i=0;i<4;i++){v+=noise2(x*f,z*f)*amp;amp*=0.5;f*=2;}
  return v/(0.5+0.25+0.125+0.0625);
}
function hash3i(x,y,z){
  let h=(x*374761393+y*2246822519+z*668265263)|0;
  h=(h^(h>>>13))*1274126177|0;
  h^=h>>>16;
  return (h>>>0)/4294967295;
}
function noise3(x,y,z){
  const xi=Math.floor(x),yi=Math.floor(y),zi=Math.floor(z);
  const fx=sm(x-xi),fy=sm(y-yi),fz=sm(z-zi);
  const c000=hash3i(xi,yi,zi),c100=hash3i(xi+1,yi,zi),c010=hash3i(xi,yi+1,zi),c110=hash3i(xi+1,yi+1,zi),
        c001=hash3i(xi,yi,zi+1),c101=hash3i(xi+1,yi,zi+1),c011=hash3i(xi,yi+1,zi+1),c111=hash3i(xi+1,yi+1,zi+1);
  const x00=c000+(c100-c000)*fx, x10=c010+(c110-c010)*fx,
```

```js
function getBlock(x,y,z){
  if(y<0||y>=HS)return 0;
  const cx=Math.floor(x/CS),cz=Math.floor(z/CS);
  const c=chunks.get(cx+','+cz);
  if(!c||!c.data)return 0;
  return c.data[((x-cx*CS)*CS+(z-cz*CS))*HS+y];
}
function setBlock(x,y,z,id){
  if(y<0||y>=HS)return;
  const cx=Math.floor(x/CS),cz=Math.floor(z/CS);
  const c=chunks.get(cx+','+cz);
  if(!c||!c.data)return;
  c.data[((x-cx*CS)*CS+(z-cz*CS))*HS+y]=id;
}
```

```js
function genChunkData(cx,cz){
  const data=new Uint8Array(CS*CS*HS);
  const heights=new Int16Array(CS*CS);
  for(let lx=0;lx<CS;lx++)for(let lz=0;lz<CS;lz++){
    const wx=cx*CS+lx, wz=cz*CS+lz;
    const H=columnHeight(wx,wz);
    heights[lz*CS+lx]=H;
    for(let y=0;y<=H&&y<HS;y++){
      let id;
      if(y===0)id=3;
      else if(y<H-3)id=3;
      else if(y<H)id=(H<=16?4:(H>=37?3:2));
      else id=(H>=46?7:(H>=37?3:(H<=16?4:1)));
      data[(lx*CS+lz)*HS+y]=id;
    }
    for(let y=3;y<H-2;y++){
      if(fractal3(wx*0.09,y*0.09,wz*0.09)>0.67)data[(lx*CS+lz)*HS+y]=0;
    }
  }
  // trees
  for(let lx=0;lx<CS;lx++)for(let lz=0;lz<CS;lz++){
    const wx=cx*CS+lx,wz=cz*CS+lz;
    const H=heights[lz*CS+lx];
    const si=data[(lx*CS+lz)*HS+H];
    if(si!==1)continue;
    if(hash2i(wx*31+7,wz*17+13)>=0.02)continue;
    if(lx<2||lx>13||lz<2||lz>13)continue;
    if(H+8>=HS)continue;
    for(let i=1;i<=4;i++)data[(lx*CS+lz)*HS+H+i]=5;
    for(let dy=5;dy<=6;dy++)for(let ox=-2;ox<=2;ox++)for(let oz=-2;oz<=2;oz++){
      const y=H+dy;
      if(data[((lx+ox)*CS+(lz+oz))*HS+y]===0)data[((lx+ox)*CS+(lz+oz))*HS+y]=6;
    }
    for(let ox=-1;ox<=1;ox++)for(let oz=-1;oz<=1;oz++){
      const y=H+7;
      if(data[((lx+ox)*CS+(lz+oz))*HS+y]===0)data[((lx+ox)*CS+(lz+oz))*HS+y]=6;
    }
    const y=H+8;
    if(data[(lx*CS+lz)*HS+y]===0)data[(lx*CS+lz)*HS+y]=6;
  }
  return data;
}
```
