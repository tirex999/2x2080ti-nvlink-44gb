  function hashInt(x, y, z) {
    var n = (x | 0) * 374761393 + (y | 0) * 668265263 + (z | 0) * 1274126177;
    n = (n ^ (n >>> 13)) | 0;
    n = (n * 1274126177) | 0;
    n = (n ^ (n >>> 16)) | 0;
    return (n >>> 0) / 4294967295;
  }
  function hash2(x, y) {
    var n = (x | 0) * 374761393 + (y | 0) * 668265263;
    n = (n ^ (n >>> 13)) | 0;
    n = (n * 1274126177) | 0;
    n = (n ^ (n >>> 16)) | 0;

  function smoothstep(t) { return t * t * (3 - 2 * t); }

  function valueNoise2(x, y) {
    var xi = Math.floor(x), yi = Math.floor(y);
    var xf = x - xi, yf = y - yi;
    var u = smoothstep(xf), v = smoothstep(yf);
    var a = hash2(xi, yi), b = hash2(xi + 1, yi);
    var c = hash2(xi, yi + 1), d = hash2(xi + 1, yi + 1);
    var top = a + (b - a) * u;
    var bot = c + (d - c) * u;
    return top + (bot - top) * v;
  }
  function valueNoise3(x, y, z) {
    var xi = Math.floor(x), yi = Math.floor(y), zi = Math.floor(z);
    var xf = x - xi, yf = y - yi, zf = z - zi;
    var u = smoothstep(xf), v = smoothstep(yf), w = smoothstep(zf);
    function h(a, b, c) { return hashInt(a, b, c); }
    var c000 = h(xi, yi, zi),     c100 = h(xi+1, yi, zi);
    var c010 = h(xi, yi+1, zi),   c110 = h(xi+1, yi+1, zi);
    var c001 = h(xi, yi, zi+1),   c101 = h(xi+1, yi, zi+1);
    var c011 = h(xi, yi+1, zi+1), c111 = h(xi+1, yi+1, zi+1);
    var x00 = c000 + (c100 - c000) * u;
    var x10 = c010 + (c110 - c010) * u;
    var x01 = c001 + (c101 - c001) * u;
    var x11 = c011 + (c111 - c011) * u;
    var y0 = x00 + (x10 - x00) * v;
    var y1 = x01 + (x11 - x01) * v;
    return y0 + (y1 - y0) * w;
  }
  function fractal2(x, y, octaves) {
    var total = 0, amp = 0.5, freq = 1, max = 0;
    for (var i = 0; i < octaves; i++) {
      total += valueNoise2(x * freq, y * freq) * amp;
      max += amp;
      amp *= 0.5;
      freq *= 2;
    }
    return total / max;
  }
  function fractal3(x, y, z, octaves) {
    var total = 0, amp = 0.5, freq = 1, max = 0;
    for (var i = 0; i < octaves; i++) {
      total += valueNoise3(x * freq, y * freq, z * freq) * amp;
      max += amp;
      amp *= 0.5;
      freq *= 2;
    }
    return total / max;
  }

  // ---------- Terrain generation ----------
  function columnHeight(wx, wz) {
    var m = fractal2(wx * 0.004, wz * 0.004, 4);
    var h = fractal2(wx * 0.02, wz * 0.02, 4);
    return Math.floor(5 + m * m * 58 + h * 10);
  }

  function generateChunkData(cx, cz) {
    var data = new Uint8Array(CHUNK_SIZE * CHUNK_SIZE * CHUNK_HEIGHT);
    var baseX = cx * CHUNK_SIZE, baseZ = cz * CHUNK_SIZE;

    for (var lx = 0; lx < CHUNK_SIZE; lx++) {
      for (var lz = 0; lz < CHUNK_SIZE; lz++) {
        var wx = baseX + lx, wz = baseZ + lz;
        var H = columnHeight(wx, wz);
        if (H > CHUNK_HEIGHT - 1) H = CHUNK_HEIGHT - 1;
        if (H < 0) H = 0;

        var surfaceBlock, dirtBlock;
        if (H >= 46) { surfaceBlock = SNOW; dirtBlock = DIRT; }
        else if (H >= 37) { surfaceBlock = STONE; dirtBlock = STONE; }
        else if (H <= 16) { surfaceBlock = SAND; dirtBlock = SAND; }
        else { surfaceBlock = GRASS; dirtBlock = DIRT; }

        for (var y = 0; y <= H; y++) {
          var idx = (lx * CHUNK_SIZE + lz) * CHUNK_HEIGHT + y;
          if (y === 0) { data[idx] = STONE; continue; }
          if (y < H - 3) { data[idx] = STONE; continue; }
          if (y < H) { data[idx] = dirtBlock; continue; }
          data[idx] = surfaceBlock; // y === H
        }

        // Caves
        for (var cy = 3; cy <= H - 2; cy++) {
          var n = fractal3(wx * 0.09, cy * 0.09, wz * 0.09, 3);
          if (n > 0.67) {
            data[(lx * CHUNK_SIZE + lz) * CHUNK_HEIGHT + cy] = AIR;
          }
        }
      }
    }

    // Trees (only if trunk fits inside this chunk)
    for (var tx = 0; tx < CHUNK_SIZE; tx++) {
      for (var tz = 0; tz < CHUNK_SIZE; tz++) {
        var wx2 = baseX + tx, wz2 = baseZ + tz;
        var H2 = columnHeight(wx2, wz2);
        if (H2 > CHUNK_HEIGHT - 1) H2 = CHUNK_HEIGHT - 1;
        var sIdx = (tx * CHUNK_SIZE + tz) * CHUNK_HEIGHT + H2;
        if (data[sIdx] !== GRASS) continue;
        if (hash2(wx2, wz2) >= 0.02) continue;
        var topY = H2 + 4;
        if (topY >= CHUNK_HEIGHT) continue;

        // Trunk
        for (var ty = H2 + 1; ty <= H2 + 4; ty++) {
          data[(tx * CHUNK_SIZE + tz) * CHUNK_HEIGHT + ty] = WOOD;
        }
        // Leaves: two 5x5 layers, one 3x3, one top
        for (var ly = H2 + 3; ly <= H2 + 4; ly++) {
          for (var dx = -2; dx <= 2; dx++) {
            for (var dz = -2; dz <= 2; dz++) {
              var nx = tx + dx, nz = tz + dz;
              if (nx < 0 || nx >= CHUNK_SIZE || nz < 0 || nz >= CHUNK_SIZE) continue;
              if (ly === H2 + 4 && (Math.abs(dx) > 1 || Math.abs(dz) > 1)) continue; // 3x3 on top
              var li = (nx * CHUNK_SIZE + nz) * CHUNK_HEIGHT + ly;
              if (data[li] === AIR) data[li] = LEAVES;
            }
          }
        }
        // Single leaf on very top
        var ti = (tx * CHUNK_SIZE + tz) * CHUNK_HEIGHT + (H2 + 5);
        if (H2 + 5 < CHUNK_HEIGHT && data[ti] === AIR) data[ti] = LEAVES;
      }
    }

    return data;
  }

  // ---------- Global block access ----------
  function getChunkData(cx, cz) {

  }
  function getBlock(wx, wy, wz) {
    if (wy < 0 || wy >= CHUNK_HEIGHT) return AIR;
    var cx = Math.floor(wx / CHUNK_SIZE);
    var cz = Math.floor(wz / CHUNK_SIZE);
    var data = getChunkData(cx, cz);
    if (!data) return AIR;
    var lx = wx - cx * CHUNK_SIZE;
    var lz = wz - cz * CHUNK_SIZE;
    return data[(lx * CHUNK_SIZE + lz) * CHUNK_HEIGHT + wy];
  }
  function setBlock(wx, wy, wz, id) {
    if (wy < 0 || wy >= CHUNK_HEIGHT) return;
    var cx = Math.floor(wx / CHUNK_SIZE);
    var cz = Math.floor(wz / CHUNK_SIZE);
    var c = chunkMap.get(cx + ',' + cz);
    if (!c) return;
    var lx = wx - cx * CHUNK_SIZE;
    var lz = wz - cz * CHUNK_SIZE;
    c.data[(lx * CHUNK_SIZE + lz) * CHUNK_HEIGHT + wy] = id;
  }

  // ---------- Meshing ----------
  var _col = new Float32Array(3);
  function buildChunkMesh(cx, cz) {
    var c = chunkMap.get(cx + ',' + cz);
    if (!c) return;
    if (c.mesh) {
      scene.remove(c.mesh);
      c.mesh.geometry.dispose();
      var i = chunkMeshes.indexOf(c.mesh);
      if (i >= 0) chunkMeshes.splice(i, 1);
      c.mesh = null;
    }
    var data = c.data;
    var baseX = cx * CHUNK_SIZE, baseZ = cz * CHUNK_SIZE;
    var positions = [], normals = [], colors = [];

    for (var lx = 0; lx < CHUNK_SIZE; lx++) {
      for (var lz = 0; lz < CHUNK_SIZE; lz++) {
        for (var y = 0; y < CHUNK_HEIGHT; y++) {
          var idx = (lx * CHUNK_SIZE + lz) * CHUNK_HEIGHT + y;
          var id = data[idx];
          if (id === AIR) continue;
          var wx = baseX + lx, wz = baseZ + lz;
          var col = BLOCK_COLORS[id];
          var r = ((col >> 16) & 255) / 255;
          var g = ((col >> 8) & 255) / 255;
          var b = (col & 255) / 255;

          for (var f = 0; f < 6; f++) {
            var face = FACES[f];
            var nb = getBlock(wx + face.dir[0], y + face.dir[1], wz + face.dir[2]);
            if (nb !== AIR) continue;
            var nx = wx + face.dir[0], ny = y + face.dir[1], nz = wz + face.dir[2];
            var cr = face.corners;
            // quad -> two triangles: (0,1,2) and (0,2,3)
            var order = [0, 1, 2, 0, 2, 3];
            for (var o = 0; o < 6; o++) {
              var corner = cr[order[o]];
              positions.push(nx + corner[0], ny + corner[1], nz + corner[2]);
              normals.push(face.dir[0], face.dir[1], face.dir[2]);
              _col[0] = r * face.shade;
              _col[1] = g * face.shade;
              _col[2] = b * face.shade;
              colors.push(_col[0], _col[1], _col[2]);
            }
          }
        }
      }
    }

    if (positions.length === 0) return;
    var geo = new THREE.BufferGeometry();
    geo.setAttribute('position', new THREE.Float32BufferAttribute(positions, 3));
    geo.setAttribute('normal', new THREE.Float32BufferAttribute(normals, 3));
    geo.setAttribute('color', new THREE.Float32BufferAttribute(colors, 3));
    var mat = scene.userData.sharedMat;
    var mesh = new THREE.Mesh(geo, mat);
    mesh.frustumCulled = true;
    scene.add(mesh);
    chunkMeshes.push(mesh);
    c.mesh = mesh;
  }

  function rebuildChunk(cx, cz) {
    var c = chunkMap.get(cx + ',' + cz);
    if (!c) return;
    buildChunkMesh(cx, cz);
  }

  function disposeChunk(cx, cz) {
    var c = chunkMap.get(cx + ',' + cz);
    if (!c) return;
    if (c.mesh) {
      scene.remove(c.mesh);
      c.mesh.geometry.dispose();
      var i = chunkMeshes.indexOf(c.mesh);
      if (i >= 0) chunkMeshes.splice(i, 1);
    }
    chunkMap.delete(cx + ',' + cz);
  }

  // ---------- World update per frame ----------
  function updateWorld() {
    var pcx = Math.floor(player.pos.x / CHUNK_SIZE);
    var pcz = Math.floor(player.pos.z / CHUNK_SIZE);

    // Generate data within GEN_RADIUS (max per frame)
    var genCount = 0;
    for (var ox = -GEN_RADIUS; ox <= GEN_RADIUS && genCount < MAX_GEN_PER_FRAME; ox++) {
      for (var oz = -GEN_RADIUS; oz <= GEN_RADIUS && genCount < MAX_GEN_PER_FRAME; oz++) {
        var cx = pcx + ox, cz = pcz + oz;
        var key = cx + ',' + cz;
        if (!chunkMap.has(key)) {
          var data = generateChunkData(cx, cz);
          chunkMap.set(key, { cx: cx, cz: cz, data: data, mesh: null });
          genCount++;
        }
      }
    }

    // Build meshes within MESH_RADIUS whose 4 neighbors have data (max per frame)
    var meshCount = 0;
    for (var mx = -MESH_RADIUS; mx <= MESH_RADIUS && meshCount < MAX_MESH_PER_FRAME; mx++) {
      for (var mz = -MESH_RADIUS; mz <= MESH_RADIUS && meshCount < MAX_MESH_PER_FRAME; mz++) {
        var cx2 = pcx + mx, cz2 = pcz + mz;
        var c = chunkMap.get(cx2 + ',' + cz2);
        if (!c || c.mesh) continue;
        var hasN = getChunkData(cx2, cz2 + 1) !== null;
        var hasS = getChunkData(cx2, cz2 - 1) !== null;
        var hasE = getChunkData(cx2 + 1, cz2) !== null;
        var hasW = getChunkData(cx2 - 1, cz2) !== null;
        if (hasN && hasS && hasE && hasW) {
          buildChunkMesh(cx2, cz2);
          meshCount++;
        }
      }
    }

    // Dispose chunks farther than CULL_RADIUS
    var toDispose = [];
    chunkMap.forEach(function (c, key) {
      var dist = Math.max(Math.abs(c.cx - pcx), Math.abs(c.cz - pcz));
      if (dist > CULL_RADIUS) toDispose.push(key);
    });
    for (var d = 0; d < toDispose.length; d++) {
      var parts = toDispose[d].split(',');
      disposeChunk(parseInt(parts[0], 10), parseInt(parts[1], 10));
    }
  }

  // ---------- Collision ----------
  function collides(px, py, pz) {
    var minX = px - PLAYER_HALF, maxX = px + PLAYER_HALF;
    var minY = py, maxY = py + PLAYER_HEIGHT;
    var minZ = pz - PLAYER_HALF, maxZ = pz + PLAYER_HALF;
    var x0 = Math.floor(minX), x1 = Math.floor(maxX);
    var y0 = Math.floor(minY), y1 = Math.floor(maxY);
    var z0 = Math.floor(minZ), z1 = Math.floor(maxZ);
    for (var bx = x0; bx <= x1; bx++) {
      for (var by = y0; by <= y1; by++) {
        for (var bz = z0; bz <= z1; bz++) {
          if (getBlock(bx, by, bz) !== AIR) return true;
        }
      }
    }
    return false;
  }

      if (keys['KeyD']) ix += 1;
    }
    var len = Math.hypot(ix, iz);
    if (len > 0) { ix /= len; iz /= len; }
    var sin = Math.sin(player.yaw), cos = Math.cos(player.yaw);
    var wx = (ix * cos + iz * sin) * WALK_SPEED;
    var wz = (iz * cos - ix * sin) * WALK_SPEED;

    player.vel.x = wx;
    player.vel.z = wz;

    // Jump
    if (locked && keys['Space'] && player.onGround) {
      player.vel.y = JUMP_VEL;
      player.onGround = false;
    }

    // Gravity
    player.vel.y -= GRAVITY * dt;
    if (player.vel.y < -50) player.vel.y = -50;

    // Move axis-separated
    var nx = player.pos.x + player.vel.x * dt;
    if (collides(nx, player.pos.y, player.pos.z)) {
      player.vel.x = 0;
    } else {
      player.pos.x = nx;
    }

    var nz = player.pos.z + player.vel.z * dt;
    if (collides(player.pos.x, player.pos.y, nz)) {
      player.vel.z = 0;
    } else {
      player.pos.z = nz;
    }

    var ny = player.pos.y + player.vel.y * dt;
    if (collides(player.pos.x, ny, player.pos.z)) {
      if (player.vel.y < 0) player.onGround = true;
      player.vel.y = 0;
    } else {
      player.pos.y = ny;
      player.onGround = false;
    }

    // Respawn if fallen
    if (player.pos.y < -20) {
      respawn();
    }
  }

  function respawn() {
    var H = columnHeight(SPAWN.x, SPAWN.z);
    player.pos.set(SPAWN.x, H + 2, SPAWN.z);
    player.vel.set(0, 0, 0);
    player.onGround = false;
  }

  // ---------- Break & Place ----------
  function updateTarget() {
    raycaster.setFromCamera(new THREE.Vector2(0, 0), camera);
    raycaster.far = REACH;
    var hits = raycaster.intersectObjects(chunkMeshes, false);
    if (hits.length > 0) {
      var hit = hits[0];
      var p = hit.point;
      var n = hit.face.normal;
      // break target
      var bx = Math.floor(p.x - n.x * 0.5);
      var by = Math.floor(p.y - n.y * 0.5);
      var bz = Math.floor(p.z - n.z * 0.5);
      targetOutline.visible = true;
      targetOutline.position.set(bx + 0.5, by + 0.5, bz + 0.5);
      player._target = { bx: bx, by: by, bz: bz };
    } else {
      targetOutline.visible = false;
      player._target = null;
    }
  }

  function doBreak() {
    if (!player._target) return;
    var t = player._target;
    if (t.by <= 0) return; // never break bedrock
    setBlock(t.bx, t.by, t.bz, AIR);
    rebuildChunk(t.bx, t.bz);
  }

  function doPlace() {
    if (!player._target) return;
    var t = player._target;
    var p = raycaster.ray.origin;
    var n = lastNormal;
    var px = Math.floor(p.x + n.x * 0.5);
    var py = Math.floor(p.y + n.y * 0.5);
    var pz = Math.floor(p.z + n.z * 0.5);
    if (getBlock(px, py, pz) !== AIR) return;
    // don't place inside player
    var overlapX = (px + 1 > player.pos.x - PLAYER_HALF) && (px < player.pos.x + PLAYER_HALF);
    var overlapY = (py + 1 > player.pos.y) && (py < player.pos.y + PLAYER_HEIGHT);
    var overlapZ = (pz + 1 > player.pos.z - PLAYER_HALF) && (pz < player.pos.z + PLAYER_HALF);
    if (overlapX && overlapY && overlapZ) return;
    setBlock(px, py, pz, HOTBAR_BLOCKS[selectedSlot]);
    rebuildChunk(px, pz);
  }

  var lastNormal = new THREE.Vector3(0, 1, 0);

  function handleEdit(type) {
    if (!locked) return;
    raycaster.setFromCamera(new THREE.Vector2(0, 0), camera);
    raycaster.far = REACH;
    var hits = raycaster.intersectObjects(chunkMeshes, false);
    if (hits.length === 0) return;
    var hit = hits[0];
    var p = hit.point;
    var n = hit.face.normal;
    lastNormal.copy(n);
    if (type === 'break') {
      var bx = Math.floor(p.x - n.x * 0.5);
      var by = Math.floor(p.y - n.y * 0.5);
      var bz = Math.floor(p.z - n.z * 0.5);
      if (by <= 0) return;
      setBlock(bx, by, bz, AIR);
      rebuildChunk(bx, bz);
    } else {
      var px = Math.floor(p.x + n.x * 0.5);
      var py = Math.floor(p.y + n.y * 0.5);
      var pz = Math.floor(p.z + n.z * 0.5);
      if (getBlock(px, py, pz) !== AIR) return;
      var overlapX = (px + 1 > player.pos.x - PLAYER_HALF) && (px < player.pos.x + PLAYER_HALF);
      var overlapY = (py + 1 > player.pos.y) && (py < player.pos.y + PLAYER_HEIGHT);
      var overlapZ = (pz + 1 > player.pos.z - PLAYER_HALF) && (pz < player.pos.z + PLAYER_HALF);
      if (overlapX && overlapY && overlapZ) return;
      setBlock(px, py, pz, HOTBAR_BLOCKS[selectedSlot]);
      rebuildChunk(px, pz);
    }
  }

  function rebuildChunkAndNeighbors(wx, wy, wz) {
    rebuildChunk(wx, wz);
    var lx = wx - Math.floor(wx / CHUNK_SIZE) * CHUNK_SIZE;
    var lz = wz - Math.floor(wz / CHUNK_SIZE) * CHUNK_SIZE;
    if (lx === 0) rebuildChunk(wx - 1, wz);
    if (lx === CHUNK_SIZE - 1) rebuildChunk(wx + 1, wz);
    if (lz === 0) rebuildChunk(wx, wz - 1);
    if (lz === CHUNK_SIZE - 1) rebuildChunk(wx, wz + 1);
  }

  // ---------- Clouds ----------
  function createClouds() {
    var cloudMat = new THREE.MeshBasicMaterial({
      color: 0xffffff, transparent: true, opacity: 0.85, depthWrite: false
    });
    for (var i = 0; i < 25; i++) {
      var w = 18 + hash2(i, 7) * 30;
      var d = 12 + hash2(i, 13) * 20;
      var geo = new THREE.BoxGeometry(w, 1.2, d);
      var mesh = new THREE.Mesh(geo, cloudMat);
      var ang = hash2(i, 23) * Math.PI * 2;
      var rad = 40 + hash2(i, 31) * 160;
      mesh.position.set(
        player.pos.x + Math.cos(ang) * rad,
        90 + hash2(i, 41) * 8,
        player.pos.z + Math.sin(ang) * rad
      );
      mesh.userData.speed = 1.5 + hash2(i, 53) * 2.5;
      scene.add(mesh);
      clouds.push(mesh);
    }
  }
  function updateClouds(dt) {
    for (var i = 0; i < clouds.length; i++) {
      var c = clouds[i];
      c.position.x += c.userData.speed * dt;
      // wrap around the player
      var dx = c.position.x - player.pos.x;
      if (dx > 220) c.position.x -= 440;
      else if (dx < -220) c.position.x += 440;
      var dz = c.position.z - player.pos.z;
      if (dz > 220) c.position.z -= 440;
      else if (dz < -220) c.position.z += 440;
    }
  }

  // ---------- Hotbar UI ----------
  function buildHotbar() {
    var bar = document.getElementById('hotbar');
    bar.innerHTML = '';
    for (var i = 0; i < HOTBAR_BLOCKS.length; i++) {
      var col = '#' + BLOCK_COLORS[HOTBAR_BLOCKS[i]].toString(16).padStart(6, '0');
      var slot = document.createElement('div');
      slot.className = 'slot' + (i === selectedSlot ? ' selected' : '');
      slot.dataset.index = i;
      slot.innerHTML =
        '<span class="num">' + (i + 1) + '</span>' +
        '<span class="swatch" style="background:' + col + '"></span>' +
        '<span class="name">' + HOTBAR_NAMES[i] + '</span>';
      slot.addEventListener('click', function () {
        selectSlot(parseInt(this.dataset.index, 10));
      });
      bar.appendChild(slot);
    }
  }
  function selectSlot(i) {
    selectedSlot = ((i % HOTBAR_BLOCKS.length) + HOTBAR_BLOCKS.length) % HOTBAR_BLOCKS.length;
    var slots = document.querySelectorAll('.slot');
    for (var k = 0; k < slots.length; k++) {
      slots[k].classList.toggle('selected', k === selectedSlot);
    }
  }

  // ---------- Input ----------
  function bindInput() {
    var canvas = renderer.domElement;

    document.addEventListener('keydown', function (e) {
      keys[e.code] = true;
      if (e.code === 'Space') e.preventDefault();
      if (e.code >= 'Digit1' && e.code <= 'Digit7') {
        selectSlot(parseInt(e.code.slice(5), 10) - 1);
      }
    });
    document.addEventListener('keyup', function (e) {
      keys[e.code] = false;
    });

    document.addEventListener('mousemove', function (e) {
      if (!locked) return;
      player.yaw -= e.movementX * 0.002;
      player.pitch -= e.movementY * 0.002;
      var lim = Math.PI / 2 - 0.05;
      if (player.pitch > lim) player.pitch = lim;
      if (player.pitch < -lim) player.pitch = -lim;
    });

    document.addEventListener('mousedown', function (e) {
      if (!locked) return;
      if (e.button === 0) handleEdit('break');
      else if (e.button === 2) handleEdit('place');
    });

    document.addEventListener('wheel', function (e) {
      if (!locked) return;
      selectSlot(selectedSlot + (e.deltaY > 0 ? 1 : -1));
    }, { passive: true });

    document.addEventListener('contextmenu', function (e) {
      e.preventDefault();
    });

    // Pointer lock
    var overlay = document.getElementById('overlay');
    var playBtn = document.getElementById('play-btn');
    function requestLock() {
      try { canvas.requestPointerLock(); } catch (err) {}
    }
    playBtn.addEventListener('click', requestLock);
    overlay.addEventListener('click', function (e) {
      if (e.target === overlay) requestLock();
    });

    document.addEventListener('pointerlockchange', function () {
      locked = (document.pointerLockElement === canvas);
      overlay.style.display = locked ? 'none' : 'flex';
      if (locked) {
        // ensure we're on the ground properly
      }
    });
  }

  // ---------- Init ----------
  function init() {
    scene = new THREE.Scene();
    scene.background = new THREE.Color(0x87ceeb);
    scene.fog = new THREE.Fog(0x87ceeb, 40, 110);

    camera = new THREE.PerspectiveCamera(75, window.innerWidth / window.innerHeight, 0.1, 400);
    camera.rotation.order = 'YXZ';

    renderer = new THREE.WebGLRenderer({ canvas: document.getElementById('game-canvas'), antialias: true });
    renderer.setSize(window.innerWidth, window.innerHeight);
    renderer.setPixelRatio(Math.min(window.devicePixelRatio, 2));

    // Lights
    var ambient = new THREE.AmbientLight(0xffffff, 0.65);
    scene.add(ambient);
    var dir = new THREE.DirectionalLight(0xffffff, 0.8);
    dir.position.set(0.5, 1, 0.35);
    scene.add(dir);

    // Shared material for all chunk meshes
    var sharedMat = new THREE.MeshLambertMaterial({ vertexColors: true });
    scene.userData.sharedMat = sharedMat;

    // Target outline
    targetOutline = new THREE.LineSegments(
      new THREE.EdgesGeometry(new THREE.BoxGeometry(1, 1, 1)),
      new THREE.LineBasicMaterial({ color: 0x000000 })
    );
    targetOutline.visible = false;
    scene.add(targetOutline);

    // Water plane
    var waterGeo = new THREE.PlaneGeometry(600, 600);
    var waterMat = new THREE.MeshLambertMaterial({
      color: 0x2f6fb0, transparent: true, opacity: 0.55, side: THREE.DoubleSide
    });
    waterPlane = new THREE.Mesh(waterGeo, waterMat);
    waterPlane.rotation.x = -Math.PI / 2;
    waterPlane.position.set(player.pos.x, WATER_LEVEL, player.pos.z);
    scene.add(waterPlane);

    createClouds();
    buildHotbar();
    bindInput();

    // Spawn
    var H = columnHeight(SPAWN.x, SPAWN.z);
    player.pos.set(SPAWN.x, H + 2, SPAWN.z);
    player.yaw = Math.PI; // look toward positive... choose a pleasant direction
    player.pitch = -0.1;

    // Pre-generate a ring around spawn so we have something to look at immediately
    var pcx = Math.floor(SPAWN.x / CHUNK_SIZE);
    var pcz = Math.floor(SPAWN.z / CHUNK_SIZE);
    for (var ox = -1; ox <= 1; ox++) {
      for (var oz = -1; oz <= 1; oz++) {
        var cx = pcx + ox, cz = pcz + oz;
        if (!chunkMap.has(cx + ',' + cz)) {
          chunkMap.set(cx + ',' + cz, { cx: cx, cz: cz, data: generateChunkData(cx, cz), mesh: null });
        }
      }
    }
    // Build the immediate central chunk mesh
    buildChunkMesh(pcx, pcz);

    clock = new THREE.Clock();
    animate();
  }

  // ---------- Main loop ----------
  function animate() {
    requestAnimationFrame(animate);
    var dt = Math.min(clock.getDelta(), 0.05);

    updateWorld();
    updatePlayer(dt);

    // Camera follows player
    camera.position.set(player.pos.x, player.pos.y + EYE_HEIGHT, player.pos.z);
    camera.rotation.y = player.yaw;
    camera.rotation.x = player.pitch;

    // Water re-centers on player
    waterPlane.position.x = player.pos.x;
    waterPlane.position.z = player.pos.z;

    updateClouds(dt);
    updateTarget();

    renderer.render(scene, camera);
  }

  // ---------- Resize ----------
  window.addEventListener('resize', function () {
    camera.aspect = window.innerWidth / window.innerHeight;
    camera.updateProjectionMatrix();
    renderer.setSize(window.innerWidth, window.innerHeight);
  });

  init();
})();
