**2D value noise:**
```javascript
function noise2D(x, y) {
  const xi = Math.floor(x), yi = Math.floor(y);
  const xf = x - xi, yf = y - yi;
  
  const v00 = hash2D(xi, yi);
  const v10 = hash2D(xi + 1, yi);
  const v01 = hash2D(xi, yi + 1);
  const v11 = hash2D(xi + 1, yi + 1);
  
  const sx = smoothstep(xf), sy = smoothstep(yf);
  
  const v0 = v00 + (v10 - v00) * sx;
  const v1 = v01 + (v11 - v01) * sx;
  
  return v0 + (v1 - v0) * sy;
}
```

**Fractal 2D (4 octaves):**
```javascript
function fractal2D(x, y) {
  let val = 0, amp = 1, freq = 1, total = 0;
  for (let i = 0; i < 4; i++) {
    val += noise2D(x * freq, y * freq) * amp;
    total += amp;
    amp *= 0.5;
    freq *= 2;
  }
  return val / total;
}
```

**3D value noise:**
```javascript
function noise3D(x, y, z) {
  const xi = Math.floor(x), yi = Math.floor(y), zi = Math.floor(z);
  const xf = x - xi, yf = y - yi, zf = z - zi;
  
  const sx = smoothstep(xf), sy = smoothstep(yf), sz = smoothstep(zf);
  
  // 8 corners
  const c000 = hash3D(xi, yi, zi);
  const c100 = hash3D(xi+1, yi, zi);
  const c010 = hash3D(xi, yi+1, zi);
  const c110 = hash3D(xi+1, yi+1, zi);
  const c001 = hash3D(xi, yi, zi+1);
  const c101 = hash3D(xi+1, yi, zi+1);
  const c011 = hash3D(xi, yi+1, zi+1);
  const c111 = hash3D(xi+1, yi+1, zi+1);
  
  // Trilinear interpolation
  const c00 = c000 + (c100 - c000) * sx;
  const c10 = c010 + (c110 - c010) * sx;
  const c01 = c001 + (c101 - c001) * sx;
  const c11 = c011 + (c111 - c011) * sx;
  
  const c0 = c00 + (c10 - c00) * sy;
  const c1 = c01 + (c11 - c01) * sy;
  
  return c0 + (c1 - c0) * sz;
}
```

```javascript
  const pcx = Math.floor(player.x / 16);
  const pcz = Math.floor(player.z / 16);
  
  // 1. Generate data for chunks within 5 (max 4 per frame)
  let genCount = 0;
  for (let dx = -5; dx <= 5 && genCount < 4; dx++) {
    for (let dz = -5; dz <= 5 && genCount < 4; dz++) {
      const cx = pcx + dx, cz = pcz + dz;
      const key = cx + ',' + cz;
      if (!chunks.has(key)) {
        const data = generateChunkData(cx, cz);
        chunks.set(key, { data, mesh: null });
        genCount++;
      }
    }
  }
  
  // 2. Build meshes for chunks within 4 whose 4 neighbors have data (max 2 per frame)
  let meshCount = 0;
  for (let dx = -4; dx <= 4 && meshCount < 2; dx++) {
    for (let dz = -4; dz <= 4 && meshCount < 2; dz++) {
      const cx = pcx + dx, cz = pcz + dz;
      const key = cx + ',' + cz;
      const chunk = chunks.get(key);
      if (!chunk || chunk.mesh) continue; // skip if no data or already has mesh
      
      // Check all 4 neighbors have data
      const neighborsReady = 
        chunks.has((cx-1)+','+cz) &&
        chunks.has((cx+1)+','+cz) &&
        chunks.has(cx+','+(cz-1)) &&
        chunks.has(cx+','+(cz+1));
      
      if (neighborsReady) {
        buildChunkMesh(cx, cz);
        meshCount++;
      }
    }
  }
  
  // 3. Cull chunks beyond 7
  for (const [key, chunk] of chunks) {
    const [cx, cz] = key.split(',').map(Number);
    const dist = Math.max(Math.abs(cx - pcx), Math.abs(cz - pcz));
    if (dist > 7) {
      if (chunk.mesh) {
        scene.remove(chunk.mesh);
        chunk.mesh.geometry.dispose();
        chunk.mesh = null;
      }
      chunks.delete(key);
    }
  }
```

Actually, in JavaScript, the `*` operator produces a floating-point result. But when followed by a bitwise operation (`^`, `>>>`, `&`), the value is coerced to a 32-bit integer. So:

```
n = 374761393;
n = (n ^ (n >>> 13)) * 1103515245;  // result is a float, possibly > 2^32
n = n ^ (n >>> 16);           // coerces n to int32, then XOR
```

The coercion to int32 happens at the `^` operation. So even if the intermediate multiplication overflows, the subsequent bitwise op wraps it correctly.

The final `(n & 0x7FFFFFFF)` ensures the result is in [0, 2^31-1], and dividing by 2^31-1 gives [0, 1]. ✓

This should give good-enough pseudo-random values for our purposes.
