  // ---------- Value noise ----------
  function hash2(x, y) {
    var n = (x * 374761393 + y * 668265263) | 0;
    n = (n ^ (n >> 13)) | 0;
    n = (n * 1274126177) | 0;
    n = (n ^ (n >> 16)) | 0;
    return (n & 0x7fffffff) / 0x7fffffff;
  }
  function hash3(x, y, z) {
    var n = (x * 374761393 + y * 668265263 + z * 1440662683) | 0;
    n = (n ^ (n >> 13)) | 0;
    n = (n * 1274126177) | 0;
    n = (n ^ (n >> 16)) | 0;
    return (n & 0x7fffffff) / 0x7fffffff;
  }
  function smoothstep(t) { return t * t * (3 - 2 * t); }
  function noise2(x, y) {
    var xi = Math.floor(x), yi = Math.floor(y);
    var xf = x - xi, yf = y - yi;
    var u = smoothstep(xf), v = smoothstep(yf);
    var a = hash2(xi, yi), b = hash2(xi + 1, yi);
    var c = hash2(xi, yi + 1), d = hash2(xi + 1, yi + 1);
    var ab = a + (b - a) * u;
    var cd = c + (d - c) * u;
    return ab + (cd - ab) * v;
  }
  function noise3(x, y, z) {
    var xi = Math.floor(x), yi = Math.floor(y), zi = Math.floor(z);
    var xf = x - xi, yf = y - yi, zf = z - zi;
    var u = smoothstep(xf), v = smoothstep(yf), w = smoothstep(zf);
    var c000 = hash3(xi, yi, zi), c100 = hash3(xi + 1, yi, zi);
    var c010 = hash3(xi, yi + 1, zi), c110 = hash3(xi + 1, yi + 1, zi);
    var c001 = hash3(xi, yi, zi + 1), c101 = hash3(xi + 1, yi, zi + 1);
    var c011 = hash3(xi, yi + 1, zi + 1), c111 = hash3(xi + 1, yi + 1, zi + 1);
    var c00 = c000 + (c100 - c000) * u;
    var c10 = c010 + (c110 - c010) * u;
    var c01 = c001 + (c101 - c001) * u;
    var c11 = c011 + (c111 - c011) * u;
    var c0 = c00 + (c10 - c00) * v;
    var c1 = c01 + (c11 - c01) * v;
    return c0 + (c1 - c0) * w;
  }
  function fbm2(x, y) {
    var total = 0, amp = 1, freq = 1, norm = 0;
    for (var i = 0; i < 4; i++) {
      total += noise2(x * freq, y * freq) * amp;
      norm += amp;
      amp *= 0.5;
      freq *= 2;
    }
    return total / norm;
  }
  function fbm3(x, y, z) {
    var total = 0, amp = 1, freq = 1, norm = 0;
    for (var i = 0; i < 3; i++) {
      total += noise3(x * freq, y * freq, z * freq) * amp;
      norm += amp;
      amp *= 0.5;
      freq *= 2;
    }
    return total / norm;
  }

  var SPAWN = { x: 8, y: 40, z: 8 };

  function chunkKey(cx, cz) { return cx + "," + cz; }
  function getChunkData(cx, cz) {
    var c = chunks.get(chunkKey(cx, cz));
    return c ? c.data : null;
  }
  function readBlock(wx, wy, wz) {
    if (wy < 0 || wy >= HEIGHT) return 0;
    var cx = Math.floor(wx / CHUNK), cz = Math.floor(wz / CHUNK);
    var lx = wx - cx * CHUNK, lz = wz - cz * CHUNK;
    var data = getChunkData(cx, cz);
    if (!data) return 0;
    return data[lx + lz * CHUNK + wy * CHUNK * CHUNK];
  }
  function writeBlock(wx, wy, wz, id) {
    if (wy < 0 || wy >= HEIGHT) return false;
    var cx = Math.floor(wx / CHUNK), cz = Math.floor(wz / CHUNK);
    var lx = wx - cx * CHUNK, lz = wz - cz * CHUNK;
    var data = getChunkData(cx, cz);
    if (!data) return false;
    var idx = lx + lz * CHUNK + wy * CHUNK * CHUNK;
    data[idx] = id;
    return true;
  }

  // ---------- Terrain generation ----------
  function terrainHeight(wx, wz) {
    var m = fbm2(wx * 0.004, wz * 0.004);
    var h = fbm2(wx * 0.02, wz * 0.02);
    var H = Math.floor(5 + m * m * 58 + h * 10);
    return H;
  }
  function generateChunk(cx, cz) {
    var data = new Uint8Array(CHUNK * CHUNK * HEIGHT);
    var ox = cx * CHUNK, oz = cz * CHUNK;
    for (var lx = 0; lx < CHUNK; lx++) {
      for (var lz = 0; lz < CHUNK; lz++) {
        var wx = ox + lx, wz = oz + lz;
        var H = terrainHeight(wx, wz);
        if (H > HEIGHT) H = HEIGHT;
        var surfaceType;
        if (H >= 46) surfaceType = 7;       // snow
        else if (H >= 37) surfaceType = 3; // stone
        else if (H <= 16) surfaceType = 4; // sand
        else surfaceType = 1;             // grass
        var subType = (H <= 16) ? 4 : ((H >= 37) ? 3 : 2); // sand/stone/dirt
        for (var y = 0; y < H; y++) {
          var id;
          if (y === 0) id = 3;                       // bedrock stone
          else if (y < H - 3) id = 3;                     // stone
          else if (y < H - 1) id = subType;              // dirt/sand/stone
          else id = surfaceType;                        // surface
          data[lx + lz * CHUNK + y * CHUNK * CHUNK] = id;
        }
        // Caves
        for (var cy = 3; cy < H - 2; cy++) {
          if (noise3(wx * 0.09, cy * 0.09, wz * 0.09) > 0.67) {
            data[lx + lz * CHUNK + cy * CHUNK * CHUNK] = 0;
          }
        }
      }
    }
    // Trees
    for (var tx = 0; tx < CHUNK; tx++) {
      for (var tz = 0; tz < CHUNK; tz++) {
        var twx = ox + tx, twz = oz + tz;
        var tH = terrainHeight(twx, twz);
        if (tH >= 46 || tH >= 37 || tH <= 16) continue; // only on grass
        // check surface is grass
        var surfIdx = tx + tz * CHUNK + (tH - 1) * CHUNK * CHUNK;
        if (data[surfIdx] !== 1) continue;
        var treeHash = hash2(twx, twz);
        if (treeHash >= 0.02) continue;
        // trunk 4 up, must fit in chunk
        if (tx < 1 || tx > CHUNK - 2 || tz < 1 || tz > CHUNK - 2) continue;
        if (tH + 4 > HEIGHT) continue;
        // place trunk
        for (var ty = 0; ty < 4; ty++) {
          data[tx + tz * CHUNK + (tH + ty) * CHUNK * CHUNK] = 5;
        }
        // leaves: two 5x5 layers, then 3x3, then 1
        function setLeaf(lxx, lyy, lzz) {
          if (lxx < 0 || lxx >= CHUNK || lzz < 0 || lzz >= CHUNK) return;
          var ly = tH + lyy;
          if (ly < 0 || ly >= HEIGHT) return;
          var li = lxx + lzz * CHUNK + ly * CHUNK * CHUNK;
          if (data[li] === 0) data[li] = 6;
        }
        // layer at tH+3 (5x5)
        for (var a = -2; a <= 2; a++) for (var b = -2; b <= 2; b++) setLeaf(tx + a, 3, tz + b);
        // layer at tH+4 (5x5)
        for (var c = -2; c <= 2; c++) for (var d = -2; d <= 2; d++) setLeaf(tx + c, 4, tz + d);
        // layer at tH+5 (3x3)
        for (var e = -1; e <= 1; e++) for (var f = -1; f <= 1; f++) setLeaf(tx + e, 5, tz + f);
        // top at tH+6 (1)
        setLeaf(tx, 6, tz);
      }
    }
    return data;
  }

  // ---------- Meshing ----------
  var sharedMat = new THREE.MeshLambertMaterial({ vertexColors: true });
  function buildChunkMesh(data, cx, cz) {
    var positions = [], normals = [], colors = [];
    var ox = cx * CHUNK, oz = cz * CHUNK;
    for (var y = 0; y < HEIGHT; y++) {
      for (var lz = 0; lz < CHUNK; lz++) {
        for (var lx = 0; lx < CHUNK; lx++) {
          var bid = data[lx + lz * CHUNK + y * CHUNK * CHUNK];
          if (bid === 0) continue;
          var col = BLOCKS[bid].color;
          var wx = ox + lx, wz = oz + lz;
          for (var f = 0; f < 6; f++) {
            var face = FACES[f];
            var nx = wx + face.dir[0], ny = y + face.dir[1], nz = wz + face.dir[2];
            var nb = readBlock(nx, ny, nz);
            if (nb !== 0) continue; // neighbor opaque
            var light = FACE_LIGHT[face.idx];
            var cr = col[0] * light, cg = col[1] * light, cb = col[2] * light;
            var base = positions.length / 3;
            // 4 verts
            for (var k = 0; k < 4; k++) {
              var vv = face.v[k];
              positions.push(wx + vv[0], y + vv[1], wz + vv[2]);
              normals.push(face.dir[0], face.dir[1], face.dir[2]);
              colors.push(cr, cg, cb);
            }
            // indices: two triangles (0,1,2)(0,2,3)
            var idxArr = buildChunkMesh._idx;
            idxArr.push(base, base + 1, base + 2, base, base + 2, base + 3);
          }
        }
      }
    }
    var geo = new THREE.BufferGeometry();
    geo.setAttribute("position", new THREE.Float32BufferAttribute(positions, 3));
    geo.setAttribute("normal", new THREE.Float32BufferAttribute(normals, 3));
    geo.setAttribute("color", new THREE.Float32BufferAttribute(colors, 3));
    geo.setIndex(buildChunkMesh._idx);
    buildChunkMesh._idx = [];
    return geo;
  }
  buildChunkMesh._idx = [];

  function createChunkEntry(cx, cz) {
    var key = chunkKey(cx, cz);
    var data = generateChunk(cx, cz);
    var entry = { data: data, mesh: null };
    chunks.set(key, entry);
    return entry;
  }
  function hasNeighborData(cx, cz) {
    return !!(getChunkData(cx + 1, cz) && getChunkData(cx - 1, cz) &&
               getChunkData(cx, cz + 1) && getChunkData(cx, cz - 1));
  }
  function buildChunkMeshFor(cx, cz) {
    var entry = chunks.get(chunkKey(cx, cz));
    if (!entry) return;
    if (entry.mesh) {
      entry.mesh.geometry.dispose();
      scene.remove(entry.mesh);
      var mi = chunkMeshes.indexOf(entry.mesh);
      if (mi >= 0) chunkMeshes.splice(mi, 1);
    }
    var geo = buildChunkMesh(entry.data, cx, cz);
    var mesh = new THREE.Mesh(geo, sharedMat);
    mesh.position.set(0, 0, 0);
    scene.add(mesh);
    chunkMeshes.push(mesh);
    entry.mesh = mesh;
  }
  function rebuildChunk(cx, cz) {
    buildChunkMeshFor(cx, cz);
  }

  // ---------- Break / Place ----------
  var targetCell = null; // {x,y,z}
  function computeTarget() {
    targetCell = null;
    if (chunkMeshes.length === 0) { outline.visible = false; return; }
    raycaster.setFromCamera(centerVec, camera);
    var hits = raycaster.intersectObjects(chunkMeshes, false);
    if (hits.length === 0) { outline.visible = false; return; }
    var hit = hits[0];
    var p = hit.point, n = hit.face.normal;
    var bx = Math.floor(p.x - n.x * 0.5), by = Math.floor(p.y - n.y * 0.5), bz = Math.floor(p.z - n.z * 0.5);
    targetCell = { x: bx, y: by, z: bz };
    // outline
    var bid = readBlock(bx, by, bz);
    if (bid !== 0) {
      outline.position.set(bx + 0.5, by + 0.5, bz + 0.5);
      outline.visible = true;
    } else {
      outline.visible = false;
    }
  }
  function editChunkOf(wx, wy, wz) {
    var cx = Math.floor(wx / CHUNK), cz = Math.floor(wz / CHUNK);
    var lx = wx - cx * CHUNK, lz = wz - cz * CHUNK;
    rebuildChunk(cx, cz);
    // neighbor if on border
    if (lx === 0) rebuildChunk(cx - 1, cz);
    if (lx === CHUNK - 1) rebuildChunk(cx + 1, cz);
    if (lz === 0) rebuildChunk(cx, cz - 1);
    if (lz === CHUNK - 1) rebuildChunk(cx, cz + 1);
  }
  function handleBreak() {
    if (!targetCell) return;
    var t = targetCell;
    if (t.y === 0) return; // bedrock unbreakable
    var id = readBlock(t.x, t.y, t.z);
    if (id === 0) return;
    writeBlock(t.x, t.y, t.z, 0);
    editChunkOf(t.x, t.y, t.z);
  }
  function handlePlace() {
    if (!targetCell) return;
    var t = targetCell;
    // place cell = p + n*0.5
    // recompute place cell from target: place adjacent along normal. We need normal.
    // Recompute from raycast
    if (chunkMeshes.length === 0) return;
    raycaster.setFromCamera(centerVec, camera);
    var hits = raycaster.intersectObjects(chunkMeshes, false);
    if (hits.length === 0) return;
    var hit = hits[0];
    var p = hit.point, n = hit.face.normal;
    var px = Math.floor(p.x + n.x * 0.5), py = Math.floor(p.y + n.y * 0.5), pz = Math.floor(p.z + n.z * 0.5);
    var existing = readBlock(px, py, pz);
    if (existing !== 0) return;
    // don't overlap player
    var pp = player.pos;
    var pminX = pp.x - player.halfW, pmaxX = pp.x + player.halfW;
    var pminZ = pp.z - player.halfW, pmaxZ = pp.z + player.halfW;
    var pminY = pp.y, pmaxY = pp.y + player.height;
    var overlap = (px + 1 > pminX && px < pmaxX) &&
                    (pz + 1 > pminZ && pz < pmaxZ) &&
                    (py + 1 > pminY && py < pmaxY);
    if (overlap) return;
    var placeId = HOTBAR_BLOCKS[selectedSlot];
    writeBlock(px, py, pz, placeId);
    editChunkOf(px, py, pz);
  }

  // ---------- Physics ----------
  var GRAVITY = 25, JUMP_V = 8.5, MOVE_SPEED = 5.5;
  function collidesAt(px, py, pz) {
    // check player box against blocks
    var minX = Math.floor(px - player.halfW), maxX = Math.floor(px + player.halfW);
    var minY = Math.floor(py), maxY = Math.floor(py + player.height);
    var minZ = Math.floor(pz - player.halfW), maxZ = Math.floor(pz + player.halfW);
    for (var bx = minX; bx <= maxX; bx++)
      for (var by = minY; by <= maxY; by++)
        for (var bz = minZ; bz <= maxZ; bz++)
          if (readBlock(bx, by, bz) !== 0) return true;
    return false;
  }
  function updatePhysics(dt) {
    var pos = player.pos, vel = player.vel;
    // horizontal movement relative to yaw
    var fwd = new THREE.Vector3(-Math.sin(player.yaw), 0, -Math.cos(player.yaw));
    var right = new THREE.Vector3(Math.cos(player.yaw), 0, -Math.sin(player.yaw));
    var move = new THREE.Vector3();
    if (keys["KeyW"]) move.add(fwd);
    if (keys["KeyS"]) move.sub(fwd);
    if (keys["KeyD"]) move.add(right);
    if (keys["KeyA"]) move.sub(right);
    if (move.lengthSq() > 0) move.normalize().multiplyScalar(MOVE_SPEED);
    vel.x = move.x;
    vel.z = move.z;
    // gravity
    vel.y -= GRAVITY * dt;
    if (vel.y < -50) vel.y = -50;
    // jump
    if (keys["Space"] && player.onGround) {
      vel.y = JUMP_V;
      player.onGround = false;
    }
    // Move per axis with collision
    // X
    var nx = pos.x + vel.x * dt;
    if (!collidesAt(nx, pos.y, pos.z)) pos.x = nx;
    else vel.x = 0;
    // Z
    var nz = pos.z + vel.z * dt;
    if (!collidesAt(pos.x, pos.y, nz)) pos.z = nz;
    else vel.z = 0;
    // Y
    var ny = pos.y + vel.y * dt;
    if (!collidesAt(pos.x, ny, pos.z)) {
      pos.y = ny;
      player.onGround = false;
    } else {
      if (vel.y < 0) player.onGround = true;
      vel.y = 0;
    }
    // fell below
    if (pos.y < -20) {
      pos.set(SPAWN.x, SPAWN.y, SPAWN.z);
      vel.set(0, 0, 0);
    }
  }

  // ---------- World update (chunk gen/build) ----------
  var clock = new THREE.Clock();
  function updateWorld() {
    var pcx = Math.floor(player.pos.x / CHUNK), pcz = Math.floor(player.pos.z / CHUNK);
    // Generate data for chunks within 5 (max 4/frame)
    var genList = [];
    for (var dx = -5; dx <= 5; dx++)
      for (var dz = -5; dz <= 5; dz++) {
        var gx = pcx + dx, gz = pcz + dz;
        if (!chunks.has(chunkKey(gx, gz))) genList.push([gx, gz]);
      }
    // sort by distance
    genList.sort(function (a, b) {
      var da = (a[0] - pcx) * (a[0] - pcx) + (a[1] - pcz) * (a[1] - pcz);
      var db = (b[0] - pcx) * (b[0] - pcx) + (b[1] - pcz) * (b[1] - pcz);
      return da - db;
    });
    var genCount = 0;
    for (var gi = 0; gi < genList.length && genCount < 4; gi++) {
      createChunkEntry(genList[gi][0], genList[gi][1]);
      genCount++;
    }
    // Build meshes for chunks within 4 whose neighbors have data (max 2/frame)
    var buildList = [];
    for (var bx = -4; bx <= 4; bx++)
      for (var bz = -4; bz <= 4; bz++) {
        var bcx = pcx + bx, bcz = pcz + bz;
        var entry = chunks.get(chunkKey(bcx, bcz));
        if (entry && !entry.mesh && hasNeighborData(bcx, bcz)) buildList.push([bcx, bcz]);
      }
    buildList.sort(function (a, b) {
      var da = (a[0] - pcx) * (a[0] - pcx) + (a[1] - pcz) * (a[1] - pcz);
      var db = (b[0] - pcx) * (b[0] - pcx) + (b[1] - pcz) * (b[1] - pcz);
      return da - db;
    });
    var buildCount = 0;
    for (var bi = 0; bi < buildList.length && buildCount < 2; bi++) {
      buildChunkMeshFor(buildList[bi][0], buildList[bi][1]);
      buildCount++;
    }
    // Remove chunks farther than 7
    for (var rx = -8; rx <= 8; rx++)
      for (var rz = -8; rz <= 8; rz++) {
        var rcx = pcx + rx, rcz = pcz + rz;
        var dist = Math.max(Math.abs(rx), Math.abs(rz));
        if (dist > 7) {
          var rentry = chunks.get(chunkKey(rcx, rcz));
          if (rentry) {
            if (rentry.mesh) {
              rentry.mesh.geometry.dispose();
              scene.remove(rentry.mesh);
              var mi = chunkMeshes.indexOf(rentry.mesh);
              if (mi >= 0) chunkMeshes.splice(mi, 1);
            }
            chunks.delete(chunkKey(rcx, rcz));
          }
        }
    }

  // ---------- Animation loop ----------
  var lastTime = performance.now();
  function animate() {
    requestAnimationFrame(animate);
    var now = performance.now();
    var dt = (now - lastTime) / 1000;
    lastTime = now;
    if (dt > 0.1) dt = 0.1;

    // physics
    updatePhysics(dt);

    // camera
    var pos = player.pos;
    camera.position.set(pos.x, pos.y + player.eye, pos.z);
    camera.rotation.set(player.pitch, player.yaw, 0);

    // water recenter
    water.position.set(pos.x, WATER_Y, pos.z);

    // clouds drift & wrap
    for (var i = 0; i < clouds.length; i++) {
      clouds[i].position.x += 0.05;
      if (clouds[i].position.x - pos.x > 150) clouds[i].position.x -= 300;
      if (clouds[i].position.x - pos.x < -150) clouds[i].position.x += 300;
      if (clouds[i].position.z - pos.z > 150) clouds[i].position.z -= 300;
      if (clouds[i].position.z - pos.z < -150) clouds[i].position.z += 300;
    }

    // world update
    updateWorld();

    // target raycast
    computeTarget();

    renderer.render(scene, camera);
  }
  animate();
})();
