```javascript
function hash2(x, y) {
  let h = x * 374761393 + y * 668265263;
  h = (h ^ (h >> 13)) * 1274126177;
  h = h ^ (h >> 16);
  return (h & 0x7FFFFFFF) / 0x7FFFFFFF;
}

function noise2(x, y) {
  const ix = Math.floor(x), iy = Math.floor(y);
  const fx = x - ix, fy = y - iy;
  const sx = fx * fx * (3 - 2 * fx);
  const sy = fy * fy * (3 - 2 * fy);
  
  const v00 = hash2(ix, iy);
  const v10 = hash2(ix + 1, iy);
  const v01 = hash2(ix, iy + 1);
  const v11 = hash2(ix + 1, iy + 1);
  
  const v0 = v00 + (v10 - v00) * sx;
  const v1 = v01 + (v11 - v01) * sx;
  return v0 + (v1 - v0) * sy;
}

function fractal2(x, y) {
  let val = 0, amp = 1, freq = 1, max = 0;
  for (let i = 0; i < 4; i++) {
    val += noise2(x * freq, y * freq) * amp;
    max += amp;
    amp *= 0.5;
    freq *= 2;
  }
  return val / max;
}
```

```javascript
function hash3(x, y, z) {
  let h = x * 374761393 + y * 668265263 + z * 2147483647;
  h = (h ^ (h >> 13)) * 1274126177;
  h = h ^ (h >> 16);

function noise3(x, y, z) {
  const ix = Math.floor(x), iy = Math.floor(y), iz = Math.floor(z);
  const fx = x - ix, fy = y - iy, fz = z - iz;
  const sx = fx * fx * (3 - 2 * fx);
  const sy = fy * fy * (3 - 2 * fy);
  const sz = fz * fz * (3 - 2 * fz);
  
  // Trilinear interpolation of 8 corners
  const c000 = hash3(ix, iy, iz);
  const c100 = hash3(ix+1, iy, iz);
  const c010 = hash3(ix, iy+1, iz);
  const c110 = hash3(ix+1, iy+1, iz);
  const c001 = hash3(ix, iy, iz+1);
  const c101 = hash3(ix+1, iy, iz+1);
  const c011 = hash3(ix, iy+1, iz+1);
  const c111 = hash3(ix+1, iy+1, iz+1);
  
  // Interpolate along x
  const v00 = c000 + (c100 - c000) * sx;
  const v10 = c010 + (c110 - c010) * sx;
  const v01 = c001 + (c101 - c001) * sx;
  const v11 = c011 + (c111 - c011) * sx;
  
  // Along y
  const v0 = v00 + (v10 - v00) * sy;
  const v1 = v01 + (v11 - v01) * sy;
  
  // Along z
  return v0 + (v1 - v0) * sz;
}

  // For caves, we might just use single octave or fewer
```

```javascript
  const pcz = Math.floor(player.z / 16);
  
  // 1. Generate data for chunks within radius 5
  let genCount = 0;
  for (let dx = -5; dx <= 5 && genCount < 4; dx++) {
    for (let dz = -5; dz <= 5 && genCount < 4; dz++) {
      const cx = pcx + dx;
      const cz = pcz + dz;
      const key = cx + ',' + cz;
      if (chunks.has(key)) continue;
      
      const data = generateChunkData(cx, cz);
      chunks.set(key, { data: data, mesh: null });
      genCount++;
    }
  }
  
  // 2. Build meshes for chunks within radius 4
  let meshCount = 0;
  for (let dx = -4; dx <= 4 && meshCount < 2; dx++) {
    for (let dz = -4; dz <= 4 && meshCount < 2; dz++) {
      const cx = pcx + dx;
      const cz = pcz + dz;
      const key = cx + ',' + cz;
      const chunk = chunks.get(key);
      if (!chunk || chunk.mesh) continue;
      
      // Check all 4 neighbors have data
      const neighbors = [
        [cx+1, cz], [cx-1, cz], [cx, cz+1], [cx, cz-1]
      ];
      let ready = true;
      for (const [nx, nz] of neighbors) {
        if (!chunks.has(nx + ',' + nz)) { ready = false; break; }
      }
      if (!ready) continue;
      
      buildChunkMesh(cx, cz);
      meshCount++;
    }
  }
  
  // 3. Remove chunks farther than 7
  for (const [key, chunk] of chunks) {
    const [cx, cz] = key.split(',').map(Number);
    const dist = Math.max(Math.abs(cx - pcx), Math.abs(cz - pcz));
    if (dist > 7) {
      if (chunk.mesh) removeChunkMesh(chunk.mesh);
      chunks.delete(key);
    }
  }
}
```

One more thing I want to verify: the `hash2` function needs to work correctly with negative integers. In JavaScript, bitwise operations convert to 32-bit signed integers. So:

```javascript
function hash2(x, y) {
  let h = (x | 0) * 374761393 + (y | 0) * 668265263;
  h = ((h ^ (h >> 13)) * 1274126177) | 0;
  h = (h ^ (h >> 16)) | 0;
  return (h & 0x7FFFFFFF) / 0x7FFFFFFF;
}
```

The `| 0` ensures we're working with 32-bit integers. The multiplication `h * 1274126177` might overflow 32 bits, but since we're doing `| 0` after, it wraps around, which is fine for a hash function.

Actually wait, in JavaScript, numbers are 64-bit floats. The `*` operator will give a floating point result. The `| 0` converts to a 32-bit integer (truncating). So:

```javascript
h = ((h ^ (h >>> 13)) * 1274126177) | 0;  // use unsigned shift
h = (h ^ (h >>> 16)) | 0;

```
