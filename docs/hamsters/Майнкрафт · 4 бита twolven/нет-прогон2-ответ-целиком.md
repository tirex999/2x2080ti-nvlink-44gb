    function ihash(x, y) {
        var h = (x * 374761393 + y * 668265263) | 0;
        h = (h ^ (h >>> 13)) | 0;
        h = (h * 1274126177) | 0;
        h = (h ^ (h >>> 16)) | 0;
        return (h & 0x7fffffff) / 0x7fffffff;
    }
    function smooth(t) { return t * t * (3 - 2 * t); }
    function vnoise2(x, y) {
        var xi = Math.floor(x), yi = Math.floor(y);
        var xf = x - xi, yf = y - yi;
        var u = smooth(xf), v = smooth(yf);
        var a = ihash(xi, yi), b = ihash(xi + 1, yi);
        var c = ihash(xi, yi + 1), d = ihash(xi + 1, yi + 1);
        var ab = a + (b - a) * u;
        var cd = c + (d - c) * u;
        return ab + (cd - ab) * v;
    }
    function fbm2(x, y) {
        var sum = 0, amp = 1, freq = 1, norm = 0;
        for (var o = 0; o < 4; o++) {
            sum += vnoise2(x * freq, y * freq) * amp;
            norm += amp;
            amp *= 0.5;
            freq *= 2;
        }
        return sum / norm;
    }
    function ihash3(x, y, z) {
        var h = (x * 374761393 + y * 668265263 + z * 1440662683) | 0;
        h = (h ^ (h >>> 13)) | 0;
        h = (h * 1274126177) | 0;
        h = (h ^ (h >>> 16)) | 0;
        return (h & 0x7fffffff) / 0x7fffffff;
    }
    function vnoise3(x, y, z) {
        var xi = Math.floor(x), yi = Math.floor(y), zi = Math.floor(z);
        var xf = x - xi, yf = y - yi, zf = z - zi;
        var u = smooth(xf), v = smooth(yf), w = smooth(zf);
        function lerp(a, b, t) { return a + (b - a) * t; }
        var c000 = ihash3(xi, yi, zi),     c100 = ihash3(xi+1, yi, zi);
        var c010 = ihash3(xi, yi+1, zi),  c110 = ihash3(xi+1, yi+1, zi);
        var c001 = ihash3(xi, yi, zi+1), c101 = ihash3(xi+1, yi, zi+1);
        var c011 = ihash3(xi, yi+1, zi+1), c111 = ihash3(xi+1, yi+1, zi+1);
        var x00 = lerp(c000, c100, u), x10 = lerp(c010, c110, u);
        var x01 = lerp(c001, c101, u), x11 = lerp(c011, c111, u);
        var y0 = lerp(x00, x10, v), y1 = lerp(x01, x11, v);
        return lerp(y0, y1, w);
    }

    function getBlock(wx, wy, wz) {

    }

    // ---------- Terrain generation ----------
    function columnHeight(x, z) {
        var m = fbm2(x * 0.004, z * 0.004);
        var h = fbm2(x * 0.02, z * 0.02);
        return Math.floor(5 + m * m * 58 + h * 10);

    function generateChunk(cx, cz) {
        var data = new Uint8Array(CHUNK_X * CHUNK_Z * CHUNK_Y);
        for (var lz = 0; lz < CHUNK_Z; lz++) {
            for (var lx = 0; lx < CHUNK_X; lx++) {
                var wx = cx * 16 + lx, wz = cz * 16 + lz;
                var H = columnHeight(wx, wz);
                if (H > CHUNK_Y - 1) H = CHUNK_Y - 1;
                // Surface type
                var surf, below;
                if (H >= 46) { surf = 7; below = 3; }
                else if (H >= 37) { surf = 3; below = 3; }
                else if (H <= 16) { surf = 4; below = 4; }
                else { surf = 1; below = 2; }

                var idx = function (y) { return y * 256 + lx * 16 + lz; };
                // y = 0 unbreakable stone
                data[idx(0)] = 3;
                // Fill up to H
                for (var y = 1; y <= H; y++) {
                    var b;
                    if (y === H) b = surf;
                    else if (y >= H - 3) b = below;
                    else b = 3; // stone
                    // Caves
                    if (y > 2 && y < H - 1) {
                        if (vnoise3(wx * 0.09, y * 0.09, wz * 0.09) > 0.67) b = 0;
                    }
                    data[idx(y)] = b;
                }
                // Trees
                if (surf === 1 && H + 7 < CHUNK_Y) {
                    var th = ihash(wx * 3 + 11, wz * 3 + 7);
                    if (th < 0.02) {
                        var baseY = H + 1;
                        // trunk 4 high
                        for (var ty = 0; ty < 4; ty++) {
                            if (baseY + ty < CHUNK_Y) data[idx(baseY + ty)] = 5;
                        }
                        // leaves: 5x5 two layers at ty=3,4 ; 3x3 at ty=5 ; 1 at ty=6
                        function setLeaf(lx2, ly, lz2) {
                            if (lx2 < 0 || lx2 >= CHUNK_X || lz2 < 0 || lz2 >= CHUNK_Z) return;
                            if (ly < 0 || ly >= CHUNK_Y) return;
                            if (data[idx(ly)] === 0) data[idx(ly)] = 6;
                        }
                        for (var ly = 3; ly <= 4; ly++) {
                            for (var dx = -2; dx <= 2; dx++)
                                for (var dz = -2; dz <= 2; dz++)
                                    setLeaf(lx + dx, baseY + ly, lz + dz);
                        }
                        for (var dx2 = -1; dx2 <= 1; dx2++)
                            for (var dz2 = -1; dz2 <= 1; dz2++)
                                setLeaf(lx + dx2, baseY + 5, lz + dz2);
                        setLeaf(lx, baseY + 6, lz);
                    }
                }
            }
        }
        var e = { data: data, mesh: null, dirty: true };
        chunks.set(chunkKey(cx, cz), e);
    }

    // ---------- Meshing ----------

    function buildChunkGeometry(cx, cz) {
        _pos.length = 0; _norm.length = 0; _col.length = 0; _idx.length = 0;
        for (var y = 0; y < CHUNK_Y; y++) {
            for (var lz = 0; lz < CHUNK_Z; lz++) {
                for (var lx = 0; lx < CHUNK_X; lx++) {
                    var off = y * 256 + lx * 16 + lz;
                    var id = getBlock(cx * 16 + lx, y, cz * 16 + lz);
                    if (id === 0) continue;
                    var col = BLOCK_COLOR[id];
                    for (var f = 0; f < 6; f++) {
                        var face = FACES[f];
                        var nx = cx * 16 + lx + face.n[0];
                        var ny = y + face.n[1];
                        var nz = cz * 16 + lz + face.n[2];
                        var nb = getBlock(nx, ny, nz);
                        // show face if neighbor air OR (leaves and neighbor not leaves)
                        var visible = (nb === 0);
                        if (id === 6 && nb !== 6) visible = true;
                        if (!visible) continue;
                        var base = _pos.length / 3;
                        var sh = face.s;
                        for (var k = 0; k < 4; k++) {
                            var co = face.c[k];
                            var px = cx * 16 + lx + co[0];
                            var py = y + co[1];
                            var pz = cz * 16 + lz + co[2];
                            _pos.push(px, py, pz);
                            _norm.push(face.n[0], face.n[1], face.n[2]);
                            _col.push(col[0] * sh, col[1] * sh, col[2] * sh);
                        }
                        _idx.push(base, base + 1, base + 2, base, base + 2, base + 3);
                    }
                }
            }
        }
        if (_idx.length === 0) return null;
        var geo = new THREE.BufferGeometry();
        geo.setAttribute("position", new THREE.Float32BufferAttribute(_pos, 3));
        geo.setAttribute("normal", new THREE.Float32BufferAttribute(_norm, 3));
        geo.setAttribute("color", new THREE.Float32BufferAttribute(_col, 3));
        geo.setIndex(_idx);
        geo.computeBoundingSphere();
        return geo;
    }

    function removeFromMeshList(mesh) {
        var i = chunkMeshes.indexOf(mesh);
        if (i >= 0) chunkMeshes.splice(i, 1);
    }
    function rebuildChunk(cx, cz) {

        if (!e) return;
        if (e.mesh) {
            scene.remove(e.mesh);
            e.mesh.geometry.dispose();
            removeFromMeshList(e.mesh);
            e.mesh = null;
        }
        var geo = buildChunkGeometry(cx, cz);
        if (geo) {
            var mesh = new THREE.Mesh(geo, material);
            scene.add(mesh);
            chunkMeshes.push(mesh);
            e.mesh = mesh;
        }
        e.dirty = false;
    }

    // ---------- Raycast targeting ----------
    var raycaster = new THREE.Raycaster();
    var target = null; // { bx,by,bz }
    function computeTarget() {
        target = null;
        if (chunkMeshes.length === 0) { outline.visible = false; return; }
        raycaster.setFromCamera(new THREE.Vector2(0, 0), camera);
        raycaster.far = REACH;
        var hits = raycaster.intersectObjects(chunkMeshes, false);
        if (hits.length === 0) { outline.visible = false; return; }
        var hit = hits[0];
        var p = hit.point;
        var n = hit.face.normal;
        // world normal from object matrix (chunks at origin, identity)
        var nn = n.clone().transformDirection(hit.object.matrixWorld);
        var nx = Math.round(nn.x), ny = Math.round(nn.y), nz = Math.round(nn.z);
        if (nx === 0 && ny === 0 && nz === 0) { nx = 1; }
        var bx = Math.floor(p.x - nx * 0.5);
        var by = Math.floor(p.y - ny * 0.5);
        var bz = Math.floor(p.z - nz * 0.5);
        target = { bx: bx, by: by, bz: bz };
        outline.visible = true;
        outline.position.set(bx + 0.5, by + 0.5, bz + 0.5);
    }

    function breakBlock() {
        if (!target) return;
        if (target.by === 0) return; // unbreakable
        var cx = Math.floor(target.bx / 16), cz = Math.floor(target.bz / 16);
        setBlock(target.bx, target.by, target.bz, 0);
        rebuildChunk(cx, cz);
        // border neighbor
        var lx = target.bx - cx * 16, lz = target.bz - cz * 16;
        if (lx === 0) rebuildChunk(cx - 1, cz);
        if (lx === 15) rebuildChunk(cx + 1, cz);
        if (lz === 0) rebuildChunk(cx, cz - 1);
        if (lz === 15) rebuildChunk(cx, cz + 1);
    }
    function placeBlock() {
        if (!target) return;
        var px = target.bx + Math.round((camera.position.x - (target.bx + 0.5)) / Math.max(0.001, 1));
        // place cell = floor(p + n*0.5): compute via neighbor of target in normal direction
        // We need the face normal used for target. Recompute from raycast is heavy; instead
        // place into the adjacent cell along the same normal. Store lastNormal.
        if (!lastNormal) return;
        var nx = Math.round(lastNormal.x), ny = Math.round(lastNormal.y), nz = Math.round(lastNormal.z);
        var bx = target.bx + nx, by = target.by + ny, bz = target.bz + nz;
        if (getBlock(bx, by, bz) !== 0) return;
        // avoid overlapping player
        if (bx === Math.floor(player.x) && bz === Math.floor(player.z)) {
            var feet = Math.floor(player.y), head = Math.floor(player.y + PLAYER_H - 0.01);
            if (by >= feet && by <= head) return;
        }
        setBlock(bx, by, bz, HOTBAR_BLOCKS[selectedSlot]);
        var cx = Math.floor(bx / 16), cz = Math.floor(bz / 16);
        rebuildChunk(cx, cz);
        var lx = bx - cx * 16, lz = bz - cz * 16;
        if (lx === 0) rebuildChunk(cx - 1, cz);
        if (lx === 15) rebuildChunk(cx + 1, cz);
        if (lz === 0) rebuildChunk(cx, cz - 1);
        if (lz === 15) rebuildChunk(cx, cz + 1);
    }

    var _origCompute = computeTarget;

        _origCompute();

    function computeTargetFull() {
        target = null; lastNormal = null;
        if (chunkMeshes.length === 0) { outline.visible = false; return; }
        raycaster.setFromCamera(new THREE.Vector2(0, 0), camera);
        raycaster.far = REACH;
        var hits = raycaster.intersectObjects(chunkMeshes, false);
        if (hits.length === 0) { outline.visible = false; return; }
        var hit = hits[0];
        var p = hit.point;
        var nn = hit.face.normal.clone().transformDirection(hit.object.matrixWorld);
        var nx = Math.round(nn.x), ny = Math.round(nn.y), nz = Math.round(nn.z);
        if (nx === 0 && ny === 0 && nz === 0) { nx = 1; }
        lastNormal = nn;
        var bx = Math.floor(p.x - nx * 0.5);
        var by = Math.floor(p.y - ny * 0.5);
        var bz = Math.floor(p.z - nz * 0.5);
        target = { bx: bx, by: by, bz: bz };
        outline.visible = true;
        outline.position.set(bx + 0.5, by + 0.5, bz + 0.5);

    // ---------- Collision ----------
    function collides(px, py, pz) {
        var minX = Math.floor(px - PLAYER_HW), maxX = Math.floor(px + PLAYER_HW);
        var minY = Math.floor(py), maxY = Math.floor(py + PLAYER_H - 0.01);
        var minZ = Math.floor(pz - PLAYER_HW), maxZ = Math.floor(pz + PLAYER_HW);
        for (var x = minX; x <= maxX; x++)
            for (var y = minY; y <= maxY; y++)
                for (var z = minZ; z <= maxZ; z++)
                    if (getBlock(x, y, z) !== 0) return true;
        return false;
    }

    var clock = new THREE.Clock();

    function generateNearby() {
        var count = 0;
        for (var dx = -5; dx <= 5 && count < 4; dx++) {
            for (var dz = -5; dz <= 5 && count < 4; dz++) {
                var cx = pcx + dx, cz = pcz + dz;
                if (!chunks.has(chunkKey(cx, cz))) {
                    generateChunk(cx, cz);
                    count++;
                }
            }
        }
    }
    function buildNearbyMeshes() {
        var count = 0;
        for (var dx = -4; dx <= 4 && count < 2; dx++) {
            for (var dz = -4; dz <= 4 && count < 2; dz++) {
                var cx = pcx + dx, cz = pcz + dz;
                var e = chunks.get(chunkKey(cx, cz));
                if (!e) continue;
                if (!e.dirty && e.mesh) continue;
                // ensure 4 neighbors have data
                if (!getChunkData(cx + 1, cz) || !getChunkData(cx - 1, cz) ||
                    !getChunkData(cx, cz + 1) || !getChunkData(cx, cz - 1)) continue;
                rebuildChunk(cx, cz);
                count++;
            }
        }
    }
    function cleanupFar() {
        for (var key of Array.from(chunks.keys())) {
            var parts = key.split(",");
            var cx = parseInt(parts[0], 10), cz = parseInt(parts[1], 10);
            var dist = Math.max(Math.abs(cx - pcx), Math.abs(cz - pcz));
            if (dist > 7) {
                var e = chunks.get(key);
                if (e.mesh) {
                    scene.remove(e.mesh);
                    e.mesh.geometry.dispose();
                    removeFromMeshList(e.mesh);
                    e.mesh = null;
                }
                chunks.delete(key);
            }
        }
    }

    function updatePlayer(dt) {
        // movement
        var forward = (keys["KeyW"] ? 1 : 0) - (keys["KeyS"] ? 1 : 0);
        var strafe = (keys["KeyD"] ? 1 : 0) - (keys["KeyA"] ? 1 : 0);
        var sin = Math.sin(player.yaw), cos = Math.cos(player.yaw);
        var mx = (-sin * forward + cos * strafe) * MOVE_SPEED;
        var mz = (-cos * forward - sin * strafe) * MOVE_SPEED;
        player.vx = mx; player.vz = mz;
        player.vy -= GRAVITY * dt;
        if (player.vy < -50) player.vy = -50;
        if (keys["Space"] && player.onGround) {
            player.vy = JUMP_V;
            player.onGround = false;
        }
        // X axis
        var nx = player.x + player.vx * dt;
        if (!collides(nx, player.y, player.z)) player.x = nx; else player.vx = 0;
        // Z axis
        var nz = player.z + player.vz * dt;
        if (!collides(player.x, player.y, nz)) player.z = nz; else player.vz = 0;
        // Y axis
        var ny = player.y + player.vy * dt;
        if (!collides(player.x, ny, player.z)) {
            player.y = ny;
            player.onGround = false;
        } else {
            if (player.vy < 0) player.onGround = true;
            player.vy = 0;
        }
        // fell off
        if (player.y < -20) respawn();
    }

    function animate() {
        requestAnimationFrame(animate);
        var dt = Math.min(clock.getDelta(), 0.05);

        pcx = Math.floor(camera.position.x / 16);
        pcz = Math.floor(camera.position.z / 16);

        generateNearby();
        buildNearbyMeshes();
        cleanupFar();

        if (locked) updatePlayer(dt);

        // camera
        camera.position.set(player.x, player.y + EYE, player.z);
        camera.rotation.y = player.yaw;
        camera.rotation.x = player.pitch;
        camera.rotation.z = 0;

        // water follow
        water.position.set(camera.position.x, WATER_Y, camera.position.z);

        // clouds drift + wrap
        for (var i = 0; i < clouds.length; i++) {
            var cl = clouds[i];
            cl.position.x += dt * (1.5 + (i % 3) * 0.5);
            if (cl.position.x - camera.position.x > 150) cl.position.x -= 300;
            if (cl.position.x - camera.position.x < -150) cl.position.x += 300;
            if (cl.position.z - camera.position.z > 150) cl.position.z -= 300;
            if (cl.position.z - camera.position.z < -150) cl.position.z += 300;
        }

        computeTargetFull();
        renderer.render(scene, camera);
    }

    window.addEventListener("resize", function () {
        camera.aspect = window.innerWidth / window.innerHeight;
        camera.updateProjectionMatrix();
        renderer.setSize(window.innerWidth, window.innerHeight);
    });

    animate();
})();
