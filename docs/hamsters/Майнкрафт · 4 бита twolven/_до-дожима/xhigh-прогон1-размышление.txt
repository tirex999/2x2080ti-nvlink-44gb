```js
function smooth(t){ return t*t*(3-2*t); }
function noise2(x, y) {
  const xi = Math.floor(x), yi = Math.floor(y);
  const xf = x - xi, yf = y - yi;
  const a = hashInt(xi, yi, 0), b = hashInt(xi+1, yi, 0);
  const c = hashInt(xi, yi+1, 0), d = hashInt(xi+1, yi+1, 0);
  const u = smooth(xf), v = smooth(yf);
  return a + (b-a)*u + (c-a)*v + (a - b - c + d)*u*v; // bilinear
}
```

Fractal 2D (4 octaves):

```js
function fractal2(x, y) {
  let sum = 0, amp = 1, freq = 1, norm = 0;
  for (let i = 0; i < 4; i++) {
    sum += noise2(x*freq, y*freq) * amp;
    norm += amp;
    amp *= 0.5;
    freq *= 2;
  }
  return sum / norm;
}
```

```js
function buildChunkGeometry(cx, cz) {
  const chunk = chunks.get(cx + "," + cz);
  const positions = [], normals = [], colors = [];
  for (let lx = 0; lx < 16; lx++) {
    for (let lz = 0; lz < 16; lz++) {
      for (let ly = 0; ly < 80; ly++) {
        const id = chunk.data[ly*256 + lz*16 + lx]; // choose layout
        if (!id) continue;
        const wx = cx*16 + lx, wy = ly, wz = cz*16 + lz;
        const col = BLOCK_COLORS[id];
        const r = ((col>>16)&255)/255, g = ((col>>8)&255)/255, b = (col&255)/255;
        for (const f of FACES) {
          const nx = wx + f.dir[0], ny = wy + f.dir[1], nz = wz + f.dir[2];
          if (getBlock(nx, ny, nz) !== 0) continue;
          // push 2 triangles
          const v = f.verts;
          for (const ii of [0,1,2, 0,2,3]) {
            positions.push(wx + v[ii][0], wy + v[ii][1], wz + v[ii][2]);
            normals.push(f.dir[0], f.dir[1], f.dir[2]);
            const sh = f.shade;
            colors.push(r*sh, g*sh, b*sh);
          }
        }
      }
    }
  }
  const geo = new THREE.BufferGeometry();
  geo.setAttribute('position', new THREE.Float32BufferAttribute(positions, 3));
  geo.setAttribute('normal', new THREE.Float32BufferAttribute(normals, 3));
  geo.setAttribute('color', new THREE.Float32BufferAttribute(colors, 3));
  return geo;
}
```

```js
function getBlock(wx, wy, wz) {
  if (wy < 0 || wy >= 80) return 0;
  const cx = Math.floor(wx / 16), cz = Math.floor(wz / 16);
```
