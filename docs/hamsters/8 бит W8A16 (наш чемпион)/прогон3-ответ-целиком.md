function mat(color, opts = {}) {

}

function cyl(rt, rb, h, seg, m) { const o = new THREE.Mesh(new THREE.CylinderGeometry(rt, rb, h, seg), m); o.castShadow = true; return o; }

/* ---------------- cage: tray, sand, bars ---------------- */

/* ---------------- hamster factory ---------------- */
const PALETTES = [
  { body: 0xf2a65a, belly: 0xffe8c9, ear: 0xd98a3d },
  { body: 0xfff3e0, belly: 0xffffff, ear: 0xf0d9b8 },

function makeHamster(pal) {
  const g = new THREE.Group();
  const bodyM = mat(pal.body), bellyM = mat(pal.belly), earM = mat(pal.ear);

  const body = new THREE.Mesh(new THREE.SphereGeometry(0.42, 8, 6), bodyM);
  body.scale.set(1, 0.9, 1.15);
  body.castShadow = true;
  g.add(body);

  const head = new THREE.Group();
  head.position.set(0, 0.12, 0.36);
  g.add(head);

  const skull = new THREE.Mesh(new THREE.SphereGeometry(0.28, 8, 6), bodyM);
  skull.castShadow = true;
  head.add(skull);

  const snout = new THREE.Mesh(new THREE.SphereGeometry(0.12, 6, 5), bellyM);
  snout.position.set(0, -0.05, 0.22);
  head.add(snout);

  const nose = new THREE.Mesh(new THREE.SphereGeometry(0.045, 6, 5), mat(0xff7fa5));
  nose.position.set(0, -0.02, 0.33);
  head.add(nose);

  const eyeM = mat(0x2b2b2b);
  for (const s of [-1, 1]) {
    const eye = new THREE.Mesh(new THREE.SphereGeometry(0.05, 6, 5), eyeM);
    eye.position.set(s * 0.11, 0.08, 0.22);
    head.add(eye);
    const cheek = new THREE.Mesh(new THREE.SphereGeometry(0.07, 6, 5), mat(0xffb3c1));
    cheek.scale.set(1, 0.7, 0.5);
    cheek.position.set(s * 0.2, -0.05, 0.18);
    head.add(cheek);
    const ear = new THREE.Mesh(new THREE.ConeGeometry(0.09, 0.15, 5), earM);
    ear.position.set(s * 0.17, 0.28, -0.02);
    ear.rotation.z = -s * 0.35;
    head.add(ear);
  }

  const tail = new THREE.Mesh(new THREE.SphereGeometry(0.08, 6, 5), bellyM);
  tail.position.set(0, 0, -0.5);
  g.add(tail);

  const legs = [];
  for (const [x, z] of [[-0.18, 0.25], [0.18, 0.25], [-0.18, -0.25], [0.18, -0.25]]) {
    const leg = new THREE.Mesh(new THREE.SphereGeometry(0.08, 6, 5), bellyM);
    leg.position.set(x, -0.34, z);
    g.add(leg);
    legs.push(leg);
  }

  return { group: g, head, legs };
}

/* ---------------- hamster AI ---------------- */
const WHEEL_STAND = new THREE.Vector3(-2.05, 0, 0);
const WHEEL_RIDE  = new THREE.Vector3(-2.28, WHEEL_Y, 0);
const BOWL_POINT  = new THREE.Vector3(1.35, 0, 1.1);
const FLOOR_Y = 0.42;

function angLerp(a, b, t) {
  let d = (b - a) % (Math.PI * 2);
  if (d > Math.PI) d -= Math.PI * 2;
  if (d < -Math.PI) d += Math.PI * 2;
  return a + d * t;
}

class Hamster {
  constructor(palette, startPos) {
    const built = makeHamster(palette);
    Object.assign(this, built);
    this.pos = startPos.clone();
    this.group.position.copy(this.pos).setY(FLOOR_Y);
    this.group.rotation.y = Math.random() * Math.PI * 2;
    this.state = 'idle';
    this.timer = 0.5 + Math.random() * 2;
    this.target = null;
    this.speed = 0.85 + Math.random() * 0.3;
    this.moving = false;
    scene.add(this.group);
  }

  chooseAction() {
    const r = Math.random();
    if (r < 0.38) {
      this.state = 'walk';
      const a = Math.random() * Math.PI * 2, rad = Math.random() * 2.7;
      this.target = new THREE.Vector3(Math.sin(a) * rad, 0, Math.cos(a) * rad);
      this.timer = 2 + Math.random() * 3;
    } else if (r < 0.58) {
      this.state = 'toWheel';
    } else if (r < 0.76) {
      this.state = 'toBowl';
    } else {
      this.state = 'idle';
      this.timer = 1 + Math.random() * 2.5;
    }
  }

  moveToward(dest, dt) {
    const dx = dest.x - this.pos.x, dz = dest.z - this.pos.z;
    const dist = Math.hypot(dx, dz);
    if (dist < 0.18) return true;
    const want = Math.atan2(dx, dz);
    this.group.rotation.y = angLerp(this.group.rotation.y, want, Math.min(1, dt * 6));
    this.pos.x += (dx / dist) * this.speed * dt;
    this.pos.z += (dz / dist) * this.speed * dt;
    return false;
  }

  update(dt, t) {
    this.moving = false;
    switch (this.state) {
      case 'idle':
        this.timer -= dt;
        if (this.timer <= 0) this.chooseAction();
        break;

      case 'walk':
        this.moving = true;
        this.timer -= dt;
        if (!this.target || this.moveToward(this.target, dt) || this.timer <= 0) {
          this.state = 'idle';
          this.timer = 0.5 + Math.random() * 1.5;
        }
        break;

      case 'toWheel':
        this.moving = true;
        if (this.moveToward(WHEEL_STAND, dt)) {
          this.state = 'onWheel';
          this.timer = 4 + Math.random() * 4;
          this.pos.z = THREE.MathUtils.clamp(this.pos.z, -0.3, 0.3);
        }
        break;

      case 'onWheel':
        this.timer -= dt;
        wheelSpeed = Math.max(wheelSpeed, 6);
        this.pos.lerp(WHEEL_RIDE, Math.min(1, dt * 8));
        this.group.rotation.y = angLerp(this.group.rotation.y, -Math.PI / 2, Math.min(1, dt * 8));
        this.group.rotation.x = -0.15;
        this.group.position.y = WHEEL_Y + Math.sin(t * 14) * 0.05;
        this.moving = true;
        if (this.timer <= 0) {
          this.state = 'walk';
          this.timer = 1.5 + Math.random();
          this.group.rotation.x = 0;
          this.pos.set(-1.7, 0, THREE.MathUtils.clamp(this.pos.z, -0.6, 0.6));
          this.target = new THREE.Vector3(
            (Math.random() - 0.5) * 4, 0, (Math.random() - 0.5) * 4);
        }
        break;

      case 'toBowl':
        this.moving = true;
        if (this.moveToward(BOWL_POINT, dt)) {
          this.state = 'eating';
          this.timer = 2 + Math.random() * 2;
          const a = Math.atan2(BOWL.x - this.pos.x, BOWL.z - this.pos.z);
          this.group.rotation.y = a;
        }
        break;

      case 'eating':
        this.timer -= dt;
        this.head.rotation.x = Math.sin(t * 12) * 0.25 - 0.15; // munching
        this.group.scale.setScalar(THREE.MathUtils.lerp(this.group.scale.x, 1.12, dt * 3));
        if (this.timer <= 0) {
          this.state = 'idle';
          this.timer = 1 + Math.random() * 2;
          this.head.rotation.x = 0;
        }
        break;
    }

    // stay inside the cage
    const len = Math.hypot(this.pos.x, this.pos.z);
    if (len > 2.9 && this.state !== 'onWheel') {
      this.pos.multiplyScalar(2.9 / len);
      if (this.state === 'walk') this.target = this.pos.clone().multiplyScalar(0.5);
    }

    if (this.state !== 'onWheel') {
      this.group.position.set(this.pos.x, FLOOR_Y, this.pos.z);
      // scurry bob + leg wiggle
      if (this.moving) {
        this.group.position.y = FLOOR_Y + Math.abs(Math.sin(t * 16)) * 0.06;
        this.group.scale.x = THREE.MathUtils.lerp(this.group.scale.x, 1, dt * 4);
        this.group.scale.z = THREE.MathUtils.lerp(this.group.scale.z, 1, dt * 4);
        this.legs.forEach((l, i) => {
          l.position.y = -0.34 + Math.sin(t * 18 + i * 1.7) * 0.04;
        });
      } else {
        this.group.scale.x = THREE.MathUtils.lerp(this.group.scale.x, 1, dt * 4);
        this.group.scale.z = THREE.MathUtils.lerp(this.group.scale.z, 1, dt * 4);
        this.legs.forEach(l => l.position.y = -0.34);
      }
    }
  }
}

/* ---------------- spawn hamsters ---------------- */
const hamsters = [];
const spots = [[1.2, 0.5], [-0.8, -1.4], [0.3, 1.8], [-1.5, 0.9]];
PALETTES.forEach((pal, i) => {
  hamsters.push(new Hamster(pal, new THREE.Vector3(spots[i][0], 0, spots[i][1])));
});

// gentle separation so they don't stack
function separate() {
  for (let i = 0; i < hamsters.length; i++) {
    for (let j = i + 1; j < hamsters.length; j++) {
      const a = hamsters[i], b = hamsters[j];
      if (a.state === 'onWheel' || b.state === 'onWheel') continue;
      const dx = b.pos.x - a.pos.x, dz = b.pos.z - a.pos.z;
      const d = Math.hypot(dx, dz);
      if (d > 0.001 && d < 0.75) {
        const push = (0.75 - d) / 2;
        a.pos.x -= (dx / d) * push; a.pos.z -= (dz / d) * push;
        b.pos.x += (dx / d) * push; b.pos.z += (dz / d) * push;
      }
    }
  }
}

/* ---------------- main loop ---------------- */
const clock = new THREE.Clock();
function animate() {
  requestAnimationFrame(animate);
  const dt = Math.min(clock.getDelta(), 0.05);
  const t = clock.elapsedTime;

  for (const h of hamsters) h.update(dt, t);
  separate();

  // wheel spins while ridden, coasts down otherwise
  wheelSpin.rotation.x -= wheelSpeed * dt;
  if (wheelSpeed > 0) wheelSpeed = Math.max(0, wheelSpeed - dt * (hamsters.some(h => h.state === 'onWheel') ? 0 : 4));

  ball.position.y = 0.18 + Math.sin(t * 2) * 0.01; // lazy wobble, purely decorative

  controls.update();
  renderer.render(scene, camera);
}
animate();

addEventListener('resize', () => {
  camera.aspect = innerWidth / innerHeight;
  camera.updateProjectionMatrix();
  renderer.setSize(innerWidth, innerHeight);
});
</script>

## What you get

| Feature | Details |
|---|---|
| **Scene** | Circular wire cage with metal bars/rings, wooden tray, sandy floor, pastel background |
| **Interactive object** | A spinning exercise wheel — hamsters walk over, climb on, sprint (wheel speeds up), then hop off; it coasts down slowly afterward |
| **Other objects** | Blue food bowl with pellets, orange half-pipe tunnel, carrot toy, red ball, scattered seeds |
| **4 hamsters** | Different fur colors, flat-shaded low-poly bodies, pink cheeks/noses, cone ears, wiggling legs |
| **AI** | Simple state machine: `idle → walk / toWheel / toBowl`, with munching (head bob + belly grow!), scurry-bob animation, and soft collision separation |
| **Camera** | OrbitControls with damping, zoom limits, and a floor-level polar clamp |

**Tips for tweaking:** change `PALETTES` for new fur colors, `WHEEL_RIDE`/`BOWL_POINT` to relocate furniture, or bump the hamster count by adding more palette entries and spawn spots.
