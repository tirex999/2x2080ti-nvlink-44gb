import * as THREE from 'three';
import { OrbitControls } from 'three/addons/controls/OrbitControls.js';

// ---------- helpers ----------
const rand  = (a, b) => a + Math.random() * (b - a);
const mat   = (color, opts = {}) => new THREE.MeshStandardMaterial({ color, flatShading: true, ...opts });

// ---------- renderer / scene / camera ----------
const renderer = new THREE.WebGLRenderer({ antialias: true, alpha: true });
renderer.setSize(innerWidth, innerHeight);
renderer.setPixelRatio(Math.min(devicePixelRatio, 2));
renderer.shadowMap.enabled = true;
renderer.shadowMap.type = THREE.PCFSoftShadowMap;
document.body.appendChild(renderer.domElement);

const scene = new THREE.Scene();

const camera = new THREE.PerspectiveCamera(45, innerWidth / innerHeight, 0.1, 100);
camera.position.set(8, 6.5, 10);

const controls = new OrbitControls(camera, renderer.domElement);
controls.target.set(0, 1, 0);
controls.enableDamping = true;
controls.dampingFactor = 0.08;
controls.minDistance = 5;
controls.maxDistance = 20;
controls.maxPolarAngle = Math.PI * 0.49;

// ---------- lights ----------
scene.add(new THREE.HemisphereLight(0xffffff, 0xffc9de, 0.9));
const sun = new THREE.DirectionalLight(0xfff2d0, 1.6);
sun.position.set(6, 10, 5);
sun.castShadow = true;
sun.shadow.mapSize.set(1024, 1024);
sun.shadow.camera.left = -8; sun.shadow.camera.right = 8;
sun.shadow.camera.top = 8;   sun.shadow.camera.bottom = -8;
scene.add(sun);

// ---------- cage: tray + glass walls ----------
const TRAY_W = 10, TRAY_D = 6, WALL_H = 2.6;

const tray = new THREE.Mesh(new THREE.BoxGeometry(TRAY_W, 0.4, TRAY_D), mat(0xffd9ec));
tray.position.y = -0.2;
tray.receiveShadow = true;
scene.add(tray);

const bedding = new THREE.Mesh(new THREE.BoxGeometry(TRAY_W - 0.4, 0.1, TRAY_D - 0.4), mat(0xf7e8c9));
bedding.position.y = 0.05;
bedding.receiveShadow = true;
scene.add(bedding);

function addWall(w, h, x, z, ry) {
  const glass = new THREE.Mesh(
    new THREE.PlaneGeometry(w, h),
    new THREE.MeshStandardMaterial({ color: 0xcdeeff, transparent: true, opacity: 0.14, side: THREE.DoubleSide })
  );
  glass.position.set(x, h / 2, z);
  glass.rotation.y = ry;
  scene.add(glass);
  const edges = new THREE.LineSegments(
    new THREE.EdgesGeometry(new THREE.PlaneGeometry(w, h)),
    new THREE.LineBasicMaterial({ color: 0xffffff, transparent: true, opacity: 0.7 })
  );
  edges.position.copy(glass.position);
  edges.rotation.copy(glass.rotation);
  scene.add(edges);
}
addWall(TRAY_W, WALL_H, 0, -TRAY_D / 2, 0);                       // back
addWall(TRAY_W, WALL_H, 0,  TRAY_D / 2, Math.PI);                 // front
addWall(TRAY_D, WALL_H, -TRAY_W / 2, 0,  Math.PI / 2);            // left
addWall(TRAY_D, WALL_H,  TRAY_W / 2, 0, -Math.PI / 2);            // right

// ---------- exercise wheel ----------
const WHEEL_POS = new THREE.Vector3(-3.1, 1.05, -1.9);
const wheel = new THREE.Group();
{
  const rimMat = mat(0xff8fb1);
  const rim = new THREE.Mesh(new THREE.TorusGeometry(1, 0.09, 6, 24), rimMat);
  rim.rotation.y = Math.PI / 2;
  rim.castShadow = true;
  wheel.add(rim);
  const hub = new THREE.Mesh(new THREE.CylinderGeometry(0.13, 0.13, 0.35, 8), rimMat);
  hub.rotation.z = Math.PI / 2;
  wheel.add(hub);
  const spokeMat = mat(0xffb3cd);
  for (let i = 0; i < 8; i++) {
    const s = new THREE.Mesh(new THREE.CylinderGeometry(0.03, 0.03, 1.95, 4), spokeMat);
    s.rotation.x = i * Math.PI / 4;   // spokes lie in the wheel's YZ plane
    wheel.add(s);
  }
  // stand
  const pole = new THREE.Mesh(new THREE.CylinderGeometry(0.08, 0.1, 1.05, 6), mat(0xd9a066));
  pole.position.y = -0.55;
  const base = new THREE.Mesh(new THREE.BoxGeometry(0.9, 0.12, 0.9), mat(0xd9a066));
  base.position.y = -1.0;
  base.castShadow = true;
  wheel.add(pole, base);
}
wheel.position.copy(WHEEL_POS);
scene.add(wheel);
let wheelSpin = 0, wheelTarget = 0;

// ---------- food bowl + pellets ----------
const BOWL_POS = new THREE.Vector3(3.3, 0, 1.9);
const bowl = new THREE.Mesh(new THREE.CylinderGeometry(0.55, 0.4, 0.3, 10), mat(0x8ecae6));
bowl.position.set(BOWL_POS.x, 0.15, BOWL_POS.z);
bowl.castShadow = true;
scene.add(bowl);

const pellets = [];
for (let i = 0; i < 10; i++) {
  const p = new THREE.Mesh(new THREE.SphereGeometry(0.07, 5, 4), mat(0xc98a4b));
  if (i < 6) {
    p.position.set(BOWL_POS.x + rand(-0.25, 0.25), 0.32, BOWL_POS.z + rand(-0.25, 0.25));
  } else {
    p.position.set(rand(-4, 4), 0.12, rand(-2.5, 2.5));
  }
  p.userData = { visible: true, respawnAt: 0 };
  scene.add(p);
  pellets.push(p);
}

// ---------- tunnel (half pipe) ----------
{

  geo.rotateZ(Math.PI / 2);   // lay it along X, opening faces down
  const tunnel = new THREE.Mesh(geo, mat(0xa8dadc, { side: THREE.DoubleSide }));
  tunnel.position.set(1.6, 0.75, -1.6);
  tunnel.castShadow = true;
  scene.add(tunnel);
}

// ---------- hay tufts ----------
for (let i = 0; i < 14; i++) {
  const h = new THREE.Mesh(new THREE.ConeGeometry(0.06, rand(0.25, 0.5), 4), mat(0xd9c37a));
  h.position.set(rand(-4.5, 4.5), 0.15, rand(-2.6, 2.6));
  h.rotation.z = rand(-0.4, 0.4);
  scene.add(h);
}

// ---------- hamster factory ----------
function makeHamster(colors) {
  const g = new THREE.Group();

  const body = new THREE.Mesh(new THREE.SphereGeometry(0.5, 7, 5), mat(colors.body));
  body.scale.set(1, 0.85, 1.25);
  body.position.y = 0.45;
  body.castShadow = true;
  g.add(body);

  const head = new THREE.Mesh(new THREE.SphereGeometry(0.33, 7, 5), mat(colors.head));
  head.position.set(0, 0.62, 0.55);
  head.castShadow = true;
  g.add(head);

  const snout = new THREE.Mesh(new THREE.SphereGeometry(0.16, 6, 4), mat(colors.belly));
  snout.position.set(0, 0.55, 0.85);
  g.add(snout);

  const nose = new THREE.Mesh(new THREE.SphereGeometry(0.05, 5, 4), mat(0xe07a9f));
  nose.position.set(0, 0.58, 1.0);
  g.add(nose);

  for (const side of [-1, 1]) {
    const eye = new THREE.Mesh(new THREE.SphereGeometry(0.06, 5, 4), mat(0x2b2b2b));
    eye.position.set(side * 0.18, 0.7, 0.8);
    g.add(eye);
    const ear = new THREE.Mesh(new THREE.ConeGeometry(0.11, 0.18, 5), mat(colors.body));
    ear.position.set(side * 0.22, 0.95, 0.42);
    ear.rotation.z = -side * 0.35;
    g.add(ear);
    const cheek = new THREE.Mesh(new THREE.SphereGeometry(0.13, 5, 4), mat(0xffc2d4));
    cheek.position.set(side * 0.28, 0.52, 0.72);
    g.add(cheek);
  }

  for (const [fx, fz] of [[-0.22, 0.55], [0.22, 0.55], [-0.22, -0.45], [0.22, -0.45]]) {
    const foot = new THREE.Mesh(new THREE.SphereGeometry(0.09, 5, 4), mat(colors.belly));
    foot.position.set(fx, 0.08, fz);
    g.add(foot);
  }

  const tail = new THREE.Mesh(new THREE.SphereGeometry(0.07, 5, 4), mat(colors.belly));
  tail.position.set(0, 0.4, -0.68);
  g.add(tail);

  return { group: g, head };
}

// ---------- hamster AI ----------
const hamsters = [];
const spawnDefs = [
  { colors: { body: 0xf4a259, head: 0xf4a259, belly: 0xfdf0d5 }, pos: [-2, 1.2] },
  { colors: { body: 0x9aa5b1, head: 0x9aa5b1, belly: 0xe8ecf1 }, pos: [ 2, -0.5] },
  { colors: { body: 0xf7e3c0, head: 0xf7e3c0, belly: 0xfff8ea }, pos: [ 0,  1.8] },
];
for (const def of spawnDefs) {
  const h = makeHamster(def.colors);
  h.group.position.set(def.pos[0], 0.02, def.pos[1]);
  scene.add(h.group);
  hamsters.push({
    group: h.group, head: h.head,
    state: 'wander', timer: rand(0.5, 2), speed: rand(0.9, 1.3),
    target: new THREE.Vector3(), moving: false, eatTimer: 0,
  });
  setState(hamsters[hamsters.length - 1], 'wander', 99);
}

function setState(h, state, dur) {
  h.state = state;
  h.timer = dur;
  if (state === 'toWheel') {
    h.target.set(WHEEL_POS.x, 0, WHEEL_POS.z + 0.15);
    h.timer = 12;
  } else if (state === 'toBowl') {
    h.target.set(BOWL_POS.x - 0.75, 0, BOWL_POS.z - 0.45);
    h.timer = 12;
  } else if (state === 'wander') {
    h.target.set(rand(-4.3, 4.3), 0, rand(-2.4, 2.4));
    h.timer = 12;
  }
}

function pickNextAction(h) {
  const r = Math.random();
  if (r < 0.25) setState(h, 'toWheel', 12);
  else if (r < 0.5) setState(h, 'toBowl', 12);
  else setState(h, 'wander', 12);
}

function eatPellet() {
  const p = pellets.find(p => p.userData.visible);
  if (p) { p.userData.visible = false; p.visible = false; p.userData.respawnAt = performance.now() / 1000 + 12; }
}

const _dir = new THREE.Vector3();
function updateHamster(h, dt, t) {
  h.timer -= dt;
  const m = h.group;

  const seek = () => {
    _dir.copy(h.target).sub(m.position); _dir.y = 0;
    const dist = _dir.length();
    if (dist < 0.2) return true;
    _dir.normalize();
    m.position.addScaledVector(_dir, h.speed * dt);
    m.rotation.y = Math.atan2(_dir.x, _dir.z);
    h.moving = true;
    return false;
  };

  switch (h.state) {
    case 'wander':
      if (seek()) setState(h, 'pause', rand(0.5, 2.5));
      break;
    case 'pause':
      if (h.timer <= 0) pickNextAction(h);
      break;
    case 'toWheel':
      if (seek()) { h.state = 'wheel'; h.timer = rand(4, 7); }
      else if (h.timer <= 0) { h.state = 'wheel'; h.timer = rand(3, 5); }
      break;
    case 'toBowl':
      if (seek()) { h.state = 'eat'; h.timer = rand(3, 5); h.eatTimer = 0.3; }
      else if (h.timer <= 0) { h.state = 'eat'; h.timer = rand(2, 4); h.eatTimer = 0.3; }
      break;
    case 'wheel':
      m.position.x = WHEEL_POS.x;
      m.position.z = WHEEL_POS.z + 0.15;
      m.rotation.y = 0;
      h.moving = true;
      if (h.timer <= 0) setState(h, 'pause', rand(0.5, 1.5));
      break;
    case 'eat':
      m.position.x = BOWL_POS.x - 0.75;
      m.position.z = BOWL_POS.z - 0.45;
      m.rotation.y = Math.atan2(BOWL_POS.x - m.position.x, BOWL_POS.z - m.position.z);
      h.head.rotation.x = Math.sin(t * 10) * 0.25;
      h.eatTimer -= dt;
      if (h.eatTimer <= 0) { eatPellet(); h.eatTimer = 0.7; }
      if (h.timer <= 0) { h.head.rotation.x = 0; setState(h, 'pause', rand(0.5, 1.5)); }
      break;
  }

  // waddle: bob + side-to-side sway while moving
  if (h.moving) {
    m.position.y = 0.02 + Math.abs(Math.sin(t * 12)) * 0.06;
    m.rotation.z = Math.sin(t * 12) * 0.07;
  } else {
    m.position.y += (0.02 - m.position.y) * Math.min(dt * 10, 1);
    m.rotation.z *= 0.9;
    if (h.state !== 'eat') h.head.rotation.x *= 0.9;
  }
  h.moving = false;
}

// ---------- main loop ----------
const clock = new THREE.Clock();
function animate() {
  requestAnimationFrame(animate);
  const dt = Math.min(clock.getDelta(), 0.05);
  const t = clock.elapsedTime;

  for (const h of hamsters) updateHamster(h, dt, t);

  // wheel spins while anyone is in it
  const riders = hamsters.filter(h => h.state === 'wheel').length;
  wheelTarget = riders > 0 ? 6 : 0;
  wheelSpin += (wheelTarget - wheelSpin) * Math.min(dt * 2, 1);
  wheel.rotation.x -= wheelSpin * dt;

  // pellets slowly regrow
  const now = t;
  for (const p of pellets) {
    if (!p.userData.visible && now >= p.userData.respawnAt) {
      p.userData.visible = true; p.visible = true;
      if (Math.random() < 0.5) {
        p.position.set(BOWL_POS.x + rand(-0.25, 0.25), 0.32, BOWL_POS.z + rand(-0.25, 0.25));
      }
    }
  }

  controls.update();
  renderer.render(scene, camera);
}
animate();

addEventListener('resize', () => {
  camera.aspect = innerWidth / innerHeight;
  camera.updateProjectionMatrix();
  renderer.setSize(innerWidth, innerHeight);
});
</script>

## What's inside

**Scene**
- Pastel tray + bedding, four frosted-glass cage walls with white frame edges
- A pink **exercise wheel** (rim, hub, 8 spokes, wooden stand)
- A blue **food bowl** with pellets, a **half-pipe tunnel**, and scattered hay tufts
- Soft shadows, warm sunlight, and a candy-colored gradient sky

**The hamsters** (Peanut, Mochi, Biscuit)
- Built from low-poly spheres/cones with `flatShading`: bodies, heads, snouts, pink cheeks, ears, tiny feet, and stubby tails
- A simple state machine: `wander → pause → (go to wheel | go to bowl | wander)`
- **In the wheel**: they stand at the bottom and the wheel smoothly spins up/down based on how many riders there are
- **At the bowl**: they turn to face it, bob their heads to eat, and pellets actually disappear — then regrow after 12 seconds
- Walking includes a bobbing hop and side-to-side waddle for extra cuteness

**Controls**
- OrbitControls with damping, zoom limits, and a floor clamp so you never go underground

Everything is dependency-light (just Three.js via CDN), runs at 60fps, and the code is organized into clearly commented sections so it's easy to tweak colors, speeds, or add a fourth hamster. 🐹
