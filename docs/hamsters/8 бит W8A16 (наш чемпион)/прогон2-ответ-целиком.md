// ---------- Hamsters ----------
function makeHamster(bodyColor, bellyColor) {
  const g = new THREE.Group();
  const bodyMat = flatMat(bodyColor), bellyMat = flatMat(bellyColor);

  const body = new THREE.Mesh(new THREE.SphereGeometry(0.32, 7, 5), bodyMat);
  body.scale.set(1, 0.85, 1.25);
  body.position.y = 0.3;
  g.add(body);

  const belly = new THREE.Mesh(new THREE.SphereGeometry(0.24, 6, 4), bellyMat);
  belly.scale.set(0.9, 0.7, 1.1);
  belly.position.set(0, 0.24, 0.1);
  g.add(belly);

  const head = new THREE.Mesh(new THREE.SphereGeometry(0.24, 7, 5), bodyMat);
  head.position.set(0, 0.42, 0.38);
  g.add(head);

  [-1, 1].forEach(s => {
    const ear = new THREE.Mesh(new THREE.SphereGeometry(0.09, 5, 4), bodyMat);
    ear.position.set(0.17 * s, 0.62, 0.3);
    g.add(ear);
    const earIn = new THREE.Mesh(new THREE.SphereGeometry(0.05, 4, 3), flatMat(0xffb3c1));
    earIn.position.set(0.17 * s, 0.63, 0.34);
    g.add(earIn);
    const eye = new THREE.Mesh(new THREE.SphereGeometry(0.045, 5, 4), flatMat(0x2b2b2b));
    eye.position.set(0.14 * s, 0.47, 0.57);
    g.add(eye);
    const cheek = new THREE.Mesh(new THREE.SphereGeometry(0.06, 5, 4), flatMat(0xffb3c1));
    cheek.position.set(0.16 * s, 0.36, 0.52);
    g.add(cheek);
    const foot = new THREE.Mesh(new THREE.BoxGeometry(0.1, 0.12, 0.16), bellyMat);
    foot.position.set(0.18 * s, 0.06, 0.15);
    g.add(foot);
  });
  const nose = new THREE.Mesh(new THREE.SphereGeometry(0.05, 5, 4), flatMat(0xff8fab));
  nose.position.set(0, 0.4, 0.62);
  g.add(nose);
  const tail = new THREE.Mesh(new THREE.ConeGeometry(0.06, 0.15, 4), bodyMat);
  tail.rotation.x = -Math.PI / 2;
  tail.position.set(0, 0.3, -0.42);
  g.add(tail);

  g.traverse(o => { if (o.isMesh) o.castShadow = true; });
  return { group: g, body, head, feet: [] };
}
// collect feet (small boxes at y≈0.06)
function collectFeet(h) {
  h.group.children.forEach(c => {
    if (c.geometry.type === 'BoxGeometry') h.feet.push(c);
  });
}

const hamsterDefs = [
  { body: 0xf4a261, belly: 0xfff1dc, pos: [-1.5, 0.5] },
  { body: 0xd9a066, belly: 0xfdf0d5, pos: [0.8, -0.8] },

];

const hamsters = hamsterDefs.map(d => {
  const h = makeHamster(d.body, d.belly);
  collectFeet(h);
  h.group.position.set(d.pos[0], SAND_Y, d.pos[1]);
  h.group.rotation.y = Math.random() * Math.PI * 2;
  h.state = 'wander';
  h.timer = 1 + Math.random() * 2;
  h.target = new THREE.Vector3();
  h.speed = 0.9 + Math.random() * 0.5;
  h.phase = Math.random() * 10;
  h.wheeling = false;
  pickWanderTarget(h);
  scene.add(h.group);
  return h;

function pickWanderTarget(h) {
  h.target.set(
    (Math.random() * 2 - 1) * BOUNDS_X,
    SAND_Y,
    (Math.random() * 2 - 1) * BOUNDS_Z
  );
}

function moveTo(h, target, speed, dt) {
  const dir = new THREE.Vector3().subVectors(target, h.group.position);
  dir.y = 0;
  const dist = dir.length();
  if (dist > 0.05) {
    dir.normalize();
    // smooth turn
    const desired = Math.atan2(dir.x, dir.z);
    let diff = desired - h.group.rotation.y;
    while (diff > Math.PI) diff -= Math.PI * 2;
    while (diff < -Math.PI) diff += Math.PI * 2;
    h.group.rotation.y += diff * Math.min(1, dt * 6);
    h.group.position.addScaledVector(dir, Math.min(speed * dt, dist));
  }
  return dist;
}

function updateHamster(h, dt, t) {
  h.timer -= dt;
  const p = h.group.position;

  switch (h.state) {
    case 'wander': {
      const d = moveTo(h, h.target, h.speed, dt);
      if (d < 0.15 || h.timer <= 0) {
        h.timer = 1 + Math.random() * 2.5;
        const r = Math.random();
        if (r < 0.22) { h.state = 'pause'; h.timer = 0.8 + Math.random() * 2; }
        else if (r < 0.32) { h.state = 'eat'; h.target = bowlPos.clone(); h.timer = 99; }
        else if (r < 0.42) { h.state = 'run'; h.target = WHEEL_POS.clone(); h.timer = 99; }
        else pickWanderTarget(h);
      }
      break;
    }
    case 'pause': {
      if (h.timer <= 0) { h.state = 'wander'; pickWanderTarget(h); }
      break;
    }
    case 'eat': {
      const d = moveTo(h, new THREE.Vector3(bowlPos.x + 0.55, SAND_Y, bowlPos.z + 0.4), h.speed, dt);
      if (d < 0.15) {
        if (h.timer === 99) h.timer = 2 + Math.random() * 2;
        h.head.position.y = 0.42 + Math.sin(t * 12) * 0.05; // munching
        h.head.rotation.x = Math.sin(t * 12) * 0.25;
        if (h.timer <= 0) {
          h.head.position.y = 0.42; h.head.rotation.x = 0;
          h.state = 'wander'; pickWanderTarget(h);
        }
      }
      break;
    }
    case 'run': {
      const d = moveTo(h, WHEEL_POS, h.speed * 1.3, dt);
      if (d < 0.15) {
        if (h.timer === 99) h.timer = 3 + Math.random() * 3;
        h.wheeling = true;
        wheelSpin += dt * 6;
        if (h.timer <= 0) {
          h.wheeling = false;
          h.state = 'wander'; pickWanderTarget(h);
        }
      }
      break;
    }
  }

  // clamp to tray
  p.x = THREE.MathUtils.clamp(p.x, -BOUNDS_X, BOUNDS_X);
  p.z = THREE.MathUtils.clamp(p.z, -BOUNDS_Z, BOUNDS_Z);

  // walk animation: bob + foot shuffle
  const moving = h.state === 'wander' || h.state === 'run' || (h.state === 'eat' && h.timer === 99);
  const freq = h.state === 'run' ? 18 : 10;
  const amp = moving ? 0.04 : 0.015;
  h.group.position.y = SAND_Y + Math.abs(Math.sin(t * freq + h.phase)) * amp;
  h.group.rotation.x = moving ? Math.sin(t * freq + h.phase) * 0.05 : 0;
  h.feet.forEach((f, i) => {
    f.position.y = 0.06 + (moving ? Math.max(0, Math.sin(t * freq + h.phase + (i % 2) * Math.PI)) * 0.05 : 0);
  });
  // idle ear twitch
  if (!moving && Math.sin(t * 0.7 + h.phase) > 0.99) h.head.rotation.z = 0.1;
  else h.head.rotation.z *= 0.9;
}

// ---------- Loop ----------
const clock = new THREE.Clock();
function animate() {
  requestAnimationFrame(animate);
  const dt = Math.min(clock.getDelta(), 0.05);
  const t = clock.elapsedTime;

  hamsters.forEach(h => updateHamster(h, dt, t));

  // wheel decays when nobody runs
  if (wheelSpin > 0.001) {
    wheelGroup.rotation.x -= wheelSpin * dt;
    wheelSpin *= (hamsters.some(h => h.wheeling) ? 1 : 0.97);
  }
  // ball wobbles gently
  ball.rotation.y += dt * 0.3;
  ball.position.y = SAND_Y + 0.22 + Math.sin(t * 2) * 0.01;

  controls.update();
  renderer.render(scene, camera);
}
animate();

addEventListener('resize', () => {
  camera.aspect = innerWidth / innerHeight;
  camera.updateProjectionMatrix();
  renderer.setSize(innerWidth, innerHeight);
});
</script>

## What you get

**Scene contents:**
- 🏠 **Cage** — pink tray, sand floor, wire bars with horizontal rails, and faint translucent glass panels, all sitting on a wooden table
- 🎡 **Exercise wheel** — hamsters run on it, spinning it (it coasts to a stop when they leave)
- 🥣 **Food bowl** — purple bowl with seeds; hamsters approach and visibly *munch* (head bobbing)
- 🛸 **Tunnel**, grass tufts, and a wobbling red ball for decoration

**Hamster behavior (simple state machine):**
- **Wander** — picks random targets, walks with smooth turning, bobs, and shuffles its feet
- **Pause** — stops and idles, occasionally twitching an ear
- **Eat** — walks to the bowl and chews for a few seconds
- **Run** — dashes to the wheel and cranks it until it gets bored

**Style:** flat-shaded geometry everywhere (low segment counts on spheres/toruses), pastel palette, soft shadows, fog, and damped orbit controls so the camera feels pleasant to drag. Everything is self-contained — just open the file and it runs.
