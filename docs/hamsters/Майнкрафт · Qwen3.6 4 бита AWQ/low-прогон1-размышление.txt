```javascript
function smoothstep(t) {
    return t * t * (3 - 2 * t);
}

function lerp(a, b, t) {
    return a + (b - a) * t;
}

function noise2D(x, y) {
    let ix = Math.floor(x), iy = Math.floor(y);
    let fx = x - ix, fy = y - iy;
    fx = smoothstep(fx);
    fy = smoothstep(fy);
    let a = hash(ix, iy);
    let b = hash(ix + 1, iy);
    let c = hash(ix, iy + 1);
    let d = hash(ix + 1, iy + 1);
    return lerp(lerp(a, b, fx), lerp(c, d, fx), fy);
}
```

```javascript
function fractalNoise2D(x, y) {
    let val = 0, amp = 1, freq = 1, totalAmp = 0;
    for (let i = 0; i < 4; i++) {
        val += noise2D(x * freq, y * freq) * amp;
        totalAmp += amp;
        amp *= 0.5;
        freq *= 2;
    }
    return val / totalAmp;
}
```

// === NOISE FUNCTIONS ===
function hash(x, y) {
    let h = (x * 374761393 + y * 668265263) | 0;
    h = ((h ^ (h >> 13)) * 1274126177) | 0;
    h = (h ^ (h >> 16)) | 0;
    return (h & 0x7fffffff) / 0x7fffffff;
}

function hash3(x, y, z) {
    let h = (x * 374761393 + y * 668265263 + z * 254431351) | 0;
    h = ((h ^ (h >> 13)) * 1274126177) | 0;
    h = (h ^ (h >> 16)) | 0;
    return (h & 0x7fffffff) / 0x7fffffff;
}

function smoothstep(t) {
    return t * t * (3 - 2 * t);
}

function lerp(a, b, t) {
    return a + (b - a) * t;
}

function noise2D(x, y) {
    let ix = Math.floor(x), iy = Math.floor(y);
    let fx = x - ix, fy = y - iy;
    fx = smoothstep(fx);
    fy = smoothstep(fy);
    let a = hash(ix, iy);
    let b = hash(ix + 1, iy);
    let c = hash(ix, iy + 1);
    let d = hash(ix + 1, iy + 1);
    return lerp(lerp(a, b, fx), lerp(c, d, fx), fy);
}

function noise3D(x, y, z) {
    let ix = Math.floor(x), iy = Math.floor(y), iz = Math.floor(z);
    let fx = x - ix, fy = y - iy, fz = z - iz;
    fx = smoothstep(fx);
    fy = smoothstep(fy);
    fz = smoothstep(fz);
    let a00 = lerp(hash(ix, iy, iz), hash(ix+1, iy, iz), fx);
    let a01 = lerp(hash(ix, iy+1, iz), hash(ix+1, iy+1, iz), fx);
    let a10 = lerp(hash(ix, iy, iz+1), hash(ix+1, iy, iz+1), fx);
    let a11 = lerp(hash(ix, iy+1, iz+1), hash(ix+1, iy+1, iz+1), fx);
    let ab0 = lerp(a00, a01, fy);
    let ab1 = lerp(a10, a11, fy);
    return lerp(ab0, ab1, fz);
}

function fractalNoise2D(x, y) {
    let val = 0, amp = 1, freq = 1, totalAmp = 0;
    for (let i = 0; i < 4; i++) {
        val += noise2D(x * freq, y * freq) * amp;
        totalAmp += amp;
        amp *= 0.5;
        freq *= 2;
    }
    return val / totalAmp;
}

function fractalNoise3D(x, y, z) {
    let val = 0, amp = 1, freq = 1, totalAmp = 0;
    for (let i = 0; i < 4; i++) {
        val += noise3D(x * freq, y * freq, z * freq) * amp;
        totalAmp += amp;
        amp *= 0.5;
        freq *= 2;
    }
    return val / totalAmp;
}
