```javascript
function fractalNoise2d(x, y, octaves) {
  let total = 0, amplitude = 1, frequency = 1, maxAmplitude = 0;
  for (let i = 0; i < octaves; i++) {
    total += noise2d(x * frequency, y * frequency) * amplitude;
    maxAmplitude += amplitude;
    amplitude *= 0.5;
    frequency *= 2;
  }
  return total / maxAmplitude;
}
```

```javascript
function updateChunks() {
  let pcx = Math.floor(player.x / 16);
  let pcz = Math.floor(player.z / 16);
  
  // Generate data for chunks within 5 chunks
  let toGenerate = [];
  for (let dx = -5; dx <= 5; dx++) {
    for (let dz = -5; dz <= 5; dz++) {
      let cx = pcx + dx, cz = pcz + dz;
      let key = getChunkKey(cx, cz);
      if (!chunks.has(key)) {
        toGenerate.push({cx, cz, key});
      }
    }
  }
  // Sort by distance (optional, for better visual loading)
  toGenerate.sort((a, b) => {
    let da = a.cx * a.cx + a.cz * a.cz;
    let db = b.cx * b.cx + b.cz * b.cz;
    return da - db;
  });
  // Generate up to 4 per frame
  for (let i = 0; i < Math.min(4, toGenerate.length); i++) {
    let {cx, cz, key} = toGenerate[i];
    chunks.set(key, {data: generateChunkData(cx, cz), mesh: null});
  }
  
  // Build meshes for chunks within 4 chunks with all neighbors
  let toMesh = [];
  for (let dx = -4; dx <= 4; dx++) {
    for (let dz = -4; dz <= 4; dz++) {
      let cx = pcx + dx, cz = pcz + dz;
      let key = getChunkKey(cx, cz);
      let chunk = chunks.get(key);
      if (chunk && !chunk.mesh) {
        // Check if all 4 neighbors have data
        if (chunks.has(getChunkKey(cx-1, cz)) &&
            chunks.has(getChunkKey(cx+1, cz)) &&
            chunks.has(getChunkKey(cx, cz-1)) &&
            chunks.has(getChunkKey(cx, cz+1))) {
          toMesh.push({cx, cz, key, chunk});
        }
      }
    }
  }
  toMesh.sort((a, b) => {
    let da = a.cx * a.cx + a.cz * a.cz;
    let db = b.cx * b.cx + b.cz * b.cz;
    return da - db;
  });
  for (let i = 0; i < Math.min(2, toMesh.length); i++) {
    let {key, chunk} = toMesh[i];
    chunk.mesh = buildMesh(chunk.cx, chunk.cz, chunk.data);
    if (chunk.mesh) scene.add(chunk.mesh);
    chunkMeshes.push(chunk.mesh);
  }
  
  // Remove chunks farther than 7 chunks
  let toRemove = [];
  for (let [key, chunk] of chunks) {
    let dx = chunk.cx - pcx, dz = chunk.cz - pcz;
    if (Math.abs(dx) > 7 || Math.abs(dz) > 7) {
      toRemove.push({key, chunk});
```

```javascript
function generateChunkData(cx, cz) {
  let data = new Uint8Array(20480); // 16*16*80
  
  // First pass: terrain
  for (let lx = 0; lx < 16; lx++) {
    for (let lz = 0; lz < 16; lz++) {
      let wx = cx * 16 + lx, wz = cz * 16 + lz;
      let m = fractalNoise2d(wx * 0.004, wz * 0.004, 4);
      let h = fractalNoise2d(wx * 0.02, wz * 0.02, 4);
      let H = Math.floor(5 + m * m * 58 + h * 10);
      
      for (let y = 0; y < 80; y++) {
        let block = 0;
        if (y === 0) block = 3;
        else if (y < H - 3) block = 3;
        else if (y < H) {
          if (H <= 16) block = 4;
          else if (H >= 37) block = 3;
          else block = 2;
        }
        else if (y === H) {
          if (H >= 46) block = 7;
          else if (H >= 37) block = 3;
          else if (H <= 16) block = 4;
          else block = 1;
        }
        
        // Caves
        if (y >= 3 && y <= H - 2 && block !== 0) {
          let cave = fractalNoise3d(wx * 0.09, y * 0.09, wz * 0.09, 3);
          if (cave > 0.67) block = 0;
        }
        
        data[y * 256 + lz * 16 + lx] = block;
      }
      
      // Trees
      if (data[H * 256 + lz * 16 + lx] === 1 && lx >= 2 && lx <= 13 && lz >= 2 && lz <= 13 && H + 8 < 80) {
        let th = hash(wx * 7 + 13, wz * 13 + 7);
        if (th < 0.02) {
          // Trunk
          for (let ty = H + 1; ty <= H + 4; ty++) {
            data[ty * 256 + lz * 16 + lx] = 5;
          }
          // Leaves
          let leafLayers = [
            {y: H + 5, size: 5},
            {y: H + 6, size: 5},
            {y: H + 7, size: 3},
            {y: H + 8, size: 1}
          ];
          for (let layer of leafLayers) {
            let half = Math.floor(layer.size / 2);
            for (let dx = -half; dx <= half; dx++) {
              for (let dz = -half; dz <= half; dz++) {
                let nx = lx + dx, nz = lz + dz;
```

```javascript
function buildMesh(cx, cz, data) {
  let positions = [], normals = [], colors = [];
  
  for (let y = 0; y < 80; y++) {
    for (let z = 0; z < 16; z++) {
      for (let x = 0; x < 16; x++) {
        let blockId = data[y * 256 + z * 16 + x];
        if (blockId === 0) continue;
        
        let wx = cx * 16 + x, wz = cz * 16 + z;
        let color = blockColors[blockId];
        
        // Check each face
        // Top face (y+1)
        if (getBlock(wx, y + 1, wz) === 0) {
          addFace(positions, normals, colors, wx, y, wz, 0, 1, 0, color, 1.0);
        }
        // Bottom face (y)
        if (getBlock(wx, y - 1, wz) === 0) {
          addFace(positions, normals, colors, wx, y, wz, 0, -1, 0, color, 0.55);
        }
        // North face (z)
        if (getBlock(wx, y, wz - 1) === 0) {
          addFace(positions, normals, colors, wx, y, wz, 0, 0, -1, color, 0.8);
        }
        // South face (z+1)
        if (getBlock(wx, y, wz + 1) === 0) {
          addFace(positions, normals, colors, wx, y, wz, 0, 0, 1, color, 0.8);
        }
        // West face (x)
        if (getBlock(wx - 1, y, wz) === 0) {
          addFace(positions, normals, colors, wx, y, wz, -1, 0, 0, color, 0.8);
        }
        // East face (x+1)
        if (getBlock(wx + 1, y, wz) === 0) {
          addFace(positions, normals, colors, wx, y, wz, 1, 0, 0, color, 0.8);
        }
      }
    }
  }
  
  if (positions.length === 0) return null;
  
  let geometry = new THREE.BufferGeometry();
  geometry.setAttribute('position', new THREE.Float32BufferAttribute(positions, 3));
  geometry.setAttribute('normal', new THREE.Float32BufferAttribute(normals, 3));
  geometry.setAttribute('color', new THREE.Float32BufferAttribute(colors, 3));
  
  let material = new THREE.MeshLambertMaterial({vertexColors: true});
  let mesh = new THREE.Mesh(geometry, material);
  mesh.chunkKey = getChunkKey(cx, cz);
  
  return mesh;
}

function addFace(positions, normals, colors, wx, wy, wz, nx, ny, nz, color, light) {
  // Add 2 triangles for the face
  // Face corners depend on the normal direction
  let corners;
  if (ny === 1) { // Top
    corners = [[wx,wy+1,wz+1],[wx+1,wy+1,wz+1],[wx+1,wy+1,wz],[wx,wy+1,wz]];
  } else if (ny === -1) { // Bottom
    corners = [[wx,wy,wz],[wx+1,wy,wz],[wx+1,wy,wz+1],[wx,wy,wz+1]];
  } else if (nz === -1) { // North
    corners = [[wx,wy+1,wz],[wx+1,wy+1,wz],[wx+1,wy,wz],[wx,wy,wz]];
  } else if (nz === 1) { // South
    corners = [[wx+1,wy,wz+1],[wx+1,wy+1,wz+1],[wx,wy+1,wz+1],[wx,wy,wz+1]];
  } else if (nx === -1) { // West
    corners = [[wx,wy+1,wz],[wx,wy+1,wz+1],[wx,wy,wz+1],[wx,wy,wz]];
  } else if (nx === 1) { // East
    corners = [[wx+1,wy,wz+1],[wx+1,wy+1,wz+1],[wx+1,wy+1,wz],[wx+1,wy,wz]];
  }
  
  // Two triangles: 0,1,2 and 0,2,3
  let tri1 = [corners[0], corners[1], corners[2]];
  let tri2 = [corners[0], corners[2], corners[3]];
  
  for (let tri of [tri1, tri2]) {
    for (let v of tri) {
      positions.push(v[0], v[1], v[2]);
      normals.push(nx, ny, nz);
      colors.push(color[0] * light, color[1] * light, color[2] * light);
    }
  }
}
```

```javascript
function hash3d(x, y, z) {
  let h = x * 374761393 + y * 668265263 + z * 2140136427;
  h = (h ^ (h >> 13)) * 1274126177;
  h = h ^ (h >> 16);
  return (h & 0x7fffffff) / 0x7fffffff;

function noise3d(x, y, z) {
  let ix = Math.floor(x), iy = Math.floor(y), iz = Math.floor(z);
  let fx = x - ix, fy = y - iy, fz = z - iz;
  let sx = smoothstep(fx), sy = smoothstep(fy), sz = smoothstep(fz);
  
  let n000 = hash3d(ix, iy, iz), n100 = hash3d(ix+1, iy, iz);
  let n010 = hash3d(ix, iy+1, iz), n110 = hash3d(ix+1, iy+1, iz);
  let n001 = hash3d(ix, iy, iz+1), n101 = hash3d(ix+1, iy, iz+1);

  let nx00 = lerp(n000, n100, sx), nx10 = lerp(n010, n110, sx);
  let nx01 = lerp(n001, n101, sx), nx11 = lerp(n011, n111, sx);
  let nxy0 = lerp(nx00, nx10, sy), nxy1 = lerp(nx01, nx11, sy);
  return lerp(nxy0, nxy1, sz);
}

function fractalNoise3d(x, y, z, octaves) {
  let total = 0, amplitude = 1, frequency = 1, maxAmplitude = 0;

    maxAmplitude += amplitude;
    amplitude *= 0.5;
    frequency *= 2;
```

```javascript
function buildMesh(cx, cz, data) {
  let positions = [], normals = [], colors = [];
  
  for (let y = 0; y < CHUNK_HEIGHT; y++) {
    for (let z = 0; z < CHUNK_SIZE; z++) {
      for (let x = 0; x < CHUNK_SIZE; x++) {
        let blockId = data[y * 256 + z * 16 + x];
        if (blockId === 0) continue;
        
        let wx = cx * CHUNK_SIZE + x, wz = cz * CHUNK_SIZE + z;
        let color = blockColors[blockId];
        
        // Helper for local or global read
        function getNeighbor(nx, ny, nz) {
          let lx = nx - cx * CHUNK_SIZE, lz = nz - cz * CHUNK_SIZE;
          if (lx >= 0 && lx < CHUNK_SIZE && lz >= 0 && lz < CHUNK_SIZE && ny >= 0 && ny < CHUNK_HEIGHT) {
            return data[ny * 256 + lz * 16 + lx];
          }
          return getBlock(nx, ny, nz);
        }
        
        // Check each face
        if (getNeighbor(wx, y + 1, wz) === 0) addFace(positions, normals, colors, wx, y, wz, 0, 1, 0, color, 1.0);
        if (getNeighbor(wx, y - 1, wz) === 0) addFace(positions, normals, colors, wx, y, wz, 0, -1, 0, color, 0.55);
        if (getNeighbor(wx, y, wz - 1) === 0) addFace(positions, normals, colors, wx, y, wz, 0, 0, -1, color, 0.8);
        if (getNeighbor(wx, y, wz + 1) === 0) addFace(positions, normals, colors, wx, y, wz, 0, 0, 1, color, 0.8);
        if (getNeighbor(wx - 1, y, wz) === 0) addFace(positions, normals, colors, wx, y, wz, -1, 0, 0, color, 0.8);
        if (getNeighbor(wx + 1, y, wz) === 0) addFace(positions, normals, colors, wx, y, wz, 1, 0, 0, color, 0.8);
      }
    }
  }
  
  if (positions.length === 0) return null;
  
  let geometry = new THREE.BufferGeometry();
  geometry.setAttribute('position', new THREE.Float32BufferAttribute(positions, 3));
  geometry.setAttribute('normal', new THREE.Float32BufferAttribute(normals, 3));
  geometry.setAttribute('color', new THREE.Float32BufferAttribute(colors, 3));
  
  return new THREE.Mesh(geometry, sharedMaterial);
}
```
