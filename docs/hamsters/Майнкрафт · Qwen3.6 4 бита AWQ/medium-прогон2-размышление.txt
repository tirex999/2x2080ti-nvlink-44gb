            else if (H <= 16) { surfaceType = SAND; subSurfaceType = SAND; }
            else { surfaceType = GRASS; subSurfaceType = DIRT; }

                } else if (y === H) {

    const chunk = chunks.get(key);
    if (!chunk || chunk.mesh) return;
    
    const ox = cx * CHUNK_SIZE;
    const oz = cz * CHUNK_SIZE;
    
    const positions = [];
    const normals = [];
    const colors = [];

        for (let lz = 0; lz < CHUNK_SIZE; lz++) {
            for (let lx = 0; lx < CHUNK_SIZE; lx++) {
                const idx = lx + lz * CHUNK_SIZE + y * CHUNK_SIZE * CHUNK_SIZE;
                const blockId = chunk.data[idx];
                if (blockId === AIR) continue;
                
                const gx = ox + lx;
                const gz = oz + lz;
                const color = BLOCK_COLORS[blockId];
                if (!color) continue;
                
                for (const face of FACES) {
                    const nx = gx + face.dir[0];
                    const ny = y + face.dir[1];
                    const nz = gz + face.dir[2];

                        const light = face.light;
                        const r = color[0] * light / 255;
                        const g = color[1] * light / 255;
                        const b = color[2] * light / 255;
                        
                        // Triangle 1: verts[0], verts[1], verts[2]
                        for (let i = 0; i < 3; i++) {
                            const v = face.verts[i];
                            positions.push(gx + v[0], y + v[1], gz + v[2]);
                            normals.push(face.normal[0], face.normal[1], face.normal[2]);
                            colors.push(r, g, b);
                        }
                        // Triangle 2: verts[1], verts[2], verts[3]
                        for (let i = 1; i < 4; i++) {
                            const v = face.verts[i];
                            positions.push(gx + v[0], y + v[1], gz + v[2]);
                            normals.push(face.normal[0], face.normal[1], face.normal[2]);
                            colors.push(r, g, b);
                        }
                    }
                }
            }
        }
    }
    
    if (positions.length === 0) {
        chunk.mesh = null;
        return;
    }
    
    const geometry = new THREE.BufferGeometry();
    geometry.setAttribute('position', new THREE.Float32BufferAttribute(positions, 3));
    geometry.setAttribute('normal', new THREE.Float32BufferAttribute(normals, 3));
    geometry.setAttribute('color', new THREE.Float32BufferAttribute(colors, 3));
    
    const material = new THREE.MeshLambertMaterial({ vertexColors: true });
    const mesh = new THREE.Mesh(geometry, material);
    scene.add(mesh);
    chunkMeshes.push(mesh);
    chunk.mesh = mesh;
}

function rebuildChunkMesh(cx, cz) {

    const chunk = chunks.get(key);

    if (idx >= 0) chunkMeshes.splice(idx, 1);
    chunk.mesh.geometry.dispose();
    chunk.mesh = null;
    
    // Build new mesh
    buildChunkMesh(cx, cz);
}

function collideAt(x, y, z) {
    const minX = Math.floor(x - 0.3), maxX = Math.floor(x + 0.3);
    const minY = Math.floor(y), maxY = Math.floor(y + 1.8);
    const minZ = Math.floor(z - 0.3), maxZ = Math.floor(z + 0.3);
    for (let bx = minX; bx <= maxX; bx++) {
        for (let by = minY; by <= maxY; by++) {
            for (let bz = minZ; bz <= maxZ; bz++) {
                if (getBlock(bx, by, bz) !== AIR) return true;
            }
        }
    }
    return false;
}

function updatePlayer(dt) {
    // Movement input
    const sinYaw = Math.sin(player.yaw);
    const cosYaw = Math.cos(player.yaw);
    let moveX = 0, moveZ = 0;
    if (keys['KeyW']) { moveX -= sinYaw; moveZ -= cosYaw; }
    if (keys['KeyS']) { moveX += sinYaw; moveZ += cosYaw; }
    if (keys['KeyA']) { moveX -= cosYaw; moveZ += sinYaw; }
    if (keys['KeyD']) { moveX += cosYaw; moveZ -= sinYaw; }
    
    // Normalize and scale
    const len = Math.sqrt(moveX * moveX + moveZ * moveZ);
    if (len > 0) {
        moveX = (moveX / len) * 5.5;
        moveZ = (moveZ / len) * 5.5;
    }
    
    // Jump
    if (keys['Space'] && player.onGround) {
        player.vy = 8.5;
        player.onGround = false;
    }
    
    // Apply gravity
    player.vy -= 25 * dt;
    
    // Calculate displacements
    const dx = moveX * dt;
    const dy = player.vy * dt;
    const dz = moveZ * dt;
    
    // Axis-separated collision
    // Move X
    if (!collideAt(player.x + dx, player.y, player.z)) {
        player.x += dx;
    } else {
        moveX = 0;
    }
    // Move Y
    player.onGround = false;
    if (!collideAt(player.x, player.y + dy, player.z)) {
        player.y += dy;
    } else {
        if (dy < 0) player.onGround = true;
        player.vy = 0;
    }
    // Move Z
    if (!collideAt(player.x, player.y, player.z + dz)) {
        player.z += dz;
    } else {
        moveZ = 0;
    }
    
    // Teleport if fallen too far
    if (player.y < -20) {
        respawnPlayer();
    }
    
    // Update camera
    camera.position.set(player.x, player.y + 1.62, player.z);
    camera.rotation.set(player.pitch, player.yaw, 0, 'YXZ');
}

function breakBlock() {
    if (!breakTarget) return;
    if (breakTarget.y === 0) return; // unbreakable
    
    const { x, y, z } = breakTarget;

    if (oldId === AIR) return;

    // Rebuild affected chunks
    const { cx, cz, lx, lz } = globalToLocal(x, z);
    rebuildChunkMesh(cx, cz);
    
    // Rebuild neighbor chunks at borders
    if (lx === 0) rebuildChunkMesh(cx - 1, cz);
    if (lx === CHUNK_SIZE - 1) rebuildChunkMesh(cx + 1, cz);
    if (lz === 0) rebuildChunkMesh(cx, cz - 1);
    if (lz === CHUNK_SIZE - 1) rebuildChunkMesh(cx, cz + 1);
}
