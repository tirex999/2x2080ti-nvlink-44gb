Main loop:
```js
const clock = new THREE.Clock();
const tmpObj = new THREE.Object3D();
const tmpV = new THREE.Vector3();
const tmpF = new THREE.Vector3();

let sunOn = true;
let frames = 0, fpsAcc = 0;

function animate() {
  requestAnimationFrame(animate);
  const dt = Math.min(clock.getDelta(), 0.05);
  const t = clock.elapsedTime;

  // Рыбки
  for (const fish of fishArray) {
    tmpF.set(0,0,0);
    // блуждание
    fish.wanderTimer -= dt;
    if (fish.wanderTimer <= 0) {
      fish.wanderTimer = rand(2, 6);
      fish.wanderDir.set(rand(-1,1), rand(-0.6,0.6), rand(-1,1)).normalize();
    }
    tmpF.addScaledVector(fish.wanderDir, 0.8);

    // избегание соседей
    for (const other of fishArray) {
      if (other === fish) continue;
      tmpV.subVectors(fish.mesh.position, other.mesh.position);
      const d = tmpV.length();
      const r = (fish.avoidanceRadius + other.avoidanceRadius) / 2;
      if (d < r && d > 0.0001) {
        tmpF.addScaledVector(tmpV, (1 - d / r) / d * 2.5);
      }
    }

    // поиск корма
    fish.targetFood = null;
    let best = 15 * 15;
    for (const food of foods) {
      const d2 = fish.mesh.position.distanceToSquared(food.mesh.position);
      if (d2 < best) { best = d2; fish.targetFood = food; }
    }
    if (fish.targetFood) {
      tmpV.subVectors(fish.targetFood.mesh.position, fish.mesh.position);
      const d = tmpV.length();
      tmpF.addScaledVector(tmpV, (d > 0.001 ? 3.2 / d : 0));
      if (d < 0.9 + fish.baseScale * 0.3) {
        // поела
        scene.remove(fish.targetFood.mesh);
        foods.splice(foods.indexOf(fish.targetFood), 1);
        const ns = Math.min(fish.mesh.scale.x * 1.05, 2.3);
        fish.mesh.scale.setScalar(ns);
        fish.targetFood = null;
        // всплеск пузырьков? small: spawn 2 mini bubbles at fish position
        spawnMiniBubbles(fish.mesh.position);
      }
    }

    // стены
    const p = fish.mesh.position;
    const M = 2.5;
    if (p.x < -HW + M) tmpF.x += (-HW + M - p.x) * 2;
    if (p.x >  HW - M) tmpF.x -= (p.x - (HW - M)) * 2;
    if (p.z < -HD + M) tmpF.z += (-HD + M - p.z) * 2;
    if (p.z >  HD - M) tmpF.z -= (p.z - (HD - M)) * 2;
    if (p.y < 2.5) tmpF.y += (2.5 - p.y) * 2.5;
    if (p.y > TANK.h - 2.5) tmpF.y -= (p.y - (TANK.h - 2.5)) * 2.5;

    fish.velocity.addScaledVector(tmpF, dt * 2.2);
    const sp = fish.speed * (0.75 + 0.25 * Math.sin(t * 0.6 + fish.phase));
    const len = fish.velocity.length();
    if (len > sp * 1.6) fish.velocity.setLength(sp * 1.6);
    else if (len < sp * 0.55 && len > 0.001) fish.velocity.setLength(sp * 0.55);

    p.addScaledVector(fish.velocity, dt);
    p.x = clamp(p.x, -HW + 1.2, HW - 1.2);
    p.y = clamp(p.y, 1.4, TANK.h - 1.4);
    p.z = clamp(p.z, -HD + 1.2, HD - 1.2);

    // ориентация
    if (fish.velocity.lengthSq() > 0.01) {
      tmpV.copy(p).add(fish.velocity);
      tmpObj.position.copy(p);
      tmpObj.lookAt(tmpV);
      fish.mesh.quaternion.slerp(tmpObj.quaternion, Math.min(1, dt * 3.5));
    }

    // анимация плавников
    const wag = Math.sin(t * fish.tailSpeed + fish.phase);
    fish.tail.rotation.z = wag * 0.5;
    fish.tail.rotation.y = Math.sin(t * fish.tailSpeed * 0.5 + fish.phase) * 0.18;
    const finFlap = Math.sin(t * fish.tailSpeed * 0.8 + fish.phase);
    fish.leftFin.rotation.x = -0.45 + finFlap * 0.3;
    fish.rightFin.rotation.x = 0.45 - finFlap * 0.3;
    fish.dorsal.rotation.z = Math.sin(t * fish.tailSpeed * 0.4 + fish.phase) * 0.1;
  }
```

One caveat: root's quaternion slerp — root has no parent transform (direct child of scene) ✓.
