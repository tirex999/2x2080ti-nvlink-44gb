function resolveCircleBox(h,c,r){
  const nx0 = c.cx - c.hx, nx1 = c.cx + c.hx, nz0 = c.cz - c.hz, nz1 = c.cz + c.hz;
  const px = clamp(h.pos.x, nx0, nx1), pz = clamp(h.pos.z, nz0, nz1);
  let dx = h.pos.x - px, dz = h.pos.z - pz;
  let d2 = dx*dx+dz*dz;
  if (d2 > r*r) return 0;
  let pen, nx, nz;
  if (d2 > 1e-9){ const d=Math.sqrt(d2); pen = r-d; nx=dx/d; nz=dz/d; }
  else { // центр внутри: выталкиваем по кратчайшей нормали
    const pxa = Math.min(h.pos.x-nx0, nx1-h.pos.x), pza = Math.min(h.pos.z-nz0, nz1-h.pos.z);
    if (pxa<pza){ pen=pxa+r; nx = (h.pos.x < c.cx? -1:1); nz=0; } else { pen=pza+r; nz=(h.pos.z<c.cz?-1:1); nx=0; }
  }
  return {pen,nx,nz};
}
