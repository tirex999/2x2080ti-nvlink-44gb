```js
function steer(h, dt) {
  // desired velocity toward goal
  const to = tmp.subVectors(h.goal, h.pos); to.y = 0;
  const d = to.length();
  if (d < h.arriveEps) { arrive }
  to.normalize();
  const desiredSpeed = h.maxSpeed * (state speed mult) * min(1, d/1.2) // slow near goal
  // accel limit
  h.speed += clamp(desiredSpeed - h.speed, -accel*dt, accel*dt)
  // yaw turn toward goal with limited rate
  h.yaw = lerpAngle(h.yaw, atan2(to.x, to.z), turnRate*dt)
  h.vel.copy(to).multiplyScalar(h.speed)
  const before = h.pos.clone()
  h.pos.addScaledVector(h.vel, dt)
  resolveCollisions(h)
  const moved = h.pos.distanceTo(before) // фактический путь
  h.stepPhase += (moved / h.stepLength) * TAU
  h.footSpeed = moved/dt  (smoothed)
}
```
