function steer(h,dt){
  let diff=((h.targetHeading-h.heading+Math.PI)%(Math.PI*2))-Math.PI;
  h.heading+=clamp(diff,-4*dt,4*dt);
  h.group.rotation.y=h.heading;

function updateHamster(h,dt,now){
  const s=dt*parseFloat(document.getElementById('speed').value);
  h.timer-=s;
  // blink
  h.blinkT-=s;if(h.blinkT<=0){h.blink=1;h.blinkT=rand(2.5,6);}
  h.blink=Math.max(0,h.blink-s*8);
  const eyeScale=1-h.blink*0.9;

  switch(h.state){
    case 'wander':{
      if(h.timer<=0){
        if(Math.random()<0.28){h.state='pause';h.timer=rand(0.8,2.4);}
        else if(now-h.lastInteract>3 && Math.random()<0.5){chooseInteract(h);}
        else{h.targetHeading=rand(0,Math.PI*2);h.timer=rand(1.5,4);}
      }
      h.speed=damp(h.speed,h.maxSpeed,3,s);
      break;
    }
    case 'pause':{
      h.speed=damp(h.speed,0,5,s);
      if(h.timer<=0){h.state='wander';h.timer=rand(1,3);h.targetHeading=h.heading+rand(-1,1);}
      break;
    }
    case 'go':{
      const dx=h.actionTarget.x-h.pos.x,dz=h.actionTarget.z-h.pos.z,dist=Math.hypot(dx,dz);
      if(dist>0.15)h.targetHeading=Math.atan2(dx,dz);
      h.speed=damp(h.speed,h.maxSpeed*1.15,3,s);
      if(dist<0.35){beginAction(h);h.state='act';}
      break;
    }
    case 'act':{
      h.actionTimer-=s;
      if(h.action==='run'){
        h.actionSpin+=s*14;h.interact.wheelState.angle=h.actionSpin;
        h.actionA=damp(h.actionA,0.6,10,s);
        h.speed=0;
      }else if(h.action==='eat'){
        h.actionA=damp(h.actionA,1,8,s);
        h.speed=0;
      }else if(h.action==='hide'){
        h.actionA=damp(h.actionA,1,8,s);
        h.speed=0;
      }else if(h.action==='spin'){
        h.actionB+=s*10;h.speed=0;
      }
      if(h.actionTimer<=0){
        h.state='wander';h.timer=rand(1,3);h.action=null;h.actionA=0;h.actionB=0;h.speed=0;
        h.lastInteract=now;
      }
      break;
    }
  }

  // integrate
  const spd=h.speed;
  h.pos.x+=Math.sin(h.heading)*spd*s;
  h.pos.z+=Math.cos(h.heading)*spd*s;
  if(h.pos.x>bounds.x){h.pos.x=bounds.x;h.targetHeading=Math.PI-h.heading;}
  if(h.pos.x<-bounds.x){h.pos.x=-bounds.x;h.targetHeading=Math.PI-h.heading;}
  if(h.pos.z>bounds.z){h.pos.z=bounds.z;h.targetHeading=-h.heading;}
  if(h.pos.z<-bounds.z){h.pos.z=-bounds.z;h.targetHeading=-h.heading;}
  steer(h,s);

  // apply transforms
  const moving=spd>0.05;
  h.bob+=s*(moving?9:2.5);
  const bobY=moving?Math.abs(Math.sin(h.bob))*0.07:Math.sin(h.bob*0.5)*0.015;
  h.group.position.set(h.pos.x,WALL+0.1+bobY,h.pos.z);

  // squash & stretch
  const sc=h.baseScale*(1+h.actionA*0.12);
  h.scale=damp(h.scale,sc,8,s);
  const sy=1+h.actionA*0.14 - (moving?Math.abs(Math.sin(h.bob))*0.03:0);
  const sxz=1-h.actionA*0.05;
  h.group.scale.set(h.scale*sxz,h.scale*sy,h.scale*sxz);

  // legs
  const amp=moving?0.7:0.02,ph=h.bob;
  h.legs[0].rotation.x=Math.sin(ph)*amp;
  h.legs[1].rotation.x=Math.sin(ph+Math.PI)*amp;
  h.legs[2].rotation.x=Math.sin(ph+Math.PI*0.5)*amp;
  h.legs[3].rotation.x=Math.sin(ph+Math.PI*1.5)*amp;

  // eyes
  h.group.traverse(o=>{}); // no-op guard
  // set eye scale via stored refs is simpler: recompute by scaling head children is complex, so we nudge head
  headEyes(h,eyeScale);
  if(moving && Math.random()<s*1.5) h.group.rotation.z=Math.sin(h.bob)*0.05; else h.group.rotation.z=damp(h.group.rotation.z,0,6,s);
  if(h.action==='spin')h.group.rotation.y=h.heading+h.actionB;
}
