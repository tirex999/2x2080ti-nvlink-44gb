function seek(h, target, speed, dt, arrive=0.25){
  const dx=target.x-h.pos.x, dz=target.z-h.pos.z;
  const dist=Math.hypot(dx,dz);
  if(dist<arrive) return true;
  const want=Math.atan2(dx,dz);
  const d=angleTo(h.yaw,want);
  h.yaw+=d*Math.min(1,dt*7);
  // only move if roughly aligned

  h.moving=true; h.turn=d;
  return false;
}

function seek(h,target,dt,speedMul=1){
  const dx=target.x-h.pos.x,dz=target.z-h.pos.z;
  const dist=Math.hypot(dx,dz);
  if(dist<0.28)return true;
  const want=Math.atan2(dx,dz);
  const d=angDiff(h.yaw,want);
  h.yaw+=d*Math.min(1,dt*7);
  const align=Math.max(0,Math.cos(d));
  const sp=h.cfg.speed*speedMul*Math.min(1,dist/0.6)*dt;
  h.pos.x+=Math.sin(h.yaw)*sp*align;
  h.pos.z+=Math.cos(h.yaw)*sp*align;
  h.moving=align>0.3;h.turn=d;
  return false;
}
function faceTo(h,tx,tz,dt){
  const d=angDiff(h.yaw,Math.atan2(tx-h.pos.x,tz-h.pos.z));
  h.yaw+=d*Math.min(1,dt*6);
}
function contain(h){
  const r=Math.hypot(h.pos.x,h.pos.z);
  if(r>WALK_R){h.pos.x*=WALK_R/r;h.pos.z*=WALK_R/r;}
}
