function makeSideFinGeo(span, chord, camber){
  const geo = new THREE.PlaneGeometry(1, 1, 8, 4);
  const p = geo.attributes.position;
  for(let i=0;i<p.count;i++){
    const u = p.getX(i)+0.5;   // 0..1 outward param
    const v = p.getY(i)+0.5;   // 0..1 chord param
    const taper = 1 - u*0.55;
    const x = u*span;
    const z = -0.12 - v*chord*taper - 0.35*u*u;   // swept back
    const y = Math.sin(u*Math.PI)*camber - v*0.12*taper;
    p.setXYZ(i, x, y, z);
  }
  geo.computeVertexNormals();
  return geo;
}
