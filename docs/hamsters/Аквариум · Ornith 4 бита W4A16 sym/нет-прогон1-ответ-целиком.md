function init() {
  // --- Сцена ---
  scene = new THREE.Scene();

  const grd = cx.createLinearGradient(0, 0, 0, 16);
  grd.addColorStop(0, '#0a2a52');
  grd.addColorStop(0.5, '#082040');
  grd.addColorStop(1, '#040e20');
  cx.fillStyle = grd; cx.fillRect(0, 0, 2, 16);
  scene.background = new THREE.CanvasTexture(c);
  scene.fog = new THREE.FogExp2(0x0a2a4a, 0.012);

  // --- Камера ---

  camera.position.set(0, 6, 52);

function createTank() {
  const glassMat = new THREE.MeshPhysicalMaterial({
    color: 0xbfe9ff, transmission: 0.95, transparent: true,
    opacity: 1, roughness: 0.05, metalness: 0, ior: 1.33,
    thickness: 0.6, side: THREE.DoubleSide, depthWrite: false
  });
  const g = new THREE.BoxGeometry(TANK.w, TANK.h, TANK.d);
  const tank = new THREE.Mesh(g, glassMat);
  tank.material.renderDepth = false;
  scene.add(tank);

  // Wireframe edges
  const edges = new THREE.LineSegments(
    new THREE.EdgesGeometry(g),
    new THREE.LineBasicMaterial({ color: 0x88ccff, transparent: true, opacity: 0.5 })
  );
  scene.add(edges);

  // Свечение рамки (тонкие боксы по рёбрам)
  const frameMat = new THREE.MeshBasicMaterial({ color: 0x5599cc, transparent: true, opacity: 0.25 });
  const tw = TANK.w, th = TANK.h, td = TANK.d, t = 0.25;
  const barData = [
    [tw, t, t, 0, th/2, td/2], [tw, t, t, 0, th/2, -td/2],
    [t, th, t, -tw/2, 0, 0], [t, th, t, tw/2, 0, 0],
    [t, th, t, 0, 0, td/2], [t, th, t, 0, 0, -td/2]
  ];
  barData.forEach(d => {
    const m = new THREE.Mesh(new THREE.BoxGeometry(d[0], d[1], d[2]), frameMat);
    m.position.set(d[3], d[4], d[5]); scene.add(m);
  });
}

// ---------- Песчаное дно ----------
function createSand() {
  const geo = new THREE.PlaneGeometry(TANK.w - 1, TANK.d - 1, 48, 32);
  geo.rotateX(-Math.PI / 2);
  const pos = geo.attributes.position;
  for (let i = 0; i < pos.count; i++) {
    const x = pos.getX(i), z = pos.getZ(i);
    const h = Math.sin(x * 0.4) * Math.cos(z * 0.35) * 0.35
            + Math.sin(x * 1.3 + z * 0.9) * 0.15
            + (Math.random() - 0.5) * 0.2;
    pos.setY(i, h - 10.2);
  }
  geo.computeVertexNormals();
  const mat = new THREE.MeshStandardMaterial({ color: 0xdcb873, roughness: 1, metalness: 0 });
  const sand = new THREE.Mesh(geo, mat);
  sand.receiveShadow = true;
  sand.position.y = 0;
  scene.add(sand);
}

// ---------- Декоративные камни ----------
function createRocks() {
  const rockMat = new THREE.MeshStandardMaterial({ color: 0x7a7268, roughness: 0.9, metalness: 0.05, flatShading: true });
  for (let i = 0; i < 8; i++) {
    const geo = new THREE.DodecahedronGeometry(1.1 + Math.random() * 1.4, 1);
    const p = geo.attributes.position;
    for (let j = 0; j < p.count; j++) {
      const f = 0.72 + Math.random() * 0.5;
      p.setXYZ(j, p.getX(j) * f, p.getY(j) * f * 0.7, p.getZ(j) * f);
    }
    geo.computeVertexNormals();
    const rock = new THREE.Mesh(geo, rockMat.clone());
    rock.material.color.setHSL(0.08, 0.15, 0.32 + Math.random() * 0.12);
    const scale = 0.9 + Math.random() * 1.3;
    rock.scale.set(scale, scale * 0.7, scale);
    rock.position.set((Math.random() - 0.5) * 28, 0.2, (Math.random() - 0.5) * 13);
    rock.rotation.set(Math.random() * Math.PI, Math.random() * Math.PI, Math.random() * Math.PI);
    rock.castShadow = true; rock.receiveShadow = true;
    scene.add(rock);
  }
}

// ---------- Водоросли ----------
function createSeaweed() {
  const seaweed = [];
  const colors = [0x2d8a4f, 0x1e6b3a, 0x3da35d, 0x267d43, 0x4caf50, 0x1b5e20];
  for (let k = 0; k < 12; k++) {
    const grp = new THREE.Group();
    const baseCol = colors[Math.floor(Math.random() * colors.length)];
    const stalks = 3 + Math.floor(Math.random() * 3);
    const hgt = 4 + Math.random() * 5;
    for (let s = 0; s < stalks; s++) {
      const pts = [];
      const segs = 8;
      const ox = (Math.random() - 0.5) * 1.2, oz = (Math.random() - 0.5) * 1.2;
      for (let i = 0; i <= segs; i++) {
        const t = i / segs;
        pts.push(new THREE.Vector3(
          ox + Math.sin(t * 2) * 1.0,
          t * hgt,
          oz + Math.cos(t * 1.5) * 0.7
        ));
      }
      const curve = new THREE.CatmullRomCurve3(pts);
      const tubeGeo = new THREE.TubeGeometry(curve, 14, 0.16 + Math.random() * 0.12, 6, false);
      const tubeMat = new THREE.MeshStandardMaterial({ color: baseCol, roughness: 0.7, metalness: 0, side: THREE.DoubleSide });
      const tube = new THREE.Mesh(tubeGeo, tubeMat);
      tube.position.set((Math.random() - 0.5) * 3, 0, (Math.random() - 0.5) * 3);
      tube.castShadow = false;
      grp.add(tube);
    }
    grp.position.set((Math.random() - 0.5) * 30, 0.2, (Math.random() - 0.5) * 15);
    grp.userData.swayPhase = Math.random() * Math.PI * 2;
    grp.userData.swaySpeed = 0.6 + Math.random() * 0.6;
    scene.add(grp);
    seaweed.push(grp);
  }
  window._seaweed = seaweed;
}

function createBubbles(n) {
  const mat = new THREE.MeshPhysicalMaterial({
    color: 0xffffff, transmission: 0.95, transparent: true,
    roughness: 0.1, metalness: 0, ior: 1.1, thickness: 0.3,
    emissive: 0x2266aa, emissiveIntensity: 0.15
  });
  for (let i = 0; i < n; i++) addBubble(mat);
}
function addBubble(mat) {
  const r = 0.15 + Math.random() * 0.4;
  const b = new THREE.Mesh(new THREE.SphereGeometry(r, 12, 12), mat.clone());
  b.position.set((Math.random() - 0.5) * (TANK.w - 4), Math.random() * TANK.h * 0.9, (Math.random() - 0.5) * (TANK.d - 4));
  b.userData = { vy: 0.6 + Math.random() * 1.2, phase: Math.random() * Math.PI * 2, amp: 0.3 + Math.random() * 0.6, r: r };
  scene.add(b);
  bubbleArray.push(b);
  updateStats();
}

const FISH_COLORS = [
  { body: 0xff8c00, fin: 0xffb347 },   // оранжевая
  { body: 0x2196f3, fin: 0x64b5f6 },    // синяя
  { body: 0xffd600, fin: 0xf44336 },    // желто-красная

function createFish() {
  const scheme = FISH_COLORS[Math.floor(Math.random() * FISH_COLORS.length)];
  const group = new THREE.Group();
  const s = 0.6 + Math.random() * 0.6; // базовый размер (0.6–1.2)

  const bodyMat = new THREE.MeshStandardMaterial({ color: scheme.body, roughness: 0.45, metalness: 0.15 });
  const finMat = new THREE.MeshStandardMaterial({ color: scheme.fin, roughness: 0.5, transparent: true, opacity: 0.85, side: THREE.DoubleSide });

  // Тело — вытянутая сфера
  const body = new THREE.Mesh(new THREE.SphereGeometry(1, 20, 16), bodyMat);
  body.scale.set(1.7, 0.95, 0.7);
  body.castShadow = true;
  group.add(body);

  // Голова (чуть светлее)
  const headMat = new THREE.MeshStandardMaterial({ color: scheme.body, roughness: 0.4, metalness: 0.1 });
  const head = new THREE.Mesh(new THREE.SphereGeometry(0.62, 16, 12), headMat);
  head.position.set(1.35, 0, 0);
  head.scale.set(1, 0.9, 0.85);
  head.castShadow = true;
  group.add(head);

  // Глаза
  const eyeMat = new THREE.MeshStandardMaterial({ color: 0xffffff, roughness: 0.2 });
  const pupilMat = new THREE.MeshStandardMaterial({ color: 0x111111, roughness: 0.1 });
  [-0.42, 0.42].forEach(oz => {
    const eye = new THREE.Mesh(new THREE.SphereGeometry(0.16, 12, 10), eyeMat);
    eye.position.set(1.6, 0.16, oz);
    group.add(eye);
    const pupil = new THREE.Mesh(new THREE.SphereGeometry(0.09, 10, 8), pupilMat);
    pupil.position.set(1.74, 0.16, oz);
    group.add(pupil);
  });

  // Хвост
  const tailShape = new THREE.Shape();
  tailShape.moveTo(0, 0);
  tailShape.lineTo(-0.7, 0.55);
  tailShape.lineTo(-0.45, 0);
  tailShape.lineTo(-0.7, -0.55);
  tailShape.closePath();
  const tailGeo = new THREE.ShapeGeometry(tailShape, 8);
  const tail = new THREE.Mesh(tailGeo, finMat.clone());
  tail.position.set(-1.65, 0, 0);
  tail.scale.set(1.1, 1.1, 1.1);
  group.add(tail);

  // Верхний плавник
  const dorsalShape = new THREE.Shape();
  dorsalShape.moveTo(0, 0);
  dorsalShape.quadraticCurveTo(0.3, 0.7, 0.8, 0.15);
  dorsalShape.lineTo(0.6, 0);
  dorsalShape.closePath();
  const dorsalGeo = new THREE.ShapeGeometry(dorsalShape, 6);
  const dorsal = new THREE.Mesh(dorsalGeo, finMat.clone());
  dorsal.position.set(-0.1, 0.75, 0);
  dorsal.rotation.z = 0.15;
  group.add(dorsal);

  // Боковые плавники
  const finShape = new THREE.Shape();
  finShape.moveTo(0, 0);
  finShape.quadraticCurveTo(-0.4, -0.5, -0.8, -0.2);
  finShape.lineTo(-0.3, 0.1);
  finShape.closePath();
  const finGeo = new THREE.ShapeGeometry(finShape, 5);
  const leftFin = new THREE.Mesh(finGeo, finMat.clone());
  leftFin.position.set(0.1, -0.35, 0.5);
  leftFin.rotation.x = -0.3;
  group.add(leftFin);
  const rightFin = new THREE.Mesh(finGeo, finMat.clone());
  rightFin.position.set(0.1, -0.35, -0.5);
  rightFin.rotation.x = 0.3;
  group.add(rightFin);

  group.scale.setScalar(s);

  // Позиция и параметры
  group.position.set((Math.random() - 0.5) * 24, (Math.random() - 0.5) * 14, (Math.random() - 0.5) * 12);
  scene.add(group);

  const speed = 2.5 + Math.random() * 3.0;
  const dir = new THREE.Vector3(Math.random() - 0.5, (Math.random() - 0.5) * 0.4, Math.random() - 0.5).normalize();
  const avoidanceRadius = 3.5 + s * 1.5;

  fishArray.push({
    mesh: group, tail, leftFin, rightFin, dorsal,
    velocity: dir.multiplyScalar(speed),
    speed, baseScale: s, tailSpeed: 6 + Math.random() * 6,
    phase: Math.random() * Math.PI * 2, targetFood: null,
    avoidanceRadius, growth: 0, wanderTimer: Math.random() * 5
  });
  updateStats();
}

function createFood(pos) {
  const geo = new THREE.SphereGeometry(0.22, 8, 6);
  const mat = new THREE.MeshStandardMaterial({ color: 0x8d5a2b, roughness: 0.8 });
  const pellet = new THREE.Mesh(geo, mat);
  pellet.position.copy(pos);
  pellet.castShadow = true;
  pellet.userData = { vy: 0, y: pos.y };
  scene.add(pellet);
  foodArray.push(pellet);
  updateStats();
}

const raycaster = new THREE.Raycaster();
const ndc = new THREE.Vector2();
const groundPlane = new THREE.Plane(new THREE.Vector3(0, 1, 0), 0);

function onCanvasClick(event) {
  // не кормить, если перетащили камеру
  if (event.shiftKey) return;
  ndc.x = (event.clientX / window.innerWidth) * 2 - 1;
  ndc.y = -(event.clientY / window.innerHeight) * 2 + 1;
  raycaster.setFromCamera(ndc, camera);

  const camDir = new THREE.Vector3();
  camera.getWorldDirection(camDir);
  const plane = new THREE.Plane();
  plane.setFromNormalAndCoplanarPoint(camDir.clone().negate(), new THREE.Vector3(0, 8, 0));

  const target = new THREE.Vector3();
  if (!raycaster.ray.intersectPlane(plane, target)) return;

  // ограничим внутри аквариума
  target.x = THREE.MathUtils.clamp(target.x, -TANK.w / 2 + 2, TANK.w / 2 - 2);
  target.z = THREE.MathUtils.clamp(target.z, -TANK.d / 2 + 2, TANK.d / 2 - 2);
  target.y = THREE.MathUtils.clamp(target.y, 3, TANK.h - 2);

  // несколько гранул
  for (let i = 0; i < 3; i++) {
    createFood(new THREE.Vector3(
      target.x + (Math.random() - 0.5) * 1.5,
      target.y,
      target.z + (Math.random() - 0.5) * 1.5
    ));
  }
}

// ---------- Обновление рыбок ----------
function updateFish(dt, time) {
  const n = fishArray.length;

  // Поиск ближайшего корма
  for (let i = 0; i < n; i++) {
    const f = fishArray[i];
    let best = null, bestD = 15 * 15; // радиус 15
    for (let j = 0; j < foodArray.length; j++) {
      const d2 = f.mesh.distanceToSquared(foodArray[j]);
      if (d2 < bestD) { bestD = d2; best = foodArray[j]; }
    }
    f.targetFood = best;
  }

  for (let i = 0; i < n; i++) {
    const f = fishArray[i];
    const acc = new THREE.Vector3();
    const mesh = f.mesh;

    // 1) Преследование корма
    if (f.targetFood) {
      const to = new THREE.Vector3().subVectors(f.targetFood.position, mesh.position);
      const d = to.length();
      if (d > 0.8) {
        acc.add(to.normalize().multiplyScalar(f.speed * 1.8));
      }
      // поедание
      if (d < 1.1 * f.baseScale) {
        scene.remove(f.targetFood);
        foodArray.splice(foodArray.indexOf(f.targetFood), 1);
        f.growth += 0.05; // +5%
        updateStats();
        f.targetFood = null;
      }
    }

    // 2) Избегание других рыбок
    for (let j = 0; j < n; j++) {
      if (i === j) continue;
      const other = fishArray[j];
      const diff = new THREE.Vector3().subVectors(mesh.position, other.mesh.position);
      const dist = diff.length();
      if (dist < f.avoidanceRadius && dist > 0.001) {
        const push = diff.multiplyScalar((f.avoidanceRadius - dist) / f.avoidanceRadius);
        acc.add(push.multiplyScalar(2.0));
      }
    }

    // 3) Отталкивание от стен
    const p = mesh.position;
    const margin = 2;
    if (p.x >  BOUNDS.x - margin) acc.x -= (p.x - (BOUNDS.x - margin)) * 3;
    if (p.x < -BOUNDS.x + margin) acc.x += (-BOUNDS.x + margin - p.x) * 3;
    if (p.y >  BOUNDS.y - margin) acc.y -= (p.y - (BOUNDS.y - margin)) * 3;
    if (p.y < -BOUNDS.y + margin) acc.y += (-BOUNDS.y + margin - p.y) * 3;
    if (p.z >  BOUNDS.z - margin) acc.z -= (p.z - (BOUNDS.z - margin)) * 3;
    if (p.z < -BOUNDS.z + margin) acc.z += (-BOUNDS.z + margin - p.z) * 3;

    // 4) Случайное блуждание
    f.wanderTimer -= dt;
    if (f.wanderTimer <= 0) {
      f.wanderTimer = 2 + Math.random() * 4;
      f.velocity.x += (Math.random() - 0.5) * 2;
      f.velocity.z += (Math.random() - 0.5) * 2;
      f.velocity.y += (Math.random() - 0.5) * 0.6;
    }

    // Интеграция скорости
    f.velocity.add(acc.multiplyScalar(dt));

    // ограничение скорости
    const sp = f.velocity.length();
    const maxSp = f.targetFood ? f.speed * 1.9 : f.speed;
    if (sp > maxSp) f.velocity.multiplyScalar(maxSp / sp);
    if (sp < f.speed * 0.3) f.velocity.multiplyScalar((f.speed * 0.3) / (sp || 1));

    // обновление позиции
    mesh.position.addScaledVector(f.velocity, dt);

    // Плавный поворот в направлении движения
    const vel = f.velocity;
    if (vel.lengthSq() > 0.01) {
      const targetAngle = Math.atan2(vel.y, Math.hypot(vel.x, vel.z));
      const yaw = Math.atan2(vel.x, vel.z);
      // вычислим целевой кватернион
      const dir = vel.clone().normalize();
      const desired = new THREE.Object3D();
      desired.position.set(0, 0, 1);
      desired.lookAt(dir.x, dir.y, dir.z);
      const q = desired.quaternion;
      mesh.quaternion.slerp(q, 1 - Math.pow(0.001, dt));
    }

    // Наклон при разворотах
    mesh.rotation.z = THREE.MathUtils.clamp(-f.velocity.y * 0.06, -0.5, 0.5);

    // --- Анимация хвоста и плавников ---
    const tailWave = Math.sin(time * f.tailSpeed + f.phase) * 0.45;
    f.tail.rotation.y = tailWave;
    f.leftFin.rotation.z = Math.sin(time * f.tailSpeed * 0.9 + f.phase + 0.5) * 0.4 - 0.2;
    f.rightFin.rotation.z = -Math.sin(time * f.tailSpeed * 0.9 + f.phase + 0.5) * 0.4 + 0.2;
    if (f.dorsal) f.dorsal.rotation.z = 0.15 + Math.sin(time * f.tailSpeed * 0.7 + f.phase) * 0.08;

    // Рост
    const ns = f.baseScale * (1 + f.growth);
    mesh.scale.setScalar(ns);
  }
}

// ---------- Обновление еды ----------
function updateFood(dt) {
  const gravity = -9.0;
  for (let i = foodArray.length - 1; i >= 0; i--) {
    const fd = foodArray[i];
    fd.userData.vy += gravity * dt;
    fd.position.y += fd.userData.vy * dt;
    // лёгкое качание
    fd.position.x += Math.sin(fd.position.y * 2 + performance.now() * 0.002) * 0.003;

    if (fd.position.y <= 0.3) {
      // достиг дна — удаляем
      scene.remove(fd);
      foodArray.splice(i, 1);
      updateStats();
    }
  }
}

// ---------- Обновление пузырей ----------
function updateBubbles(dt, time) {
  const halfW = TANK.w / 2 - 1, halfD = TANK.d / 2 - 1;
  for (let i = 0; i < bubbleArray.length; i++) {
    const b = bubbleArray[i];
    b.position.y += b.userData.vy * dt;
    b.position.x += Math.sin(time * 1.5 + b.userData.phase) * b.userData.amp * dt;
    b.position.z += Math.cos(time * 1.2 + b.userData.phase) * b.userData.amp * dt;

    if (b.position.y > TANK.h - 1) {
      // сброс вниз
      b.position.y = 0.5;
      b.position.x = (Math.random() - 0.5) * halfW * 2;
      b.position.z = (Math.random() - 0.5) * halfD * 2;
    }
  }
}

// ---------- Анимация водорослей ----------
function updateSeaweed(time) {
  const sw = window._seaweed;
  if (!sw) return;
  for (let g = 0; g < sw.length; g++) {
    const grp = sw[g];
    const sway = Math.sin(time * grp.userData.swaySpeed + grp.userData.swayPhase) * 0.06;
    grp.rotation.x = sway;
    grp.rotation.z = Math.cos(time * grp.userData.swaySpeed * 0.8 + grp.userData.swayPhase) * 0.05;
  }
}

// ---------- Цикл анимации ----------
let fpsFrames = 0, fpsTime = 0, currentFps = 0;
function animate() {
  requestAnimationFrame(animate);
  const dt = Math.min(clock.getDelta(), 0.05);
  const time = clock.elapsedTime;

  updateFish(dt, time);
  updateFood(dt);
  updateBubbles(dt, time);
  updateSeaweed(time);

  // пульсация подводных огней
  pointL1.intensity = 0.6 + Math.sin(time * 1.5) * 0.15;
  pointL2.intensity = 0.5 + Math.sin(time * 1.1 + 1) * 0.15;

  controls.update();
  renderer.render(scene, camera);

  // FPS
  fpsFrames++; fpsTime += dt;
  if (fpsTime >= 0.5) { currentFps = Math.round(fpsFrames / fpsTime); fpsFrames = 0; fpsTime = 0; document.getElementById('fpsCount').textContent = currentFps; }
}
