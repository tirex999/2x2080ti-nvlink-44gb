Value noise 2D:
```js
function noise2(x, y) {
  const xi = Math.floor(x), yi = Math.floor(y);
  const xf = x - xi, yf = y - yi;
  const u = xf*xf*(3-2*xf), v = yf*yf*(3-2*yf);
  const a = hash3(xi, yi, 0);
  const b = hash3(xi+1, yi, 0);
  const c = hash3(xi, yi+1, 0);
  const d = hash3(xi+1, yi+1, 0);
  return a + (b-a)*u + (c-a)*v + (a - b - c + d)*u*v;
}
```

```js
function fbm2(x, y) {

  for (let i = 0; i < 4; i++) {
    sum += amp * noise2(x*freq, y*freq);
    norm += amp;
    amp *= 0.5;
    freq *= 2;
  }
  return sum / norm;
}
```

3D value noise:
```js
function noise3(x, y, z) {
  const xi = Math.floor(x), yi = Math.floor(y), zi = Math.floor(z);
  const xf = x-xi, yf=y-yi, zf=z-zi;
  const u = xf*xf*(3-2*xf), v = yf*yf*(3-2*yf), w = zf*zf*(3-2*zf);
  // 8 corners
  const c000 = hash3(xi,yi,zi), c100 = hash3(xi+1,yi,zi), c010 = hash3(xi,yi+1,zi), c110 = hash3(xi+1,yi+1,zi);
  const c001 = hash3(xi,yi,zi+1), c101 = hash3(xi+1,yi,zi+1), c011 = hash3(xi,yi+1,zi+1), c111 = hash3(xi+1,yi+1,zi+1);
  const x00 = c000 + (c100-c000)*u;
  const x10 = c010 + (c110-c010)*u;
  const x01 = c001 + (c101-c001)*u;
  const x11 = c011 + (c111-c011)*u;
  const y0 = x00 + (x10-x00)*v;
  const y1 = x01 + (x11-x01)*v;
  return y0 + (y1-y0)*w;
}
```

```js
function updateWorld() {
  const pcx = Math.floor(playerPos.x / 16), pcz = Math.floor(playerPos.z / 16);
  
  // Generate: within radius 5, max 4 per frame, sort by distance
  const genQueue = [];
  for (let dx = -5; dx <= 5; dx++)
    for (let dz = -5; dz <= 5; dz++) {
      if (dx*dx + dz*dz > 25) continue;
      const key = (pcx+dx)+','+(pcz+dz);
      if (!chunks.has(key)) genQueue.push([pcx+dx, pcz+dz, dx*dx+dz*dz]);
    }
  genQueue.sort((a,b) => a[2]-b[2]);
  for (let i = 0; i < Math.min(4, genQueue.length); i++) {
    const [cx, cz] = genQueue[i];
    const data = generateChunkData(cx, cz);
    chunks.set(cx+','+cz, { cx, cz, data, mesh: null });
  }
  
  // Mesh: within radius 4, neighbors have data, max 2 per frame
  const meshQueue = [];
  for (let dx = -4; dx <= 4; dx++)
    for (let dz = -4; dz <= 4; dz++) {
      if (dx*dx + dz*dz > 16) continue;
      const cx = pcx+dx, cz = pcz+dz;
      const key = cx+','+cz;
      const ch = chunks.get(key);
      if (!ch || ch.mesh) continue;
      if (chunks.has((cx+1)+','+cz) && chunks.has((cx-1)+','+cz) &&
          chunks.has(cx+','+(cz+1)) && chunks.has(cx+','+(cz-1))) {
        meshQueue.push([cx, cz, dx*dx+dz*dz]);
      }
    }
  meshQueue.sort((a,b)=>a[2]-b[2]);
  for (let i = 0; i < Math.min(2, meshQueue.length); i++) {
    buildChunkMesh(chunks.get(meshQueue[i][0]+','+meshQueue[i][1]));
  }
  
  // Unload: beyond radius 7
  for (const [key, ch] of chunks) {
    const dx = ch.cx - pcx, dz = ch.cz - pcz;
    if (dx*dx + dz*dz > 49) {
      if (ch.mesh) {
        scene.remove(ch.mesh);
        const idx = meshList.indexOf(ch.mesh);
        if (idx>=0) meshList.splice(idx,1);
        ch.mesh.geometry.dispose();
      }
      chunks.delete(key);
    }
  }
}
```

```js
function generateChunkData(cx, cz) {
  const data = new Uint8Array(16*16*80);
  const baseX = cx*16, baseZ = cz*16;
  const heights = new Int16Array(256);
  for (let lx = 0; lx < 16; lx++) {
    for (let lz = 0; lz < 16; lz++) {
      const wx = baseX + lx, wz = baseZ + lz;
      const m = fbm2(wx*0.004, wz*0.004);
      const h = fbm2(wx*0.02, wz*0.02);
      let H = Math.floor(5 + m*m*58 + h*10);
      H = Math.max(2, Math.min(79, H));
      heights[lz*16+lx] = H;
      const beach = H <= 16, peak = H >= 37;
      const sub = beach ? 4 : (peak ? 3 : 2);
      const surf = beach ? 4 : (H >= 46 ? 7 : (peak ? 3 : 1));
      data[(lz<<4)|lx] = 3; // y=0
      for (let y = 1; y <= H; y++) {
        let id;
        if (y === H) id = surf;
        else if (y > H-3) id = sub;
        else id = 3;
        if (y >= 3 && y <= H-2 && fbm3(wx*0.09, y*0.09, wz*0.09) > 0.67) id = 0;
        data[(y<<8)|(lz<<4)|lx] = id;
      }
    }
  }
  // trees
  for (let lx = 2; lx <= 13; lx++) {
    for (let lz = 2; lz <= 13; lz++) {
      const wx = baseX+lx, wz = baseZ+lz;
      const H = heights[lz*16+lx];
      const sIdx = (H<<8)|(lz<<4)|lx;
      if (data[sIdx] !== 1) continue; // grass only
      if (hash3(wx, 0, wz) >= 0.02) continue;
      if (H + 7 > 79) continue;
      // trunk
      for (let y = H+1; y <= H+4; y++) data[(y<<8)|(lz<<4)|lx] = 5;
      // leaves
      for (let dy = 4; dy <= 5; dy++) {
        for (let dx = -2; dx <= 2; dx++) for (let dz = -2; dz <= 2; dz++) {
          const i = ((H+dy)<<8)|((lz+dz)<<4)|(lx+dx);
          if (data[i] === 0) data[i] = 6;
        }
      }
      for (let dx = -1; dx <= 1; dx++) for (let dz = -1; dz <= 1; dz++) {
        const i = ((H+6)<<8)|((lz+dz)<<4)|(lx+dx);
        if (data[i] === 0) data[i] = 6;
      }
      const iT = ((H+7)<<8)|(lz<<4)|lx;
      if (data[iT] === 0) data[iT] = 6;
    }
  }
  return data;
}
```

```js
function moveAxis(axis, dist) {
  const STEP = 0.1;
  let rem = dist;
  while (rem > 1e-9 || rem < -1e-9) {
    const d = Math.abs(rem) > STEP ? Math.sign(rem)*STEP : rem;
    playerPos[axis] += d;
    if (collides(playerPos.x, playerPos.y, playerPos.z)) {
      playerPos[axis] -= d;
      return false;
    }
    rem -= d;
  }
  return true;
}
```

```js
function buildChunkMesh(ch){
  const pos=[], nor=[], col=[];
  const bx = ch.cx*CHUNK, bz = ch.cz*CHUNK;
  for (let y=0;y<HEIGHT;y++){
    for (let lz=0;lz<CHUNK;lz++){

        if (!id) continue;
        const wx = bx+lx, wz = bz+lz;
        const rgb = BLOCK_RGB[id];
        for (let f=0;f<6;f++){
          const F = FACES[f];
          if (getBlock(wx+F.nx, y+F.ny, wz+F.nz) !== 0) continue;
          const r = rgb.r*F.b, g = rgb.g*F.b, b = rgb.b*F.b;
          for (let v=0; v<6; v++){
            const c = F.c[VORDER[v]];
            pos.push(wx+c[0], y+c[1], wz+c[2]);
            nor.push(F.nx, F.ny, F.nz);
            col.push(r,g,b);
          }
        }
      }
    }
  }
  const geo = new THREE.BufferGeometry();
  geo.setAttribute("position", new THREE.Float32BufferAttribute(pos,3));
  geo.setAttribute("normal",   new THREE.Float32BufferAttribute(nor,3));
  geo.setAttribute("color",    new THREE.Float32BufferAttribute(col,3));
```

```js
function rebuildChunk(cx,cz){
```

```js
function findSpawn(){
  for (let ox=0; ox<4; ox++) for (let oz=0; oz<4; oz++){
    const x = 8+ox, z = 8+oz;
    let ty = 0;
    for (let y=HEIGHT-1; y>=0; y--){

    }
    const py = ty+1;
    if (!collides(x, py, z)) return new THREE.Vector3(x, py, z);
  }
  return new THREE.Vector3(8, HEIGHT, 8);
}
```

```js
function updateWorld(){
  const pcx = Math.floor(playerPos.x/CHUNK), pcz = Math.floor(playerPos.z/CHUNK);
  // 1) generate
  const gen = [];
  for (let dx=-GEN_RADIUS; dx<=GEN_RADIUS; dx++)
    for (let dz=-GEN_RADIUS; dz<=GEN_RADIUS; dz++){
      if (dx*dx+dz*dz > GEN_RADIUS*GEN_RADIUS) continue;
      const cx=pcx+dx, cz=pcz+dz, k=cx+","+cz;
      if (!chunks.has(k)) gen.push([cx,cz,dx*dx+dz*dz]);
    }
  gen.sort((a,b)=>a[2]-b[2]);
  for (let i=0;i<MAX_GEN_PER_FRAME && i<gen.length;i++){
    const [cx,cz]=gen[i];
    chunks.set(cx+","+cz, {cx, cz, data: generateChunkData(cx,cz), mesh:null});
  }
  // 2) mesh
  const mes = [];
  for (let dx=-MESH_RADIUS; dx<=MESH_RADIUS; dx++)
    for (let dz=-MESH_RADIUS; dz<=MESH_RADIUS; dz++){
      if (dx*dx+dz*dz > MESH_RADIUS*MESH_RADIUS) continue;
      const cx=pcx+dx, cz=pcz+dz, ch=chunks.get(cx+","+cz);
      if (!ch || ch.mesh) continue;
      if (chunks.has((cx+1)+","+cz) && chunks.has((cx-1)+","+cz) &&
          chunks.has(cx+","+(cz+1)) && chunks.has(cx+","+(cz-1)))
        mes.push([cx,cz,dx*dx+dz*dz]);
    }
  mes.sort((a,b)=>a[2]-b[2]);
  for (let i=0;i<MAX_MESH_PER_FRAME && i<mes.length;i++){
    buildChunkMesh(chunks.get(mes[i][0]+","+mes[i][1]));
  }
  // 3) unload
  for (const [k,ch] of chunks){
    const dx=ch.cx-pcx, dz=ch.cz-pcz;
    if (dx*dx+dz*dz > UNLOAD_RADIUS*UNLOAD_RADIUS){
      if (ch.mesh){
        scene.remove(ch.mesh);
        const i = meshList.indexOf(ch.mesh);
        if (i>=0) meshList.splice(i,1);
        ch.mesh.geometry.dispose();
      }
      chunks.delete(k);
    }
  }
}
```
