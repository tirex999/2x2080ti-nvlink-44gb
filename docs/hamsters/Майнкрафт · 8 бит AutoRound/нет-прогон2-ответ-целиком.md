  function hashInt(x, y) {
    var h = Math.imul(x | 0, 374761393) ^ Math.imul(y | 0, 668265263);
    h = Math.imul(h ^ (h >>> 13), 1274126177);
    h ^= h >>> 16;
    return (h >>> 0) / 4294967296;
  }
  function hash3(x, y, z) {
    var h = Math.imul(x | 0, 374761393) ^ Math.imul(y | 0, 668265263) ^ Math.imul(z | 0, 2246822519);
    h = Math.imul(h ^ (h >>> 13), 1274126177);
    h ^= h >>> 16;
    return (h >>> 0) / 4294967296;

  function smooth(t) { return t * t * (3 - 2 * t); }
  function noise2(x, y) {
    var xi = Math.floor(x), yi = Math.floor(y);
    var xf = x - xi, yf = y - yi;
    var u = smooth(xf), v = smooth(yf);
    var a = hashInt(xi, yi), b = hashInt(xi + 1, yi);
    var c = hashInt(xi, yi + 1), d = hashInt(xi + 1, yi + 1);
    return a + (b - a) * u + (c - a) * v + (a - b - c + d) * u * v;
  }
  function noise3(x, y, z) {
    var xi = Math.floor(x), yi = Math.floor(y), zi = Math.floor(z);
    var xf = x - xi, yf = y - yi, zf = z - zi;
    var u = smooth(xf), v = smooth(yf), w = smooth(zf);
    var c000 = hash3(xi, yi, zi),     c100 = hash3(xi + 1, yi, zi);
    var c010 = hash3(xi, yi + 1, zi), c110 = hash3(xi + 1, yi + 1, zi);
    var c001 = hash3(xi, yi, zi + 1), c101 = hash3(xi + 1, yi, zi + 1);
    var c011 = hash3(xi, yi + 1, zi + 1), c111 = hash3(xi + 1, yi + 1, zi + 1);
    var x00 = c000 + (c100 - c000) * u, x10 = c010 + (c110 - c010) * u;
    var x01 = c001 + (c101 - c001) * u, x11 = c011 + (c111 - c011) * u;
    var y0 = x00 + (x10 - x00) * v, y1 = x01 + (x11 - x01) * v;
    return y0 + (y1 - y0) * w;
  }
  function fractal2(x, y) {
    var f = 0, amp = 0.5, freq = 1, norm = 0;
    for (var i = 0; i < 4; i++) {
      f += noise2(x * freq, y * freq) * amp;
      norm += amp; amp *= 0.5; freq *= 2;
    }
    return f / norm;
  }
  function fractal3(x, y, z) {
    var f = 0, amp = 0.5, freq = 1, norm = 0;
    for (var i = 0; i < 4; i++) {
      f += noise3(x * freq, y * freq, z * freq) * amp;
      norm += amp; amp *= 0.5; freq *= 2;
    }
    return f / norm;
  }

    var lx = wx - cx * CHUNK, lz = wz - cz * CHUNK;

  }

  }

  // ============================ Terrain generation ============================
  function genChunk(cx, cz) {
    var data = new Uint8Array(CHUNK * CHUNK * HEIGHT);
    var base = cx * CHUNK, basez = cz * CHUNK;
    var heights = new Int16Array(CHUNK * CHUNK);

    for (var lx = 0; lx < CHUNK; lx++) {
      for (var lz = 0; lz < CHUNK; lz++) {
        var wx = base + lx, wz = basez + lz;
        var m = fractal2(wx * 0.004, wz * 0.004);
        var h = fractal2(wx * 0.02, wz * 0.02);
        var H = Math.floor(5 + m * m * 58 + h * 10);
        if (H < 1) H = 1;
        if (H > HEIGHT - 1) H = HEIGHT - 1;
        heights[lx * 16 + lz] = H;

        var sub, surf;
        if (H <= 16) { sub = 4; surf = 4; }
        else if (H >= 46) { sub = 3; surf = 7; }
        else if (H >= 37) { sub = 3; surf = 3; }
        else { sub = 2; surf = 1; }

        for (var y = 0; y <= H; y++) {
          var id;
          if (y === 0) id = 3;                       // bedrock-ish stone
          else if (y < H - 3) id = 3;                // stone body
          else if (y < H) id = sub;                  // 3 sub-surface layers
          else id = surf;                            // surface
          data[y * 256 + lx * 16 + lz] = id;
        }
        // Caves
        for (var cy = 3; cy <= H - 2; cy++) {
          if (noise3(wx * 0.09, cy * 0.09, wz * 0.09) > 0.67)
            data[cy * 256 + lx * 16 + lz] = 0;
        }
      }
    }

    // Trees (trunk fully inside this chunk)
    for (var tx = 0; tx < CHUNK; tx++) {
      for (var tz = 0; tz < CHUNK; tz++) {
        var gwx = base + tx, gwz = basez + tz;
        if (hashInt(gwx, gwz) >= 0.02) continue;
        var gy = heights[tx * 16 + tz];
        if (gy >= HEIGHT - 7) continue;
        if (data[gy * 256 + tx * 16 + tz] !== 1) continue; // needs grass
        for (var ty = 1; ty <= 4; ty++)
          data[(gy + ty) * 256 + tx * 16 + tz] = 5;       // trunk
        var ly = gy + 4;
        for (var dx = -2; dx <= 2; dx++)
          for (var dz = -2; dz <= 2; dz++)
            for (var dy = 0; dy < 2; dy++) {
              var yy = ly + dy;
              if (yy >= HEIGHT) continue;
              var ix = tx + dx, iz = tz + dz;
              if (ix < 0 || ix >= CHUNK || iz < 0 || iz >= CHUNK) continue;
              if (data[yy * 256 + ix * 16 + iz] === 0) data[yy * 256 + ix * 16 + iz] = 6;
            }
        for (var dx2 = -1; dx2 <= 1; dx2++)
          for (var dz2 = -1; dz2 <= 1; dz2++) {
            var yy2 = ly + 2;
            if (yy2 >= HEIGHT) continue;
            var ix2 = tx + dx2, iz2 = tz + dz2;
            if (ix2 < 0 || ix2 >= CHUNK || iz2 < 0 || iz2 >= CHUNK) continue;
            if (data[yy2 * 256 + ix2 * 16 + iz2] === 0) data[yy2 * 256 + ix2 * 16 + iz2] = 6;
          }
        var ty2 = ly + 3;
        if (ty2 < HEIGHT && data[ty2 * 256 + tx * 16 + tz] === 0)
          data[ty2 * 256 + tx * 16 + tz] = 6;
      }
    }
    return data;
  }

  function buildChunkMesh(c) {

    for (var lx = 0; lx < CHUNK; lx++) {

  // ============================ Collision ============================
  function collideAxis(axis, amount) {
    if (amount === 0) return;
    player[axis] += amount;
    var minX = player.x - PH, maxX = player.x + PH;
    var minY = player.y, maxY = player.y + PHH;
    var minZ = player.z - PH, maxZ = player.z + PH;
    var bx0 = Math.floor(minX), bx1 = Math.floor(maxX);
    var by0 = Math.floor(minY), by1 = Math.floor(maxY);
    var bz0 = Math.floor(minZ), bz1 = Math.floor(maxZ);
    for (var bx = bx0; bx <= bx1; bx++)
      for (var by = by0; by <= by1; by++)
        for (var bz = bz0; bz <= bz1; bz++) {
          if (getBlock(bx, by, bz) === 0) continue;
          if (minX < bx + 1 && maxX > bx && minY < by + 1 && maxY > by && minZ < bz + 1 && maxZ > bz) {
            if (axis === "x") player.x -= amount;
            else if (axis === "y") { player.y -= amount; player.vy = 0; if (amount < 0) player.onGround = true; }
            else player.z -= amount;
            return;
          }
        }
  }
  function updatePlayer(dt) {
    var f = (keys["w"] ? 1 : 0) - (keys["s"] ? 1 : 0);
    var r = (keys["d"] ? 1 : 0) - (keys["a"] ? 1 : 0);
    var len = Math.hypot(f, r);
    if (len > 0) { f /= len; r /= len; }
    var sy = Math.sin(player.yaw), cy = Math.cos(player.yaw);
    var mx = (-sy * f + cy * r) * SPEED;
    var mz = (-cy * f - sy * r) * SPEED;
    player.vx = mx; player.vz = mz;
    player.vy -= GRAVITY * dt;
    if (player.vy < -60) player.vy = -60;
    if (keys[" "] && player.onGround) { player.vy = JUMP; player.onGround = false; }
    player.onGround = false;
    collideAxis("y", player.vy * dt);
    collideAxis("x", player.vx * dt);
    collideAxis("z", player.vz * dt);
    if (player.y < -20) {
      player.x = SPAWN_X; player.z = SPAWN_Z;
      player.y = computeSpawnY();
      player.vx = player.vy = player.vz = 0;
    }
  }

      var p = hit.point, n = hit.face.normal;
      var bx = Math.floor(p.x - n.x * 0.5), by = Math.floor(p.y - n.y * 0.5), bz = Math.floor(p.z - n.z * 0.5);
      target = { x: bx, y: by, z: bz, px: Math.floor(p.x + n.x * 0.5), py: Math.floor(p.y + n.y * 0.5), pz: Math.floor(p.z + n.z * 0.5) };
      outline.visible = true;
      outline.position.set(bx + 0.5, by + 0.5, bz + 0.5);
    } else {
      target = null;
      outline.visible = false;
    }
  }
  function overlapsPlayer(bx, by, bz) {
    return bx + 1 > player.x - PH && bx < player.x + PH &&
           by + 1 > player.y && by < player.y + PHH &&
           bz + 1 > player.z - PH && bz < player.z + PH;
  }
  function doBreak() {
    if (!target || target.y === 0) return;
    setBlock(target.x, target.y, target.z, 0);
    var lx = target.x - Math.floor(target.x / CHUNK) * CHUNK;
    var lz = target.z - Math.floor(target.z / CHUNK) * CHUNK;
    rebuildChunk(Math.floor(target.x / CHUNK), Math.floor(target.z / CHUNK));
    if (lx === 0) rebuildChunk(Math.floor((target.x - 1) / CHUNK), Math.floor(target.z / CHUNK));
    if (lx === 15) rebuildChunk(Math.floor((target.x + 1) / CHUNK), Math.floor(target.z / CHUNK));
    if (lz === 0) rebuildChunk(Math.floor(target.x / CHUNK), Math.floor((target.z - 1) / CHUNK));
    if (lz === 15) rebuildChunk(Math.floor(target.x / CHUNK), Math.floor((target.z + 1) / CHUNK));
  }
  function doPlace() {
    if (!target) return;
    var bx = target.px, by = target.py, bz = target.pz;
    if (by < 0 || by >= HEIGHT) return;
    if (getBlock(bx, by, bz) !== 0) return;
    if (overlapsPlayer(bx, by, bz)) return;
    setBlock(bx, by, bz, selected + 1);
    var lx = bx - Math.floor(bx / CHUNK) * CHUNK;
    var lz = bz - Math.floor(bz / CHUNK) * CHUNK;
    rebuildChunk(Math.floor(bx / CHUNK), Math.floor(bz / CHUNK));
    if (lx === 0) rebuildChunk(Math.floor((bx - 1) / CHUNK), Math.floor(bz / CHUNK));
    if (lx === 15) rebuildChunk(Math.floor((bx + 1) / CHUNK), Math.floor(bz / CHUNK));
    if (lz === 0) rebuildChunk(Math.floor(bx / CHUNK), Math.floor((bz - 1) / CHUNK));
    if (lz === 15) rebuildChunk(Math.floor(bx / CHUNK), Math.floor((bz + 1) / CHUNK));
  }

  // ============================ World streaming ============================
  function streamWorld() {
    var pcx = Math.floor(player.x / CHUNK), pcz = Math.floor(player.z / CHUNK);

    // Generate data (radius 5, max 4/frame)
    var gen = 0;
    for (var dx = -5; dx <= 5 && gen < 4; dx++)
      for (var dz = -5; dz <= 5 && gen < 4; dz++) {
        var cx = pcx + dx, cz = pcz + dz;
        if (!chunks.has(chunkKey(cx, cz))) {
          chunks.set(chunkKey(cx, cz), { cx: cx, cz: cz, data: genChunk(cx, cz), mesh: null, dirty: true });
          gen++;
        }
      }

    // Build meshes (radius 4, neighbors present, max 2/frame)
    var built = 0;
    for (var gx = -4; gx <= 4 && built < 2; gx++)
      for (var gz = -4; gz <= 4 && built < 2; gz++) {
        var ccx = pcx + gx, ccz = pcz + gz;
        var c = chunks.get(chunkKey(ccx, ccz));
        if (!c || c.mesh) continue;
        if (hasData(ccx - 1, ccz) && hasData(ccx + 1, ccz) && hasData(ccx, ccz - 1) && hasData(ccx, ccz + 1)) {
          buildChunkMesh(c);
          built++;
        }
      }

    // Rebuild dirty chunks near player (up to 2/frame)
    var rebuilt = 0;
    for (var rx = -4; rx <= 4 && rebuilt < 2; rx++)
      for (var rz = -4; rz <= 4 && rebuilt < 2; rz++) {
        var rcx = pcx + rx, rcz = pcz + rz;
        var rc = chunks.get(chunkKey(rcx, rcz));
        if (rc && rc.dirty && rc.mesh) { rebuildChunk(rcx, rcz); rebuilt++; }
      }

    // Remove far chunks (radius > 7)
    var toDel = [];
    chunks.forEach(function (c, k) {
      if (Math.abs(c.cx - pcx) > 7 || Math.abs(c.cz - pcz) > 7) toDel.push(k);
    });
    for (var i = 0; i < toDel.length; i++) {
      var dc = chunks.get(toDel[i]);
      if (dc.mesh) {
        scene.remove(dc.mesh);
        var mi = chunkMeshes.indexOf(dc.mesh);
        if (mi >= 0) chunkMeshes.splice(mi, 1);
        dc.mesh.geometry.dispose();
        dc.mesh = null;
      }
      chunks.delete(toDel[i]);
    }
  }

  // ============================ Input ============================
  var overlay = document.getElementById("overlay");
  function setLocked(v) {
    locked = v;
    document.body.classList.toggle("playing", v);
    overlay.classList.toggle("hidden", v);

    if (!locked) return;

  document.addEventListener("keydown", function (e) {

  // ============================ Hotbar UI ============================
  var hotbar = document.getElementById("hotbar");

  for (var si = 0; si < 7; si++) {
    (function (i) {

  function selectSlot(i) {
    selected = i;
    for (var j = 0; j < slots.length; j++) slots[j].classList.toggle("selected", j === i);
  }
  selectSlot(0);

  // ============================ Main loop ============================
  var last = performance.now();
  function animate(now) {
    requestAnimationFrame(animate);
    var dt = (now - last) / 1000; last = now;
    if (dt > 0.05) dt = 0.05;

    streamWorld();
    updatePlayer(dt);

    camera.position.set(player.x, player.y + EYE, player.z);
    camera.rotation.set(player.pitch, player.yaw, 0);

    // Water follows player
    water.position.set(Math.floor(player.x), WATER_Y, Math.floor(player.z));

    // Clouds drift + wrap
    for (var ci = 0; ci < clouds.length; ci++) {
      var cl = clouds[ci];
      cl.position.x += cl.userData.v * dt;
      if (cl.position.x - player.x > 180) cl.position.x -= 360;
      else if (cl.position.x - player.x < -180) cl.position.x += 360;
      if (cl.position.z - player.z > 180) cl.position.z -= 360;
      else if (cl.position.z - player.z < -180) cl.position.z += 360;
    }

    if (locked) updateTarget();

    renderer.render(scene, camera);
  }
  requestAnimationFrame(animate);
})();
