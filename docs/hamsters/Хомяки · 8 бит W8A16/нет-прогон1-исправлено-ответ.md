const HALF_W=CAGE.w/2-0.9, HALF_D=CAGE.d/2-0.9;
function updateHamster(h,dt,t){
  h.blinkTimer-=dt;
  if(h.blinkTimer<=0){h.blink=0.14;h.blinkTimer=rand(2.5,6);}
  if(h.blink>0)h.blink-=dt;
  const eyeScale=h.blink>0?0.12:1;
  h.eyes.forEach(e=>e.scale.y=lerp(e.scale.y,eyeScale,0.5));

  h.scalePulse=damp(h.scalePulse,1,8,dt);
  if(h.pet>0)h.pet-=dt;

  h.timer-=dt;
  if(h.state==='wander' && h.timer<=0){
    const r=Math.random();
    if(r<0.34){h.state='go';h.target=pick(interactables);h.timer=rand(3,5);h.speed=h.maxSpeed*1.25;}
    else if(r<0.5){h.state='pause';h.timer=rand(0.8,2.2);h.speed=0;}
    else {h.state='wander';h.timer=rand(1.5,3.5);h.heading=rand(0,Math.PI*2);h.speed=h.maxSpeed*rand(0.4,0.8);}
  }

  let desiredSpeed=h.speed;
  if(h.state==='go'&&h.target){
    const tp=h.target.pos;
    const dx=tp.x-h.pos.x, dz=tp.z-h.pos.z;
    const dist=Math.hypot(dx,dz);
    h.targetHeading=Math.atan2(dx,dz);
    if(dist<h.target.radius){
      if(h.target.type==='wheel'){h.state='wheel';h.timer=rand(3,5);WHEEL.speed=Math.max(WHEEL.speed,6);}
      else if(h.target.type==='tunnel'){h.state='tunnel';h.timer=rand(2.5,4);}
      else if(h.target.type==='bowl'){h.state='eat';h.timer=rand(2.5,4);doBite();}
    } else {
      h.speed=h.maxSpeed*1.3;
    }
  }
  if(h.state==='wheel'){
    const a=Math.atan2(h.pos.z-WHEEL.pos.z,h.pos.x-WHEEL.pos.x);
    const na=a+dt*1.4;
    const rr=WHEEL.radius+0.35;
    h.pos.x=lerp(h.pos.x,WHEEL.pos.x+Math.cos(na)*rr,0.15);
    h.pos.z=lerp(h.pos.z,WHEEL.pos.z+Math.sin(na)*rr,0.15);
    h.heading=na+Math.PI/2;
    h.speed=h.maxSpeed*1.6;
    WHEEL.speed=Math.max(WHEEL.speed,6);
    if(h.timer<=0){h.state='wander';h.timer=rand(1.5,3);h.speed=h.maxSpeed*0.5;}
  }
  if(h.state==='tunnel'){
    const tx=TUNNEL.pos.x+Math.sin(t*3)*0.9;
    h.pos.x=lerp(h.pos.x,tx,0.1);h.pos.z=lerp(h.pos.z,TUNNEL.pos.z,0.1);
    h.heading=tx>h.pos.x?0:Math.PI;
    h.speed=h.maxSpeed*0.9;
    if(h.timer<=0){h.state='wander';h.timer=rand(1.5,3);h.speed=h.maxSpeed*0.5;}
  }
  if(h.state==='eat'){
    h.pos.x=lerp(h.pos.x,BOWL.pos.x+0.9,0.12);
    h.pos.z=lerp(h.pos.z,BOWL.pos.z,0.12);
    h.heading=Math.atan2(BOWL.pos.x-h.pos.x,BOWL.pos.z-h.pos.z);
    h.speed=0;
    if(h.timer<=0){h.state='wander';h.timer=rand(1.5,3);h.speed=h.maxSpeed*0.5;}
  }

  if(h.state!=='wheel'&&h.state!=='tunnel'&&h.state!=='eat'){
    if(h.state==='wander'||h.state==='go') h.heading=dampAngle(h.heading,h.targetHeading||h.heading,4,dt);
    h.pos.x+=Math.sin(h.heading)*desiredSpeed*dt;
    h.pos.z+=Math.cos(h.heading)*desiredSpeed*dt;
    if(h.pos.x>HALF_W){h.pos.x=HALF_W;h.heading=Math.PI-h.heading;}
    if(h.pos.x<-HALF_W){h.pos.x=-HALF_W;h.heading=Math.PI-h.heading;}
    if(h.pos.z>HALF_D){h.pos.z=HALF_D;h.heading=-h.heading;}
    if(h.pos.z<-HALF_D){h.pos.z=-HALF_D;h.heading=-h.heading;}
  }
  h.pos.y=FLOOR_Y;

  h.group.rotation.y=h.heading;
  const moving=desiredSpeed>0.05||h.state==='wheel'||h.state==='tunnel';
  const bobAmt=moving?0.06:0.02;
  h.bob+=dt*(moving?10:3);
  h.group.position.y=FLOOR_Y+Math.abs(Math.sin(h.bob))*bobAmt;
  const sy=h.scalePulse*(h.state==='eat'?1+Math.sin(t*10)*0.06:1);
  h.group.scale.set(1/sy, sy*(h.pet>0?1+Math.sin(t*18)*0.05:1), 1/sy);
  if(h.pet>0)h.group.rotation.z=Math.sin(t*18)*0.12;else h.group.rotation.z=damp(h.group.rotation.z,0,8,dt);
}
