      const dx = h.target.x - p.x, dz = h.target.z - p.z;
      const dist = Math.hypot(dx, dz);
      if(dist < 0.55){
        if(h.wheel && h.target === wheelSpot){ h.state='atWheel'; h.timer=rand(4,8); }
        else { h.state='pause'; h.timer=rand(1,3); }
      } else {
        const ta = Math.atan2(dx, dz);
        let da = ta - h.angle;
        while(da > Math.PI) da -= Math.PI*2;
        while(da < -Math.PI) da += Math.PI*2;
        h.angle += da * Math.min(1, dt*6);
        p.x += Math.sin(h.angle) * h.speed * dt;
        p.z += Math.cos(h.angle) * h.speed * dt;
      }
    } else if(h.state === 'atWheel'){
      anyAtWheel = true;
      h.timer -= dt;
      const p = h.g.position;
      // face the wheel and wobble
      const ta = Math.atan2(wx - p.x, wz - p.z);
      let da = ta - h.angle;
      while(da > Math.PI) da -= Math.PI*2;
      while(da < -Math.PI) da += Math.PI*2;
      h.angle += da * Math.min(1, dt*4);
      p.x = wheelSpot.x + Math.sin(t*3 + h.phase) * 0.15;
      p.z = wheelSpot.z + Math.cos(t*2.2 + h.phase) * 0.15;
      if(h.timer <= 0){
        h.state = 'walk';
        h.target.set(rand(-5.6,5.6), 0, rand(-3.6,3.6));
      }
    }
    // clamp
    const p = h.g.position;
    p.x = THREE.MathUtils.clamp(p.x, -BOUNDS.x, BOUNDS.x);
    p.z = THREE.MathUtils.clamp(p.z, -BOUNDS.z, BOUNDS.z);
    // bob
    const moving = h.state !== 'pause';
    p.y = GROUND + (moving ? 0.035*Math.abs(Math.sin(t*11 + h.phase)) : 0);
    h.g.rotation.y = h.angle;
  }
  // wheel spin
  const target = anyAtWheel ? 5.5 : 0.7;
  wheelSpeed += (target - wheelSpeed) * Math.min(1, dt*1.5);
  wheelSpin += wheelSpeed * dt;
  spinGroup.rotation.z = wheelSpin;
}
