function steer(h, dt){
  // repulsion
  let px = 0, pz = 0;
  for(let i=0;i<OBSTACLES.length;i++){
    if(i === h.ignoreObs) continue;
    const o = OBSTACLES[i];
    const dx = h.pos.x - o.x, dz = h.pos.z - o.z;
    const d = Math.hypot(dx,dz);
    const min = o.r + 0.35;
    if(d < min && d > 0.0001){
      const f = (1 - d/min) * 2.2;
      px += (dx/d)*f; pz += (dz/d)*f;
    }
  }
  const dirX = Math.sin(h.yaw), dirZ = Math.cos(h.yaw);
  let vx = dirX*h.speed + px, vz = dirZ*h.speed + pz;
  h.pos.x = THREE.MathUtils.clamp(h.pos.x + vx*dt, BOUNDS.minX, BOUNDS.maxX);
  h.pos.z = THREE.MathUtils.clamp(h.pos.z + BOUNDS... );
}

function updateHamster(h, dt, t){
  h.modeT += dt;
  let moving = false;
  switch(h.mode){
    case 'walk': {
      const dx = h.target.x - h.pos.x, dz = h.target.z - h.pos.z;
      const d = Math.hypot(dx,dz);
      if(d < 0.22){ h.decide(dt); break; }
      const desired = Math.atan2(dx,dz);
      h.yaw = lerpAngle(h.yaw, desired, 1-Math.exp(-5*dt));
      h.phase += dt*7;
      moving = true;
      break;
    }
    case 'nap': {
      h.squash = lerp(h.squash, 0.62, 1-Math.exp(-4*dt));
      h.eyeClose = lerp(h.eyeClose, 1, 1-Math.exp(-8*dt));
      if(h.modeT > h.dur) h.decide(dt);
      break;
    }
    case 'groom': {
      h.wiggle = 1;
      if(h.modeT > h.dur) h.decide(dt);
      break;
    }
    case 'sniff': {
      h.sniffAmt = 1;
      if(h.modeT > h.dur) h.decide(dt);
      break;
    }
    case 'eat': {
      if(h.sub === 'go'){
        const dx = EAT_SPOT.x - h.pos.x, dz = EAT_SPOT.z - h.pos.z;
        const d = Math.hypot(dx,dz);
        if(d < 0.2){
          h.sub = 'munch'; h.modeT = 0;
          h.cheekPuff = 1;
        } else {
          h.yaw = lerpAngle(h.yaw, Math.atan2(dx,dz), 1-Math.exp(-5*dt));
          h.phase += dt*7; moving = true;
        }
      } else { // munch
        h.cheekPuff = 1;
        h.munchPhase = (h.munchPhase||0) + dt;
        const dip = Math.max(0, Math.sin((h.munchPhase)*5.5));
        h.dip = dip;
        h.chompT = (h.chompT||0) + dt;
        if(h.chompT > 0.5){
          h.chompT = 0;
          if(bowl.count > 0) bowl.removeSeed();
        }
        if(h.modeT > h.dur){
          h.cheekPuff = 0; h.dip = 0;
          h.decide(dt);
        }
      }
      break;
    }
    case 'wheel': {
      if(h.sub === 'go'){
        const dx = WHEEL_SPOT.x - h.pos.x, dz = WHEEL_SPOT.z - h.pos.z;
        const d = Math.hypot(dx,dz);
