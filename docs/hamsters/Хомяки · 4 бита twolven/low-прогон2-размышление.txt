function mat(color, opts = {}) {
  return new THREE.MeshStandardMaterial({
    color,
    roughness: opts.roughness ?? 0.7,
    metalness: opts.metalness ?? 0.05,
    ...opts
  });

// --- Build Wheel ---
function buildWheel() {
  const group = new THREE.Group();
  const wheelMat = mat(0x44cc88, { metalness: 0.1, roughness: 0.5 });
  const spokeMat = mat(0x33aa77, { metalness: 0.1 });

  // Outer ring
  const ringGeo = new THREE.TorusGeometry(0.7, 0.06, 6, 12);
  const ring = new THREE.Mesh(ringGeo, wheelMat);
  ring.castShadow = true;
  group.add(ring);

  // Inner hub
  const hubGeo = new THREE.CylinderGeometry(0.1, 0.1, 0.15, 6);
  const hub = new THREE.Mesh(hubGeo, spokeMat);
  hub.rotation.x = Math.PI / 2;
  group.add(hub);

  // Spokes
  for (let i = 0; i < 5; i++) {
    const angle = (i / 5) * Math.PI * 2;
    const spokeGeo = new THREE.CylinderGeometry(0.03, 0.03, 0.6, 4);
    const spoke = new THREE.Mesh(spokeGeo, spokeMat);
    spoke.position.set(Math.cos(angle) * 0.35, Math.sin(angle) * 0.35, 0);
    spoke.lookAt(0, 0, 0);
    spoke.rotateX(Math.PI / 2);
    group.add(spoke);
  }

  // Stand
  const standMat = mat(0x666666, { metalness: 0.3 });
  const standGeo = new THREE.CylinderGeometry(0.06, 0.08, 0.5, 6);
  const stand = new THREE.Mesh(standGeo, standMat);
  stand.position.y = -0.95;
  stand.castShadow = true;
  group.add(stand);

  // Base
  const baseGeo = new THREE.CylinderGeometry(0.25, 0.3, 0.08, 8);
  const base = new THREE.Mesh(baseGeo, standMat);
  base.position.y = -1.22;
  group.add(base);

  group.position.set(2, 1.2, -1.2);
  return group;
}

  const cap1 = new THREE.Mesh(capGeo, tunnelMat);
  cap1.position.set(0.6, 0.35, 0);

// --- Animation Loop ---
const clock = new THREE.Clock();
let wheelSpinSpeed = 0;

function animate() {
  requestAnimationFrame(animate);
  const dt = Math.min(clock.getDelta(), 0.05);
  const time = clock.elapsedTime;

  // Update hamsters
  hamsters.forEach((h, idx) => {
    h.stateTimer -= dt;

    // Bob animation
    h.bobPhase += dt * 8;
    const bobY = h.state === 'walking' ? Math.abs(Math.sin(h.bobPhase)) * 0.04 : 0;
    h.mesh.position.y = bobY;

    // Slight idle wobble
    if (h.state === 'paused') {
      h.mesh.rotation.z = Math.sin(time * 2 + idx) * 0.03;
    } else {
      h.mesh.rotation.z *= 0.9;
    }

    // State machine
    if (h.stateTimer <= 0) {
      // Choose next state
      const r = Math.random();
      
      // Check proximity to objects
      const distToWheel = Math.hypot(h.mesh.position.x - 2, h.mesh.position.z - (-1.2));
      const distToBowl = Math.hypot(h.mesh.position.x - (-1.5), h.mesh.position.z - 1);
      
      if (distToWheel < 0.8 && r < 0.4) {
        h.state = 'wheeling';
        h.stateTimer = 2 + Math.random() * 2;
      } else if (distToBowl < 0.7 && r < 0.5) {
        h.state = 'eating';
        h.stateTimer = 1.5 + Math.random() * 1.5;
      } else if (r < 0.3) {
        h.state = 'walking';
        h.stateTimer = 1 + Math.random() * 2;
      } else if (r < 0.5) {
        h.state = 'paused';
        h.stateTimer = 0.5 + Math.random() * 1.5;
      } else {
        h.state = 'turning';
        h.stateTimer = 0.4 + Math.random() * 0.4;
        h.targetAngle = h.angle + (Math.random() - 0.5) * Math.PI * 1.5;
      }
    }

    // Execute state
    switch (h.state) {
      case 'walking': {
        h.angle += (Math.random() - 0.5) * 0.02;
        h.mesh.position.x += Math.sin(h.angle) * h.speed * dt;
        h.mesh.position.z += Math.cos(h.angle) * h.speed * dt;
        break;
      }
      case 'turning': {
        const diff = h.targetAngle - h.angle;
        h.angle += diff * dt * 3;
        break;
      }
      case 'eating': {
        // Face the bowl
        const targetA = Math.atan2(-1.5 - h.mesh.position.x, 1 - h.mesh.position.z);
        h.angle += (targetA - h.angle) * dt * 3;
        // Nod
        h.mesh.rotation.x = Math.sin(time * 5) * 0.08;
        break;
      }
      case 'wheeling': {
        // Move in circle around wheel
        const wheelCenter = { x: 2, z: -1.2 };
        const orbitAngle = time * 2 + idx;
        const orbitRadius = 0.4;
