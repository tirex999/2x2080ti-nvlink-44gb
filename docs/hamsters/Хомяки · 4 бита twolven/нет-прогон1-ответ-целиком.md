import * as THREE from 'three';

/* ============================================================
   0. PALETTE + HELPERS

const C = {
  sun:'#ffd93d', peach:'#ffb26b', cream:'#fff0d0',
  orange:'#ff9f45', brown:'#a9743f',
  pink:'#ff9ec4', rose:'#ff7ea8',

};
const rand = (a,b)=> a + Math.random()*(b-a);
const pick = arr => arr[(Math.random()*arr.length)|0];
const clamp = THREE.MathUtils.clamp;
const lerp  = THREE.MathUtils.lerp;
// shortest-angle approach
function angleLerp(a,b,t){ let d=((b-a+Math.PI*3)%(Math.PI*2))-Math.PI; return a+d*t; }

function mat(color, opts={}){
  return new THREE.MeshStandardMaterial({
    color:new THREE.Color(color),
    roughness: opts.rough ?? 0.72,
    metalness: opts.metal ?? 0.04,
    flatShading: opts.flat ?? true,
    emissive: opts.emissive ? new THREE.Color(opts.emissive) : new THREE.Color(0x000000),
    emissiveIntensity: opts.ei ?? 0,
    side: opts.side ?? THREE.FrontSide
  });
}
function box(w,h,d,m,x=0,y=0,z=0){ const o=new THREE.Mesh(new THREE.BoxGeometry(w,h,d),m); o.position.set(x,y,z); return o; }
function cyl(rt,rb,h,m,seg=14,open=false){ const g=new THREE.CylinderGeometry(rt,rb,h,seg,1,open); return new THREE.Mesh(g,m); }
function sph(r,m,ws=10,hs=8){ return new THREE.Mesh(new THREE.SphereGeometry(r,ws,hs),m); }

renderer.outputColorSpace = THREE.SRGBColorSpace;
renderer.toneMapping = THREE.ACESFilmicToneMapping;

const scene = new THREE.Scene();

  const m = axis==='x'? cyl(r,r,len,metal,8) : axis==='y'? cyl(r,r,len,metal,8) : cyl(r,r,len,metal,8);
  if(axis==='z') m.rotation.x=Math.PI/2;
  if(axis==='y') m.rotation.z=Math.PI/2;
  m.position.copy(pos); cage.add(m); return m;
}
const fT = FLOOR_Y+WALL_H; // frame top

[[-W/2,-D/2],[W/2,-D/2],[-W/2,D/2],[W/2,D/2]].forEach(([x,z])=>{

function buildHamster(o){
  const g = new THREE.Group();
  const fur = mat(o.fur,{rough:.85});
  const belly = mat(o.belly,{rough:.9});
  const earM = mat(o.ear,{rough:.85});
  const noseM = mat(o.nose,{rough:.7});
  const eyeM = mat(C.ink,{rough:.35,flat:false});
  const shine = mat('#ffffff',{rough:.2,flat:false});
  shine.transparent=true; shine.opacity=.9;

  // body
  const body = sph(0.30, fur, 14, 10);
  body.scale.set(1.12, 0.92, 1.28);
  body.position.y = 0.30;
  body.castShadow = true;
  g.add(body);
  // belly patch
  const bellyM = sph(0.24, belly, 12, 9);
  bellyM.scale.set(1.0,0.82,1.18);
  bellyM.position.set(0,0.26,0.09);
  g.add(bellyM);
  // haunch
  const haunch = sph(0.20, fur, 12, 9);
  haunch.scale.set(1.1,0.95,1.0);
  haunch.position.set(0,0.26,-0.22);
  haunch.castShadow=true;
  g.add(haunch);
  // tail
  const tail = sph(0.06, belly, 8, 6);
  tail.position.set(0,0.22,-0.42);
  g.add(tail);

  // head
  const head = sph(0.21, fur, 14, 11);
  head.scale.set(1.05,0.95,1.0);
  head.position.set(0,0.52,0.24);
  head.castShadow=true;
  g.add(head);
  // cheeks
  [-1,1].forEach(s=>{
    const ch = sph(0.09, belly, 10, 8);
    ch.position.set(s*0.13,0.46,0.34);
    g.add(ch);
  });
  // ears
  [-1,1].forEach(s=>{
    const ear = sph(0.09, earM, 10, 8);
    ear.scale.set(0.8,1.0,0.5);
    ear.position.set(s*0.13,0.70,0.10);
    ear.rotation.z = s*-0.25;
    g.add(ear);
    const earIn = sph(0.05, mat(o.earIn||o.ear,{rough:.85}),8,6);
    earIn.scale.set(0.7,0.9,0.4);
    earIn.position.set(s*0.13,0.70,0.14);
    g.add(earIn);
  });
  // eyes
  [-1,1].forEach(s=>{
    const eye = sph(0.05, eyeM, 10, 8);
    eye.position.set(s*0.085,0.56,0.40);
    g.add(eye);
    const sh = sph(0.018, shine, 8, 6);
    sh.position.set(s*0.075,0.585,0.44);
    g.add(sh);
  });
  // nose
  const nose = sph(0.045, noseM, 10, 8);
  nose.scale.set(1,0.8,1);
  nose.position.set(0,0.47,0.44);
  g.add(nose);
  // mouth
  const mouth = new THREE.Mesh(new THREE.TorusGeometry(0.03,0.008,6,12,Math.PI), mat(C.ink,{rough:.5,flat:false}));
  mouth.position.set(0,0.43,0.44);
  mouth.rotation.z=Math.PI;
  g.add(mouth);
  // whiskers
  const wMat = mat('#ffffff',{rough:.6,flat:false}); wMat.transparent=true; wMat.opacity=.75;
  [-1,1].forEach(s=>{
    for(let i=0;i<3;i++){
      const w = new THREE.Mesh(new THREE.CylinderGeometry(0.004,0.002,0.16,4), wMat);
      w.position.set(s*0.09, 0.45-i*0.01, 0.42);
      w.rotation.z = s*(0.5+i*0.12);
      w.rotation.y = s*0.2;
      g.add(w);
    }
  });

  // legs (pivot at hip)
  function leg(x,z){
    const pivot = new THREE.Group();
    pivot.position.set(x,0.24,z);
    const foot = cyl(0.055,0.05,0.24, fur, 8);
    foot.position.y = -0.12;
    foot.castShadow=true;
    pivot.add(foot);
    const pad = sph(0.05, belly, 8, 6);
    pad.position.set(0,-0.24,0.02);
    pivot.add(pad);
    g.add(pivot);
    return pivot;
  }
  const legs = {
    fl: leg( 0.13, 0.16), fr: leg(-0.13, 0.16),
    bl: leg( 0.13,-0.16), br: leg(-0.13,-0.16)
  };

  g.userData = {
    legs, body, head,
    phase: rand(0,Math.PI*2),
    blink: rand(0.5,2),
    blinkT: 0,
    lean: 0
  };
  return g;
}

function updateHamster(h, dt){
  const u = h.mesh.userData;
  h.phase += dt*3;
  h.bob   += dt*6;

  // blink
  u.blinkT -= dt;
  if(u.blinkT<=0){
    u.blink = 0.12;
    u.blinkT = rand(1.5,4.5);
  }
  u.blink = Math.max(0, u.blink-dt*1.5);
  const eyeScaleY = 1 - u.blink*8;
  // apply blink by scaling eye meshes (find them by material color)
  // simpler: store references when building. We'll just scale whole head slightly? No—do it via userData.eyes
  // (added below in build step)

  // ---- STATE MACHINE ----
  h.timer -= dt;
  const t = h.target ? setTargetPoint(h.target) : null;

  if(h.state==='walk'){
    if(t){
      // steer toward target
      const dx=t.x-h.pos.x, dz=t.z-h.pos.z;
      const dist=Math.hypot(dx,dz);
      const desired=Math.atan2(dx,dz);
      h.heading = angleLerp(h.heading, desired, Math.min(1,dt*2.2));
      if(dist<0.35){
        h.state='interact'; h.timer=rand(1.2,2.6);
      } else {
        h.pos.x += Math.sin(h.heading)*h.speed*dt;
        h.pos.z += Math.cos(h.heading)*h.speed*dt;
      }
    } else {
      // wander within bounds
      h.pos.x += Math.sin(h.heading)*h.speed*dt;
      h.pos.z += Math.cos(h.heading)*h.speed*dt;
      // soft bounds
      const bx=W/2-0.5, bz=D/2-0.5;
      if(h.pos.x>bx){ h.heading=angleLerp(h.heading,Math.atan2(-1,0),dt*3); }
      if(h.pos.x<-bx){ h.heading=angleLerp(h.heading,Math.atan2(1,0),dt*3); }
      if(h.pos.z>bz){ h.heading=angleLerp(h.heading,Math.atan2(0,-1),dt*3); }
      if(h.pos.z<-bz){ h.heading=angleLerp(h.heading,Math.atan2(0,1),dt*3); }
      // random turn
      if(h.timer<=0){
        h.timer=rand(1.5,3.5);
        h.heading += rand(-1.4,1.4);
      }
    }
  }
  else if(h.state==='pause'){
    if(h.timer<=0){ h.state='walk'; h.timer=rand(1.5,3); }
  }
  else if(h.state==='turn'){
    h.heading += h.pauseDir * dt * 2.2;
    h.timer -= dt;
    if(h.timer<=0){ h.state='walk'; h.timer=rand(1.5,3); }
  }
  else if(h.state==='interact'){
    // little hop / nibble in place
    h.timer -= dt;
    if(h.timer<=0){
      h.state='walk';
      h.timer=rand(1.5,3);
      // face away from target
      if(t){
        const dx=h.pos.x-t.x, dz=h.pos.z-t.z;
        h.heading=Math.atan2(dx,dz);
      }
    }
  }

  // ---- APPLY TRANSFORMS ----
  // position (with bob)
  const bobY = Math.abs(Math.sin(h.bob)) * 0.03 * (h.state==='walk'?1:0.4);
  h.mesh.position.set(h.pos.x, h.pos.y + bobY, h.pos.z);
  // heading
  h.mesh.rotation.y = h.heading;
  // lean forward when walking
  const targetLean = h.state==='walk' ? 0.12 : 0;
  u.lean = lerp(u.lean, targetLean, Math.min(1,dt*4));
  h.mesh.rotation.x = u.lean;

  // leg swing
  const swing = h.state==='walk' ? Math.sin(h.phase) * 0.55 : 0;
  u.legs.fl.rotation.x =  swing;
  u.legs.fr.rotation.x = -swing;
  u.legs.bl.rotation.x = -swing;
  u.legs.br.rotation.x =  swing;

  // body squash & stretch
  const sq = 1 + Math.sin(h.phase*2)*0.04*(h.state==='walk'?1:0.3);
  u.body.scale.y = 0.92*sq;
  u.head.position.y = 0.52 + Math.sin(h.phase)*0.015;
}

  const et = clock.elapsedTime;

    el.textContent=f;
    fpsFrames=0; fpsTimer=0;

    // wheel spin

    // plant sway
