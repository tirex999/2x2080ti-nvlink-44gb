function updateHamster(h, dt){
  const u = h.userData;
  u.bob += dt * 8;
  // bobbing while walking
  if(u.state==='walk'){
    const dir = new THREE.Vector3().subVectors(u.target, h.position);
    dir.y = 0;
    const d = dir.length();
    if(d < 0.25){ u.state='pause'; u.t = 0.5+Math.random()*1.5; }
    else {
      dir.normalize();
      // smooth turn
      const targetAngle = Math.atan2(dir.x, dir.z);
      let da = targetAngle - h.rotation.y;
      da = Math.atan2(Math.sin(da), Math.cos(da));
      h.rotation.y += da * Math.min(1, dt*4);
      h.position.addScaledVector(dir, u.speed*dt);
    }
    h.position.y = HAMSTER_Y + Math.abs(Math.sin(u.bob))*0.05;
  } else if(u.state==='pause'){
    u.t -= dt;
    h.position.y = HAMSTER_Y + Math.sin(u.bob*0.5)*0.02;
    if(u.t<=0){
      if(hamsterCanWheel(h) && Math.random()<0.5){ u.state='toWheel'; }
      else { u.state='walk'; u.target = randPoint(); }
    }
  }
  // keep inside cage circle
  const r = Math.hypot(h.position.x, h.position.z);
  if(r > 2.85 && u.state==='walk'){
    // steer toward center
    h.rotation.y = Math.atan2(-h.position.x, -h.position.z);
  }
}

```
function updateHamster(h, dt){
  const u = h.userData;
  u.bob += dt * (u.state==='running' ? 14 : u.state==='wander'||u.state==='goEat'||u.state==='toWheel' ? 9 : 3);

  switch(u.state){
    case 'idle':
      u.t -= dt;
      h.position.y = BASE_Y + Math.sin(u.bob*0.5)*0.02;
      if(u.t <= 0) nextDecision(h);
      break;
    case 'wander':
    case 'goEat':
    case 'toWheel':
      stepToward(h, u, dt);
      h.position.y = BASE_Y + Math.abs(Math.sin(u.bob))*0.06;
      break;
    case 'eating':
      u.t -= dt;
      h.position.y = BASE_Y + Math.abs(Math.sin(u.bob))*0.03;
      h.rotation.x = Math.sin(u.bob*1.5)*0.15; // nibble
      if(u.t <= 0){
        h.rotation.x = 0;
        if(u.eatSeed){ scene.remove(u.eatSeed); u.eatSeed.dispose?.(); u.eatSeed = null; }
        u.state = 'idle'; u.t = 0.5 + Math.random();
      }
      break;
    case 'running':
      u.t -= dt;
      h.position.y = BASE_Y + Math.abs(Math.sin(u.bob))*0.09;
      if(u.t <= 0){ u.state='idle'; u.t = 1 + Math.random(); }
      break;
  }
  // containment
  const r = Math.hypot(h.position.x, h.position.z);
  if(r > CAGE_R - 0.2 && ['wander','goEat','toWheel'].includes(u.state)){
    u.target = new THREE.Vector3(h.position.x*0.25, 0, h.position.z*0.25);
  }
}

stepToward:

```
function stepToward(h, u, dt){
  const dir = _v1.subVectors(u.target, h.position);
  dir.y = 0;
  const d = dir.length();
  if(d < 0.28){
    if(u.state === 'wander'){ u.state='idle'; u.t = 0.6 + Math.random()*1.6; }
    else if(u.state === 'goEat'){
      h.rotation.y = Math.atan2(u.target.x - h.position.x, u.target.z - h.position.z);
      u.state='eating'; u.t = 1.2 + Math.random();
    }
    else if(u.state === 'toWheel'){
      h.rotation.y = Math.atan2(WHEEL_POS.x - h.position.x, WHEEL_POS.z - h.position.z);
      u.state='running'; u.t = 3 + Math.random()*3;
    }
    return;
  }
  dir.normalize();
  const ta = Math.atan2(dir.x, dir.z);
  let da = ta - h.rotation.y;
  da = Math.atan2(Math.sin(da), Math.cos(da));
  h.rotation.y += da * Math.min(1, dt*5);
  h.position.addScaledVector(dir, u.speed*dt);
}
