function islandHeight(x,z){
  const r=Math.hypot(x,z);
  let h=1.55*Math.exp(-Math.pow(r/3.0,2.0));
  h+=0.14*Math.sin(x*1.7+0.5)*Math.cos(z*1.5);
  h+=0.09*Math.sin(x*3.3+2.0)*Math.sin(z*3.1+1.0);
  h-=0.06*Math.sin(r*4.0);
  return h;
}
function buildIsland(){
  const R=5.0, N=150;
  const pos=[],uv=[],idx=[],col=[];
  const cSand=new THREE.Color(0xe7cf9c),cSandWet=new THREE.Color(0xc9a878),
        cGrass=new THREE.Color(0x74a045),cGrassDark=new THREE.Color(0x5c8a37),
        cRock=new THREE.Color(0x8b8579);
  for(let j=0;j<=N;j++){
    for(let i=0;i<=N;i++){
      const x=(i/N*2-1)*R, z=(j/N*2-1)*R;
      const h=islandHeight(x,z);
      pos.push(x,h,z); uv.push(i/N,j/N);
      const r=Math.hypot(x,z);
      const g=smoothstep(1.15,1.95,h);
      const rock=smoothstep(0.75,0.28,r);
      const c=new THREE.Color().copy(cSand).lerp(cSandWet,smoothstep(1.2,0.95,h));
      c.lerp(cGrass,g);
      c.lerp(cGrassDark,smoothstep(1.7,2.4,h)*0.7);
      c.lerp(cRock,rock*0.55);
      col.push(c.r,c.g,c.b);
    }
  }
  for(let j=0;j<N;j++)for(let i=0;i<N;i++){
    const a=j*(N+1)+i,b=a+1,c=a+N+1,d=c+1;
    idx.push(a,c,b, b,c,d);
  }
  const geo=new THREE.BufferGeometry();
  geo.setAttribute('position',new THREE.Float32BufferAttribute(pos,3));
  geo.setAttribute('uv',new THREE.Float32BufferAttribute(uv,2));
  geo.setAttribute('color',new THREE.Float32BufferAttribute(col,3));
  geo.setIndex(idx); geo.computeVertexNormals();
  const mesh=new THREE.Mesh(geo,M.sand.clone());
  mesh.material.vertexColors=true; mesh.receiveShadow=true; mesh.castShadow=true;
  return mesh;
}

function applyTime(dt){
  const target=state.timeBlend;
  state.timeBlend=lerp(state.timeBlend,target,1-Math.exp(-dt*2.2));
  const b=state.timeBlend;
  // blend between day(0) and night(2) through sunset(1)
  let A,B,t;
  if(b<1){A=TIMES.day;B=TIMES.sunset;t=b;} else {A=TIMES.sunset;B=TIMES.night;t=b-1;}
  const mix=(a,bb,tt)=>lerp(a,bb,tt);
  sun.position.set(mix(A.sunPos[0],B.sunPos[0],t),mix(A.sunPos[1],B.sunPos[1],t),mix(A.sunPos[2],B.sunPos[2],t));
  sun.color.setHex(A.sunColor).lerp(cTmp.setHex(B.sunColor),t);
  sun.intensity=mix(A.sunInt,B.sunInt,t);
  hemi.color.setHex(A.hemiSky).lerp(cTmp.setHex(B.hemiSky),t);
  hemi.groundColor.setHex(A.hemiGround).lerp(cTmp.setHex(B.hemiGround),t);
  hemi.intensity=mix(A.hemiInt,B.hemiInt,t);
  skyMat.uniforms.uTop.value.setHex(A.skyTop).lerp(cTmp.setHex(B.skyTop),t);
  skyMat.uniforms.uMid.value.setHex(A.skyMid).lerp(cTmp.setHex(B.skyMid),t);
  skyMat.uniforms.uBot.value.setHex(A.skyBot).lerp(cTmp.setHex(B.skyBot),t);
  renderer.setClearColor(cTmp.setHex(A.fog).lerp(cTmp.setHex(B.fog),t),1);
  renderer.toneMappingExposure=mix(A.exposure,B.exposure,t);
  const day=mix(A.day,B.day,t);
  const u=waterUniforms;
  u.uDay.value=day;
  u.uSunDir.value.set(mix(A.waterSun[0],B.waterSun[0],t),mix(A.waterSun[1],B.waterSun[1],t),mix(A.waterSun[2],B.waterSun[2],t)).normalize();
  u.uSunColor.value.setHex(A.waterSunColor).lerp(cTmp.setHex(B.waterSunColor),t);
  u.uSkyTop.value.setHex(A.wSkyTop).lerp(cTmp.setHex(B.wSkyTop),t);

  if(seabedMat.userData.shader)seabedMat.userData.shader.uniforms.uDay.value=day;
  // artificial lights
  const evening=1-day;
  winMat.emissiveIntensity=evening*1.6;

    this.ctx=new AC();
    this.master=this.ctx.createGain(); this.master.gain.value=state.audio.volume; this.master.connect(this.ctx.destination);
    this.musicGain=this.ctx.createGain(); this.musicGain.gain.value=0.0; this.musicGain.connect(this.master);
    this.ambGain=this.ctx.createGain(); this.ambGain.gain.value=0.0; this.ambGain.connect(this.master);
    this._ambience();
  },
  _noiseBuffer(){
    const b=this.ctx.createBuffer(1,this.ctx.sampleRate*2,this.ctx.sampleRate);
    const d=b.getChannelData(0); for(let i=0;i<d.length;i++)d[i]=Math.random()*2-1; return b;
  },
  _ambience(){
    // wind: noise -> bandpass -> LFO gain
    const src=this.ctx.createBufferSource(); src.buffer=this._noiseBuffer(); src.loop=true;
    const bp=this.ctx.createBiquadFilter(); bp.type='bandpass'; bp.frequency.value=500; bp.Q.value=0.6;
    const g=this.ctx.createGain(); g.gain.value=0.12;
    const lfo=this.ctx.createOscillator(); lfo.frequency.value=0.12; const lg=this.ctx.createGain(); lg.gain.value=0.06;
    lfo.connect(lg).connect(g.gain);
    src.connect(bp).connect(g).connect(this.ambGain); src.start(); lfo.start();
    // water: lowpass noise + slow swell
    const src2=this.ctx.createBufferSource(); src2.buffer=this._noiseBuffer(); src2.loop=true;
    const lp=this.ctx.createBiquadFilter(); lp.type='lowpass'; lp.frequency.value=320;
    const g2=this.ctx.createGain(); g2.gain.value=0.16;
    const lfo2=this.ctx.createOscillator(); lfo2.frequency.value=0.18; const lg2=this.ctx.createGain(); lg2.gain.value=0.08;
    lfo2.connect(lg2).connect(g2.gain);
    src2.connect(lp).connect(g2).connect(this.ambGain); src2.start(); lfo2.start();
    this.windFilter=bp;
  },
  setVolume(v){state.audio.volume=v; if(this.master)this.master.gain.setTargetAtTime(v,this.ctx.currentTime,0.05);},
  setWind(w){if(this.windFilter)this.windFilter.frequency.setTargetAtTime(300+w*900,this.ctx.currentTime,0.2);},
  toggleMusic(){
    this.init();
    if(this.ctx.state==='suspended')this.ctx.resume();
    this.musicOn=!this.musicOn;
    this.musicGain.gain.setTargetAtTime(this.musicOn?0.9:0.0,this.ctx.currentTime,0.6);

  },
  _startMusic(){
    this.musicStarted=true;
    const scale=[0,3,5,7,10,12,15]; // A minor pentatonic-ish
    const root=110;
    const self=this;
    let step=0;
    function schedule(){
      if(!self.musicOn)return;
      const t=self.ctx.currentTime;
      // pad chord
      if(step%8===0){
        const chord=[0,7,12,15].map(s=>root*Math.pow(2,s/12));
        chord.forEach(f=>{
          const o=self.ctx.createOscillator(); o.type='sine'; o.frequency.value=f;
          const g=self.ctx.createGain(); g.gain.setValueAtTime(0,t);
          g.gain.linearRampToValueAtTime(0.06,t+1.2); g.gain.linearRampToValueAtTime(0,t+6);
          o.connect(g).connect(self.musicGain); o.start(t); o.stop(t+6.2);
        });
      }
      // melody
      const n=pick(scale);
      const f=root*2*Math.pow(2,n/12);
      const o=self.ctx.createOscillator(); o.type='triangle'; o.frequency.value=f;
      const g=self.ctx.createGain(); g.gain.setValueAtTime(0,t);
      g.gain.linearRampToValueAtTime(0.09,t+0.05); g.gain.exponentialRampToValueAtTime(0.001,t+1.4);
      const lp=self.ctx.createBiquadFilter(); lp.type='lowpass'; lp.frequency.value=1800;
      o.connect(lp).connect(g).connect(self.musicGain); o.start(t); o.stop(t+1.5);
      step++;
      setTimeout(schedule,1000*(step%8===0?1.0:0.75));
    }
    schedule();
  },
  splash(){
    if(!this.ctx)return;
    const t=this.ctx.currentTime;
    const src=this.ctx.createBufferSource(); src.buffer=this._noiseBuffer();
    const bp=this.ctx.createBiquadFilter(); bp.type='bandpass'; bp.frequency.setValueAtTime(1400,t); bp.frequency.exponentialRampToValueAtTime(200,t+0.5); bp.Q.value=1.2;
    const g=this.ctx.createGain(); g.gain.setValueAtTime(0.35,t); g.gain.exponentialRampToValueAtTime(0.001,t+0.5);
    src.connect(bp).connect(g).connect(this.master); src.start(t); src.stop(t+0.6);
  },
  click(){
    if(!this.ctx)return;
    const t=this.ctx.currentTime;
    const o=this.ctx.createOscillator(); o.type='sine'; o.frequency.setValueAtTime(600,t); o.frequency.exponentialRampToValueAtTime(300,t+0.08);
    const g=this.ctx.createGain(); g.gain.setValueAtTime(0.08,t); g.gain.exponentialRampToValueAtTime(0.001,t+0.1);
    o.connect(g).connect(this.master); o.start(t); o.stop(t+0.12);
  }
};

function updateRay(dt){
  const R=state.ray; if(!R.active)return;
  R.t+=dt/2.0;

    r.g.rotation.set(lerp(0,-0.9,k),Math.atan2(R.dir.x,R.dir.z),0);

    const land=R.start.clone().add(R.dir.clone().multiplyScalar(2.6)).setY(R.start.y);

  }
}
