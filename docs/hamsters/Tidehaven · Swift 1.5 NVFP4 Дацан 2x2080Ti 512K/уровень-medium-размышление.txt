Water shader vertex:

```
uniform float uTime; uniform float uWind;
varying vec3 vPos; varying vec3 vNormal; varying vec3 vWorld;
void main(){
  vec3 p = position;
  vec3 n = normal;
  if(n.y > 0.5){ // top face (after rotate geometry local? transform geometry so up is local Y)
    float w = uWind;
    float x = p.x, z = p.z;
    p.y += sin(x*1.6 + uTime*1.4)*0.05*w + sin(z*2.1 - uTime*1.1)*0.04*w + sin((x+z)*1.1+uTime*0.7)*0.03*w;
    // approximate normal
    float dx = cos(x*1.6+uTime*1.4)*0.08*w + cos((x+z)*1.1+uTime*0.7)*0.033*w;
    float dz = cos(z*2.1-uTime*1.1)*0.084*w + cos((x+z)*1.1+uTime*0.7)*0.033*w;
    n = normalize(vec3(-dx,1.0,-dz));
  }
  vNormal=n; vWorld=(modelMatrix*vec4(p,1.)).xyz;
  gl_Position = projectionMatrix*modelViewMatrix*vec4(p,1.);
```
