function createWheel() {
    const group = new THREE.Group();
    
    // Rim - torus
    const rimGeo = new THREE.TorusGeometry(WHEEL_RADIUS, WHEEL_TUBE_RADIUS, 8, 32);
    const rimMat = new THREE.MeshStandardMaterial({ color: 0x8B4513 });
    const rim = new THREE.Mesh(rimGeo, rimMat);
    group.add(rim);
    
    // Spokes
    for (let i = 0; i < 8; i++) {
        const angle = (i / 8) * Math.PI * 2;
        const spokeGeo = new THREE.CylinderGeometry(0.01, 0.01, WHEEL_RADIUS, 6);
        const spoke = new THREE.Mesh(spokeGeo, rimMat);
        spoke.position.set(Math.cos(angle) * WHEEL_RADIUS/2, Math.sin(angle) * WHEEL_RADIUS/2, 0);
        spoke.rotation.z = angle - Math.PI/2;
        group.add(spoke);
    }
    
    // Stand
    // ...
    
    return group;
}
