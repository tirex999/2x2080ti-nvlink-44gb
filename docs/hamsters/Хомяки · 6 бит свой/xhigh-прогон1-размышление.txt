function clampOut(v){

function updateHamster(h,dt,time){
  h.t+=dt; h.desireCd=Math.max(0,h.desireCd-dt);

  switch(h.state){
    case 'wander':
      maybeDesire(h,dt);
      if(h.state==='wander' && (!h.target || h.pos.distanceTo(h.target)<.2 || h.t>7)){
        if(Math.random()<.5) setState(h,'pause');
        else { pickTarget(h); setState(h,'wander'); }
      }
      break;
    case 'pause':
      maybeDesire(h,dt);
      if(h.state==='pause' && h.t>1+Math.random()*1.6){ pickTarget(h); setState(h,'wander'); }
      break;
    case 'towheel':
      if(h.pos.distanceTo(h.target)<.28){
        if(wheel.occupant && wheel.occupant!==h){ pickTarget(h); setState(h,'wander'); }
        else { wheel.occupant=h; h.wheelDur=4+Math.random()*5; h.pos.copy(WHEEL_RUN); setState(h,'wheel'); }
      }
      break;
    case 'wheel':
      if(h.t>h.wheelDur){ wheel.occupant=null; h.pos.copy(WHEEL_EXIT); pickTarget(h); setState(h,'wander'); }
      break;
    case 'tofood':{
      const p=h.foodTarget;
      if(!p||!p.alive){ pickTarget(h); setState(h,'wander'); }
      else if(h.pos.distanceTo(p.pos)<.35){ p.alive=false; h.eatPellet=p; setState(h,'eat'); }
      break;
    }
    case 'eat':
      if(h.t>3.2){ if(h.eatPellet) removePellet(h.eatPellet); h.eatPellet=null; pickTarget(h); setState(h,'wander'); }
      break;
  }

  /* speed & movement */
  let want=0;
  if(h.state==='wander') want=h.baseSpeed;
  else if(h.state==='towheel'||h.state==='tofood') want=1.15;
  else if(h.state==='wheel') want=1.4;
  h.speed+=(want-h.speed)*Math.min(1,dt*4);

  if(h.state!=='wheel'&&h.target){
    const dx=h.target.x-h.pos.x, dz=h.target.z-h.pos.z, d=Math.hypot(dx,dz);

    h.pos.z+=Math.cos(h.heading)*h.speed*dt;
  }
  if(h.state==='wheel') h.heading=WHEEL_YAW;

  /* keep inside cage & out of furniture */
  const rr=Math.hypot(h.pos.x,h.pos.z)||.001;
  if(rr>WALK_R){ h.pos.x*=WALK_R/rr; h.pos.z*=WALK_R/rr; }
  if(h.state!=='wheel') for(const s of SOLIDS){
    const dx=h.pos.x-s.x, dz=h.pos.z-s.z, d=Math.hypot(dx,dz)||.001;
    if(d<s.r){ h.pos.x=s.x+dx/d*s.r; h.pos.z=s.z+dz/d*s.r; }
  }

  /* ---- visuals ---- */
  const moving=h.speed>.08;
  const freq=6+h.speed*8;
  h.bob=moving?Math.abs(Math.sin(time*freq+h.phase))*(.03+.05*h.speed):Math.sin(time*2+h.phase)*.012;
  h.bounce=Math.max(0,h.bounce-dt*1.8);
  const yOff=Math.sin(h.bounce*Math.PI)*.28;
  const baseY=h.state==='wheel'?SAND_Y+.19:SAND_Y;
  h.group.position.set(h.pos.x,baseY+h.bob+yOff,h.pos.z);
  h.group.rotation.y=h.heading;
  h.group.rotation.z=moving?Math.sin(time*freq+h.phase)*.06:0;
  h.body.scale.set(1+h.bob*.5,.82-h.bob*.7,1.12+h.bob*.4);

  h.feet.forEach((f,i)=>{
    const ph=time*(8+h.speed*9)+h.phase+((i===0||i===3)?0:Math.PI);
    f.position.y=.05+(moving?Math.abs(Math.sin(ph))*.05:0);
  });

  let hx=0,hz=0;
  if(h.state==='pause'){ hx=Math.sin(time*2.3+h.phase)*.22; hz=Math.sin(time*1.4+h.phase)*.18; }
  else if(h.state==='eat'){ hx=.55+Math.sin(time*9)*.1; }
  else if(h.state==='wheel'){ hx=.12; }
  h.head.rotation.x+=(hx-h.head.rotation.x)*Math.min(1,dt*6);
  h.head.rotation.z+=(hz-h.head.rotation.z)*Math.min(1,dt*6);

  const puffT=h.state==='eat'?1:0;
