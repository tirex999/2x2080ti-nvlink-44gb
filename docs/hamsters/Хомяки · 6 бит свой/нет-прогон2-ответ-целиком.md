const INTERACTIONS=[
  {pos:WHEEL_POS,r:0.9,kind:'wheel',label:'is doing laps on the wheel'},
  {pos:TUNNEL_POS,r:0.9,kind:'tunnel',label:'is exploring the tunnel'},
  {pos:BOWL_POS,r:0.85,kind:'eat',label:'is munching on seeds'},
  {pos:house.position.clone().setY(FLOOR_Y),r:0.95,kind:'nap',label:'is napping in the hideout'},
];
const WALLX=W/2-0.5, WALLZ=D/2-0.5;

function nearestInteraction(pos){
  let best=null,bd=1e9;
  for(const it of INTERACTIONS){
    const d=pos.distanceTo(it.pos);
    if(d<it.r+0.4&&d<bd){bd=d;best=it;}
  }
  return best;
}
function randomPoint(){
  let x,z,tries=0;
  do{
    x=rand(-WALLX,WALLX);z=rand(-WALLZ,WALLZ);tries++;
  }while(Math.hypot(x-0.6,z+0.2)<1.0&&tries<8); // avoid wheel hub
  return new THREE.Vector3(x,FLOOR_Y,z);
}

function behavior(h,dt){
  h.timer-=dt;
  const p=h.pos;
  switch(h.state){
    case 'idle':
      if(h.timer<=0){
        const it=nearestInteraction(p);
        if(it&&Math.random()<0.7){
          h.targetPos.copy(it.pos);
          h.targetHeading=Math.atan2(it.pos.x-p.x,it.pos.z-p.z);
          h.state='go';
        }else{
          h.targetPos.copy(randomPoint());
          h.targetHeading=Math.atan2(h.targetPos.x-p.x,h.targetPos.z-p.z);
          h.state='wander';
        }
      }
      break;
    case 'wander':
    case 'go':
      {
        const dx=h.targetPos.x-p.x,dz=h.targetPos.z-p.z;
        const dist=Math.hypot(dx,dz);
        if(dist<0.18){
          if(h.state==='go'){
            const it=nearestInteraction(p);
            if(it){h.state='interact';h.timer=it.kind==='wheel'?rand(3,5):it.kind==='eat'?rand(2.5,4):rand(3,6);}
            else{h.state='idle';h.timer=rand(0.5,2);}
          }else{h.state='idle';h.timer=rand(0.6,2.2);}
          break;
        }
        const desired=Math.atan2(dx,dz);
        let diff=((desired-h.heading+Math.PI)%(2*Math.PI))-Math.PI;
        h.heading+=clamp(diff,-2.2*dt,2.2*dt);
        const spd=(h.state==='go'?1.5:1.0);
        p.x+=Math.sin(h.heading)*spd*dt;
        p.z+=Math.cos(h.heading)*spd*dt;
        p.x=clamp(p.x,-WALLX,WALLX);p.z=clamp(p.z,-WALLZ,WALLZ);
        h.speed=spd;
      }
      break;
    case 'interact':
      {
        const it=nearestInteraction(p);
        if(!it){h.state='idle';h.timer=1;break;}
        const dx=it.pos.x-p.x,dz=it.pos.z-p.z,dist=Math.hypot(dx,dz);
        const face=Math.atan2(dx,dz);
        let diff=((face-h.heading+Math.PI)%(2*Math.PI))-Math.PI;
        h.heading+=clamp(diff,-2.5*dt,2.5*dt);
        h.speed=0;
        if(it.kind==='wheel'){
          if(dist>0.75){ // climb onto wheel
            const s=1.4*dt;p.x+=Math.sin(face)*s;p.z+=Math.cos(face)*s;
            h.group.position.y=FLOOR_Y+0.15;
          }else{
            h.group.position.y=FLOOR_Y+0.95; // standing on rim
            wheelSpin+=(1.1+Math.sin(h.walkPhase)*0.1)*dt*2.4;
            h.walkPhase+=dt*9;
          }
        }else if(it.kind==='tunnel'&&dist<0.55){
          // peek in / out
          h.group.rotation.x=Math.sin(h.timer*3)*0.15;
        }else if(it.kind==='eat'){
          h.group.rotation.x=0.35; // head down
        }else{ // nap
          h.group.rotation.x=0;
        }
        if(h.timer<=0){
          h.group.position.y=FLOOR_Y;
          h.group.rotation.x=0;
          h.state='idle';h.timer=rand(0.5,1.8);
        }
      }
      break;
  }
  // write transforms
  h.group.position.set(p.x,h.group.position.y,p.z);
  h.group.rotation.y=h.heading;
}

function animateHamster(h,dt){
  // walk cycle
  const moving=h.speed>0.05;
  if(moving)h.walkPhase+=dt*h.speed*9;
  const amp=moving?0.5:0;
  const swing=Math.sin(h.walkPhase);
  h.fL.grp.rotation.x=swing*amp;
  h.fR.grp.rotation.x=-swing*amp;
  h.bL.grp.rotation.x=-swing*amp;
  h.bR.grp.rotation.x=swing*amp;
  // body bob
  const bob=moving?Math.abs(Math.sin(h.walkPhase))*0.04:Math.sin(performance.now()/500+h.heading)*0.01;
  h.group.scale.y=1+bob*2;
  // squash & stretch spring
  h.squashVel+=(-h.squash)*120*dt;
  h.squashVel*=Math.pow(0.02,dt);
  h.squash+=h.squashVel*dt;
  const sq=clamp(h.squash,0.7,1.35);
  h.group.scale.x=1+(1-sq)*0.5;
  h.group.scale.z=1+(1-sq)*0.5;
  // blink
  h.blinkTimer-=dt;
  if(h.blinkTimer<=0){h.blink=1;h.blinkTimer=rand(2,5);}
  if(h.blink>0){
    h.blink-=dt*8;
    const s=Math.max(0.05,Math.abs(h.blink));
    h.group.children.forEach(()=>{});
  }
}

  const n=kind==='heart'?6:8;

    const color=kind==='heart'?'#ff6f91':'#ffd166';

    particles.push({
      mesh:m,vel:new THREE.Vector3(Math.sin(a)*sp,rand(1.2,2.4),Math.cos(a)*sp),
      life:1,maxLife:1,spin:rand(-6,6),kind
    });
  }
}
function updateParticles(dt){
  for(let i=particles.length-1;i>=0;i--){
    const p=particles[i];
    p.vel.y-=4*dt;
    p.mesh.position.addScaledVector(p.vel,dt);
    p.mesh.rotation.x+=p.spin*dt;p.mesh.rotation.y+=p.spin*dt;
    p.life-=dt/p.maxLife;
    const s=clamp(p.life,0,1);
    p.mesh.scale.setScalar(s*1.1);
    p.mesh.material.opacity=s;
    if(p.life<=0){scene.remove(p.mesh);p.mesh.material.dispose();particles.splice(i,1);}
  }
}
