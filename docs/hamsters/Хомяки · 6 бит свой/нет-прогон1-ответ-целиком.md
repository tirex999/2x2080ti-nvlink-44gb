const matCache = {};
function mat(color, o={}){
  const key = color + '|' + (o.flatShading?1:0) + '|' + (o.metalness||0) + '|' + (o.roughness||1) + '|' + (o.emissive||'') + '|' + (o.opacity||1);
  if(matCache[key]) return matCache[key];
  const m = new THREE.MeshStandardMaterial({
    color, roughness:o.roughness ?? 0.9, metalness:o.metalness ?? 0,
    flatShading:o.flatShading ?? false, transparent:(o.opacity!=null && o.opacity<1), opacity:o.opacity ?? 1
  });
  if(o.emissive) m.emissive = new THREE.Color(o.emissive);
  matCache[key] = m; return m;
}

    const e = mesh(sphere(0.06,8,8), eyeMat, s*0.14,0.72,0.66, g);
    mesh(sphere(0.02,6,6), mat('#ffffff',{roughness:0.2}), s*0.16,0.75,0.71, g);
  });
  // nose
  mesh(sphere(0.05,6,6), mat('#e86a86',{flatShading:false}), 0,0.56,0.82, g);

  [-1,1].forEach(s=>{
    const ch = mesh(ico(0.13,0), mat('#ffb0a0',{flatShading:true}), s*0.2,0.58,0.5, g);
    ch.scale.y = 0.8;
  });
  // feet
  const footMat = mat(pal.ear,{flatShading:true});
  const f = [];
  [[-0.2,0.16,0.22],[0.2,0.16,0.22],[-0.2,0.16,-0.22],[0.2,0.16,-0.22]].forEach(p=>{
    const ft = mesh(ico(0.12,0), footMat, p[0],p[1],p[2], g); ft.scale.set(1,0.7,1.3); f.push(ft);
  });
  // tail
  mesh(ico(0.1,0), mat(pal.ear,{flatShading:true}), 0,0.5,-0.55, g);

  g.userData = {
    body, head, f,
    pos:new THREE.Vector3(), dir:new THREE.Vector3(1,0,0),

  };

  return g;
}

function updateHamster(h, dt, t){
  const u = h.userData;
  u.timer -= dt;
  const distTo = p => u.pos.distanceTo(p);

  // ---- Wheel ----
  if(u.onWheel){
    u.wheelAngVel += 1.3*dt;
    if(u.wheelAngVel>2.4) u.wheelAngVel=2.4;
    u.wheelAngle += u.wheelAngVel*dt;
    wheelSpin += u.wheelAngVel*dt;
    const a = u.wheelAngle;
    h.position.set(3.4 - Math.cos(a)*1.35, 1.55 + Math.sin(a)*1.35, -2.2);
    h.quaternion.setFromAxisAngle(new THREE.Vector3(0,1,0), -a + Math.PI/2);
    u.bob = Math.max(0, u.bob - dt*2);
    if(u.timer<=0){ u.onWheel=false; u.state='wander'; u.timer=2+Math.random()*2; u.goal=randPoint(); }
    return;
  }

  // ---- State machine ----
  switch(u.state){
    case 'wander':
      u.targetSpeed = 0.9;
      if(!u.goal || u.pos.distanceTo(u.goal)<0.7){
        // chance to interact with a nearby object
        const near = [];
        if(distTo(bowlPos)<2.2) near.push('eat');
        if(distTo(hayPos)<2.2) near.push('hay');
        if(distTo(archEntry)<1.6) near.push('tunnel');
        if(distTo(wheel.position)<2.4) near.push('wheel');
        if(near.length && Math.random()<0.6){
          u.state = pick(near); u.timer = 1.6+Math.random()*1.4;
          u.goal = (u.state==='eat'?bowlPos:u.state==='hay'?hayPos:u.state==='tunnel'?archEntry:wheel.position).clone();
        } else {
          u.goal = randPoint();
        }
      }
      break;
    case 'eat':
      u.targetSpeed = distTo(bowlPos)>0.8 ? 1.1 : 0.15;
      if(u.timer<=0){ u.state='wander'; u.timer=2+Math.random()*2; u.goal=randPoint(); }
      break;
    case 'hay':
      u.targetSpeed = distTo(hayPos)>0.8 ? 1.1 : 0.15;
      if(u.timer<=0){ u.state='wander'; u.timer=2+Math.random()*2; u.goal=randPoint(); }
      break;
    case 'tunnel':
      u.targetSpeed = distTo(archEntry)>0.5 ? 1.1 : 0.2;
      if(u.timer<=0){ u.state='wander'; u.timer=2+Math.random()*2; u.goal=randPoint(); }
      break;
    case 'wheel':
      u.targetSpeed = distTo(wheel.position)>0.9 ? 1.4 : 0.2;
      if(distTo(wheel.position)<0.9 && u.timer<=0){
        u.onWheel=true; u.timer=3+Math.random()*3; u.wheelAngVel=0.6;
        toast(`${u.name} is on the wheel! 🎡`);
      }
      break;
  }

  // ---- Steering toward goal ----
  if(u.goal){
    const toGoal = u._look.copy(u.goal).sub(u.pos); toGoal.y=0;
    if(toGoal.lengthSq()>0.0001){
      toGoal.normalize();
      u.dir.lerp(toGoal, 1-Math.pow(0.001, dt)).normalize();
    }
  }
  // wander drift
  u.idleDir += (Math.random()-0.5)*dt*2;
  u.dir.applyAxisAngle(new THREE.Vector3(0,1,0), u.idleDir*0.4).normalize();

  // ---- Move ----
  u.speed += (u.targetSpeed-u.speed)*Math.min(1, dt*4);
  u.pos.addScaledVector(u.dir, u.speed*dt);
  // keep in bounds
  if(u.pos.x<bounds.minX){u.pos.x=bounds.minX; u.dir.x=Math.abs(u.dir.x);}
  if(u.pos.x>bounds.maxX){u.pos.x=bounds.maxX; u.dir.x=-Math.abs(u.dir.x);}
  if(u.pos.z<bounds.minZ){u.pos.z=bounds.minZ; u.dir.z=Math.abs(u.dir.z);}
  if(u.pos.z>bounds.maxZ){u.pos.z=bounds.maxZ; u.dir.z=-Math.abs(u.dir.z);}
  h.position.set(u.pos.x, 0, u.pos.z);

  // face direction (smooth)
  u._q.setFromUnitVectors(new THREE.Vector3(0,0,1), u.dir);
  h.quaternion.slerp(u._q, Math.min(1, dt*6));

  // ---- Animations ----
  u.walkPhase += u.speed*dt*9;
  const moving = u.speed>0.25;
  u.bob = Math.max(0, u.bob-dt*2);
  const hop = moving ? Math.abs(Math.sin(u.walkPhase))*0.06 : Math.sin(t*2+h.id)*0.015;
  u.body.position.y = 0.5 + hop + u.bob*0.12;
  // squash & stretch
  const sq = 1 + (moving?Math.sin(u.walkPhase)*0.03:0) + u.bob*0.05;
  u.body.scale.set(1.25/sq, 0.95*sq, 1.05/sq);
  u.head.position.y = 0.66 + hop*0.5;
  u.head.rotation.z = moving ? Math.sin(u.walkPhase)*0.08 : Math.sin(t*1.5+h.id)*0.04;
  u.head.rotation.x = (u.state==='eat'||u.state==='hay') ? 0.3 : 0;
  // feet shuffle
  u.f.forEach((ft,i)=>{
    const s = (i<2?1:-1);
    ft.position.z = (i<2?0.22:-0.22) + (moving?Math.sin(u.walkPhase + (i%2?Math.PI:0))*0.08*s:0);
    ft.position.y = 0.16 + (moving?Math.max(0,Math.sin(u.walkPhase+(i%2?Math.PI:0)))*0.05:0);
  });
}

const clock = new THREE.Clock();
function animate(){
  requestAnimationFrame(animate);
  const dt = Math.min(clock.getDelta(), 0.05);
  const t = clock.elapsedTime;

  // camera easing
  if(ctrl.auto && !ctrl.dragging) ctrl.tTheta += dt*0.12;
  ctrl.theta  += (ctrl.tTheta-ctrl.theta)*Math.min(1,dt*6);
  ctrl.phi    += (ctrl.tPhi-ctrl.phi)*Math.min(1,dt*6);
  ctrl.radius += (ctrl.tRadius-ctrl.radius)*Math.min(1,dt*5);
  applyCamera();

  // wheel inertia
  wheelSpin *= (1 - Math.min(1,dt*0.6));
  wheelRim.rotation.z -= wheelSpin;

  // hamsters
  for(const h of hamsters) updateHamster(h, dt, t);

  // grow-in scale
  for(const h of hamsters){
    const s = h.userData.scale;
    h.scale.setScalar(THREE.MathUtils.lerp(h.scale.x, s, Math.min(1,dt*4)));
  }

  // ambient
  for(const c of clouds){ c.position.x += c.userData.v*dt; if(c.position.x>24) c.position.x=-24; }
  for(const f of floaters){
    f.rotation.x += f.userData.rs*dt; f.rotation.y += f.userData.rs*0.7*dt;
    f.position.y += Math.sin(t*f.userData.sp + f.userData.ph)*dt*0.3;
  }
  for(const s of sparkles){
    const k = 0.5+0.5*Math.sin(t*2+s.userData.ph);
    s.scale.setScalar(0.5+k); s.material.opacity = 0.3+k*0.7;
  }

  renderer.render(scene, camera);
}
