function updateFish(dt, t){
  const n=fishArray.length;
  // pairwise avoidance
  for(let i=0;i<n;i++){
    const a=fishArray[i];
    for(let j=i+1;j<n;j++){
      const b=fishArray[j];
      _v1.subVectors(a.mesh.position, b.mesh.position);
      const d=_v1.length();
      const rr=(a.avoidanceRadius+b.avoidanceRadius)*0.5*(a.scaleCur+b.scaleCur)*0.5;
      if(d<rr && d>0.0001){
        _v1.multiplyScalar(1/d);
        const push=(rr-d)/rr;
        a.velocity.addScaledVector(_v1, push*6*dt);
        b.velocity.addScaledVector(_v1, -push*6*dt);
      }
    }
  }
  for(let i=0;i<n;i++){
    const f=fishArray[i];
    const p=f.mesh.position;
    // food

      for(const fd of foods){
        const d2=p.distanceToSquared(fd.mesh.position);
        if(d2<best){best=d2; f.targetFood=fd;}

    _v2.copy(f.velocity).normalize().multiplyScalar(f.speed);
