Value noise 2D:
```js
function vnoise2(x,z){
  const xi=Math.floor(x), zi=Math.floor(z);
  const xf=x-xi, zf=z-zi;
  const u=xf*xf*(3-2*xf), v=zf*zf*(3-2*zf);
  const a=hash2(xi,zi), b=hash2(xi+1,zi), c=hash2(xi,zi+1), d=hash2(xi+1,zi+1);
  return a+(b-a)*u+(c-a)*v+(a-b-c+d)*u*v;
}
```

```js
  const m=fractal2(x*0.004,z*0.004);
  const h=fractal2(x*0.02,z*0.02);
  return Math.floor(5+m*m*58+h*10);

function genChunk(cx,cz){
  const data=new Uint8Array(16*16*80);
  // columns
  for(let lx=0;lx<16;lx++)for(let lz=0;lz<16;lz++){
    const wx=cx*16+lx, wz=cz*16+lz;
    const H=terrainHeight(wx,wz);
    for(let y=0;y<=H&&y<HEIGHT;y++){
      let id;
      if(y===0)id=3;
      else if(y===H) id = H>=46?7 : H>=37?3 : H<=16?4 : 1;
      else if(y>=H-3) id = H<=16?4 : H>=37?3 : 2;
      else id=3;
      data[lx+lz*16+y*256]=id;
    }
    // caves
    for(let y=3;y<=H-2;y++){
      if(fractal3(wx*0.09,y*0.09,wz*0.09)>0.67) data[lx+lz*16+y*256]=0;
    }
  }
  // trees
  for(let lx=2;lx<=13;lx++)for(let lz=2;lz<=13;lz++){
    const wx=cx*16+lx, wz=cz*16+lz;
    if(treeHash(wx,wz)>=0.02)continue;

    if(data[(lx)+(lz)*16+H*256]!==1)continue;
    if(H+7>=HEIGHT)continue;
    // trunk
    for(let i=1;i<=4;i++) data[lx+lz*16+(H+i)*256]=5;
    // leaves 5x5 at H+4,H+5
```
