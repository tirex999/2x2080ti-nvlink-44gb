```javascript
function createHamster(color, bellyColor) {
    const group = new THREE.Group();
    
    // Body - squished sphere
    const bodyGeo = new THREE.SphereGeometry(0.5, 7, 5);
    const bodyMat = new THREE.MeshLambertMaterial({ color, flatShading: true });
    const body = new THREE.Mesh(bodyGeo, bodyMat);
    body.scale.set(1, 0.8, 1.2); // wider, shorter, longer
    group.add(body);
    
    // Head
    const headGeo = new THREE.SphereGeometry(0.35, 6, 5);
    const head = new THREE.Mesh(headGeo, bodyMat);
    head.position.set(0, 0.2, 0.45);
    group.add(head);
    
    // Ears
    const earGeo = new THREE.SphereGeometry(0.1, 5, 4);
    const earMat = new THREE.MeshLambertMaterial({ color: 0xFFB6C1, flatShading: true });
    const leftEar = new THREE.Mesh(earGeo, earMat);
    leftEar.position.set(-0.18, 0.5, 0.35);
    group.add(leftEar);
    const rightEar = new THREE.Mesh(earGeo, earMat);
    rightEar.position.set(0.18, 0.5, 0.35);
    group.add(rightEar);
    
    // Eyes
    const eyeGeo = new THREE.SphereGeometry(0.05, 5, 4);
    const eyeMat = new THREE.MeshLambertMaterial({ color: 0x000000 });
    const leftEye = new THREE.Mesh(eyeGeo, eyeMat);
    leftEye.position.set(-0.12, 0.3, 0.65);
    group.add(leftEye);
    const rightEye = new THREE.Mesh(eyeGeo, eyeMat);
    rightEye.position.set(0.12, 0.3, 0.65);
    group.add(rightEye);
    
    // Nose
    const noseGeo = new THREE.SphereGeometry(0.04, 4, 3);
    const noseMat = new THREE.MeshLambertMaterial({ color: 0xFF69B4 });
    const nose = new THREE.Mesh(noseGeo, noseMat);
    nose.position.set(0, 0.22, 0.72);
    group.add(nose);
    
    // Cheeks
    const cheekGeo = new THREE.SphereGeometry(0.12, 5, 4);
    const cheekMat = new THREE.MeshLambertMaterial({ color: lightenColor(color), flatShading: true });
    const leftCheek = new THREE.Mesh(cheekGeo, cheekMat);
    leftCheek.position.set(-0.2, 0.15, 0.5);
    group.add(leftCheek);
    const rightCheek = new THREE.Mesh(cheekGeo, cheekMat);
    rightCheek.position.set(0.2, 0.15, 0.5);
    group.add(rightCheek);
    
    // Legs (4 small cylinders)
    const legGeo = new THREE.CylinderGeometry(0.06, 0.06, 0.15, 5);
    const legMat = new THREE.MeshLambertMaterial({ color: darkenColor(color), flatShading: true });
    
    const legPositions = [
        [-0.2, -0.35, 0.25],  // front left
        [0.2, -0.35, 0.25],   // front right
        [-0.2, -0.35, -0.25], // back left
        [0.2, -0.35, -0.25]   // back right
    ];
    
    const legs = [];
    legPositions.forEach(pos => {
        const leg = new THREE.Mesh(legGeo, legMat);
        leg.position.set(...pos);
        group.add(leg);
        legs.push(leg);
    });
    
    // Tail
    const tailGeo = new THREE.SphereGeometry(0.06, 4, 3);
    const tail = new THREE.Mesh(tailGeo, bodyMat);
    tail.position.set(0, 0, -0.6);
    group.add(tail);
    
    group.userData = { legs, body, head };
    return group;
}
```

```javascript
class Hamster {
    constructor(mesh, position, color) {
        this.mesh = mesh;
        this.state = 'IDLE';
        this.stateTimer = 0;
        this.walkSpeed = 0.5 + Math.random() * 0.3;
        this.turnSpeed = 2 + Math.random();
        this.targetAngle = 0;
        this.legs = mesh.userData.legs;
        this.body = mesh.userData.body;
        this.time = Math.random() * 100;
        
        // Bounds for walking
        this.bounds = { minX: -3.2, maxX: 3.2, minZ: -2.2, maxZ: 2.2 };
        
        this.mesh.position.copy(position);
        this.mesh.rotation.y = Math.random() * Math.PI * 2;
    }
    
    update(delta, time) {
        this.time += delta;
        this.stateTimer += delta;
        
        switch (this.state) {
            case 'IDLE':
                this.idle(delta);
                if (this.stateTimer > 1 + Math.random() * 3) {
                    this.chooseState();
                }
                break;
            case 'WALK':
                this.walk(delta);
                break;
            case 'TURN':
                this.turn(delta);
                if (this.stateTimer > 0.5 + Math.random() * 0.5) {
                    this.setState('WALK');
                }
                break;
            case 'POPCORN':
                this.popcorn(delta);
                if (this.stateTimer > 0.4) {
                    this.setState('IDLE');
                }
                break;
        }
    }
    
    // ... etc
}
```

```html
const CAGE_BOUNDS = { xMin: -2.8, xMax: 2.8, zMin: -1.8, zMax: 1.8 };
const FLOOR_Y = 0.15;

class HamsterAgent {
    constructor(mesh, startPos, isWheelHamster = false) {
        this.mesh = mesh;
        this.legs = mesh.userData.legs;
        this.body = mesh.userData.body;
        this.head = mesh.userData.head;
        this.isWheelHamster = isWheelHamster;

        this.state = isWheelHamster ? 'WHEEL' : 'IDLE';
        this.stateTimer = 0;
        this.walkSpeed = 0.6 + Math.random() * 0.4;
        this.time = Math.random() * 100;
        this.jumpVel = 0;
        this.baseY = FLOOR_Y;

        if (!isWheelHamster) {
            this.mesh.position.set(startPos.x, FLOOR_Y, startPos.z);
            this.mesh.rotation.y = Math.random() * Math.PI * 2;
        }
    }

    setState(s) {
        this.state = s;
        this.stateTimer = 0;
    }

    chooseState() {
        const r = Math.random();
        if (r < 0.4) this.setState('WALK');
        else if (r < 0.65) this.setState('TURN');
        else if (r < 0.75) this.setState('POPCORN');
        else this.setState('IDLE');
    }

    update(delta) {
        this.time += delta;
        this.stateTimer += delta;

        if (this.isWheelHamster) {
            this.wheelRun(delta);
            return;
        }

        switch (this.state) {
            case 'IDLE': this.animIdle(delta); break;
            case 'WALK': this.animWalk(delta); break;
            case 'TURN': this.animTurn(delta); break;
            case 'POPCORN': this.animPopcorn(delta); break;
        }

        // Breathing
        const breathe = 1 + Math.sin(this.time * 3) * 0.02;
        this.body.scale.set(1.0 * breathe, 0.85, 1.15 * breathe);
    }

    animIdle(delta) {
        // Gentle sway
        this.mesh.rotation.z = Math.sin(this.time * 1.5) * 0.03;
        // Head tilt occasionally
        if (this.stateTimer > 2 && Math.random() < 0.01) {
            this.head.rotation.z = (Math.random() - 0.5) * 0.3;
        }
        if (this.stateTimer > 1.5 + Math.random() * 3) {
            this.head.rotation.z = 0;
            this.chooseState();
        }
    }

    animWalk(delta) {
        // Move forward
        const dir = new THREE.Vector3(0, 0, 1).applyAxisAngle(
            new THREE.Vector3(0, 1, 0), this.mesh.rotation.y
        );
        this.mesh.position.addScaledVector(dir, this.walkSpeed * delta);
        this.mesh.position.y = FLOOR_Y + Math.sin(this.time * 10) * 0.025;
        this.mesh.rotation.z = Math.sin(this.time * 8) * 0.04;

        // Animate legs
        this.legs.forEach((leg, i) => {
            const phase = (i < 2 ? 0 : Math.PI) + (i % 2 === 0 ? 0 : Math.PI);
            leg.rotation.x = Math.sin(this.time * 12 + phase) * 0.4;
        });

        // Check bounds
        const p = this.mesh.position;
        if (p.x < CAGE_BOUNDS.xMin || p.x > CAGE_BOUNDS.xMax ||
            p.z < CAGE_BOUNDS.zMin || p.z > CAGE_BOUNDS.zMax) {
            this.setState('TURN');
        }

        // Random state change
        if (this.stateTimer > 1.5 + Math.random() * 2.5) {
            this.chooseState();
        }
    }

    animTurn(delta) {
        this.mesh.rotation.y += (2 + Math.random()) * delta;
        this.mesh.rotation.z = Math.sin(this.time * 6) * 0.06;
        if (this.stateTimer > 0.5 + Math.random() * 0.5) {
            this.setState('WALK');
        }
    }

    animPopcorn(delta) {
        const t = this.stateTimer;
        if (t < 0.35) {
            this.mesh.position.y = FLOOR_Y + Math.sin((t / 0.35) * Math.PI) * 0.35;
            this.mesh.rotation.x = Math.sin((t / 0.35) * Math.PI) * 0.3;
        } else {
            this.mesh.position.y = FLOOR_Y;
            this.mesh.rotation.x = 0;
        }
        // Legs tuck in
        this.legs.forEach(leg => {
            leg.rotation.x = -0.5;
        });
        if (t > 0.5) {
            this.legs.forEach(leg => leg.rotation.x = 0);
            this.setState('IDLE');
        }
    }

    wheelRun(delta) {
        // Animate legs rapidly
        this.legs.forEach((leg, i) => {
            const phase = (i < 2 ? 0 : Math.PI) + (i % 2 === 0 ? 0 : Math.PI);
            leg.rotation.x = Math.sin(this.time * 18 + phase) * 0.5;
        });
        // Slight body bob
        this.mesh.position.y = this.baseY + Math.sin(this.time * 18) * 0.015;
        // Occasional speed change
        if (Math.random() < 0.005) {
            this.walkSpeed = 0.5 + Math.random() * 1.0;
        }
    }
}
```
