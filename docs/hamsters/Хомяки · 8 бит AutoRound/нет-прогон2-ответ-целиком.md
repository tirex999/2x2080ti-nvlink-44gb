  const g = new THREE.Group();
  const m = new THREE.MeshStandardMaterial({color:0xffffff, roughness:1, flatShading:true});
  [[0,0,0,1],[1.1,0.1,0.3,0.75],[-1.05,0,0.2,0.7],[0.3,0.55,-0.2,0.6]].forEach(([cx,cy,cz,r])=>{

    puff.position.set(cx*s, cy*s, cz*s);
    g.add(puff);
  });
  g.position.set(x,y,z);
  g.userData.speed = 0.15 + Math.random()*0.2;
  scene.add(g); clouds.push(g);
}

const BOUND_X = W/2-1.2, BOUND_Z = D/2-1.2;
const raycaster = new THREE.Raycaster();
const pointer = new THREE.Vector2();

function randomTarget(h){
  let tries=0;
  do{
    h.target.set((Math.random()*2-1)*BOUND_X, 1.1, (Math.random()*2-1)*BOUND_Z);
    tries++;
  } while(tries<8 && h.group.position.distanceTo(h.target) < 1.5);
}

function updateHamster(h, dt){
  h.walkPhase += dt * (h.state==='walk'? 10 : 3);

  // blinking
  h.blinkTimer -= dt;
  if(h.blinkTimer<=0){ h.blink = 0.12; h.blinkTimer = 2+Math.random()*3; }
  if(h.blink>0){ h.blink-=dt; h.head.scale.y = 0.2; } else h.head.scale.y = 1;

  // pet squash recovery
  if(h.pop>0){ h.pop = Math.max(0, h.pos - dt*3); }
  const sq = 1 + Math.sin(h.pop*Math.PI)*0.18;
  h.group.scale.setScalar((0.95+ (h.def.name==='Mochi'?0.1:0)) * (h.def.name==='Coco'?0.9:1) );
  h.group.scale.y *= (1 - Math.sin(h.pop*Math.PI)*0.15);
  h.group.scale.x *= (1 + Math.sin(h.pop*Math.PI)*0.12);
  h.group.scale.z *= (1 + Math.sin(h.pop*Math.PI)*0.12);

  const pos = h.group.position;
  const toTarget = h.target.clone().sub(pos); toTarget.y = 0;
  const dist = toTarget.length();
  const desired = Math.atan2(toTarget.x, toTarget.z);

  // --- state machine ---
  switch(h.state){
    case 'wander':
      h.turnTimer -= dt;
      if(dist < 0.4 || h.turnTimer <= 0){
        if(Math.random() < 0.35){ h.state='pause'; h.pauseTimer = 1+Math.random()*2; }
        else randomTarget(h);
        h.turnTimer = 2+Math.random()*3;
      }
      break;
    case 'pause':
      h.pauseTimer -= dt;
      if(h.pauseTimer<=0){ h.state='wander'; randomTarget(h); }
      break;
    case 'goWheel':
    case 'goBowl':
    case 'goCarrot':
    case 'goTunnel':
      if(dist < 0.5){
        if(h.state==='goWheel'){ h.state='rideWheel'; h.interactTimer=2.5+Math.random()*2; }
        else if(h.state==='goBowl'){ h.state='eatBowl'; h.interactTimer=1.5+Math.random(); }
        else if(h.state==='goCarrot'){ h.state='nibbleCarrot'; h.interactTimer=1.2+Math.random(); }
        else if(h.state==='goTunnel'){ h.state='digTunnel'; h.interactTimer=1.5+Math.random(); }
      }
      break;
    case 'rideWheel':
      wheelVel = 0.9;
      h.interactTimer -= dt;
      if(h.interactTimer<=0){ h.state='wander'; randomTarget(h); }
      break;
    case 'eatBowl':
    case 'nibbleCarrot':
    case 'digTunnel':
      h.interactTimer -= dt;
      if(h.state==='digTunnel' && Math.random()<dt*3) spawnSandDust(pos);
      if(h.interactTimer<=0){ h.state='wander'; randomTarget(h); }
      break;
  }

  // --- movement ---
  let moving = false;
  if(['wander','goWheel','goBowl','goCarrot','goTunnel'].includes(h.state)){
    if(dist > 0.15){
      const step = Math.min(dist, h.speed*dt);
      pos.addScaledVector(toTarget.normalize(), step);
      // steer smoothly toward target
      let d = desired - h.group.rotation.y;
      d = Math.atan2(Math.sin(d), Math.cos(d));
      h.group.rotation.y += d * Math.min(1, dt*6);
      moving = true;
    }
  }
  if(h.state==='rideWheel') h.group.rotation.y += dt*0.5;

  // keep in bounds
  pos.x = THREE.MathUtils.clamp(pos.x, -BOUND_X, BOUND_X);
  pos.z = THREE.MathUtils.clamp(pos.z, -BOUND_Z, BOUND_Z);
  pos.y = 1.1 + Math.abs(Math.sin(h.walkPhase))* (moving?0.05:0.015);

  // --- foot scuttle ---
  const amp = moving ? 0.22 : 0.05;
  h.feet.forEach((f,i)=>{
    const phase = (i<2?0:Math.PI) + (i%2?Math.PI/2:0);
    f.position.y = 0.12 + Math.abs(Math.sin(h.walkPhase+phase))*amp;
    f.position.z = f.userData.baseZ + Math.sin(h.walkPhase+phase)*(moving?0.12:0.02);
  });
  h.tail.rotation.x = Math.sin(h.walkPhase*0.5)*0.3;
  h.head.rotation.y = Math.sin(h.walkPhase*0.3)*0.12;

  h.mood = h.state==='rideWheel'?'zooming':
           ['eatBowl','nibbleCarrot'].includes(h.state)?'eating':
           h.state==='digTunnel'?'digging':
           h.state==='pause'?'resting':'happy';
}

}
