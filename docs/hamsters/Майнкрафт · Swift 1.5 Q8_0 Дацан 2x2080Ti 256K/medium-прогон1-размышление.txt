function makeMesh(c){
  const positions=[],normals=[],colors=[];
  const baseX=c.cx*16, baseZ=c.cz*16;
  for(let y=0;y<HEIGHT;y++)for(let lz=0;lz<16;lz++)for(let lx=0;lx<16;lx++){
    const id=c.data[idx(lx,y,lz)];
    if(!id)continue;
    const x=baseX+lx, z=baseZ+lz;
    const col=RGB[id];
    for(const f of FACES){
      if(getBlock(x+f.n[0],y+f.n[1],z+f.n[2])===0){
        const [r,g,b]=[col[0]*f.b,col[1]*f.b,col[2]*f.b];
        const q=f.v, order=[0,1,2,0,2,3];
        for(const oi of order){
          const v=q[oi];
          positions.push(x+v[0],y+v[1],z+v[2]);
          normals.push(f.n[0],f.n[1],f.n[2]);
          colors.push(r,g,b);
        }
      }
    }
  }
  const geo=new THREE.BufferGeometry();
  geo.setAttribute('position',new THREE.Float32BufferAttribute(positions,3));
  geo.setAttribute('normal',new THREE.Float32BufferAttribute(normals,3));
  geo.setAttribute('color',new THREE.Float32BufferAttribute(colors,3));
  const mesh=new THREE.Mesh(geo,mat);
  mesh.userData.cx=c.cx; mesh.userData.cz=c.cz;
  scene.add(mesh); chunkMeshes.push(mesh);
  c.mesh=mesh;
}
function removeMesh(c){
  if(!c.mesh)return;
  scene.remove(c.mesh);
  c.mesh.geometry.dispose();

  if(i>=0)chunkMeshes.splice(i,1);
  c.mesh=null;
}
function rebuildChunk(cx,cz){
  const c=chunks.get(key(cx,cz));

}
