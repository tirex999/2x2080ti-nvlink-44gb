  for(let lx=0;lx<16;lx++)for(let lz=0;lz<16;lz++){
    const x=cx*16+lx, z=cz*16+lz;
    const m=fractal2(x*0.004,z*0.004);
    const h=fractal2(x*0.02,z*0.02);
    const H=Math.floor(5+m*m*58+h*10);
    heights[lz*16+lx]=H;
    const surf=H>=46?7:H>=37?3:H<=16?4:1;
    const under=H<=16?4:H>=37?3:2;
    for(let y=0;y<=H;y++){
      let id;
      if(y===0||y<H-3)id=3;
      else if(y<H)id=under;
      else id=surf;
      data[(y*16+lz)*16+lx]=id;
    }
    for(let y=3;y<H-1;y++){
      if(noise3(x*0.09,y*0.09,z*0.09)>0.67)data[(y*16+lz)*16+lx]=0;
    }
  }
  // trees
  for(let lx=2;lx<14;lx++)for(let lz=2;lz<14;lz++){
    const H=heights[lz*16+lx];
    const si=(H*16+lz)*16+lx;
    if(data[si]!==1)continue;
    const x=cx*16+lx, z=cz*16+lz;
    if(hash2(x,z)>=0.02)continue;
    const put=(xx,yy,zz,id,airOnly)=>{
      if(yy<0||yy>=HEIGHT)return;
      const i=(yy*16+zz)*16+xx;
      if(airOnly&&data[i]!==0)return;
      data[i]=id;
    };
    for(let i=1;i<=4;i++)put(lx,H+i,lz,5,false);
    for(let dy=4;dy<=5;dy++)for(let dx=-2;dx<=2;dx++)for(let dz=-2;dz<=2;dz++)put(lx+dx,H+dy,lz+dz,6,true);
    for(let dx=-1;dx<=1;dx++)for(let dz=-1;dz<=1;dz++)put(lx+dx,H+6,lz+dz,6,true);
    put(lx,H+7,lz,6,true);
  }
  chunks.set(cx+','+cz,{data:data,mesh:null});
}
