smoothNoise2:

function smoothNoise2(x, z){
  const xi = Math.floor(x), zi = Math.floor(z);
  const xf = x - xi, zf = z - zi;
  const u = xf*xf*(3-2*xf), v = zf*zf*(3-2*zf);
  const a = hash2(xi,zi), b = hash2(xi+1,zi), c = hash2(xi,zi+1), d = hash2(xi+1,zi+1);
  return a*(1-u)*(1-v)+b*u*(1-v)+c*(1-u)*v+d*u*v;
}

function hash3(x,y,z){

function buildChunkMesh(c, cx, cz){
  const pos=[], nor=[], col=[];
  const color = new THREE.Color();
  for(let lx=0;lx<16;lx++)for(let lz=0;lz<16;lz++)for(let y=0;y<80;y++){
    const id=c.data[(lx*16+lz)*80+y];
    if(!id) continue;
    const wx=cx*16+lx, wz=cz*16+lz;
    color.setHex(BLOCK_COLORS[id]);
    for(const f of FACES){
      const nx=wx+f.dir[0], ny=y+f.dir[1], nz=wz+f.dir[2];
      if(getBlock(nx,ny,nz)!==0) continue;
      const shade = f.dir[1]===1?1.0 : f.dir[1]===-1?0.55 : 0.8;
      const r=color.r*shade, g=color.g*shade, b=color.b*shade;
      for(const i of [0,1,2,0,2,3]){
        const cn=f.corners[i];
        pos.push(wx+cn[0], y+cn[1], wz+cn[2]);
        nor.push(f.dir[0],f.dir[1],f.dir[2]);
        col.push(r,g,b);
      }
    }
  }
  const geo=new THREE.BufferGeometry();
  geo.setAttribute('position', new THREE.Float32BufferAttribute(pos,3));
  geo.setAttribute('normal', new THREE.Float32BufferAttribute(nor,3));
  geo.setAttribute('color', new THREE.Float32BufferAttribute(col,3));
  const mesh=new THREE.Mesh(geo, voxelMat);
  mesh.userData={cx,cz};
  scene.add(mesh);
  c.mesh=mesh;
  chunkMeshes.push(mesh);
}

  // build
  const toBuild=[];
  for(let dx=-4;dx<=4;dx++)for(let dz=-4;dz<=4;dz++){
    const cx=pcx+dx, cz=pcz+dz, c=chunks.get(cx+","+cz);
    if(c && !c.mesh && chunks.has(cx+1+","+cz)&&chunks.has(cx-1+","+cz)&&chunks.has(cx+","+(cz+1))&&chunks.has(cx+","+(cz-1)))
      toBuild.push([dx*dx+dz*dz,cx,cz]);
  }
  toBuild.sort((a,b)=>a[0]-b[0]);
  for(let i=0;i<Math.min(2,toBuild.length);i++) buildChunkMesh(chunks.get(toBuild[i][1]+","+toBuild[i][2]), toBuild[i][1], toBuild[i][2]);
  // remove
  for(const [key,c] of Array.from(chunks)){
    const parts=key.split(","); const cx=+parts[0], cz=+parts[1];
    if(Math.max(Math.abs(cx-pcx),Math.abs(cz-pcz))>7){
      if(c.mesh){ scene.remove(c.mesh); c.mesh.geometry.dispose(); const i=chunkMeshes.indexOf(c.mesh); if(i>=0) chunkMeshes.splice(i,1); c.mesh=null; }
      chunks.delete(key);
    }
  }
}
