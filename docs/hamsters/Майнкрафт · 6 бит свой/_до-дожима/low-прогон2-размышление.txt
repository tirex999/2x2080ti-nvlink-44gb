2D value noise:
```javascript
function noise2D(x, z) {
    const xi = Math.floor(x);
    const zi = Math.floor(z);
    const xf = x - xi;
    const zf = z - zi;
    
    const sx = smoothstep(xf);
    const sz = smoothstep(zf);
    
    const v00 = hash2D(xi, zi);
    const v10 = hash2D(xi + 1, zi);
    const v01 = hash2D(xi, zi + 1);
    const v11 = hash2D(xi + 1, zi + 1);
    
    const a = v00 + sx * (v10 - v00);
    const b = v01 + sx * (v11 - v01);
    return a + sz * (b - a);
}
```

Fractal 2D (4 octaves):
```javascript
function fractal2D(x, z) {
    let val = 0;
    let amp = 1;
    let freq = 1;
    let max = 0;
    for (let i = 0; i < 4; i++) {
        val += noise2D(x * freq, z * freq) * amp;
        max += amp;
        amp *= 0.5;
        freq *= 2;
    }
    return val / max;
}
```

```javascript
function generateChunkData(cx, cz) {
    const data = new Uint8Array(CHUNK_SIZE * CHUNK_HEIGHT * CHUNK_SIZE);

            const wz = cz * CHUNK_SIZE + lz;

            const h = fractal2D(wx * 0.02, wz * 0.02);

            let surfaceId, subSurfaceId;
            if (H >= 46) {
                surfaceId = 7; // snow
                subSurfaceId = 3; // stone
            } else if (H >= 37) {
                surfaceId = 3; // stone
                subSurfaceId = 3; // stone
            } else if (H <= 16) {
                surfaceId = 4; // sand
                subSurfaceId = 4; // sand
            } else {
                surfaceId = 1; // grass
                subSurfaceId = 2; // dirt
            }
            
            for (let y = 0; y <= H && y < CHUNK_HEIGHT; y++) {

                if (y === 0) {
                    id = 3; // unbreakable stone
                } else if (y === H) {
                    id = surfaceId;
                } else if (y >= H - 2) {
                    id = subSurfaceId;
                } else {
                    id = 3; // stone

        }

                }
            }
```

```javascript
                    const ny = y + face.dir[1];
                    const nz = wz + face.dir[2];
```

```javascript
                    const vi = positions.length / 3;
                    for (let i = 0; i < 4; i++) {
                        const c = face.corners[i];
                        positions.push(wx + c[0], y + c[1], wz + c[2]);
                        normals.push(face.dir[0], face.dir[1], face.dir[2]);
                        colors.push(
                            color[0] * face.bright,
                            color[1] * face.bright,
                            color[2] * face.bright
                        );
                    }
                    indices.push(vi, vi+1, vi+2, vi, vi+2, vi+3);
                }
            }
        }
    }
    
    if (indices.length === 0) {
        chunk.mesh = null;
        return;
    }
    
    const geometry = new THREE.BufferGeometry();
    geometry.setAttribute('position', new THREE.Float32BufferAttribute(positions, 3));
    geometry.setAttribute('normal', new THREE.Float32BufferAttribute(normals, 3));
    geometry.setAttribute('color', new THREE.Float32BufferAttribute(colors, 3));
    geometry.setIndex(indices);
    
    const material = sharedMaterial; // MeshLambertMaterial with vertexColors: true
    const mesh = new THREE.Mesh(geometry, material);
    scene.add(mesh);
    chunkMeshes.push(mesh);
    chunk.mesh = mesh;
```

```javascript
function collides(px, py, pz) {
    // Check if player box at (px, py, pz) overlaps any solid block
    // Player box: x from px-HW to px+HW, y from py to py+HEIGHT, z from pz-HW to pz+HW
    const minX = Math.floor(px - PLAYER_HALF_W);
    const maxX = Math.floor(px + PLAYER_HALF_W);
    const minY = Math.floor(py);
    const maxY = Math.floor(py + PLAYER_HEIGHT);
    const minZ = Math.floor(pz - PLAYER_HALF_W);
    const maxZ = Math.floor(pz + PLAYER_HALF_W);
    
    for (let x = minX; x <= maxX; x++) {
        for (let y = minY; y <= maxY; y++) {
            for (let z = minZ; z <= maxZ; z++) {
                if (getBlock(x, y, z) !== 0) return true;
            }
        }
    }
    return false;
}
```

```javascript
function updatePlayer(dt) {
    // Apply gravity
    player.vy -= GRAVITY * dt;
    
    // Movement direction
    let mx = 0, mz = 0;
    if (keys['KeyW']) mz -= 1;
    if (keys['KeyS']) mz += 1;
    if (keys['KeyA']) mx -= 1;
    if (keys['KeyD']) mx += 1;
    
    // Normalize and apply yaw
    const len = Math.sqrt(mx*mx + mz*mz);
    if (len > 0) {
        mx /= len;
        mz /= len;
        const sin = Math.sin(player.yaw);
        const cos = Math.cos(player.yaw);
        const dx = mx * cos - mz * sin;
        const dz = mx * sin + mz * cos;
        player.vx = dx * MOVE_SPEED;
        player.vz = dz * MOVE_SPEED;
    } else {
        player.vx = 0;
        player.vz = 0;
    }
    
    // Jump
    if (keys['Space'] && player.onGround) {
        player.vy = JUMP_VEL;
        player.onGround = false;
    }
    
    // Move X
    let newX = player.x + player.vx * dt;
    if (!collides(newX, player.y, player.z)) {
        player.x = newX;
    } else {
        player.vx = 0;
    }
    
    // Move Z
    let newZ = player.z + player.vz * dt;
    if (!collides(player.x, player.y, newZ)) {
        player.z = newZ;
    } else {
        player.vz = 0;
    }
    
    // Move Y
    let newY = player.y + player.vy * dt;
    if (!collides(player.x, newY, player.z)) {
        player.y = newY;
        player.onGround = false;
    } else {
        if (player.vy < 0) {
            player.onGround = true;
        }
        player.vy = 0;
    }
    
    // Fall check
    if (player.y < -20) {
        player.x = 8;
        player.y = 50;
        player.z = 8;
        player.vx = 0;
        player.vy = 0;
        player.vz = 0;
    }
    
    // Update camera
    camera.position.set(player.x, player.y + EYE_HEIGHT, player.z);
    camera.rotation.order = 'YXZ';
    camera.rotation.y = player.yaw;
    camera.rotation.x = player.pitch;
}
```
