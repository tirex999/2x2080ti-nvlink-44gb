function moveToward(h,target,dt,speed) {
 const dx=target.x-h.position.x, dz=target.z-h.position.z;
 const dist=Math.hypot(dx,dz);
 if(dist<.05) return false;
 const nx=dx/dist, nz=dz/dist;
 // smooth rotate
 const desired=Math.atan2(nx,nz);
 let diff=desired - h.rotation.y;
 while(diff>Math.PI) diff-=Math.PI*2;
 while(diff<-Math.PI) diff+=Math.PI*2;
 h.rotation.y += diff*Math.min(1, dt*8);
 h.position.x += nx*speed*dt;
 h.position.z += nz*speed*dt;
 return true;
}
