```javascript
function smoothstep(t) { return t * t * (3 - 2 * t); }
function noise2D(x, y) {
    let ix = Math.floor(x), iy = Math.floor(y);
    let fx = x - ix, fy = y - iy;
    fx = smoothstep(fx); fy = smoothstep(fy);
    let a = hash(ix, iy, 0), b = hash(ix+1, iy, 0);
    let c = hash(ix, iy+1, 0), d = hash(ix+1, iy+1, 0);
    return a + (b-a)*fx + (c-a)*fy + (a-b-c+d)*fx*fy;
}
```

```javascript
function fractal2D(x, y) {
    let sum = 0, amp = 1, freq = 1, max = 0;
    for (let i = 0; i < 4; i++) {
        sum += noise2D(x * freq, y * freq) * amp;
        max += amp;
        amp *= 0.5;
        freq *= 2;
    }
    return sum / max;
}
```

```javascript
function noise3D(x, y, z) {
    let ix = Math.floor(x), iy = Math.floor(y), iz = Math.floor(z);
    let fx = x - ix, fy = y - iy, fz = z - iz;
    fx = smoothstep(fx); fy = smoothstep(fy); fz = smoothstep(fz);
    // 8 corners
    let n000 = hash(ix, iy, iz), n100 = hash(ix+1, iy, iz);
    let n010 = hash(ix, iy+1, iz), n110 = hash(ix+1, iy+1, iz);
    let n001 = hash(ix, iy, iz+1), n101 = hash(ix+1, iy, iz+1);
    let n011 = hash(ix, iy+1, iz+1), n111 = hash(ix+1, iy+1, iz+1);
    // Trilinear interpolation
    let nx00 = n000 + (n100 - n000) * fx;
    let nx10 = n010 + (n110 - n010) * fx;
    let nx01 = n001 + (n101 - n001) * fx;
    let nx11 = n011 + (n111 - n011) * fx;
    let nxy0 = nx00 + (nx10 - nx00) * fy;
    let nxy1 = nx01 + (nx11 - nx01) * fy;
    return nxy0 + (nxy1 - nxy0) * fz;
}
```
