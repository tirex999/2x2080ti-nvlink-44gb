For 2D noise with smoothstep interpolation:
```javascript
function smoothstep(t) { return t * t * (3 - 2 * t); }
function noise2D(x, y) {
    let ix = Math.floor(x), iy = Math.floor(y);
    let fx = x - ix, fy = y - iy;
    fx = smoothstep(fx); fy = smoothstep(fy);
    let a = hash(ix, iy), b = hash(ix+1, iy), c = hash(ix, iy+1), d = hash(ix+1, iy+1);
    return a + (b-a)*fx + (c-a)*fy + (a-b-c+d)*fx*fy;
}
```

For fractal with 4 octaves:
```javascript
function fractal2D(x, y) {
    let sum = 0, amp = 1, freq = 1, total = 0;
    for (let i = 0; i < 4; i++) {
        sum += noise2D(x * freq, y * freq) * amp;
        total += amp;
        amp *= 0.5;
        freq *= 2;
    }
    return sum / total;
}
```
