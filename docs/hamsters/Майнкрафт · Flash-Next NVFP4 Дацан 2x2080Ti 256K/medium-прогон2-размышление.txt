```js
function noise2(x,z){ // smooth value noise at integer lattice
  const xi=Math.floor(x), zi=Math.floor(z);
  const xf=x-xi, zf=z-zi;
  const u=xf*xf*(3-2*xf), v=zf*zf*(3-2*zf);
  const a=hash2(xi,zi), b=hash2(xi+1,zi), c=hash2(xi,zi+1), d=hash2(xi+1,zi+1);
  return a+(b-a)*u+(c-a)*v+(a-b-c+d)*u*v;
}
function fbm2(x,z){
  let amp=1, freq=1, sum=0, norm=0;
  for(let i=0;i<4;i++){ sum+=amp*noise2(x*freq,z*freq); norm+=amp; amp*=0.5; freq*=2; }
  return sum/norm;
}
```

```js
function updateWorld(){
  const pcx=Math.floor(player.pos.x/16), pcz=Math.floor(player.pos.z/16);
  // generate
  const need=[];
  for(let dx=-5;dx<=5;dx++)for(let dz=-5;dz<=5;dz++){
    const cx=pcx+dx,cz=pcz+dz,k=cx+','+cz;
    if(!chunks.has(k)) need.push([dx*dx+dz*dz,cx,cz]);
  }
  need.sort((a,b)=>a[0]-b[0]);
  for(let i=0;i<need.length && i<4;i++){
    const k=need[i][1]+','+need[i][2];
    chunks.set(k,{cx:need[i][1],cz:need[i][2],data:genChunk(need[i][1],need[i][2]),mesh:null});
  }
  // mesh
  const cand=[];
  for(let dx=-4;dx<=4;dx++)for(let dz=-4;dz<=4;dz++){
    const cx=pcx+dx,cz=pcz+dz,ch=chunks.get(cx+','+cz);
    if(ch&&ch.data&&!ch.mesh && chunks.has((cx+1)+','+cz)&&chunks.has((cx-1)+','+cz)&&chunks.has(cx+','+(cz+1))&&chunks.has(cx+','+(cz-1))) cand.push([dx*dx+dz*dz,cx,cz]);
  }
  cand.sort((a,b)=>a[0]-b[0]);
  for(let i=0;i<cand.length&&i<2;i++) buildMesh(cand[i][1],cand[i][2]);
  // unload
  for(const [k,ch] of chunks){
    if(Math.max(Math.abs(ch.cx-pcx),Math.abs(ch.cz-pcz))>7){
      removeChunk(k,ch);
    }
  }
}
```
