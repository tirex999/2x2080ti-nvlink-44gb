function buildChunkMesh(chunk) { const cx = ..., cz = ...; const positions = []; const normals = []; const colors = []; const data = chunk.data; for (let ly=0; ly<80; ly++) { for (let lz=0; lz<16; lz++) { for (let lx=0; lx<16; lx++) { const idx = (ly*16 + lz)*16 + lx; const id = data[idx]; if (id === 0) continue; const wx = cx*16 + lx; const wy = ly; const wz = cz*16 + lz; for (const face of FACES) { const n = face[1]; const nx = wx+n[0], ny = wy+n[1], nz = wz+n[2]; if (getBlock(nx,ny,nz) === 0) { // add face const col = blockColors[id]; const light = face[2]; const r = (col>>16)&255, g=(col>>8)&255, b=col&255; for (const corner of face[3]) { positions.push(wx+corner[0], wy+corner[1], wz+corner[2]); normals.push(n[0],n[1],n[2]); colors.push(r*light/255, g*light/255, b*light/255); } } } } } } const geo = new THREE.BufferGeometry(); geo.setAttribute('position', new THREE.Float32BufferAttribute(positions,3)); geo.setAttribute('normal', new THREE.Float32BufferAttribute(normals,3)); geo.setAttribute('color', new THREE.Float32BufferAttribute(colors,3)); const mat = blockMaterial; // shared const mesh = new THREE.Mesh(geo, mat); mesh.position.set(0,0,0); // at origin (chunk local) return mesh; }
