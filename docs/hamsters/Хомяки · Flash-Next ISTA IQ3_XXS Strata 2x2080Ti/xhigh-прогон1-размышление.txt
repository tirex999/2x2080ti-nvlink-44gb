function updateHamster(h, dt, time) {
 const u = h.userData;
 u.timer -= dt;
 const pos = h.position;

 // bowl obstacle if not seeking/eating
 if (u.state !== 'seekBowl' && u.state !== 'eat') {
   const dx = pos.x - bowlTarget.x; const dz = pos.z - bowlTarget.z; const dist = Math.hypot(dx,dz);
   if (dist < 1.35) {
     const away = Math.atan2(dx,dz);
     u.heading = angleLerp(u.heading, away, 4*dt);
     // push out slightly
     if (dist < 1.2) { pos.x += (dx/dist)*0.5*dt; pos.z += (dz/dist)*0.5*dt; }
   }
 }

 if (u.state === 'wander') {
   const dir = new THREE.Vector3(Math.sin(u.heading),0,Math.cos(u.heading));
   pos.addScaledVector(dir, u.speed*dt);
   // boundary
   if (Math.abs(pos.x) > bounds.x || Math.abs(pos.z) > bounds.z) {
      const toCenter = new THREE.Vector3(-pos.x,0,-pos.z);
      u.heading = angleLerp(u.heading, Math.atan2(toCenter.x, toCenter.z), 8*dt);
   }
   // separation
   for (const other of hamsters) if (other !== h) {
      const dx = pos.x - other.position.x; const dz = pos.z - other.position.z; const d = Math.hypot(dx,dz);
      if (d < 1.0 && d > 0.001) {
         const away = Math.atan2(dx,dz);
         u.heading = angleLerp(u.heading, away, 3*dt);
         pos.x += (dx/d)*0.35*dt; pos.z += (dz/d)*0.35*dt;
      }
   }
   if (u.timer <= 0) {
      const r = Math.random();
      if (r < 0.22) { u.state='pause'; u.timer=0.7+Math.random()*1.4; }
      else if (r < 0.42) { u.heading += (Math.random()-0.5)*1.8; u.timer=1+Math.random()*2; }
      else if (r < 0.62) { u.state='seekBowl'; u.timer=10; }
      else { u.timer=1+Math.random()*2; }
   }
 } else if (u.state === 'pause') {
   // slight turn
   u.heading += Math.sin(time*0.7+u.phase)*0.15*dt;
   if (u.timer <= 0) { u.state='wander'; u.timer=1+Math.random()*2; }
 } else if (u.state === 'seekBowl') {
   const to = new THREE.Vector3(bowlTarget.x - pos.x, 0, bowlTarget.z - pos.z);
   const dist = to.length();
   if (dist > 0.001) {
      const desired = Math.atan2(to.x, to.z);
      u.heading = angleLerp(u.heading, desired, 4*dt);
      if (dist > 1.35) {
         to.normalize();
         pos.addScaledVector(to, u.speed*1.15*dt);
      } else {
         u.state='eat'; u.timer=2+Math.random()*2.5;
      }
   } else { u.state='eat'; u.timer=2; }
   if (u.timer <= 0) { u.state='wander'; u.timer=1+Math.random()*2; }
 } else if (u.state === 'eat') {
   // stand near bowl, maybe face bowl
   const to = new THREE.Vector3(bowlTarget.x - pos.x, 0, bowlTarget.z - pos.z);
   if (to.lengthSq() > 0.01) u.heading = angleLerp(u.heading, Math.atan2(to.x,to.z), 3*dt);
   if (u.timer <= 0) { u.state='wander'; u.timer=1+Math.random()*2; }
 }

 // clamp
 pos.x = Math.max(-bounds.x, Math.min(bounds.x, pos.x));
 pos.z = Math.max(-bounds.z, Math.min(bounds.z, pos.z));
 pos.y = 0;

 // face
 h.rotation.y = angleLerp(h.rotation.y, u.heading, 6*dt);

 // animation
 const moving = (u.state === 'wander' || u.state === 'seekBowl') && u.speed > 0.05;
 const walkPhase = time * (6 + u.speed*4) + u.phase;
 const bob = moving ? Math.sin(walkPhase)*0.045 : Math.sin(time*2.2+u.phase)*0.012;
 u.body.position.y = 0.58 + bob;
 u.head.position.y = 0.64 + bob*1.15;
 u.tail.position.y = 0.48 + bob*0.5;

 const legOffsets = [0, Math.PI, Math.PI, 0];
 for (let i=0;i<u.legs.length;i++) {
   const leg = u.legs[i];
   if (moving) leg.rotation.x = Math.sin(walkPhase + legOffsets[i]) * 0.75;
   else leg.rotation.x = Math.sin(time*2.5 + u.phase + i) * 0.06;
 }

 u.leftEar.rotation.z = 0.35 + Math.sin(time*3.1+u.phase)*0.14;
 u.rightEar.rotation.z = -0.35 - Math.sin(time*3.1+u.phase+0.7)*0.14;
 u.tail.rotation.y = Math.sin(time*4.2+u.phase)*0.25;

 if (u.state === 'eat') {
   u.head.position.y += Math.sin(time*11+u.phase)*0.05;
   u.head.rotation.x = Math.sin(time*11+u.phase)*0.18;
   u.leftCheek.scale.setScalar(1 + Math.sin(time*8+u.phase)*0.08);
   u.rightCheek.scale.setScalar(1 + Math.sin(time*8+u.phase+0.4)*0.08);
 } else {
   u.head.rotation.x = Math.sin(time*1.5+u.phase)*0.04;
   u.leftCheek.scale.setScalar(1);
   u.rightCheek.scale.setScalar(1);
 }
}

function angleLerp(a,b,t) {
 let diff = b - a;

 while (diff < -Math.PI) diff += Math.PI*2;
 return a + diff * Math.min(1, t);

function angleLerp(a, b, t) {
  let diff = b - a;
  while (diff > Math.PI) diff -= Math.PI * 2;
  while (diff < -Math.PI) diff += Math.PI * 2;
  return a + diff * Math.min(1, t);
}

function applySeparation(h, dt) {
  const pos = h.position;
  for (const other of hamsters) {
    if (other === h) continue;
    const dx = pos.x - other.position.x;
    const dz = pos.z - other.position.z;
    const d = Math.hypot(dx, dz);
    if (d < 1.0 && d > 0.001) {
      const push = (1.0 - d) * 0.8 * dt;
      pos.x += (dx / d) * push;
      pos.z += (dz / d) * push;
      if (h.userData.state === 'wander' || h.userData.state === 'seekBowl') {
        h.userData.heading = angleLerp(h.userData.heading, Math.atan2(dx, dz), 2 * dt);
      }
    }
  }
}

function updateHamster(h, dt, time) {
  const u = h.userData;
  const pos = h.position;
  u.timer -= dt;

  if (u.state !== 'seekBowl' && u.state !== 'eat') {
    const dx = pos.x - bowlTarget.x;
    const dz = pos.z - bowlTarget.z;
    const dist = Math.hypot(dx, dz);
    if (dist < 1.35 && dist > 0.001) {
      u.heading = angleLerp(u.heading, Math.atan2(dx, dz), 4 * dt);
      if (dist < 1.2) {
        pos.x += (dx / dist) * 0.5 * dt;
        pos.z += (dz / dist) * 0.5 * dt;
      }
    }
  }

  if (u.state === 'wander') {
    const dir = new THREE.Vector3(Math.sin(u.heading), 0, Math.cos(u.heading));
    pos.addScaledVector(dir, u.speed * dt);

    if (Math.abs(pos.x) > bounds.x || Math.abs(pos.z) > bounds.z) {
      const toCenter = new THREE.Vector3(-pos.x, 0, -pos.z);
      u.heading = angleLerp(u.heading, Math.atan2(toCenter.x, toCenter.z), 8 * dt);
    }

    if (u.timer <= 0) {
      const r = Math.random();
      if (r < 0.22) {
        u.state = 'pause';
        u.timer = 0.7 + Math.random() * 1.4;
      } else if (r < 0.42) {
        u.heading += (Math.random() - 0.5) * 1.8;
        u.timer = 1 + Math.random() * 2;
      } else if (r < 0.62) {
        u.state = 'seekBowl';
        u.timer = 10;
      } else {
        u.timer = 1 + Math.random() * 2;
      }
    }
  } else if (u.state === 'pause') {
    u.heading += Math.sin(time * 0.7 + u.phase) * 0.15 * dt;
    if (u.timer <= 0) {
      u.state = 'wander';
      u.timer = 1 + Math.random() * 2;
    }
  } else if (u.state === 'seekBowl') {
    const to = new THREE.Vector3(bowlTarget.x - pos.x, 0, bowlTarget.z - pos.z);
