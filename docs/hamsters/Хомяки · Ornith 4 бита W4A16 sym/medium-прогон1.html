javascript function createCage() { const cage = new THREE.Group(); const trayMat = new THREE.MeshStandardMaterial({ color: 0xfdf3e7, roughness: 0.9, flatShading: true }); // Bottom const bottom = new THREE.Mesh(new THREE.BoxGeometry(6.6, 0.3, 6.6), trayMat); bottom.position.y = 0.15; bottom.receiveShadow = true; cage.add(bottom); // Lower solid walls (height 0.7) const lowH = 0.7; const t = 0.2; const walls = [ { w: 6.6, d: t, x: 0, z: 3.2 }, { w: 6.6, d: t, x: 0, z: -3.2 }, { w: t, d: 6.6, x: 3.2, z: 0 }, { w: t, d: 6.6, x: -3.2, z: 0 }, ]; walls.forEach(w => { const wall = new THREE.Mesh(new THREE.BoxGeometry(w.w, lowH, w.d), trayMat); wall.position.set(w.x, 0.3 + lowH/2, w.z); wall.receiveShadow = true; wall.castShadow = true; cage.add(wall); }); // Transparent upper walls (height 1.8) const glassMat = new THREE.MeshPhysicalMaterial({ color: 0xcdefff, transparent: true, opacity: 0.14, roughness: 0.05, metalness: 0, transmission: 0.2 }); const glassH = 1.8; const glassY = 0.3 + lowH + glassH/2; const glassWalls = [ { w: 6.6, d: 0.1, x: 0, z: 3.2 }, { w: 6.6, d: 0.1, x: 0, z: -3.2 }, { w: 0.1, d: 6.6, x: 3.2, z: 0 }, { w: 0.1, d: 6.6, x: -3.2, z: 0 }, ]; glassWalls.forEach(w => { const g = new THREE.Mesh(new THREE.BoxGeometry(w.w, glassH, w.d), glassMat); g.position.set(w.x, glassY, w.z); cage.add(g); }); // Top bar frame (metal) const barMat = new THREE.MeshStandardMaterial({ color: 0xd4d4d4, metalness: 0.7, roughness: 0.4 }); const barR = 0.05; const topY = 0.3 + lowH + glassH; // top frame edges const frame = [ { len: 6.6, x: 0, z: 3.2, rotZ: 0, axis: 'x' }, { len: 6.6, x: 0, z: -3.2, rotZ: 0, axis: 'x' }, { len: 6.6, x: 3.2, z: 0, rotZ: 0, axis: 'z' }, { len: 6.6, x: -3.2, z: 0, rotZ: 0, axis: 'z' }, ]; frame.forEach(f => { const bar = new THREE.Mesh(new THREE.CylinderGeometry(barR, barR, f.len, 8), barMat); bar.position.set(f.x, topY, f.z); if (f.axis === 'x') bar.rotation.z = Math.PI/2; else bar.rotation.x = Math.PI/2; cage.add(bar); }); // Vertical bars on top frame for (let i = -3; i <= 3; i++) { // along front/back const vb = new THREE.Mesh(new THREE.CylinderGeometry(barR, barR, 0.5, 8), barMat); vb.position.set(i * 0.9, topY - 0.25, 3.2); cage.add(vb); const vb2 = vb.clone(); vb2.position.z = -3.2; cage.add(vb2); } for (let i = -3; i <= 3; i++) { const vb = new THREE.Mesh(new THREE.CylinderGeometry(barR, barR, 0.5, 8), barMat); vb.position.set(3.2, topY - 0.25, i * 0.9); cage.add(vb); const vb2 = vb.clone(); vb2.position.x = -3.2; cage.add(vb2); } return cage; }
