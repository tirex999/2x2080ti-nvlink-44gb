function makeHamster(bodyColor, bellyColor, opts = {}) {
  const g = new THREE.Group();
  const bodyM = new THREE.MeshStandardMaterial({ color: bodyColor, roughness: .75, flatShading: true });
  const bellyM = new THREE.MeshStandardMaterial({ color: bellyColor, roughness: .75, flatShading: true });
  const darkM = new THREE.MeshStandardMaterial({ color: 0x2a2018, roughness: .4, flatShading: true });
  const pinkM = new THREE.MeshStandardMaterial({ color: 0xff9fb0, roughness: .5, flatShading: true });
  const noseM = new THREE.MeshStandardMaterial({ color: 0xff7f9c, roughness: .5, flatShading: true });

  // body
  const body = new THREE.Mesh(new THREE.SphereGeometry(1, 14, 10), bodyM);
  body.scale.set(1.28, 0.95, 0.92); body.castShadow = true; g.add(body);
  // belly
  const belly = new THREE.Mesh(new THREE.SphereGeometry(0.72, 12, 8), bellyM);
  belly.scale.set(1.05, 0.78, 0.88); belly.position.set(0.14, -0.32, 0); g.add(belly);
  // head
  const head = new THREE.Mesh(new THREE.SphereGeometry(0.62, 14, 10), bodyM);
  head.position.set(1.02, 0.24, 0); head.castShadow = true; g.add(head);
  // cheeks
  for (const s of [-1, 1]) {
    const ck = new THREE.Mesh(new THREE.SphereGeometry(0.34, 10, 8), bodyM);
    ck.position.set(0.82, 0.06, s * 0.46); g.add(ck);
  }
  // ears
  for (const s of [-1, 1]) {
    const ear = new THREE.Mesh(new THREE.CylinderGeometry(0.19, 0.22, 0.22, 8), bodyM);
    ear.rotation.z = Math.PI / 2; ear.position.set(0.92, 0.74, s * 0.34); g.add(ear);
    const i = new THREE.Mesh(new THREE.CylinderGeometry(0.11, 0.12, 0.24, 8), pinkM);
    i.rotation.z = Math.PI / 2; i.position.set(0.92, 0.74, s * 0.34); g.add(i);
  }
  // eyes
  const eyeG = new THREE.Group();
  for (const s of [-1, 1]) {
    const w = new THREE.Mesh(new THREE.SphereGeometry(0.15, 10, 8), new THREE.MeshStandardMaterial({ color: 0xffffff, flatShading: true }));
    w.position.set(1.42, 0.34, s * 0.27); g.add(w);
    const e = new THREE.Mesh(new THREE.SphereGeometry(0.105, 10, 8), darkM);
    e.position.set(1.52, 0.34, s * 0.27); g.add(e);
    const gl = new THREE.Mesh(new THREE.SphereGeometry(0.035, 6, 6),
      new THREE.MeshStandardMaterial({ color: 0xffffff, emissive: 0xffffff, emissiveIntensity: .3 }));
    gl.position.set(1.56, 0.38, s * 0.24); g.add(gl);
    eyeG.add(w, e, gl);
  }
  // nose
  const nose = new THREE.Mesh(new THREE.SphereGeometry(0.085, 8, 6), noseM);
  nose.position.set(1.6, 0.2, 0); g.add(nose);
  // tiny feet
  const footGeo = new THREE.SphereGeometry(0.14, 8, 6);
  const feet = [];
  const fp = [[0.5, -0.72, 0.38],[0.5, -0.72, -0.38],[-0.45, -0.72, 0.38],[-0.45, -0.72, -0.38]];

  }
  // tail
  const tail = new THREE.Mesh(new THREE.SphereGeometry(0.13, 8, 6), bellyM);
  tail.position.set(-1.3, -0.05, 0); g.add(tail);

  g.userData.parts = { head, eyeG, body, belly, nose, feet, tail };
  return g;
}

function buildWheel() {
  const w = new THREE.Group();
  const rimM = new THREE.MeshStandardMaterial({ color: 0xff7043, roughness: .5, flatShading: true });
  const rungM = new THREE.MeshStandardMaterial({ color: 0xffb74d, roughness: .6, flatShading: true });
  const hubM = new THREE.MeshStandardMaterial({ color: 0xef5350, roughness: .5, flatShading: true });
  const spokeM = new THREE.MeshStandardMaterial({ color: 0x90a4ae, roughness: .6, flatShading: true });

  const R = 1.9, nRung = 14;
  const spin = new THREE.Group();
  for (let i = 0; i < nRung; i++) {
    const a = (i / nRung) * Math.PI * 2;
    const r = new THREE.Mesh(new THREE.CylinderGeometry(0.06, 0.06, 0.75, 6), rungM);
    r.rotation.x = Math.PI / 2;
    r.position.set(Math.cos(a) * R, Math.sin(a) * R, 0);
    spin.add(r);
  }
  for (const s of [-1, 1]) {
    const rim = new THREE.Mesh(new THREE.TorusGeometry(R, 0.11, 8, 24), rimM);
    rim.position.z = s * 0.42; spin.add(rim);
    const disk = new THREE.Mesh(new THREE.CylinderGeometry(R, R, 0.06, 24), rimM);
    disk.rotation.x = Math.PI / 2; disk.position.z = s * 0.42; spin.add(disk);
  }
  const hub = new THREE.Mesh(new THREE.CylinderGeometry(0.22, 0.22, 0.9, 10), hubM);
  hub.rotation.x = Math.PI / 2; spin.add(hub);
  w.add(spin);

  // stand
  const stand = new THREE.Group();
  for (const s of [-1, 1]) {
    const leg = new THREE.Mesh(new THREE.BoxGeometry(0.16, 2, 0.16), spokeM);
    leg.position.set(s * 0.45, 1, 0); stand.add(leg);
    const base = new THREE.Mesh(new THREE.BoxGeometry(0.9, 0.14, 0.55), spokeM);
    base.position.set(s * 0.45, 0.07, 0); stand.add(base);
  }
  w.add(stand);

  w.position.set(-4.2, 2.05, -3.8);
  w.userData.spin = spin;
  w.userData.isWheel = true;
  scene.add(w);
  return w;
}

function updateHamster(hs, dt) {
  const u = hs.userData, p = hs.userData.parts;

  if (u.state === 'idle') {
    u.timer -= dt; u.headBob += dt * 1.5;
    p.head.rotation.y = Math.sin(u.headBob) * 0.25;

    return;

  u.timer -= dt; u.retargetTimer -= dt;

  if (u.state === 'run' && pellets.length) {
    let near = null, nd = Infinity;
    for (const pl of pellets) {
      const d = pl.position.distanceTo(hs.position);
      if (d < nd) { nd = d; near = pl; }
    }
    if (near) {
      if (nd < 0.7) {
        const idx = pellets.indexOf(near);
        if (idx >= 0) { pellets.splice(idx, 1); scene.remove(near); }
        u.state = 'idle'; u.timer = rnd(1.2, 2.4); u.headBob = Math.random() * 6;
        showToast(`${u.name} is munching! 😋`);
        spawnParticles(near.position, 0xffe082);
        return;
      }
      u.dir = Math.atan2(near.position.x - hs.position.x, near.position.z - hs.position.z);
      u.speed = 2.4;
    } else { u.state = 'wander'; u.timer = rnd(1.5, 3); }
  }

  if (u.state === 'wander') {
    if (u.timer <= 0) {
      if (Math.random() < 0.28) { u.state = 'idle'; u.timer = rnd(2, 4.5); }
      else u.dir += rnd(-1.4, 1.4);
    }
    if (u.retargetTimer <= 0) { u.tx = rnd(-PLAY, PLAY); u.tz = rnd(-PLAY, PLAY); u.retargetTimer = rnd(2, 4); }
    const dx = u.tx - hs.position.x, dz = u.tz - hs.position.z;
    const dist = Math.hypot(dx, dz);
    if (dist > 0.4) u.dir = Math.atan2(dx, dz); else u.timer = rnd(0.5, 2);
  }

  const spd = u.speed;
  hs.position.x += Math.sin(u.dir) * spd * dt;
  hs.position.z += Math.cos(u.dir) * spd * dt;

  const lim = PLAY + 0.3;
  if (hs.position.x > lim || hs.position.x < -lim || hs.position.z > lim || hs.position.z < -lim) {
    u.dir = rnd(0, Math.PI * 2);
  }

  let dh = u.dir - hs.rotation.y;
  while (dh > Math.PI) dh -= Math.PI * 2;
  while (dh < -Math.PI) dh += Math.PI * 2;
  hs.rotation.y += dh * Math.min(1, dt * 6);

  u.walkPhase += dt * spd * 7;
  const sw = Math.sin(u.walkPhase), s = u.walkScale;
  p.feet[0].position.y = -0.72 * s + Math.max(0, sw) * 0.16 * s;
  p.feet[1].position.y = -0.72 * s + Math.max(0, -sw) * 0.16 * s;
  p.feet[2].position.y = -0.72 * s + Math.max(0, -sw) * 0.16 * s;
  p.feet[3].position.y = -0.72 * s + Math.max(0, sw) * 0.16 * s;
  p.feet[0].position.x += Math.sin(u.dir) * sw * 0.06;
  p.feet[1].position.x += Math.sin(u.dir) * sw * 0.06;
  p.feet[2].position.x -= Math.sin(u.dir) * sw * 0.06;
  p.feet[3].position.x -= Math.sin(u.dir) * sw * 0.06;
  p.feet.forEach(f => f.scale.y = 0.6 + Math.max(0, Math.sin(u.walkPhase)) * 0.5);

  p.head.position.y = 0.24 * s + Math.abs(Math.sin(u.walkPhase)) * 0.04;
  p.head.rotation.y = Math.sin(u.walkPhase * 0.5) * 0.12;
  p.body.position.y = Math.abs(Math.sin(u.walkPhase)) * 0.045;
  p.belly.position.y = -0.32 * s + Math.abs(Math.sin(u.walkPhase)) * 0.03;

renderer.domElement.addEventListener('pointerdown', (e) => {
  mouse.x = (e.clientX / innerWidth) * 2 - 1;
  mouse.y = -(e.clientY / innerHeight) * 2 + 1;
  raycaster.setFromCamera(mouse, camera);

  if (raycaster.intersectObject(apple, false).length) {
    showToast('🍎 Snacks are out! The hamsters love apples!');
    hamsters.forEach(h => { if (h.userData.state === 'idle') h.userData.pausal = false;
      h.userData.state = 'run'; h.userData.timer = rnd(1.2, 2); });
    return;
  }
  if (raycaster.intersectObject(bowl, false).length) {
    showToast('🌾 Yum! New treats scattered!');
    hamsters.forEach(h => { h.userData.state = 'run'; h.userData.timer = rnd(1, 2); });
    return;
  }

  const plane = new THREE.Plane(new THREE.Vector3(0, 1, 0), 0);
  const hit = new THREE.Vector3();
  raycaster.ray.intersectPlane(plane, hit);
  if (!hit) return;

  let best = -1, bestD = Infinity;
  hamsters.forEach((h, i) => {
    const d = h.position.distanceTo(hit);
    if (d < bestD) { bestD = d; best = i; }
  });
  const drop = (best >= 0 ? hamsters[best].position : hit).clone();
  drop.y = 0.15;
  drop.x = clamp(drop.x, -PLAY, PLAY); drop.z = clamp(drop.z, -PLAY, PLAY);

  const pellet = new THREE.Mesh(new THREE.IcosahedronGeometry(0.16, 0),
    new THREE.MeshStandardMaterial({ color: 0xcaa15a, roughness: .9, flatShading: true }));
  pellet.position.copy(drop); scene.add(pellet); pellets.push(pellet);

  if (best >= 0) {
    hamsters[best].userData.state = 'run';
    hamsters[best].userData.timer = rnd(1, 1.8);
    showToast(`${NAMES[best]} sees a tasty treat!`);
  } else showToast('😋 A treat dropped!');
  spawnParticles(drop, 0xffd54f);
});

/* ---------- Behaviour ---------- */
function updateHamster(hs, dt) {
  const u = hs.userData, p = hs.userData.parts;

  if (u.state === 'idle') {
    u.timer -= dt; u.headBob += dt * 1.5;
    p.head.rotation.y = Math.sin(u.headBob) * 0.25;
    p.head.rotation.x = Math.sin(u.headBob * 1.3) * 0.1;
    p.feet.forEach(f => f.scale.y = 0.6);
    if (u.timer <= 0) { u.state = 'wander'; u.dir = rnd(0, Math.PI * 2); u.timer = rnd(2, 5); }
    return;
  }

  u.timer -= dt; u.retargetTimer -= dt;

  if (u.state === 'run' && pellets.length) {
    let near = null, nd = Infinity;
    for (const pl of pellets) {
      const d = pl.position.distanceTo(hs.position);
      if (d < nd) { nd = d; near = pl; }
    }
    if (near) {
      if (nd < 0.7) {
        const idx = pellets.indexOf(near);
        if (idx >= 0) { pellets.splice(idx, 1); scene.remove(near); }
        u.state = 'idle'; u.timer = rnd(1.2, 2.4); u.headBob = Math.random() * 6;
        showToast(`${u.name} is munching! 😋`);
        spawnParticles(near.position, 0xffe082);
        return;
      }
      u.dir = Math.atan2(near.position.x - hs.position.x, near.position.z - hs.position.z);
      u.speed = 2.4;
    } else { u.state = 'wander'; u.timer = rnd(1.5, 3); }
  }

  if (u.state === 'wander') {
    if (u.timer <= 0) {
      if (Math.random() < 0.28) { u.state = 'idle'; u.timer = rnd(2, 4.5); }
      else u.dir += rnd(-1.4, 1.4);
    }
    if (u.retargetTimer <= 0) { u.tx = rnd(-PLAY, PLAY); u.tz = rnd(-PLAY, PLAY); u.retargetTimer = rnd(2, 4); }
    const dx = u.tx - hs.position.x, dz = u.tz - hs.position.z;
    const dist = Math.hypot(dx, dz);
    if (dist > 0.4) u.dir = Math.atan2(dx, dz); else u.timer = rnd(0.5, 2);
  }

  const spd = u.speed;
  hs.position.x += Math.sin(u.dir) * spd * dt;
  hs.position.z += Math.cos(u.dir) * spd * dt;

  const lim = PLAY + 0.3;
  if (hs.position.x > lim || hs.position.x < -lim || hs.position.z > lim || hs.position.z < -lim) {
    u.dir = rnd(0, Math.PI * 2);
  }

  let dh = u.dir - hs.rotation.y;
  while (dh > Math.PI) dh -= Math.PI * 2;
  while (dh < -Math.PI) dh += Math.PI * 2;
  hs.rotation.y += dh * Math.min(1, dt * 6);

  u.walkPhase += dt * spd * 7;
  const sw = Math.sin(u.walkPhase), s = u.walkScale;
  p.feet[0].position.y = -0.72 * s + Math.max(0, sw) * 0.16 * s;
  p.feet[1].position.y = -0.72 * s + Math.max(0, -sw) * 0.16 * s;
  p.feet[2].position.y = -0.72 * s + Math.max(0, -sw) * 0.16 * s;
  p.feet[3].position.y = -0.72 * s + Math.max(0, sw) * 0.16 * s;
  p.feet[0].position.x += Math.sin(u.dir) * sw * 0.06;
  p.feet[1].position.x += Math.sin(u.dir) * sw * 0.06;
  p.feet[2].position.x -= Math.sin(u.dir) * sw * 0.06;
  p.feet[3].position.x -= Math.sin(u.dir) * sw * 0.06;
  p.feet.forEach(f => f.scale.y = 0.6 + Math.max(0, Math.sin(u.walkPhase)) * 0.5);

  p.head.position.y = 0.24 * s + Math.abs(Math.sin(u.walkPhase)) * 0.04;
  p.head.rotation.y = Math.sin(u.walkPhase * 0.5) * 0.12;
  p.body.position.y = Math.abs(Math.sin(u.walkPhase)) * 0.045;
  p.belly.position.y = -0.32 * s + Math.abs(Math.sin(u.walkPhase)) * 0.03;
}

/* ---------- Main loop ---------- */
const clock = new THREE.Clock();
let applePhase = 0;

function animate() {
  requestAnimationFrame(animate);
  const dt = Math.min(clock.getDelta(), 0.05);
  const t = clock.elapsedTime;

  hamsters.forEach(hs => updateHamster(hs, dt));

  let someone = false;
  for (const hs of hamsters) if (hs.position.distanceTo(wheel.position) < 1.1) someone = true;
  wheel.userData.spin.rotation.x += someone ? dt * 5 : 0;

  applePhase += dt;
  apple.position.y = 0.55 + Math.sin(applePhase * 1.6) * 0.08;
  apple.rotation.y += dt * 0.6;

  for (const pl of pellets) { pl.rotation.y += dt; pl.position.y = 0.15 + Math.sin(t * 2 + pl.position.x) * 0.03; }

  for (let i = particles.length - 1; i >= 0; i--) {
    const p = particles[i];
    p.userData.life -= dt;
    p.userData.vel.y -= 6 * dt;
    p.position.addScaledVector(p.userData.vel, dt);
    p.rotation.x += dt * 4; p.rotation.y += dt * 3;
    p.scale.setScalar(Math.max(0.001, p.userData.life));
    if (p.userData.life <= 0) { scene.remove(p); particles.splice(i, 1); }
  }

  controls.update();
  renderer.render(scene, camera);
}

## What you get

**The scene** — A wooden cage with corner posts and little red finials, a blue floor tray filled with tan bedding (scattered tufts), a pitched-roof house, a water bottle, potted plants, and scattered treat pellets.

**Five hamsters** — Each built from spheres/cones with a rounded body, chubby cheeks, round ears with pink insides, big shiny eyes, a pink nose, little feet, and a stubby tail. They come in different colors and sizes (Biscuit, Mochi, Peanut, Nugget, Waffles).

**Autonomous behavior** — Each hamster cycles through three states:
- **Wander** — pick a random destination, walk toward it, periodically change direction
- **Idle** — stop, bob and twitch their head as if sniffing/looking around
- **Run** — dash at high speed (triggered by treats/snacks)

They also have full walk animation (alternating feet, body bounce, head bob) and smoothly turn to face their heading, bouncing off the cage walls.

**Interactive objects:**
- 🎡 **Exercise wheel** — actually spins when a hamster wanders into it
- 🍎 **Apple** — click it and *all* hamsters get excited and run around
- 🍥 **Food bowl** — click to scatter treats
- 🚇 **Tunnel** — decorative archway
- **Click anywhere** on the floor → a treat drops and the nearest hamster rushes over to eat it (with a little golden particle burst)
