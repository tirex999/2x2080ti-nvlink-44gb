**updateHamster details:**

```js
function updateHamster(h, dt){
  const p = h.root.position;
  const prev = h._prev; // Vector3 stored
  let movedX = p.x - prev.x, movedZ = p.z - prev.z;
  const movedH = Math.hypot(movedX, movedZ);
  prev.copy(p);

  h.stateT += dt;
  switch(h.state){
    case 'idle':
      if(h.stateT > h.idleDur) decide(h);
      break;
    case 'walk': {
      const t = h.plan.target;
      const dx = t.x - p.x, dz = t.z - p.z;
      const d = Math.hypot(dx, dz);
      if(d < 0.055){ onArrive(h); break; }
      const want = Math.atan2(dx, dz);
      h.heading = turnToward(h.heading, want, 4.0*dt);
      const sp = h.plan.speed || 0.26;
      p.x += Math.sin(h.heading)*sp*dt;
      p.z += Math.cos(h.heading)*sp*dt;
      break;
    }
    case 'wheelEnter': {
      const k = easeInOut(clamp01(h.stateT / WHEEL.enterDur));
      p.lerpVectors(h.plan.from, WHEEL.innerPoint, k);
      h.heading = lerpAngle(h.plan.fromHeading, h.plan.runHeading, k);
      // legs: slow step
      h.legPhase += (0.12*dt / h.strideLen) * TAU; // small crawl speed
      break;
    }
    case 'wheelRun': {
      // speed profile
      const dur = h.plan.dur;
      let v;
      const A = 0.9, B = 0.7; // accel, decel
      if(h.stateT < A) v = h.plan.vMax * (h.stateT/A);
      else if(h.stateT > dur - B) v = h.plan.vMax * clamp01((dur - h.stateT)/B);
      else v = h.plan.vMax;
      h.runV = v;
      p.set(WHEEL.cx, WHEEL.cy - R_KOLESA + Math.abs(Math.sin(h.legPhase))*0.007, WHEEL.cz);
      h.heading = h.plan.runHeading;
      h.legPhase += (v*dt / h.strideLen) * TAU;
      if(h.stateT >= dur){
        h.state='wheelExit'; h.stateT=0;
```

```js
  const p = h.root.position;
  const before = h._prev; // copy of previous frame position
  h.stateT += dt;

  switch(h.state){
    case 'idle':
      if(h.stateT >= h.idleDur) decide(h);
      break;
    case 'walk': {
      const t = h.plan.target;
      const dx = t.x-p.x, dz = t.z-p.z, d = Math.hypot(dx,dz);
      if(d < 0.055){ onArrive(h); break; }
      h.heading = turnToward(h.heading, Math.atan2(dx,dz), 4.2*dt);
      const sp = h.plan.speed;
      p.x += Math.sin(h.heading)*sp*dt;
      p.z += Math.cos(h.heading)*sp*dt;
      break;
    }
    case 'wheelEnter': {
      const k = easeIO(clamp01(h.stateT/WHEEL.enterDur));
      p.lerpVectors(h.plan.from, WHEEL.innerPoint, k);
      h.heading = lerpAngle(h.plan.fromHeading, h.plan.runHeading, k);
      if(k>=1){ h.state='wheelRun'; h.stateT=0; h.runV=0; }
      break;
    }
    case 'wheelRun': {
      const dur = h.plan.dur, A=0.9, B=0.7, vm=h.plan.vMax;
      h.runV = h.stateT<A ? vm*h.stateT/A : (h.stateT>dur-B ? vm*clamp01((dur-h.stateT)/B) : vm);
      p.set(WHEEL.cx, WHEEL.cy - R_KOLESA + Math.abs(Math.sin(h.legPhase))*0.006, WHEEL.cz);
      h.heading = h.plan.runHeading;
      if(h.stateT>=dur){
        h.state='wheelExit'; h.stateT=0;
        h.plan.from = p.clone(); h.plan.fromHeading = h.heading;
        wheel.owner = null; logEvent('wheel_exit',{who:h.name});
      }
      break;
    }
    case 'wheelExit': {
      const k = easeIO(clamp01(h.stateT/WHEEL.exitDur));
```

```js
function repel(){
  for(let i=0;i<HAMSTERS.length;i++){
    for(let j=i+1;j<HAMSTERS.length;j++){
      const a=HAMSTERS[i], b=HAMSTERS[j];
      const pa=a.root.position, pb=b.root.position;
      if(Math.abs(pa.y-pb.y) > 0.12) continue;
      const dx=pb.x-pa.x, dz=pb.z-pa.z, d=Math.hypot(dx,dz);
      const minD=0.155;
      if(d>=minD || d<1e-5) continue;
      const nx=dx/d, nz=dz/d, push=(minD-d);
      const fa=PUSHABLE.has(a.state), fb=PUSHABLE.has(b.state);
      if(fa&&fb){ pa.x-=nx*push*0.5; pa.z-=nz*push*0.5; pb.x+=nx*push*0.5; pb.z+=nz*push*0.5; }
      else if(fa){ pa.x-=nx*push; pa.z-=nz*push; }
      else if(fb){ pb.x+=nx*push; pb.z+=nz*push; }
    }
  }
}
```

```js
function updateWheel(dt){
  const runner = HAMSTERS.find(h=>h.state==='wheelRun');
  if(runner){
    // ω = v/R: обод под лапами уходит назад со скоростью лап (без проскальзывания)
    const target = runner.runDir * runner.runV / R_KOLESA;
    wheel.omega += (target - wheel.omega) * Math.min(1, dt/0.04);
  } else {
    wheel.omega *= Math.exp(-2.0*dt);           // трение: пустое колесо гаснет
    if(Math.abs(wheel.omega) < 0.008) wheel.omega = 0;
  }
  wheel.angle += wheel.omega*dt;
  wheelSpin.rotation.z = wheel.angle;
}
```
