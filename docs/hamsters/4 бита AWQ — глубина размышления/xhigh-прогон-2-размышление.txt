```js
function updateHamster(h, dt, t){
  const p = h.root.position;
  // pose lerp
  const T = POSE[h.state] || POSE.WANDER;
  const k = Math.min(1, dt*6);
  h.pose.pitch += (T.pitch - h.pose.pitch)*k;
  h.pose.head += (T.head - h.pose.head)*k;
  h.pose.cheek += (T.cheek - h.pose.cheek)*k;
  h.pose.amp += (T.amp - h.pose.amp)*k;
  h.pose.freq += (T.freq - h.pose.freq)*k;

  let moving = 0;
  switch(h.state){
    case 'WANDER': case 'TO_WHEEL': case 'TO_BOWL': {
      const dx=h.target.x-p.x, dz=h.target.z-p.z;
      const d=Math.hypot(dx,dz);
      if(d<.28){ arrive(h); break; }
      const want=Math.atan2(dx,dz);
      h.yaw = lerpAngle(h.yaw, want, Math.min(1, dt*5));
      const sp = h.speed * (h.state==='WANDER'?1:.9);
      p.x += Math.sin(h.yaw)*sp*dt;
      p.z += Math.cos(h.yaw)*sp*dt;
      moving = 1;
      h.phase += dt*h.pose.freq;
      break;
    }
    case 'PAUSE':
      h.timer-=dt;
      h.yaw += Math.sin(t*.6+h.seed)*dt*.25;
      h.pose.head = Math.sin(t*.9+h.seed)*.35;  // look around — but pose lerp fights this...
```

Hmm, conflict: pose.head is lerped toward T.head, but I want look-around during pause. Solution: separate headLook value: headRotY = sin(...)*.4 when paused, else 0; headRotX = pose.head + wobble. Keep h.look (lerped) separately. Simplify:

headGrp.rotation.y = h.lookY; headGrp.rotation.x = h.pose.head + (state EATING? sin(t*12)*.07 : 0) + sin(h.phase*2)*.05*moving.

h.lookY target: paused → sin(t*.8+seed)*.45; else 0. Lerp h.lookY toward target.

    case 'PAUSE': timer; on end chooseNext; moving=0; break;
    case 'RUNNING': {
      // settle at spot
      p.lerp(WHEEL_SPOT, Math.min(1,dt*8));
      h.yaw = lerpAngle(h.yaw, -Math.PI/2, Math.min(1,dt*6));
      h.phase += dt*h.pose.freq;
      h.timer -= dt;
      if(h.timer<=0){ h.runner=false; wheel.runner=null; chooseNext(h); }
      break;
    }
    case 'EATING': {
      p.lerp(BOWL_SPOT, Math.min(1,dt*8));
      h.yaw = lerpAngle(h.yaw, Math.atan2(BOWL.x-p.x, BOWL.z-p.z), Math.min(1,dt*6));
      h.timer-=dt;
      if(h.timer<=0){ if(pellets.length) eatPellet(h); chooseNext(h); }
      break;
    }
  }
  // clamp
  p.x=clamp(p.x,-BX,BX); p.z=clamp(p.z,-BZ,BZ);
  h.root.rotation.y=h.yaw;
  // apply pose to meshes
  h.bodyGrp.rotation.x = h.pose.pitch;
  h.bodyGrp.position.y = .42 + Math.sin(h.phase*2)*.025*h.pose.amp + (h.state==='RUNNING'? Math.abs(Math.sin(h.phase))*.04:0);
  h.bodyGrp.rotation.z = Math.sin(h.phase)*.05*h.pose.amp;
  h.bodyGrp.scale.y = 1 + Math.sin(t*2.2+h.seed)*.02;  // breathing
  h.headGrp.rotation.x = h.pose.head + (h.state==='EATING'? Math.sin(t*11)*.08:0) + Math.sin(h.phase*2+.5)*.05*h.pose.amp;
  h.headGrp.rotation.y = h.lookY;
  h.legs.forEach((L,i)=>{ L.rotation.x = Math.sin(h.phase + (i%2? Math.PI:0) + (i>1? Math.PI/2:0))*h.pose.amp; });
```
