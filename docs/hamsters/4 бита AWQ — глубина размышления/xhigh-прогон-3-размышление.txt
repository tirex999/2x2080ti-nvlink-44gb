function step(h, dt, dest, speed){
  const v = new THREE.Vector3().subVectors(dest, h.group.position); v.y=0;
  const d = v.length();
  if (d > 0.001){
    v.normalize();
    h.group.position.addScaledVector(v, Math.min(d, speed*dt));
    const targetYaw = Math.atan2(v.x, v.z);
    h.dirYaw = dampAngle(h.dirYaw, targetYaw, dt*8);

  h.group.rotation.y = h.dirYaw;
}

function dampAngle(cur, target, k){
  let d = (target - cur) % (Math.PI*2);
  if (d > Math.PI) d -= Math.PI*2;
  if (d < -Math.PI) d += Math.PI*2;

}
