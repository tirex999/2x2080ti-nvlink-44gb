attribute vec2 aParam;   // (u, s)
attribute float aFace;   // +1 top, -1 bottom

float mSpan(float u){
  float a = pow(smoothstep(0.0, 0.42, u), 0.62);
  float b = pow(1.0 - smoothstep(0.42, 1.0, u), 1.25);
  return min(a, b) * 1.12;
}
float mThick(float u){
  return (0.14 + 0.13*sin(3.14159*clamp(u*1.35,0.0,1.0))) * (1.0 - smoothstep(0.75,1.0,u)*0.75);
}
vec3 mShape(vec2 q){
  float u = q.x, s = q.y;
  float sp = mSpan(u);
  float x = s*sp;
  float z = 0.80 - u*2.35 - 0.42*abs(s)*smoothstep(0.02,0.5,u);
  float k = max(0.0, 1.0 - s*s);
  float y = (aFace>0.0) ? mThick(u)*pow(k,0.55) : -mThick(u)*0.42*pow(k,0.45);
  y += 0.06*sin(3.14159*u)*cos(s*1.4);
  return vec3(x,y,z);
}
vec3 mPos(vec2 q){
  vec3 p = mShape(q);
  float u=q.x, s=q.y, as=abs(s);
  float t = uTime;
  // wing flap: travelling wave from root to tip
  float flap = sin(t*2.2 + u*1.0) * pow(as,1.45) * mSpan(u) * 0.34 * uFlap;
  p.y += flap;
  p.y += sin(t*3.3 + u*6.0 + as*4.0) * 0.018 * pow(as,2.0) * uFlap;
  // tail sway
  float tw = pow(max(u-0.62,0.0)/0.38, 2.0);
  p.x += sin(t*1.5 - u*2.2) * 0.17 * tw;
  // body bob
  p.y += sin(t*2.2 + 0.4)*0.03*uFlap;
  return p;
}
