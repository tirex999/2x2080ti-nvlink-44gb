```js
function updateFish(f, dt, t) {
  const p = f.mesh.position;
  // food targeting
  let food = null;
  if (foodItems.length) {
    let best = 15;
    for (const fi of foodItems) {
      const d = p.distanceTo(fi.mesh.position);
      if (d < best) { best = d; food = fi; }
    }
  }
  f.targetFood = food;
  const steer = new THREE.Vector3();
  if (food) {
    steer.copy(food.mesh.position).sub(p).normalize().multiplyScalar(f.speed * 1.8);
  } else {
    f.wanderTimer -= dt;
    if (f.wanderTimer <= 0) {
      f.wanderTimer = 2 + Math.random()*4;
      f.wanderDir.set(rand(-1,1), rand(-0.35,0.35), rand(-1,1)).normalize();
    }
    steer.copy(f.wanderDir).multiplyScalar(f.speed);
  }
  // separation
  for (const o of fishArray) {
    if (o === f) continue;
    const dx = p.x-o.mesh.position.x, dy=..., dz=...;
    const d2 = dx*dx+dy*dy+dz*dz;
    const r = f.avoidanceRadius;
    if (d2 < r*r && d2 > 0.0001) {
      const d = Math.sqrt(d2);
      const push = (1 - d/r);
      steer.x += (dx/d)*push*4; ... etc
    }
  }
  // walls
  const W = 3.0;
  if (p.x > BX-W) steer.x -= ((p.x-(BX-W))/W)*10;
  if (p.x < -BX+W) steer.x += ((-BX+W-p.x)/W)*10;
  same y (top stronger), z.
  // gentle centering vertically
  steer.y += (0 - p.y)*0.02? small
  
  // integrate
  f.velocity.lerp(steer, 1 - Math.exp(-dt*2.5));
  const sp = f.velocity.length();
  const maxSp = food ? f.speed*2.2 : f.speed*1.4;
  if (sp > maxSp) f.velocity.multiplyScalar(maxSp/sp);
  p.addScaledVector(f.velocity, dt);
  // hard clamp + reflect
  if (p.x > BX) { p.x = BX; f.velocity.x = -Math.abs(f.velocity.x)*0.6; } etc.
  
  // eating
  if (food) {
    const d = p.distanceTo(food.mesh.position);
    if (d < 0.9 + f.mesh.scale.x*0.4) eatFood(f, food);
  }
  
  // orientation
  const v = f.velocity;
  const len = v.length() || 1;
  const yaw = Math.atan2(v.x, v.z);
  const pitch = Math.asin(THREE.MathUtils.clamp(v.y/len, -1, 1));
  const roll = Math.sin(t*f.tailSpeed*0.5 + f.phase) * 0.12;
  _euler.set(pitch, yaw, roll, 'YXZ');
  _quat.setFromEuler(_euler);
  f.mesh.quaternion.slerp(_quat, 1 - Math.exp(-dt*4));
  
  // animation
  f.tail.rotation.z = Math.sin(t*f.tailSpeed + f.phase) * 0.55;
  f.leftFin.rotation.z = f.baseL + Math.sin(t*f.tailSpeed*0.8 + f.phase) * 0.3;
  f.rightFin.rotation.z = -f.baseL - Math.sin(t*f.tailSpeed*0.8 + f.phase + 0.5) * 0.3;
  f.dorsal.rotation.x = -0.35 + Math.sin(t*f.tailSpeed*0.6 + f.phase) * 0.1;
}
```

Wait, pitch sign: earlier derived pitch = asin(vy/len) with positive X rotation mapping +Z→up. Verify: rotation about X by θ: z-axis (0,0,1) → (0, sinθ, cosθ). For velocity (0,1,0) we need sinθ=1 → θ=90° = asin(1). ✓.

Yaw: rotation about Y by φ: (0,0,1) → (sinφ, 0, cosφ). Velocity (1,0,0) → φ=90°=atan2(1,0)=90° ✓.

Euler order 'YXZ': applies Y (yaw) then X (pitch) then Z (roll) — standard FPS convention, correct for this decomposition.

```js
function updateFish(f, dt, t) {
  const p = f.mesh.position;
  // food targeting
  let food = null;
  if (foodItems.length) {
    let best = 15;
    for (const fi of foodItems) {
      const d = p.distanceTo(fi.mesh.position);
      if (d < best) { best = d; food = fi; }
    }
  }
  f.targetFood = food;
  const steer = new THREE.Vector3();
  if (food) {
    steer.copy(food.mesh.position).sub(p).normalize().multiplyScalar(f.speed * 1.8);
  } else {
    f.wanderTimer -= dt;
    if (f.wanderTimer <= 0) {
      f.wanderTimer = 2 + Math.random()*4;
      f.wanderDir.set(rand(-1,1), rand(-0.35,0.35), rand(-1,1)).normalize();
    }
    steer.copy(f.wanderDir).multiplyScalar(f.speed);
  }
  // separation
  for (const o of fishArray) {
    if (o === f) continue;
    const dx = p.x-o.mesh.position.x, dy=..., dz=...;
    const d2 = dx*dx+dy*dy+dz*dz;
    const r = f.avoidanceRadius;
    if (d2 < r*r && d2 > 0.0001) {
      const d = Math.sqrt(d2);
      const push = (1 - d/r);
      steer.x += (dx/d)*push*4; ... etc
    }
  }
  // walls
  const W = 3.0;
  if (p.x > BX-W) steer.x -= ((p.x-(BX-W))/W)*10;
  if (p.x < -BX+W) steer.x += ((-BX+W-p.x)/W)*10;
  same y (top stronger), z.
  // gentle centering vertically
  steer.y += (0 - p.y)*0.02? small
  
  // integrate
  f.velocity.lerp(steer, 1 - Math.exp(-dt*2.5));
  const sp = f.velocity.length();
  const maxSp = food ? f.speed*2.2 : f.speed*1.4;
  if (sp > maxSp) f.velocity.multiplyScalar(maxSp/sp);
  p.addScaledVector(f.velocity, dt);
  // hard clamp + reflect
  if (p.x > BX) { p.x = BX; f.velocity.x = -Math.abs(f.velocity.x)*0.6; } etc.
  
  // eating
  if (food) {
    const d = p.distanceTo(food.mesh.position);
    if (d < 0.9 + f.mesh.scale.x*0.4) eatFood(f, food);
  }
  
  // orientation
  const v = f.velocity;
  const len = v.length() || 1;
  const yaw = Math.atan2(v.x, v.z);
  const pitch = Math.asin(THREE.MathUtils.clamp(v.y/len, -1, 1));
  const roll = Math.sin(t*f.tailSpeed*0.5 + f.phase) * 0.12;
  _euler.set(pitch, yaw, roll, 'YXZ');
  _quat.setFromEuler(_euler);
  f.mesh.quaternion.slerp(_quat, 1 - Math.exp(-dt*4));
  
  // animation
  f.tail.rotation.z = Math.sin(t*f.tailSpeed + f.phase) * 0.55;
  f.leftFin.rotation.z = f.baseL + Math.sin(t*f.tailSpeed*0.8 + f.phase) * 0.3;
  f.rightFin.rotation.z = -f.baseL - Math.sin(t*f.tailSpeed*0.8 + f.phase + 0.5) * 0.3;
  f.dorsal.rotation.x = -0.35 + Math.sin(t*f.tailSpeed*0.6 + f.phase) * 0.1;
}
```

Wait, pitch sign: earlier derived pitch = asin(vy/len) with positive X rotation mapping +Z→up. Verify: rotation about X by θ: z-axis (0,0,1) → (0, sinθ, cosθ). For velocity (0,1,0) we need sinθ=1 → θ=90° = asin(1). ✓.

Yaw: rotation about Y by φ: (0,0,1) → (sinφ, 0, cosφ). Velocity (1,0,0) → φ=90°=atan2(1,0)=90° ✓.

Euler order 'YXZ': applies Y (yaw) then X (pitch) then Z (roll) — standard FPS convention, correct for this decomposition.

```js
      f.wanderDir.set(rand(-1,1),rand(-0.4,0.4),rand(-1,1)).normalize();
    }
    steer.copy(f.wanderDir).multiplyScalar(f.speed);
  }
  // separation
  for(const o of fishArray){
    if(o===f) continue;
    const op=o.mesh.position;
    const dx=p.x-op.x, dy=p.y-op.y, dz=p.z-op.z;
    const r=f.avoidanceRadius;
    const d2=dx*dx+dy*dy+dz*dz;
    if(d2<r*r && d2>1e-4){
      const d=Math.sqrt(d2), k=(1-d/r)*5;
      steer.x+=dx/d*k; steer.y+=dy/d*k; steer.z+=dz/d*k;
    }
  }
  // walls
  const M=3, K=12;
  if(p.x> BX-M) steer.x-=(p.x-(BX-M))/M*K;
  if(p.x<-BX+M) steer.x+=((-BX+M)-p.x)/M*K;
  if(p.y> BY-M) steer.y-=(p.y-(BY-M))/M*K;
  if(p.y<-BY+M) steer.y+=((-BY+M)-p.y)/M*K;
  if(p.z> BZ-M) steer.z-=(p.z-(BZ-M))/M*K;
  if(p.z<-BZ+M) steer.z+=((-BZ+M)-p.z)/M*K;
  steer.y -= p.y*0.03; // gentle centering

  f.velocity.lerp(steer, 1-Math.exp(-dt*2.2));
  const sp=f.velocity.length();
  const maxS = food ? f.speed*2.3 : f.speed*1.5;
  if(sp>maxS) f.velocity.multiplyScalar(maxS/sp);
  if(sp<0.4) f.velocity.multiplyScalar(0.4/Math.max(sp,0.001)); // min speed
  p.addScaledVector(f.velocity,dt);
  // hard clamp
  if(p.x> BX){p.x= BX; f.velocity.x*=-0.5;} else if(p.x<-BX){p.x=-BX; f.velocity.x*=-0.5;}
  if(p.y> BY){p.y= BY; f.velocity.y*=-0.5;} else if(p.y<-BY){p.y=-BY; f.velocity.y*=-0.5;}
  if(p.z> BZ){p.z= BZ; f.velocity.z*=-0.5;} else if(p.z<-BZ){p.z=-BZ; f.velocity.z*=-0.5;}

  // eat
  if(food){
    if(p.distanceTo(food.mesh.position) < 1.0 + f.mesh.scale.x*0.35){
      removeFood(food);
      if(f.mesh.scale.x < 2.2) f.mesh.scale.multiplyScalar(1.05);
      for(let i=0;i<4;i++) addBubble(p);
      stats.eaten++;
      floatText(p, '+5%');
      updateStats(true);
    }
  }

  // orientation
  const v=f.velocity, len=v.length()||1;
  _euler.set(Math.asin(clamp(v.y/len,-1,1)), Math.atan2(v.x,v.z), Math.sin(t*f.tailSpeed*0.5+f.phase)*0.12);
  _quat.setFromEuler(_euler);
  f.mesh.quaternion.slerp(_quat, 1-Math.exp(-dt*4));

  // animation
  const wag=Math.sin(t*f.tailSpeed+f.phase);
  f.tail.rotation.z = wag*0.55;
  f.leftFin.rotation.z  = f.baseLF  + Math.sin(t*f.tailSpeed*0.8+f.phase)*0.3;
  f.rightFin.rotation.z = f.baseRF - Math.sin(t*f.tailSpeed*0.8+f.phase+0.6)*0.3;
  f.dorsal.rotation.x = -0.35 + Math.sin(t*f.tailSpeed*0.6+f.phase)*0.1;
}
const _steer = new THREE.Vector3();
```
