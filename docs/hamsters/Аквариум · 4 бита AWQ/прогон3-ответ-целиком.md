const SCHEMES = [

function createFish(x, y, z) {
  const scheme = SCHEMES[Math.floor(Math.random() * SCHEMES.length)];
  const g = new THREE.Group();

  // Тело — вытянутая сфера
  const bodyGeo = new THREE.SphereGeometry(1, 14, 10);
  bodyGeo.scale(0.65, 0.5, 1.35);
  const body = new THREE.Mesh(bodyGeo, new THREE.MeshStandardMaterial({
    color: scheme.body, roughness: 0.35, metalness: 0.15
  }));
  body.castShadow = true;
  g.add(body);

  // Глаза
  [-1, 1].forEach(side => {
    const eye = new THREE.Mesh(eyeGeo, eyeMat);
    eye.position.set(side * 0.32, 0.1, 0.85);
    const pupil = new THREE.Mesh(pupilGeo, pupilMat);
    pupil.position.set(side * 0.42, 0.1, 0.92);
    g.add(eye, pupil);
  });

  // Хвост (пивот для махания)
  const tailPivot = new THREE.Group();
  tailPivot.position.set(0, 0, -1.25);
  const tailGeo = new THREE.ConeGeometry(0.5, 0.9, 8);
  tailGeo.rotateX(-Math.PI / 2);
  tailGeo.scale(1, 1, 0.3);
  const tail = new THREE.Mesh(tailGeo, new THREE.MeshStandardMaterial({
    color: scheme.fin, roughness: 0.5, side: THREE.DoubleSide
  }));
  tail.position.z = -0.45;
  tailPivot.add(tail);
  g.add(tailPivot);

  // Верхний плавник
  const dorsalGeo = new THREE.ConeGeometry(0.35, 0.7, 6);
  dorsalGeo.scale(1, 1, 0.25);
  const dorsal = new THREE.Mesh(dorsalGeo, new THREE.MeshStandardMaterial({
    color: scheme.fin, roughness: 0.5, side: THREE.DoubleSide
  }));
  dorsal.position.set(0, 0.5, -0.15);
  g.add(dorsal);

  // Боковые плавники
  const finGeo = new THREE.ConeGeometry(0.25, 0.55, 6);
  finGeo.scale(1, 1, 0.2);
  const leftFin = new THREE.Mesh(finGeo, new THREE.MeshStandardMaterial({
    color: scheme.fin, roughness: 0.5, side: THREE.DoubleSide }));
  leftFin.position.set(0.5, -0.1, 0.2);
  leftFin.rotation.z = -Math.PI / 2.4;
  const rightFin = leftFin.clone();
  rightFin.position.x = -0.5;
  rightFin.rotation.z = Math.PI / 2.4;
  g.add(leftFin, rightFin);

  const scale = 0.6 + Math.random() * 0.6;
  g.scale.setScalar(scale);
  g.position.set(x, y, z);
  scene.add(g);

  return {
    mesh: g, tail: tailPivot, leftFin, rightFin, dorsal,
    velocity: new THREE.Vector3((Math.random()-0.5), 0, (Math.random()-0.5)).normalize(),
    wanderDir: new THREE.Vector3((Math.random()-0.5), 0, (Math.random()-0.5)).normalize(),
    wanderTimer: Math.random() * 3,
    speed: 2 + Math.random() * 2.5,
    tailSpeed: 6 + Math.random() * 6,
    phase: Math.random() * Math.PI * 2,
    targetFood: null,
    avoidanceRadius: 2.5 + Math.random() * 1.5
  };
}

// ==================== ОБНОВЛЕНИЕ СИМУЛЯЦИИ ====================
const tmpVec = new THREE.Vector3();
const tmpVec2 = new THREE.Vector3();

function updateFish(f, dt, t) {
  const pos = f.mesh.position;

  // Выбор цели: корм или блуждание
  let moveDir;
  if (f.targetFood && f.targetFood.active) {
    moveDir = tmpVec.copy(f.targetFood.mesh.position).sub(pos).normalize();
  } else {
    f.wanderTimer -= dt;
    if (f.wanderTimer <= 0) {
      f.wanderDir.set(Math.random()-0.5, (Math.random()-0.5)*0.5, Math.random()-0.5).normalize();
      f.wanderTimer = 2 + Math.random() * 4;
    }
    moveDir = tmpVec.copy(f.wanderDir);
  }
  f.velocity.copy(moveDir).multiplyScalar(f.speed);

  // Поиск корма (радиус 15)
  if (!f.targetFood) {
    for (const fd of foodArray) {
      if (fd.active && pos.distanceTo(fd.mesh.position) < 15) { f.targetFood = fd; break; }
    }
  }

  // Избегание столкновений
  for (const o of fishArray) {
    if (o === f) continue;
    tmpVec2.copy(o.mesh.position).sub(pos);
    const d = tmpVec2.length();
    if (d < f.avoidanceRadius && d > 0.001) {
      f.velocity.addScaledVector(tmpVec2, -(f.avoidanceRadius - d) / d * 2);
    }
  }

  // Отражение от стен
  if (pos.x >  INNER.x)  { pos.x =  INNER.x;  f.velocity.x = -Math.abs(f.velocity.x); }
  if (pos.x < -INNER.x)  { pos.x = -INNER.x;  f.velocity.x =  Math.abs(f.velocity.x); }
  if (pos.y >  INNER.yMax) { pos.y =  INNER.yMax; f.velocity.y = -Math.abs(f.velocity.y); }
  if (pos.y <  INNER.yMin) { pos.y =  INNER.yMin; f.velocity.y =  Math.abs(f.velocity.y); }
  if (pos.z >  INNER.z)  { pos.z =  INNER.z;  f.velocity.z = -Math.abs(f.velocity.z); }
  if (pos.z < -INNER.z)  { pos.z = -INNER.z;  f.velocity.z =  Math.abs(f.velocity.z); }

  pos.addScaledVector(f.velocity, dt);

  // Поворот в направлении движения
  if (f.velocity.lengthSq() > 0.001) {
    const targetYaw = Math.atan2(f.velocity.x, f.velocity.z);
    let dy = targetYaw - f.mesh.rotation.y;
    while (dy > Math.PI) dy -= Math.PI * 2;
    while (dy < -Math.PI) dy += Math.PI * 2;
    f.mesh.rotation.y += dy * Math.min(1, dt * 4);
    f.mesh.rotation.x = THREE.MathUtils.clamp(f.velocity.y * 0.25, -0.4, 0.4);
  }

  // Анимация хвоста и плавников
  const flap = Math.sin(t * f.tailSpeed + f.phase);
  f.tail.rotation.y = flap * 0.6;
  f.leftFin.rotation.z  = -Math.PI / 2.4 + Math.sin(t * f.tailSpeed * 0.7 + f.phase) * 0.25;
  f.rightFin.rotation.z =  Math.PI / 2.4 - Math.sin(t * f.tailSpeed * 0.7 + f.phase + 1) * 0.25;
  f.dorsal.rotation.x = Math.sin(t * f.tailSpeed * 0.5 + f.phase) * 0.15;

  // Поглощение корма
  if (f.targetFood && f.targetFood.active) {
    if (pos.distanceTo(f.targetFood.mesh.position) < 1.0) {
      f.targetFood.active = false;
      scene.remove(f.targetFood.mesh);
      const idx = foodArray.indexOf(f.targetFood);
      if (idx >= 0) foodArray.splice(idx, 1);
      f.targetFood = null;
      // Рост на 5%
      const s = Math.min(f.mesh.scale.x * 1.05, 2.5);
      f.mesh.scale.setScalar(s);
    }
  }
}

function updateFood(dt) {
  for (let i = foodArray.length - 1; i >= 0; i--) {
    const fd = foodArray[i];
    fd.vy -= 8 * dt; // гравитация
    fd.mesh.position.y += fd.vy * dt;
    fd.mesh.rotation.y += dt * 2;
    if (fd.mesh.position.y <= 0.4) {
      scene.remove(fd.mesh);
      foodArray.splice(i, 1);
    }
  }
}

function updateBubbles(dt, t) {
  for (const b of bubbles) {
    b.mesh.position.y += b.speed * dt;
    b.mesh.position.x = b.baseX + Math.sin(t * 1.5 + b.phase) * b.amp;
    b.mesh.position.z = b.baseZ + Math.cos(t * 1.2 + b.phase) * b.amp * 0.7;
    if (b.mesh.position.y > INNER.yMax) {
      b.mesh.position.y = INNER.yMin;
      b.baseX = (Math.random() - 0.5) * 32;
      b.baseZ = (Math.random() - 0.5) * 17;
    }
  }
}

function updateAlgae(t) {
  for (const a of algae) {
    a.mesh.rotation.x = Math.sin(t * a.speed + a.phase) * 0.08;
    a.mesh.rotation.z = Math.cos(t * a.speed * 0.8 + a.phase) * 0.08;
  }
}

const clock = new THREE.Clock();
let frames = 0, fpsTimer = 0;

function animate() {
  requestAnimationFrame(animate);
  const dt = Math.min(clock.getDelta(), 0.05);
  const t = clock.elapsedTime;

  for (const f of fishArray) updateFish(f, dt, t);
  updateFood(dt);
  updateBubbles(dt, t);
  updateAlgae(t);

  controls.update();
  renderer.render(scene, camera);

  // FPS
  frames++;
  fpsTimer += dt;
  if (fpsTimer >= 1) {
    elFps.textContent = frames;
    elFishCount.textContent = fishArray.length;
    elFoodCount.textContent = foodArray.length;
    frames = 0; fpsTimer = 0;
  }
}
