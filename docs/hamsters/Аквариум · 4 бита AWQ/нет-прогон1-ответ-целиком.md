    // ---------- tank bounds ----------
    const B={hx:18,hy:12,hz:10};

    {

      mesh.position.set(x,SAND_Y,z);
      mesh.castShadow=true;
      scene.add(mesh);
      weeds.push({mesh,base:mesh.rotation.z,phase:rand(0,TAU),amp:rand(0.05,0.14)});
    }

    // ---------- update loops ----------
    const _dir=new THREE.Vector3(),_diff=new THREE.Vector3(),_t=new THREE.Vector3();
    function updateFish(dt,time){
      for(const f of fishArray){
        // find food
        let best=null,bestD=15*15;
        for(const fd of foods){
          const d=f.mesh.position.distanceToSquared(fd.mesh.position);
          if(d<bestD){bestD=d;best=fd;}
        }
        f.targetFood=best;

        if(f.targetFood){
          _t.copy(f.targetFood.mesh.position).sub(f.mesh.position);
          const dist=_t.length();
          if(dist<0.9){
            scene.remove(f.targetFood.mesh);
            f.targetFood.mesh.geometry.dispose();
            f.targetFood.mesh.material.dispose();
            f.targetFood=null;
            foods.splice(foods.indexOf(f.targetFood||foods[0]),1); // handled below
            // remove eaten food
            for(let i=foods.length-1;i>=0;i--) if(foods[i].mesh===f.targetFood) foods.splice(i,1);
            f.scale=Math.min(f.scale*1.05,2.4);
          }else{
            _t.normalize();
            f.velocity.lerp(_t,0.06);
            f.velocity.multiplyScalar(f.speed*1.9);
          }
        }else{
          f.wanderTimer-=dt;
          if(f.wanderTimer<=0){
            f.wanderTimer=rand(1.5,4);
            _t.set(rand(-1,1),rand(-0.5,0.5),rand(-1,1)).normalize();
            f.velocity.lerp(_t,0.4);
          }
          f.velocity.multiplyScalar(f.speed);
        }

        // avoid other fish
        for(const o of fishArray){
          if(o===f) continue;
          _diff.copy(f.mesh.position).sub(o.mesh.position);
          const d=_diff.length();
          if(d<f.avoidanceRadius&&d>0.001){
            _diff.multiplyScalar((f.avoidanceRadius-d)/f.avoidanceRadius/d);
            f.velocity.add(_diff);
          }
        }

        // walls
        const p=f.mesh.position;
        if(p.x>FISH.hx) f.velocity.x-=0.5; else if(p.x<-FISH.hx) f.velocity.x+=0.5;
        if(p.y>FISH.hy) f.velocity.y-=0.5; else if(p.y<-FISH.hy) f.velocity.y+=0.5;
        if(p.z>FISH.hz) f.velocity.z-=0.5; else if(p.z<-FISH.hz) f.velocity.z+=0.5;

        // integrate
        f.velocity.clampLength(0,f.speed*2.2);
        p.addScaledVector(f.velocity,dt);
        p.x=clamp(p.x,-FISH.hx,FISH.hx);
        p.y=clamp(p.y,-FISH.hy,FISH.hy);
        p.z=clamp(p.z,-FISH.hz,FISH.hz);

        // face direction
        if(f.velocity.lengthSq()>0.001){
          _dir.copy(f.velocity).normalize();
          const yaw=Math.atan2(_dir.x,_dir.z);
          const pitch=Math.asin(clamp(_dir.y,-1,1));
          f.mesh.rotation.y=yaw;
          f.mesh.rotation.x=-pitch;
        }

        // animate
        f.phase+=f.tailSpeed*dt;
        f.tail.rotation.y=Math.sin(f.phase)*0.7;
        f.leftFin.rotation.z=Math.sin(f.phase*0.7)*0.35;
        f.rightFin.rotation.z=-Math.sin(f.phase*0.7)*0.35;
        f.mesh.scale.setScalar(f.scale);
      }
    }

    function updateFood(dt){
      for(let i=foods.length-1;i>=0;i--){
        const fd=foods[i];
        fd.vel.y-=9.8*dt*0.5;           // gravity
        fd.vel.multiplyScalar(0.94);     // water drag
        fd.mesh.position.addScaledVector(fd.vel,dt);
        if(fd.mesh.position.y<=FEED_Y){
          fd.mesh.position.y=FEED_Y;
          fd.vel.set(0,0,0);
          // remove after resting a bit
          fd.rest=(fd.rest||0)+dt;
          if(fd.rest>6){
            scene.remove(fd.mesh);
            fd.mesh.geometry.dispose(); fd.mesh.material.dispose();
            foods.splice(i,1);
            continue;
          }
        }
      }
    }

    function updateBubbles(dt,time){
      for(const b of bubbles){
        b.mesh.position.y+=b.vy*dt;
        b.mesh.position.x=b.baseX+Math.sin(time*1.5+b.ph)*b.amp;
        b.mesh.position.z=b.baseZ+Math.cos(time*1.3+b.ph)*b.amp;
        if(b.mesh.position.y>B.hy-0.5){
          b.mesh.position.y=SAND_Y+rand(0.5,2);
          b.baseX=rand(-B.hx+2,B.hx-2);
          b.baseZ=rand(-B.hz+2,B.hz-2);
          b.mesh.position.x=b.baseX; b.mesh.position.z=b.baseZ;
        }
      }
    }

    function updateWeeds(time){
      for(const w of weeds){
        w.mesh.rotation.x=Math.sin(time*0.8+w.phase)*w.amp;
        w.mesh.rotation.z=w.base+Math.cos(time*0.7+w.phase)*w.amp;
      }
    }

    // ---------- stats ----------
    const elFish=document.getElementById('stat-fish');
    const elFps=document.getElementById('stat-fps');
    const elGrow=document.getElementById('stat-grow');
    let frames=0,fpsAcc=0;

    // ---------- loop ----------
    const clock=new THREE.Clock();
    function animate(){
      requestAnimationFrame(animate);
      const dt=Math.min(clock.getDelta(),0.05);
      const time=clock.elapsedTime;

      updateFish(dt,time);
      updateFood(dt);
      updateBubbles(dt,time);
      updateWeeds(time);

      // gentle light drift
      pl1.position.x=-14+Math.sin(time*0.3)*4;
      pl2.position.z=10+Math.cos(time*0.25)*4;

      controls.update();
      renderer.render(scene,camera);

      // fps
      frames++; fpsAcc+=dt;
      if(fpsAcc>=0.5){
        elFps.textContent=Math.round(frames/fpsAcc);
        frames=0; fpsAcc=0;
      }
      elFish.textContent=fishArray.length;
      let avg=0; for(const f of fishArray) avg+=f.scale/f.baseScale;
      elGrow.textContent='×'+(avg/fishArray.length).toFixed(2);
    }
    animate();

    addEventListener('resize',()=>{
      camera.aspect=innerWidth/innerHeight;
      camera.updateProjectionMatrix();
      renderer.setSize(innerWidth,innerHeight);
    });
  })();
  </script>
</body>

Что внутри и как это работает:

- **Рыбки** — 15 на старте, каждая из 8 цветовых схем, с телом, глазами, хвостом и плавниками; индивидуальный масштаб, скорость, фаза анимации. Хвост машет, плавники колеблются, рыбка плавно разворачивается по направлению движения (yaw + pitch).
- **ИИ**: отталкивание от соседей, мягкий отскок от стенок, периодическое блуждание, преследование корма (радиус 15), рост +5% при поедании (с ограничением).
- **Окружение**: стеклянный объём (`transmission: 0.95`) с wireframe-рамкой, песчаное дно с процедурными неровностями, 8 деформированных камней, 12 покачивающихся водорослей на `TubeGeometry` + `CatmullRomCurve3`.
- **Пузыри** — 30 на старте, поднимаются с покачиванием и сбрасываются на дно.
- **Кормление**: клик по воде (raycast по объёму) роняет корм с гравитацией и сопротивлением воды; корм исчезает съеденным или после отдыха на дне.
- **Свет**: Ambient + Directional (тени 2048², PCFSoft) + два подводных PointLight; кнопка «Свет» переключает интенсивность солнца.
- **Камера**: OrbitControls с damping, zум 10–60, ограничение полярного угла.
- **UI**: стеклянные панели (blur), статистика (рыбки / FPS / средний рост), три кнопки, подсказки управления.

Дополнительно добавил туман, виньетку и лёгкий шум для глубины, адаптив под мобильные и дельта-тайминг во всех анимациях.
