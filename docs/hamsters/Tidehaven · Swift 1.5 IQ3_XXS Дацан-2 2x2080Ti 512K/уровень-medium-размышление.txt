uniform float uTime; uniform float uWave; uniform float uTide; uniform float uShoreR;
varying vec3 vWorld; varying float vH; varying vec3 vNormalW;
float wave(vec2 p, float t){
  float a = uWave*0.28+0.05;
  float d = length(p);
  float damp = smoothstep(6.0,16.0,d);
  float h = sin(p.x*0.55 + t*1.4) + sin(p.y*0.4 - t*1.1) * 0.8 + sin((p.x+p.y)*0.3 + t*0.7)*0.6;
  return h*a*damp;
}
void main(){
  vec3 pos = position;
  vec2 p = pos.xz; // plane is XY; after rotateX(-90), geometry is already baked? Apply rotateX(-Math.PI/2) to geometry so positions are x,z with y=0.
  float h = wave(p,uTime);
  pos.y += h;
  vH=h;
  // normal via finite diff
  float e=0.6;
  float hx = wave(p+vec2(e,0.),uTime)-wave(p-vec2(e,0.),uTime);
