```javascript
function generateChunk(cx, cz) {
    const data = new Uint8Array(16 * 80 * 16);
    for (let lx = 0; lx < 16; lx++) {
        for (let lz = 0; lz < 16; lz++) {
            const wx = cx * 16 + lx;
            const wz = cz * 16 + lz;
            const H = getHeight(wx, wz);
            for (let y = 0; y < 80; y++) {
                let block = 0;
                if (y === 0) block = 3; // unbreakable stone
                else if (y < H - 3) block = 3; // stone
                else if (y < H) {
                    // 3 layers under surface
                    if (H <= 16) block = 4; // sand
                    else if (H >= 37) block = 3; // stone
                    else block = 2; // dirt
                } else if (y === H) {
                    // surface
                    if (H >= 46) block = 7; // snow
                    else if (H >= 37) block = 3; // stone
                    else if (H <= 16) block = 4; // sand
                    else block = 1; // grass
                }
                // Caves
                if (y > 0 && y < H - 2 && y >= 3) {
                    if (noise3D(wx * 0.09, y * 0.09, wz * 0.09) > 0.67) {
                        block = 0;
                    }
                }
                data[lx * 80 * 16 + y * 16 + lz] = block;
            }
        }
    }
    // Trees
    for (let lx = 2; lx < 14; lx++) {
        for (let lz = 2; lz < 14; lz++) {
            const wx = cx * 16 + lx;
            const wz = cz * 16 + lz;
            const H = getHeight(wx, wz);
            if (H > 16 && H < 37) { // grass terrain
                const surfaceBlock = data[lx * 80 * 16 + H * 16 + lz];
                if (surfaceBlock === 1 && hash2(wx, wz) < 0.02) {
                    placeTree(data, lx, H, lz);
                }
            }
        }
    }
    return data;
}
```

The readBlock helper:
```javascript
function getBlock(wx, wy, wz) {
    if (wy < 0 || wy >= 80) return 0;
    const cx = Math.floor(wx / 16);
    const cz = Math.floor(wz / 16);
    const chunk = chunks.get(cx + "," + cz);
    if (!chunk) return 0;
    const lx = wx - cx * 16;
    const lz = wz - cz * 16;
    return chunk.data[lx * 80 * 16 + wy * 16 + lz];
}
```
