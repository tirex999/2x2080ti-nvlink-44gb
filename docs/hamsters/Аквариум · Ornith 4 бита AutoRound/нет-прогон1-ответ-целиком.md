<script>
(function () {
  "use strict";

  /* ============================================================
     SCENE / RENDERER / CAMERA
     ============================================================ */
  const scene = new THREE.Scene();
  scene.background = new THREE.Color(0x061224);
  scene.fog = new THREE.FogExp2(0x0a2038, 0.012);

  const renderer = new THREE.WebGLRenderer({ antialias: true, powerPreference: "high-performance" });
  renderer.setSize(window.innerWidth, window.innerHeight);
  renderer.setPixelRatio(Math.min(window.devicePixelRatio, 2));
  renderer.shadowMap.enabled = true;
  renderer.shadowMap.type = THREE.PCFSoftShadowMap;
  renderer.outputEncoding = THREE.sRGBEncoding;
  document.body.appendChild(renderer.domElement);

  const camera = new THREE.PerspectiveCamera(58, window.innerWidth / window.innerHeight, 0.1, 200);
  camera.position.set(0, 6, 46);

  const controls = new THREE.OrbitControls(camera, renderer.domElement);
  controls.enableDamping = true;
  controls.dampingFactor = 0.07;
  controls.minDistance = 10;
  controls.maxDistance = 60;
  controls.maxPolarAngle = Math.PI / 1.8;
  controls.target.set(0, 2, 0);

  /* ============================================================
     TANK DIMENSIONS
     ============================================================ */
  const TW = 36, TH = 24, TD = 20;               // width / height / depth
  const HW = TW / 2, HH = TH / 2, HD = TD / 2;

  /* ============================================================
     LIGHTING
     ============================================================ */
  const ambient = new THREE.AmbientLight(0x404040, 0.4);
  scene.add(ambient);

  const sun = new THREE.DirectionalLight(0xffffff, 0.95);
  sun.position.set(14, 34, 16);
  sun.castShadow = true;
  sun.shadow.mapSize.set(2048, 2048);
  sun.shadow.camera.near = 1;
  sun.shadow.camera.far = 100;
  const sd = 34;
  sun.shadow.camera.left = -sd; sun.shadow.camera.right = sd;
  sun.shadow.camera.top = sd; sun.shadow.camera.bottom = -sd;
  sun.shadow.bias = -0.0004;
  scene.add(sun);

  const waterLight1 = new THREE.PointLight(0x00bcd4, 0.9, 70);
  waterLight1.position.set(-12, 4, 8);
  scene.add(waterLight1);

  const waterLight2 = new THREE.PointLight(0x2962ff, 0.7, 70);
  waterLight2.position.set(12, -4, -8);
  scene.add(waterLight2);

  /* ============================================================
     GLASS TANK
     ============================================================ */
  const glassMat = new THREE.MeshPhysicalMaterial({
    color: 0xaad8ff, metalness: 0, roughness: 0.05,
    transmission: 0.95, transparent: true, opacity: 0.28,
    ior: 1.33, thickness: 1.2, side: THREE.DoubleSide,
    clearcoat: 1, clearcoatRoughness: 0.05
  });
  const tank = new THREE.Mesh(new THREE.BoxGeometry(TW, TH, TD), glassMat);
  tank.material.transparent = true;
  scene.add(tank);

  const edgeMat = new THREE.LineBasicMaterial({ color: 0x6fbfe0, transparent: true, opacity: 0.55 });
  const edgeGeo = new THREE.EdgesGeometry(new THREE.BoxGeometry(TW, TH, TD));
  const edgeLines = new THREE.LineSegments(edgeGeo, edgeMat);
  scene.add(edgeLines);

  /* ============================================================
     PEBBLES / ROCKS
     ============================================================ */
  const rockMat = new THREE.MeshStandardMaterial({ color: 0x6b6560, roughness: 0.92, metalness: 0.05, flatShading: true });
  const rocks = [];
  for (let i = 0; i < 8; i++) {
    const g = new THREE.DodecahedronGeometry(0.9 + Math.random() * 1.6, 0);
    const pos = g.attributes.position;
    for (let v = 0; v < pos.count; v++) {
      pos.setXYZ(v, pos.getX(v) * (0.7 + Math.random() * 0.5),
                        pos.getY(v) * (0.7 + Math.random() * 0.5),
                        pos.getZ(v) * (0.7 + Math.random() * 0.5));
    }
    g.computeVertexNormals();
    const m = new THREE.Mesh(g, rockMat.clone());
    m.material.color.setHSL(0.08, 0.15, 0.28 + Math.random() * 0.12);
    const sx = (Math.random() - 0.5) * (TW - 6);
    const sz = (Math.random() - 0.5) * (TD - 6);
    m.position.set(sx, 0.3 + Math.random() * 0.6, sz);
    m.rotation.set(Math.random() * Math.PI, Math.random() * Math.PI, Math.random() * Math.PI);
    m.scale.y *= 0.6 + Math.random() * 0.5;
    m.castShadow = true; m.receiveShadow = true;
    scene.add(m); rocks.push(m);
  }

  /* ============================================================
     SAND FLOOR
     ============================================================ */
  const sandMat = new THREE.MeshStandardMaterial({ color: 0xc2a668, roughness: 1, metalness: 0, flatShading: true });
  const sandGeo = new THREE.PlaneGeometry(TW - 0.5, TD - 0.5, 30, 22);
  sandGeo.rotateX(-Math.PI / 2);
  const sPos = sandGeo.attributes.position;
  for (let i = 0; i < sPos.count; i++) {
    sPos.setY(i, (Math.random() - 0.5) * 0.55);
  }
  sandGeo.computeVertexNormals();
  const sand = new THREE.Mesh(sandGeo, sandMat);
  sand.position.y = -HH + 0.15;
  sand.receiveShadow = true;
  scene.add(sand);

  /* ============================================================
     SEAWEED
     ============================================================ */
  const seaweed = [];
  function makeSeaweed() {
    const segs = 7 + Math.floor(Math.random() * 4);
    const h = 3.5 + Math.random() * 4.5;
    const base = new THREE.Vector3((Math.random() - 0.5) * (TW - 4), -HH + 0.4, (Math.random() - 0.5) * (TD - 4));
    const pts = [];
    let x = 0, z = 0;
    for (let i = 0; i <= segs; i++) {
      const t = i / segs;
      pts.push(new THREE.Vector3(x, t * h, z));
      x += Math.sin(t * 2.2) * 0.9;
      z += Math.cos(t * 1.7) * 0.6;
    }
    const curve = new THREE.CatmullRomCurve3(pts);
    const geo = new THREE.TubeGeometry(curve, segs * 4, 0.16 + Math.random() * 0.12, 5, false);
    const mat = new THREE.MeshStandardMaterial({
      color: new THREE.Color().setHSL(0.28 + Math.random() * 0.08, 0.6, 0.28 + Math.random() * 0.12),
      roughness: 0.7, metalness: 0, side: THREE.DoubleSide, flatShading: true
    });
    const m = new THREE.Mesh(geo, mat);
    m.position.copy(base);
    m.castShadow = true;
    scene.add(m);
    seaweed.push({ mesh: m, phase: Math.random() * Math.PI * 2, spd: 0.6 + Math.random() * 0.8 });
  }
  for (let i = 0; i < 12; i++) makeSeaweed();

  /* ============================================================
     FISH CREATION
     ============================================================ */
  const fishColors = [
    { body: 0xff8c1a, fin: 0xffb347 },   // orange
    { body: 0x1e90ff, fin: 0x63b3ff },    // blue
    { body: 0xffd700, fin: 0xff3030 },    // yellow-red
    { body: 0x9b30ff, fin: 0xc77dff },    // purple
    { body: 0xff2d2d, fin: 0xff6b6b },    // red
    { body: 0x2ecc71, fin: 0x6dffb0 },    // green
    { body: 0xff69b4, fin: 0xffa0d0 },    // pink
    { body: 0xffc300, fin: 0xffe066 }     // gold
  ];

  function createFish() {
    const c = fishColors[Math.floor(Math.random() * fishColors.length)];
    const sc = 0.6 + Math.random() * 0.6;
    const grp = new THREE.Group();

    const bodyMat = new THREE.MeshStandardMaterial({ color: c.body, roughness: 0.28, metalness: 0.12, flatShading: true });
    const finMat  = new THREE.MeshStandardMaterial({ color: c.fin, roughness: 0.4, metalness: 0.05, transparent: true, opacity: 0.85, flatShading: true });

    const body = new THREE.Mesh(new THREE.SphereGeometry(1, 16, 12), bodyMat);
    body.scale.set(1.5, 0.75, 0.7);
    body.castShadow = true;
    grp.add(body);

    // head
    const head = new THREE.Mesh(new THREE.SphereGeometry(0.55, 12, 10), bodyMat);
    head.position.set(1.25, 0.05, 0);
    head.scale.set(1, 0.9, 0.9);
    head.castShadow = true;
    grp.add(head);

    // eyes
    const eyeMat = new THREE.MeshStandardMaterial({ color: 0xffffff, roughness: 0.2 });
    const pupilMat = new THREE.MeshStandardMaterial({ color: 0x111111, roughness: 0.4 });
    [-1, 1].forEach(s => {
      const eye = new THREE.Mesh(new THREE.SphereGeometry(0.16, 10, 8), eyeMat);
      eye.position.set(1.45, 0.18, s * 0.34);
      grp.add(eye);
      const pupil = new THREE.Mesh(new THREE.SphereGeometry(0.09, 8, 6), pupilMat);
      pupil.position.set(1.58, 0.18, s * 0.34);
      grp.add(pupil);
    });

    // tail
    const tailShape = new THREE.Shape();
    tailShape.moveTo(0, 0);
    tailShape.lineTo(-0.7, 0.55);
    tailShape.lineTo(-0.95, 0.15);
    tailShape.lineTo(-0.75, -0.5);
    tailShape.lineTo(-0.4, 0);
    tailShape.closePath();
    const tailGeo = new THREE.ShapeGeometry(tailShape, 8);
    const tail = new THREE.Mesh(tailGeo, finMat);
    tail.position.set(-1.5, 0, 0);
    grp.add(tail);

    // dorsal fin
    const dShape = new THREE.Shape();
    dShape.moveTo(0, 0);
    dShape.lineTo(0.25, 0.55);
    dShape.lineTo(0.55, 0.05);
    dShape.closePath();
    const dorsal = new THREE.Mesh(new THREE.ShapeGeometry(dShape, 6), finMat);
    dorsal.position.set(-0.1, 0.62, 0);
    dorsal.rotation.z = -0.15;
    grp.add(dorsal);

    // side fins
    const fShape = new THREE.Shape();
    fShape.moveTo(0, 0);
    fShape.lineTo(-0.4, -0.45);
    fShape.lineTo(0.3, -0.5);
    fShape.lineTo(0.45, -0.05);
    fShape.closePath();
    const leftFin = new THREE.Mesh(new THREE.ShapeGeometry(fShape, 6), finMat);
    leftFin.position.set(0.1, -0.35, 0.55);
    leftFin.rotation.x = -0.3;
    grp.add(leftFin);
    const rightFin = new THREE.Mesh(new THREE.ShapeGeometry(fShape, 6), finMat);
    rightFin.position.set(0.1, -0.35, -0.55);
    rightFin.rotation.x = 0.3;
    grp.add(rightFin);

    grp.scale.setScalar(sc);
    grp.position.set(
      (Math.random() - 0.5) * (TW - 8),
      (Math.random() - 0.5) * (TH - 8),
      (Math.random() - 0.5) * (TD - 8)
    );
    scene.add(grp);

    const dir = new THREE.Vector3(Math.random() - 0.5, (Math.random() - 0.5) * 0.4, Math.random() - 0.5).normalize();
    return {
      mesh: grp, tail, leftFin, rightFin, dorsal,
      velocity: dir.multiplyScalar(3 + Math.random() * 3),
      speed: 3 + Math.random() * 3.5,
      tailSpeed: 4 + Math.random() * 5,
      phase: Math.random() * Math.PI * 2,
      targetFood: null,
      avoidanceRadius: 3.5 + Math.random() * 2.5,
      size: sc
    };
  }

  const fishArray = [];
  for (let i = 0; i < 15; i++) fishArray.push(createFish());

  /* ============================================================
     BUBBLES
     ============================================================ */
  const bubbles = [];
  const bubbleMat = new THREE.MeshPhysicalMaterial({
    color: 0xcffefe, metalness: 0, roughness: 0.02, transmission: 0.98,
    transparent: true, opacity: 0.35, ior: 1.1, thickness: 0.4
  });
  function makeBubble(y) {
    const r = 0.18 + Math.random() * 0.4;
    const m = new THREE.Mesh(new THREE.SphereGeometry(r, 12, 10), bubbleMat);
    m.position.set((Math.random() - 0.5) * (TW - 4), y === undefined ? -HH + 1 + Math.random() * (TH - 2) : y, (Math.random() - 0.5) * (TD - 4));
    scene.add(m);
    bubbles.push({ mesh: m, r, spd: 1.5 + Math.random() * 2.5, ph: Math.random() * Math.PI * 2 });
  }
  for (let i = 0; i < 30; i++) makeBubble();

  /* ============================================================
     FOOD SYSTEM
     ============================================================ */
  const foods = [];
  const foodMat = new THREE.MeshStandardMaterial({ color: 0x8b5e34, roughness: 0.8, metalness: 0.1, flatShading: true });

  function dropFood(pos) {
    const r = 0.25 + Math.random() * 0.2;
    const m = new THREE.Mesh(new THREE.IcosahedronGeometry(r, 0), foodMat);
    m.position.copy(pos);
    m.castShadow = true;
    scene.add(m);
    foods.push({ mesh: m, vy: 0, r });
  }

  /* ============================================================
     INTERACTION (raycast click to feed)
     ============================================================ */
  const raycaster = new THREE.Raycaster();
  const ndc = new THREE.Vector2();
  const groundPlane = new THREE.Plane(new THREE.Vector3(0, 1, 0), 0);
  const hitPoint = new THREE.Vector3();
  let lastDown = null;

  renderer.domElement.addEventListener('pointerdown', e => { lastDown = { x: e.clientX, y: e.clientY }; });
  renderer.domElement.addEventListener('pointerup', e => {
    if (!lastDown) return;
    const moved = Math.hypot(e.clientX - lastDown.x, e.clientY - lastDown.y);
    const wasLeft = e.button === 0;
    lastDown = null;
    if (moved > 6 || !wasLeft) return;

    ndc.x = (e.clientX / window.innerWidth) * 2 - 1;
    ndc.y = -(e.clientY / window.innerHeight) * 2 + 1;
    raycaster.setFromCamera(ndc, camera);

    // try tank front face first
    const tankFace = new THREE.Plane(new THREE.Vector3(0, 0, 1), HD);
    let ok = raycaster.intersectPlane(tankFace, hitPoint);
    if (!ok) {
      // fallback: project onto sand plane (bottom of tank)
      ok = raycaster.intersectPlane(groundPlane, hitPoint);
    }
    if (ok && hitPoint.x > -HW && hitPoint.x < HW && hitPoint.z > -HD && hitPoint.z < HD) {
      const y = Math.max(hitPoint.y - 2, -HH + 2);
      dropFood(new THREE.Vector3(hitPoint.x, TH - 1.5, hitPoint.z));
      showTip("🍤 Корм опущен! Рыбки плывут на обед");
    }
  });

  // hide tooltip after a few seconds
  setTimeout(() => { const t = document.getElementById('tooltip'); if (t) t.style.opacity = '0'; }, 7000);

  function showTip(msg) {
    const t = document.getElementById('tooltip');
    if (!t) return;
    t.textContent = msg;
    t.style.opacity = '1';
    clearTimeout(showTip._t);
    showTip._t = setTimeout(() => { t.style.opacity = '0'; }, 2500);
  }

  /* ============================================================
     UI BUTTONS
     ============================================================ */
  document.getElementById('addFishBtn').addEventListener('click', () => {
    if (fishArray.length >= 60) return;
    fishArray.push(createFish());
  });
  document.getElementById('addBubblesBtn').addEventListener('click', () => {
    for (let i = 0; i < 10; i++) makeBubble();
  });
  let lightOn = true;
  document.getElementById('lightBtn').addEventListener('click', function () {
    lightOn = !lightOn;
    sun.intensity = lightOn ? 0.95 : 0.12;
    this.textContent = lightOn ? "💡 Свет: ВКЛ" : "💡 Свет: ВЫКЛ";
    this.classList.toggle('on', lightOn);
  });

  /* ============================================================
     BEHAVIOR HELPERS
     ============================================================ */
  const _v = new THREE.Vector3();
  const _v2 = new THREE.Vector3();
  const _v3 = new THREE.Vector3();

  function clampTank(p, pad) {
    p.x = THREE.MathUtils.clamp(p.x, -HW + pad, HW - pad);
    p.y = THREE.MathUtils.clamp(p.y, -HH + pad, HH - pad);
    p.z = THREE.MathUtils.clamp(p.z, -HD + pad, HD - pad);
  }

  /* ============================================================
     ANIMATION LOOP
     ============================================================ */
  const clock = new THREE.Clock();
  let frames = 0, accTime = 0, fps = 0;

  function animate() {
    requestAnimationFrame(animate);
    const dt = Math.min(clock.getDelta(), 0.05);
    const t = clock.elapsedTime;

    // ---- FPS ----
    frames++; accTime += dt;
    if (accTime >= 0.5) { fps = Math.round(frames / accTime); frames = 0; accTime = 0; }

    // ---- FOOD physics ----
    for (let i = foods.length - 1; i >= 0; i--) {
      const f = foods[i];
      f.vy -= 6 * dt;
      f.mesh.position.y += f.vy * dt;
      f.mesh.rotation.x += dt * 2;
      f.mesh.rotation.z += dt * 1.5;

      let eaten = false;
      for (const fish of fishArray) {
        if (fish.mesh.position.distanceTo(f.mesh.position) < fish.size * 1.6 + f.r) {
          eaten = true;
          // grow
          const ns = fish.size * 1.05;
          fish.size = Math.min(ns, 3);
          fish.mesh.scale.setScalar(fish.size);
          // brief swim toward food already tracked
          break;
        }
      }

      if (eaten) { scene.remove(f.mesh); foods.splice(i, 1); continue; }
      if (f.mesh.position.y - f.r <= -HH + 0.3) { scene.remove(f.mesh); foods.splice(i, 1); }
    }

    // ---- FISH behavior ----
    for (let i = 0; i < fishArray.length; i++) {
      const fish = fishArray[i];
      const m = fish.mesh;
      const vel = fish.velocity;

      // seek nearest food within radius 15
      fish.targetFood = null;
      let bestDist = 15;
      for (const f of foods) {
        const d = m.position.distanceTo(f.mesh.position);
        if (d < bestDist) { bestDist = d; fish.targetFood = f.mesh.position; }
      }

      _v.copy(vel).normalize(); // direction

      if (fish.targetFood) {
        // steer toward food
        _v2.copy(fish.targetFood).sub(m.position).normalize();
        vel.lerp(_v2.multiplyScalar(fish.speed), 0.06);
      } else {
        // random wandering
        if (Math.random() < 0.012) {
          vel.x += (Math.random() - 0.5) * 3;
          vel.y += (Math.random() - 0.5) * 2;
          vel.z += (Math.random() - 0.5) * 3;
        }
      }

      // avoid other fish
      for (let j = 0; j < fishArray.length; j++) {
        if (j === i) continue;
        _v3.copy(m.position).sub(fishArray[j].mesh.position);
        const d2 = _v3.lengthSq();
        const ar = fish.avoidanceRadius * (fish.size + fishArray[j].size);
        if (d2 < ar * ar && d2 > 0.0001) {
          const inv = 1 / Math.sqrt(d2);
          _v3.multiplyScalar(inv * (ar - Math.sqrt(d2)) * 0.5);
          vel.add(_v3);
        }
      }

      // wall repulsion (smooth)
      const margin = 3.5;
      if (m.position.x < -HW + margin) vel.x += (-HW + margin - m.position.x) * 0.5;
      if (m.position.x >  HW - margin) vel.x += ( HW - margin - m.position.x) * 0.5;
      if (m.position.y < -HH + margin) vel.y += (-HH + margin - m.position.y) * 0.5;
      if (m.position.y >  HH - margin) vel.y += ( HH - margin - m.position.y) * 0.5;
      if (m.position.z < -HD + margin) vel.z += (-HD + margin - m.position.z) * 0.5;
      if (m.position.z >  HD - margin) vel.z += (HD - margin - m.position.z) * 0.5;

      // clamp speed
      const sp = vel.length();
      const maxSp = fish.speed * 2.4;
      if (sp > maxSp) vel.multiplyScalar(maxSp / sp);
      if (sp < fish.speed * 0.35) vel.multiplyScalar(fish.speed * 0.35 / (sp || 1));

      // move
      m.position.addScaledVector(vel, dt);
      clampTank(m.position, 0.8);

      // orient fish to facing direction of travel
      if (sp > 0.01) {
        _v.copy(vel).normalize();
        const angle = Math.atan2(_v.x, _v.z);
        m.rotation.y = THREE.MathUtils.lerp(m.rotation.y, angle, 0.12);
      }

      // slight vertical drift / bobbing
      m.position.y += Math.sin(t * fish.tailSpeed + fish.phase) * 0.15 * dt;

      // tail wag
      const wag = Math.sin(t * fish.tailSpeed + fish.phase) * 0.45;
      fish.tail.rotation.z = wag;
      // fin movement
      fish.leftFin.rotation.x = -0.3 + Math.sin(t * fish.tailSpeed * 0.8 + fish.phase) * 0.5;
      fish.rightFin.rotation.x =  0.3 - Math.sin(t * fish.tailSpeed * 0.8 + fish.phase) * 0.5;
      fish.dorsal.rotation.z = -0.15 + Math.sin(t * fish.tailSpeed * 0.5 + fish.phase) * 0.12;
    }

    // ---- BUBBLES ----
    for (const b of bubbles) {
      b.mesh.position.y += b.spd * dt;
      b.mesh.position.x += Math.sin(t * 1.3 + b.ph) * 0.4 * dt;
      b.mesh.position.z += Math.cos(t * 1.1 + b.ph) * 0.3 * dt;
      if (b.mesh.position.y > HH - 0.5) {
        b.mesh.position.y = -HH + 1 + Math.random() * (TH - 3);
        b.mesh.position.x = (Math.random() - 0.5) * (TW - 4);
        b.mesh.position.z = (Math.random() - 0.5) * (TD - 4);
      }
    }

    // ---- Seaweed sway ----
    for (const s of seaweed) {
      s.mesh.rotation.x = Math.sin(t * s.spd + s.phase) * 0.06;
      s.mesh.rotation.z = Math.cos(t * s.spd * 0.8 + s.phase) * 0.05;
    }

    controls.update();
    renderer.render(scene, camera);

    // ---- stats ----
    document.getElementById('fishCount').textContent = fishArray.length;
    document.getElementById('bubbleCount').textContent = bubbles.length;
    document.getElementById('foodCount').textContent = foods.length;
    document.getElementById('fpsCount').textContent = fps;
  }

  animate();

  /* ============================================================
     RESIZE
     ============================================================ */
  window.addEventListener('resize', () => {
    camera.aspect = window.innerWidth / window.innerHeight;
    camera.updateProjectionMatrix();
    renderer.setSize(window.innerWidth, window.innerHeight);
  });

})();
