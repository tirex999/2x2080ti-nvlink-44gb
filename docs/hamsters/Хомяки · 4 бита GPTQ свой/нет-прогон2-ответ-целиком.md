  constructor(name, colors, pos, facing){

  pickState(){
    const r = Math.random();
    if(r < 0.55) this.setState('wander');
    else if(r < 0.75) this.setState('pause');
    else if(r < 0.9) this.setState('turn');
    else this.setState('interact');

  setState(s){
    this.state = s;

      let t;
      do { t = new THREE.Vector3((Math.random()*2-1)*W*0.4, GY, (Math.random()*2-1)*D*0.4); }
      while(t.distanceTo(this.pos) < 0.5);
      this.wanderTarget = t;
      this.stateTimer = 3 + Math.random()*3;
    } else if(s==='pause'){ this.stateTimer = 0.8 + Math.random()*1.6; }
    else if(s==='turn'){ this.stateTimer = 0.6 + Math.random()*0.6; }
    else if(s==='interact'){
      const targets = [
        {kind:'wheel', pos:new THREE.Vector3(-1.5, GY, -0.5)},
        {kind:'tunnel', pos:new THREE.Vector3(1.5, GY, 1.0)},
        {kind:'bowl',   pos:new THREE.Vector3(0.2, GY, 1.5)},
      ];
      const t = targets[Math.floor(Math.random()*targets.length)];
      this.interactTarget = t;
      this.stateTimer = 0;
    }
  }
  update(dt){
    this.stateTimer -= dt;
    const targetSpeed = this.speed * (this.state==='walk'||this.state==='wander' ? 1 : 0);
    this.speed += (targetSpeed - this.speed) * Math.min(1, dt*6);

    if(this.state === 'wander' && this.wanderTarget){
      const dir = this.wanderTarget.clone().sub(this.pos); dir.y = 0;
      const dist = dir.length();
      if(dist < 0.25){ this.pickState(); }
      else {
        dir.normalize();
        const desired = Math.atan2(dir.x, dir.z);
        this.facing = angleLerp(this.facing, desired, Math.min(1, dt*this.turnRate));
        this.pos.addScaledVector(dir, this.speed*dt);
      }
    } else if(this.state === 'turn'){
      this.facing += dt * 3.2 * (Math.random()<0.5?1:-1);
      if(this.stateTimer <= 0) this.pickState();
    } else if(this.state === 'interact'){
      const t = this.interactTarget;
      const dir = t.pos.clone().sub(this.pos); dir.y = 0;
      const dist = dir.length();
      if(dist > 0.4){
        dir.normalize();
        const desired = Math.atan2(dir.x, dir.z);
        this.facing = angleLerp(this.facing, desired, Math.min(1, dt*this.turnRate));
        this.pos.addScaledVector(dir, this.speed*dt);
      } else {
        // arrived -> face object center & act
        const center = new THREE.Vector3(t.kind==='wheel'?-1.5:t.kind==='tunnel'?1.5:0.2, GY, t.kind==='wheel'?-1.1:t.kind==='tunnel'?1.0:1.5);
        const d = center.clone().sub(this.pos); d.y=0;
        this.facing = angleLerp(this.facing, Math.atan2(d.x,d.z), Math.min(1,dt*5));
        this.interactTimer += dt;
        if(this.interactTimer > 1.6){ this.interactTimer = 0; this.pickState(); }
      }
    } else {
      // pause / walk-idle
      if(this.stateTimer <= 0) this.pickState();
    }

    // clamp inside cage
    this.pos.x = THREE.MathUtils.clamp(this.pos.x, -W*0.42, W*0.42);
    this.pos.z = THREE.MathUtils.clamp(this.pos.z, -D*0.42, D*0.42);

    // apply transform
    this.group.position.x = this.pos.x;
    this.group.position.z = this.pos.z;
    this.group.rotation.y = this.facing;

    // bobbing + leg wiggle
    const moving = this.speed > 0.1;
    this.walkPhase += dt * (moving ? 10*this.speed : 3);
    const bob = moving ? Math.abs(Math.sin(this.walkPhase))*0.03 : Math.sin(this.walkPhase*0.5)*0.015;
    this.group.position.y = GY + bob + this.hop;

    // hop physics
    if(this.hopVel !== 0 || this.hop > 0){
      this.hop += this.hopVel*dt;
      this.hopVel -= 9*dt;
      if(this.hop <= 0){ this.hop = 0; this.hopVel = 0; }
    }

    // squash & stretch
    const sx = 1 + (this.hop>0? -0.12*Math.min(1,this.hop*2):0);
    const sy = 1 + (this.hop>0?  0.15*Math.min(1,this.hop*2):0);
    this.group.scale.set(sx, sy, sx);
  }
  hop(){ this.hopVel = 2.2; this.hop = Math.max(this.hop, 0.001); }
}
function angleLerp(a,b,t){
  let d = ((b - a + Math.PI) % (Math.PI*2)) - Math.PI;
  if(d < -Math.PI) d += Math.PI*2;
  return a + d*t;
}
