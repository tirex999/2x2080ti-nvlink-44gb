import * as THREE from 'three';
import { OrbitControls } from 'three/addons/controls/OrbitControls.js';

/* ---------- tiny helpers ---------- */
const rand   = (a,b)=> a + Math.random()*(b-a);
const randi  = (a,b)=> Math.floor(rand(a,b+1));
const pick   = arr => arr[Math.floor(Math.random()*arr.length)];
const clamp  = (v,a,b)=> Math.max(a,Math.min(b,v));
const damp   = THREE.MathUtils.damp;
function angDiff(a,b){ let d=(a-b)%(Math.PI*2); if(d>Math.PI)d-=Math.PI*2; if(d<-Math.PI)d+=Math.PI*2; return d; }

/* ---------- renderer / scene / camera ---------- */
const canvas = document.getElementById('scene');
let renderer;

  renderer = new THREE.WebGLRenderer({canvas, antialias:true});

function makeSky(){
  const c = document.createElement('canvas'); c.width=8; c.height=256;
  const x = c.getContext('2d');
  const g = x.createLinearGradient(0,0,0,256);
  g.addColorStop(0,'#ffe7b0'); g.addColorStop(.55,'#ffcf95'); g.addColorStop(1,'#ff9f6e');
  x.fillStyle=g; x.fillRect(0,0,8,256);
  const t = new THREE.CanvasTexture(c); t.colorSpace = THREE.SRGBColorSpace;
  return t;

function legGeo(h, r){ return new THREE.CylinderGeometry(r*.8, r, h, 6); }
function box(w,h,d,x,y,z,mat){ const m=new THREE.Mesh(new THREE.BoxGeometry(w,h,d),mat); m.position.set(x,y,z); m.castShadow=true; m.receiveShadow=true; return m; }

function makeCage(){
  const g = new THREE.Group();
  const W=12, D=8, H=4.4;
  // base tray
  const tray = new THREE.Mesh(new THREE.BoxGeometry(W,H*.4,D), M.wood);
  tray.position.y = H*.2; tray.castShadow=true; tray.receiveShadow=true; g.add(tray);
  const lip = new THREE.Mesh(new THREE.BoxGeometry(W+.3,.35,D+.3), M.woodD);
  lip.position.y = H*.4+.05; g.add(lip);
  // posts
  const ph=H*.85, pr=.14, pw=W/2-.25, pd=D/2-.25;
  [[pw,pd],[-pw,pd],[pw,-pd],[-pw,-pd]].forEach(([x,z])=>{
    const p=new THREE.Mesh(legGeo(ph,pr), M.metal); p.position.set(x,H*.4+ph/2,z); p.castShadow=true; g.add(p);
  });
  // horizontal rails (front/back/top)
  const rail=(w,d,x,y,z,mat)=>{const m=new THREE.Mesh(new THREE.BoxGeometry(w,.11,d),mat);m.position.set(x,y,z);m.castShadow=true;g.add(m);return m;};
  [H*.55, H*.8].forEach(y=>{ rail(W-.3,.1,0,y,pd,M.metal); rail(W-.3,.1,0,y,-pd,M.metal); });
  rail(W-.3,.1,0,H*.85,pd,M.mint); rail(W-.3,.1,0,H*.85,-pd,M.mint);
  rail(.1,D-.3,pw,H*.85,0,M.mint); rail(.1,D-.3,-pw,H*.85,0,M.mint);
  // side mesh panels
  const panel=(x,ry)=>{ const m=new THREE.Mesh(new THREE.PlaneGeometry(D-.5,H*.7),
    new THREE.MeshBasicMaterial({color:0xffffff, transparent:true, opacity:.12, side:THREE.DoubleSide}));
    m.position.set(x,H*.4+H*.45,0); m.rotation.y=ry; g.add(m); };
  panel(pw,Math.PI/2); panel(-pw,Math.PI/2);
  return g;
}

function makeWheel(pos){
  const g = new THREE.Group();
  const R=1.15, tube=.13;
  const ring = new THREE.Mesh(new THREE.TorusGeometry(R,tube,10,26), M.coral);
  ring.castShadow=true; g.add(ring);
  const hub = new THREE.Mesh(new THREE.CylinderGeometry(.3,.3,.18,12), M.sun);
  hub.rotation.x=Math.PI/2; g.add(hub);
  const spokes=[];
  for(let i=0;i<6;i++){
    const a=i/6*Math.PI*2;
    const s=new THREE.Mesh(new THREE.CylinderGeometry(.05,.05,R,6), M.coral);
    s.position.set(Math.cos(a)*R/2, Math.sin(a)*R/2, 0);
    s.rotation.z=a+Math.PI/2; s.castShadow=true; g.add(s); spokes.push(s);
  }
  // stand
  const base=new THREE.Mesh(new THREE.BoxGeometry(1.7,.22,1.7), M.blue); base.position.y=.11; base.castShadow=true; g.add(base);
  const post=new THREE.Mesh(legGeo(1.5,.14), M.metal); post.position.set(0,1.5/2+.22,0); post.castShadow=true; g.add(post);
  const arm=new THREE.Mesh(new THREE.BoxGeometry(.16,.16,1.1), M.metal); arm.position.set(0,2.15,.6); arm.castShadow=true; g.add(arm);
  const axle=new THREE.Mesh(new THREE.CylinderGeometry(.09,.09,.5,8), M.metal); axle.rotation.x=Math.PI/2; axle.position.set(0,2.15,.85); g.add(axle);
  g.userData.spokes=spokes; g.userData.ring=ring;
  g.position.copy(pos);
  return g;
}

function makeTunnel(pos, ry){
  const g=new THREE.Group();
  const len=2.6, r=.55;
  const tube=new THREE.Mesh(new THREE.CylinderGeometry(r,r,len,12,1,true), M.mint);
  tube.material.side=THREE.DoubleSide; tube.rotation.z=Math.PI/2; tube.castShadow=true; tube.receiveShadow=true; g.add(tube);
  const cap=new THREE.Mesh(new THREE.CircleGeometry(r,12), M.mint); cap.position.x=len/2; cap.rotation.y=Math.PI/2; g.add(cap);
  // hay peeking out
  for(let i=0;i<7;i++){
    const h=new THREE.Mesh(new THREE.ConeGeometry(.16,.5,5), M.hay);
    const a=rand(0,Math.PI*2);
    h.position.set(len/2-.05, Math.cos(a)*r*.6, Math.sin(a)*r*.6);
    h.rotation.set(rand(-.5,.5),rand(0,6),rand(-.5,.5)); g.add(h);
  }
  g.rotation.y=ry; g.position.copy(pos);
  return g;
}

function makeHamster(bodyHex, bellyHex, cheekHex){
  const g = new THREE.Group();
  const bodyMat = new THREE.MeshStandardMaterial({color:bodyHex, roughness:.85, flatShading:true});
  const bellyMat= new THREE.MeshStandardMaterial({color:bellyHex,roughness:.85, flatShading:true});
  const cheekMat= new THREE.MeshStandardMaterial({color:cheekHex, roughness:.8,  flatShading:true});
  const dark    = new THREE.MeshStandardMaterial({color:0x3a2a24, roughness:.6});
  const earMat  = new THREE.MeshStandardMaterial({color:0xffc7d8, roughness:.8, flatShading:true});
  const noseMat = new THREE.MeshStandardMaterial({color:0xff6f91, roughness:.6});

  const body = new THREE.Mesh(new THREE.SphereGeometry(1,7,6), bodyMat);
  body.scale.set(1.18,1.0,1.42); body.position.y=.72; body.castShadow=true; g.add(body);
  const belly = new THREE.Mesh(new THREE.SphereGeometry(1,7,6), bellyMat);
  belly.scale.set(1.02,.86,1.2); belly.position.set(0,.66,.16); g.add(belly);

  // head
  const head = new THREE.Mesh(new THREE.SphereGeometry(.6,7,6), bodyMat);
  head.scale.set(1,.95,1); head.position.set(0,.86,.98); head.castShadow=true; g.add(head);
  // snout
  const snout = new THREE.Mesh(new THREE.SphereGeometry(.34,6,5), bellyMat);
  snout.scale.set(1,.8,1.2); snout.position.set(0,.72,1.42); g.add(snout);
  // cheeks
  [-1,1].forEach(s=>{
    const ch=new THREE.Mesh(new THREE.SphereGeometry(.3,6,5), cheekMat);
    ch.scale.set(1,.9,1); ch.position.set(s*.42,.62,1.02); g.add(ch);
  });
  // eyes
  [-1,1].forEach(s=>{
    const e=new THREE.Mesh(new THREE.SphereGeometry(.12,6,5), dark);
    e.position.set(s*.28,.98,1.28); g.add(e);
    const sh=new THREE.Mesh(new THREE.SphereGeometry(.045,5,4), new THREE.MeshBasicMaterial({color:0xffffff}));
    sh.position.set(s*.24,1.03,1.4); g.add(sh);
  });
  // nose
  const nose=new THREE.Mesh(new THREE.SphereGeometry(.11,6,5), noseMat); nose.position.set(0,.72,1.7); g.add(nose);
  // ears
  [-1,1].forEach(s=>{
    const er=new THREE.Mesh(new THREE.SphereGeometry(.24,6,5), bodyMat);
    er.scale.set(1,1.1,1); er.position.set(s*.34,1.34,.7); er.castShadow=true; g.add(er);
    const ei=new THREE.Mesh(new THREE.SphereGeometry(.15,6,5), earMat);
    ei.position.set(s*.34,1.36,.82); g.add(ei);
  });
  // tail
  const tail=new THREE.Mesh(new THREE.SphereGeometry(.16,6,5), bodyMat);
  tail.scale.set(1,1,1.6); tail.position.set(0,.66,-1.25); g.add(tail);
  // legs
  const legs=[];
  const legPos=[[.4,.1,1.0],[-.4,.1,1.0],[.42,.1,-.95],[-.42,.1,-.95]];
  legPos.forEach(([x,y,z])=>{
    const leg=new THREE.Mesh(legGeo(.55,.16), bodyMat);
    leg.position.set(x,y,z); leg.castShadow=true; g.add(leg); legs.push(leg);
  });

  // whiskers (thin lines)
  const wMat=new THREE.LineBasicMaterial({color:0x8a6a5a, transparent:true, opacity:.5});
  [-1,1].forEach(s=>{
    for(let k=-1;k<=1;k++){
      const geo=new THREE.BufferGeometry().setFromPoints([
        new THREE.Vector3(s*.1,.72,1.55),
        new THREE.Vector3(s*.9,.72+k*.16,1.95)
      ]);
      g.add(new THREE.Line(geo,wMat));
    }
  });

  g.traverse(o=>{ if(o.isMesh) o.receiveShadow=false; });
  return { group:g, body, head, legs, parts:[body,head] };
}

function wanderTo(h){
  // pick a fresh target inside the play area
  h.target.set(rand(-4.4,4.4), 0, rand(-3.1,3.1));
  // occasionally head toward an object
  const roll = Math.random();
  if(roll<0.28){ h.target.copy(wheelCenter).add(new THREE.Vector3(rand(-1,1),0,rand(-1,1))); h.target.y=0; }
  else if(roll<0.44){ h.target.copy(tunnelPos).add(new THREE.Vector3(rand(-.6,.6),0,rand(-.6,.6))); h.target.y=0; }
  h.timer = rand(3,6);
}

function updateHamster(h, dt, t){
  const P = h.pos;

  // decay temporary effects
  h.squish = Math.max(0, h.squish - dt*2.5);
  if(h.boop>0){
    h.boop = Math.max(0, h.boop - dt*1.6);
    if(h.boopJump>0) h.boopJump = Math.max(0, h.boopJump - dt*2.2);
  }

  // ---- state machine ----
  switch(h.state){
    case 'wander':
      h.timer -= dt;
      const toT = h.target.clone().sub(P); toT.y=0;
      const dist = toT.length();
      if(dist < 0.4 || h.timer<=0){
        if(dist<0.4 && Math.random()<0.5){ h.state='pause'; h.pause=rand(1.2,3); h.speed=0; break; }
        wanderTo(h);
      }
      if(dist>0.05){
        const want = Math.atan2(toT.x, toT.z);
        h.yaw += angDiff(want, h.yaw) * Math.min(1, dt*4);
        h.speed = damp(h.speed, 2.4, 6, dt);
      } else h.speed = damp(h.speed, 0, 8, dt);
      break;

    case 'pause':
      h.speed = damp(h.speed, 0, 8, dt);
      h.pause -= dt;
      if(h.pause<=0){ h.state='wander'; wanderTo(h); }
      break;

    case 'eat':
      h.eatTimer -= dt;
      const toF = h.eatPos.clone().sub(P); toF.y=0;
      if(toF.length()>0.35){
        const want = Math.atan2(toF.x, toF.z);
        h.yaw += angDiff(want, h.yaw) * Math.min(1, dt*4);
        h.speed = damp(h.speed, 2.2, 6, dt);
      } else {
        h.speed = damp(h.speed, 0, 8, dt);
        h.yaw += angDiff(h.eatYaw, h.yaw) * Math.min(1, dt*6);
      }
      if(h.eatTimer<=0){ h.state='wander'; wanderTo(h); }
      break;

    case 'wheel':
      h.wheelTimer -= dt;
      // keep hamster glued to the wheel surface
      const a = h.wheelAng;
      const wx = wheelCenter.x + Math.sin(a)*1.15;
      const wz = wheelCenter.z + Math.cos(a)*1.15;
      P.x += (wx-P.x)*Math.min(1,dt*8);
      P.z += (wz-P.z)*Math.min(1,dt*8);
      // face the tangent direction of travel
      const dir = h.wheelDir;
      const tx = Math.cos(a)*dir, tz = -Math.sin(a)*dir;
      const want = Math.atan2(tx, tz);
      h.yaw += angDiff(want, h.yaw) * Math.min(1, dt*6);
      h.speed = 3.0;
      // advance around the wheel
      h.wheelAng += dir * (h.speed/1.15) * dt;
      if(h.wheelTimer<=0){
        h.state='wander';
        // hop off to the ground near the wheel
        P.x = wheelCenter.x + rand(-1.5,1.5);
        P.z = wheelCenter.z + rand(-1.5,1.5);
        P.y = 0;
        wanderTo(h);
      }
      break;
  }

  // join wheel when close enough (only while wandering/eating)
  if(h.state==='wander' || h.state==='pause'){
    const dw = Math.hypot(P.x-wheelCenter.x, P.z-wheelCenter.z);
    if(dw < 1.0 && h.wheel===null && Math.random()<0.02){
      enterWheel(h);
    }
  }

  // ---- integrate movement (ground states only) ----
  if(h.state!=='wheel'){
    const fx = Math.sin(h.yaw), fz = Math.cos(h.yaw);
    P.x += fx * h.speed * dt;
    P.z += fz * h.speed * dt;
    P.y = 0;
    // keep inside the tray
    P.x = clamp(P.x, -4.6, 4.6);
    P.z = clamp(P.z, -3.3, 3.3);
    // soft bounce off walls -> turn around
    if(Math.abs(P.x)>4.4 || Math.abs(P.z)>3.1){
      h.yaw += Math.PI * Math.min(1, dt*3);
    }
  }

  // ---- apply to node ----
  h.group.position.set(P.x, P.y, P.z);
  h.group.rotation.y = h.yaw;

  // bob while moving
  const moving = clamp(h.speed/2.4, 0, 1);
  const bob = Math.sin(t*10 + h.bobPhase) * 0.05 * moving;
  h.group.position.y += bob;

  // legs scissor
  const swing = Math.sin(t*11 + h.bobPhase) * 0.5 * moving;
  h.legs[0].rotation.x =  swing;
  h.legs[1].rotation.x = -swing;
  h.legs[2].rotation.x = -swing;
  h.legs[3].rotation.x =  swing;

  // head bob
  h.head.rotation.z = Math.sin(t*6 + h.bobPhase) * 0.05 * moving;

  // eating munch
  let munch = 0;
  if(h.state==='eat' && h.speed<0.5) munch = Math.sin(t*14)*0.12;
  h.head.rotation.x = munch;

  // blinking
  h.blinkTimer -= dt;
  if(h.blinkTimer<=0){ h.blink=1; h.blinkTimer=rand(2.5,6); }
  h.blink = Math.max(0, h.blink - dt*8);
  const eyeS = 1 - h.blink*0.9;
  h.parts.forEach(()=>{}); // placeholder
  // scale eyes via stored? we didn't store eyes; approximate via head squash is fine.

  // boop squish + jump
  const sq = 1 + h.squish*0.35;
  h.group.scale.set(1/sq, sq, 1/sq);
  if(h.boopJump>0){
    h.group.position.y += Math.sin((1-h.boopJump)*Math.PI) * 0.5;
  }
}

}

A few notes on what you're getting:

- **The scene** is a warm, cozy cage: wooden tray, metal bars with mint top-rails, a coral running wheel on a blue stand, a mint hay tunnel, a pink kibble bowl, a potted plant, and two hay tufts. Lighting uses a hemisphere fill plus a shadow-casting "sun" for that soft, friendly look.
- **Three hamsters** (Nutmeg, Peanut, Mochi) each have their own colors, chonky flat-shaded bodies, blush cheeks, blinking-ish faces, scissoring legs, and a bobbing gait.
- **Autonomous AI**: each one wanders to random targets, pauses to "think," and — the fun part — will randomly **climb onto the wheel** and run in place while the wheel actually spins (and slows back down when they hop off).
- **Interactivity**: drag to orbit, scroll to zoom, and click/tap any hamster to boop it (it squishes and hops). Buttons let you **boop everyone**, trigger **feed time** (they all trot to the bowl and munch), and toggle **auto-orbit**. The camera gently auto-orbits whenever you've been idle for a few seconds.
