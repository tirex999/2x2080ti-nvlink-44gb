function steerTo(h, tx, tz, speed, dt){
  const dx = tx - h.group.position.x, dz = tz - h.group.position.z;
  const dist = Math.hypot(dx,dz);
  if(dist < .25) return true;
  const desired = Math.atan2(dx, dz);
  h.yaw = angLerp(h.yaw, desired, 1 - Math.exp(-7*dt));
  h.group.rotation.y = h.yaw;
  const fx = Math.sin(h.yaw), fz = Math.cos(h.yaw);
  h.group.position.x += fx*speed*dt;
  h.group.position.z += fz*speed*dt;
  resolveKeepouts(h, h.ignore);
  clampBounds(h);
  h.phase += dt * speed * 2.6;
  return false;
}

function updateHamster(h, dt, t){
  h.t += dt;
  h.movingThisFrame = false;
  let done=false;
  switch(h.state){
   case 'IDLE':
     if(h.t > h.dur) done=true; break;
   case 'WANDER':
     if(steerTo(h, h.target.x, h.target.z, 2.3, dt)) done=true;
     else h.movingThisFrame = true;
     if(h.t > h.dur) done=true; break;
   case 'EAT':
     if(!h.atBowl){
        if(steerTo(h, BOWL.x-1.15, BOWL.z, 2.6, dt)) { h.atBowl=true; h.eatClock=0; }
        else h.movingThisFrame=true;
     } else {
        h.faceToward(BOWL.x, BOWL.z, dt);
        h.eatClock += dt;
        if(h.eatClock > .7){ h.eatClock=0; tryEat(h); }
        if(h.t > h.dur || seeds.length===0) done=true;
     }
     break;
   case 'WHEEL':
     if(!h.atWheel){
       if(steerTo(h, WHEEL.x+1.35, WHEEL.z, 2.8, dt)){ h.atWheel=true; h.runClock=0; }
       else h.movingThisFrame=true;
     } else {
       h.yaw = angLerp(h.yaw, Math.PI/2, 1-Math.exp(-8*dt));
       h.group.rotation.y = h.yaw;
       h.phase += dt*14;
       h.movingThisFrame = true;
       if(h.t > h.dur) done=true;
     }
     break;
   case 'TUNNEL':
     if(h.phaseT === 0){
       if(steerTo(h, TUN_IN.x, TUN_IN.z, 2.6, dt)){ h.phaseT=1; h.t=0; h.dur=.55; h.group.scale.setScalar(.01); }
       else h.movingThisFrame=true;
     } else if(h.phaseT===1){
       if(h.t>h.dur){ // pop out
         h.group.position.set(TUN_OUT.x, 0, TUN_OUT.z);
         h.yaw = Math.atan2(TUN_OUT.x-TUN_IN.x, TUN_OUT.z-TUN_IN.z);
         h.phaseT=2; h.t=0; h.dur=.5; h.hop=1;
       }
     } else {
       if(h.hop>0){ h.hop=Math.max(0,h.hop-dt*2.2); h.group.position.y = Math.sin((1-h.hop)*Math.PI)*.5; }

     }
     break;
   case 'CHASE':
     if(steerTo(h, ball.position.x, ball.position.z, 3.3, dt)) done=true;
     else h.movingThisFrame=true;
     if(h.t>h.dur || ballSpeed() < .5) done=true;

   case 'BOING':

     if(h.t>h.dur){ h.group.position.y=0; done=true; }

function updateHamster(h, dt, t){
  h.t+=dt; h.moving=false; let done=false;
  const p=h.group.position;
  switch(h.state){
    case 'IDLE': if(h.t>h.dur) done=true; break;
    case 'WANDER':
      if(steerTo(h,h.target.x,h.target.z,2.3,dt)) done=true;
      if(h.t>h.dur) done=true;
      break;
    case 'EAT':
      if(!h.atBowl){
        if(steerTo(h,BOWL_POS.x-1.2,BOWL_POS.z,2.6,dt)) { h.atBowl=true; h.eatClock=0; }
      } else {
        faceToward(h,BOWL_POS.x,BOWL_POS.z,dt);
        h.eatClock+=dt;
        if(h.eatClock>.7){ h.eatClock=0; if(!tryEat(h)) done=true; }
        if(h.t>h.dur) done=true;
      }
      break;
    case 'WHEEL':
      if(!h.atWheel){
        if(steerTo(h,WHEEL_POS.x+1.3,WHEEL_POS.z,2.8,dt)) h.atWheel=true;
      } else {
        faceToward(h,WHEEL_POS.x,WHEEL_POS.z,dt);
        h.phase+=dt*13; h.moving=true;
        if(h.t>h.dur) done=true;
      }
      break;
    case 'TUNNEL':
      if(h.phaseT===0){
        if(steerTo(h,TUN_IN.x,TUN_IN.z,2.6,dt)){ h.phaseT=1; h.t=0; h.dur=.55; h.group.scale.setScalar(.01); }
      } else if(h.phaseT===1){
        if(h.t>h.dur){
          p.set(TUN_OUT.x,0,TUN_OUT.z);
          h.yaw=Math.atan2(TUN_OUT.x-TUN_IN.x,TUN_OUT.z-TUN_IN.z);
          h.group.rotation.y=h.yaw;
          h.phaseT=2; h.t=0; h.dur=.55; h.hop=1;
          h.group.scale.setScalar(1);

      } else {

        p.y=Math.sin((1-h.hop)*Math.PI)*.45;

      }

function steerTo(h,tx,tz,speed,dt){
  const p=h.group.position;
  const dx=tx-p.x, dz=tz-p.z, dist=Math.hypot(dx,dz);
  if(dist<.28) return true;
  h.yaw=angLerp(h.yaw,Math.atan2(dx,dz),1-Math.exp(-7*dt));
  h.group.rotation.y=h.yaw;
  p.x+=Math.sin(h.yaw)*speed*dt;
  p.z+=Math.cos(h.yaw)*speed*dt;
  resolveKeepouts(h);
  clampBounds(h);
  h.phase+=dt*speed*2.4;
  h.moving=true;
  return false;
}
