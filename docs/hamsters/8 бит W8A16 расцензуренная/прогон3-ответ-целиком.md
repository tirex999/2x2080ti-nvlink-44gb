// ---------- Ground + Cage ----------
const CAGE_R = 4.5;

const ground = new THREE.Mesh(

const tray = new THREE.Mesh(

tray.position.y = -0.18;
tray.receiveShadow = true;
scene.add(tray);

const barMat = new THREE.MeshStandardMaterial({ color: 0xd8dde2, metalness: 0.6, roughness: 0.4 });
const barGeo = new THREE.BoxGeometry(0.05, 2.3, 0.05);
for (let i = 0; i < 26; i++) {

  bar.position.set(Math.cos(a) * CAGE_R, 1.15, Math.sin(a) * CAGE_R);
  bar.rotation.y = -a;
  scene.add(bar);
}
const ringGeo = new THREE.TorusGeometry(CAGE_R, 0.07, 6, 40);
[0.05, 2.3].forEach(y => {
  const ring = new THREE.Mesh(ringGeo, barMat);
  ring.rotation.x = Math.PI / 2;
  ring.position.y = y;
  scene.add(ring);
});

// ---------- Food bowl ----------
const BOWL_POS = new THREE.Vector3(-3.0, 0, 2.2);
const bowl = new THREE.Group();

);
bowlBase.position.y = 0.15;
bowl.add(bowlBase);

for (let i = 0; i < 9; i++) {

// ---------- Hamster factory ----------
const FUR_COLORS = [0xf5a25d, 0xd9d9d9, 0xf7e0b5, 0xb98a5e, 0xffc2d1];
const hamsters = [];

function makeHamster(color) {
  const g = new THREE.Group();
  const fur = new THREE.MeshStandardMaterial({ color, flatShading: true, roughness: 1 });
  const white = new THREE.MeshStandardMaterial({ color: 0xfff6ea, flatShading: true, roughness: 1 });
  const dark = new THREE.MeshStandardMaterial({ color: 0x2b2b2b });
  const pink = new THREE.MeshStandardMaterial({ color: 0xff9eb5, flatShading: true });

  const body = new THREE.Mesh(new THREE.SphereGeometry(0.42, 7, 5), fur);
  body.scale.set(1, 0.9, 1.35);
  body.position.y = 0.42;
  g.add(body);

  const belly = new THREE.Mesh(new THREE.SphereGeometry(0.34, 7, 5), white);
  belly.scale.set(0.9, 0.75, 1.15);
  belly.position.set(0, 0.32, 0.18);
  g.add(belly);

  const earGeo = new THREE.SphereGeometry(0.11, 5, 4);
  [-0.2, 0.2].forEach(x => {
    const e = new THREE.Mesh(earGeo, fur);
    e.position.set(x, 0.8, -0.12);
    g.add(e);
    const ei = new THREE.Mesh(earGeo, pink);
    ei.scale.setScalar(0.55);
    ei.position.set(x, 0.82, -0.06);
    g.add(ei);
  });

  const eyeGeo = new THREE.SphereGeometry(0.055, 5, 4);
  [-0.16, 0.16].forEach(x => {
    const e = new THREE.Mesh(eyeGeo, dark);
    e.position.set(x, 0.55, 0.36);
    g.add(e);
  });

  const nose = new THREE.Mesh(new THREE.SphereGeometry(0.05, 5, 4), pink);
  nose.position.set(0, 0.42, 0.46);
  g.add(nose);

  const tail = new THREE.Mesh(new THREE.SphereGeometry(0.07, 5, 4), fur);
  tail.position.set(0, 0.35, -0.55);
  g.add(tail);

  // Legs (animated)
  const legGeo = new THREE.BoxGeometry(0.1, 0.18, 0.12);
  const legs = [];
  [[-0.18, 0.28], [0.18, 0.28], [-0.18, -0.28], [0.18, -0.28]].forEach(([x, z]) => {
    const pivot = new THREE.Group();
    pivot.position.set(x, 0.18, z);
    const leg = new THREE.Mesh(legGeo, pink);
    leg.position.y = -0.08;
    pivot.add(leg);
    g.add(pivot);
    legs.push(pivot);
  });

  g.traverse(o => { o.castShadow = true; });
  return { group: g, legs, body };
}

function spawnHamster() {
  const color = FUR_COLORS[Math.floor(Math.random() * FUR_COLORS.length)];
  const h = makeHamster(color);
  const a = Math.random() * Math.PI * 2;
  const r = 1 + Math.random() * 2.5;
  h.group.position.set(Math.cos(a) * r, 0, Math.sin(a) * r);
  scene.add(h.group);
  hamsters.push({
    ...h,
    pos: h.group.position,
    heading: Math.random() * Math.PI * 2,
    turn: 0,
    speed: 0.7 + Math.random() * 0.4,
    state: 'wander',
    timer: 1 + Math.random() * 3,
    phase: Math.random() * 10,
    target: null
  });
}
for (let i = 0; i < 4; i++) spawnHamster();
document.getElementById('addBtn').onclick = () => { if (hamsters.length < 9) spawnHamster(); };

// ---------- Behavior ----------
const tmpV = new THREE.Vector3();

function pickNext(h) {
  const roll = Math.random();
  if (roll < 0.5) { h.state = 'wander'; h.timer = 2 + Math.random() * 4; h.turn = (Math.random() - 0.5) * 2; }
  else if (roll < 0.75) { h.state = 'pause'; h.timer = 1 + Math.random() * 2.5; }
  else if (roll < 0.9) { h.state = 'eat'; h.timer = 4 + Math.random() * 3; h.target = BOWL_POS.clone().add(tmpV.set(0.5, 0, 0.35)); }
  else { h.state = 'wheel'; h.timer = 4 + Math.random() * 3; h.target = WHEEL_POS.clone().add(tmpV.set(0, 0, 0.9)); }
}

function updateHamster(h, dt, t) {
  h.timer -= dt;
  if (h.timer <= 0) pickNext(h);

  let moving = false;

  if (h.state === 'wander') {
    h.heading += h.turn * dt;
    h.pos.x += Math.sin(h.heading) * h.speed * dt;
    h.pos.z += Math.cos(h.heading) * h.speed * dt;
    moving = true;
  }
  else if (h.state === 'eat' || h.state === 'wheel') {
    tmpV.subVectors(h.target, h.pos);
    tmpV.y = 0;
    const dist = tmpV.length();
    if (dist > 0.55) {
      const want = Math.atan2(tmpV.x, tmpV.z);
      let diff = want - h.heading;
      while (diff > Math.PI) diff -= Math.PI * 2;
      while (diff < -Math.PI) diff += Math.PI * 2;
      h.heading += Math.max(-2.5 * dt, Math.min(2.5 * dt, diff));
      h.pos.x += Math.sin(h.heading) * h.speed * dt;
      h.pos.z += Math.cos(h.heading) * h.speed * dt;
      moving = true;
    } else {
      h.heading = Math.atan2(tmpV.x, tmpV.z);
      // Eating: nibble bob. Wheel: excited hop.
      h.group.position.y = h.state === 'wheel'
        ? Math.abs(Math.sin(t * 10 + h.phase)) * 0.08
        : Math.sin(t * 6 + h.phase) * 0.02;
      h.group.rotation.x = h.state === 'eat' ? Math.sin(t * 6 + h.phase) * 0.12 : 0;
    }
  }
  else { // pause — little idle wiggle
    h.group.rotation.x = Math.sin(t * 3 + h.phase) * 0.05;
    h.group.position.y = 0;
  }

  // Keep inside cage
  const d = Math.hypot(h.pos.x, h.pos.z);
  if (d > CAGE_R - 0.6) {
    const pull = (d - (CAGE_R - 0.6)) * dt * 3;
    h.pos.x -= (h.pos.x / d) * pull;
    h.pos.z -= (h.pos.z / d) * pull;
    if (h.state === 'wander') h.turn *= -1;
  }

  if (!moving) h.group.position.y *= 0.9;
  h.group.rotation.y = h.heading;

  // Leg + body animation
  const amp = moving ? 0.7 : 0.1;
  h.legs.forEach((leg, i) => {
    leg.rotation.x = Math.sin(t * 12 + h.phase + (i % 2 ? Math.PI : 0)) * amp;
  });
  h.body.scale.y = 0.9 + Math.sin(t * 12 + h.phase) * (moving ? 0.04 : 0.015);
}

// ---------- Main loop ----------
const clock = new THREE.Clock();
function animate() {
  requestAnimationFrame(animate);
  const dt = Math.min(clock.getDelta(), 0.05);
  const t = clock.elapsedTime;

  hamsters.forEach(h => updateHamster(h, dt, t));

  // Wheel spins fast if a hamster is using it
  const busy = hamsters.some(h => h.state === 'wheel' && h.pos.distanceTo(WHEEL_POS) < 2);
  wheelSpin.rotation.z += (busy ? 7 : 0.4) * dt;

  // Bowl gently wiggles when someone eats
  const eating = hamsters.some(h => h.state === 'eat' && h.pos.distanceTo(BOWL_POS) < 1.5);
  bowl.rotation.y = eating ? Math.sin(t * 8) * 0.06 : 0;

  controls.update();
  renderer.render(scene, camera);
}
animate();

addEventListener('resize', () => {
  camera.aspect = innerWidth / innerHeight;
  camera.updateProjectionMatrix();
  renderer.setSize(innerWidth, innerHeight);
});
</script>

## What you get

**Scene contents**
- 🏠 **Cage** — 26 wire bars with top/bottom rings around a sandy bedding tray, sitting on a pastel-green ground
- 🥣 **Food bowl** with seeds — wiggles when a hamster is eating
- 🎡 **Exercise wheel** — idles slowly, spins fast when a hamster "uses" it
- 🛤️ **Yellow tunnel**, tiny house with a round door, and scattered wood shavings

**Hamster behavior** (simple state machine, no libraries beyond Three.js)
- **Wander** — walks with random gentle turning, legs pump, body bobs
- **Pause** — stops and does an idle wiggle
- **Eat** — walks to the bowl, faces it, and nibbles with a head-bob
- **Wheel** — trots to the wheel, hops excitedly, and the wheel speeds up

**Fun extras**
- Each hamster gets a random fur color (golden, gray, cream, brown, or pink!) and random personality (speed, timing)
- A "+ Add Hamster" button (up to 9 total)
- Soft shadows, fog, and damped orbit controls (drag to rotate, scroll to zoom)

Everything uses `flatShading` low-poly geometry (low-segment spheres, boxes, cones) for that cute chunky look, and the whole thing stays lightweight and stable.
