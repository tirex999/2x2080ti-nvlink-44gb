function moveTo(h, target, speed, dt){
  const v = new THREE.Vector3().subVectors(target, h.group.position);
  v.y = 0;
  const d = v.length();
  if(d < 0.15) return true;
  v.normalize();
  h.group.position.addScaledVector(v, speed*dt);
  // smooth yaw
  const targetYaw = Math.atan2(v.x, v.z);
  h.yaw = lerpAngle(h.yaw, targetYaw, Math.min(1, dt*8));
  h.group.rotation.y = h.yaw;
  h.walkPhase += dt * 12;
  h.group.position.y = 0.08 + Math.abs(Math.sin(h.walkPhase))*0.07;
  // slight tilt
  h.group.rotation.z = Math.sin(h.walkPhase)*0.05;
  return false;
}
