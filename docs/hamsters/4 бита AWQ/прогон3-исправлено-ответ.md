

Причина: опора-ножка колеса была дочерним объектом **вращающейся** группы `wheel`, поэтому при анимации `wheel.rotation.x` она описывала полный круг, «молотом» размахиваясь по песку и по хомяку у колеса (отсюда куча розовых «коробок» и обломков у обода, глаз между спицами); плюс сами хомяки ставились на `TRAY_TOP + 0.02`, то есть на ~0.1 **ниже** поверхности песка (`TRAY_TOP + 0.12`) — из-за чего они тонули в подстилке и выглядели как плоские комки, лежащие плашмя (а точка у колеса `WHEEL_POINT` была слишком близко к ободу, из-за чего тело пересекало колесо).

Исправления: нога вынесена из вращающейся группы (колесо крутится вокруг своей оси отдельно), хомяки поставлены на уровень песка, точка у колеса отодвинута от обода, границы блуждания сужены, чтобы зверьки не въезжали в колесо.

function randomPoint() {
  return new THREE.Vector3(
    (Math.random() * 2 - 1) * BOUNDS.x,
    0,
    (Math.random() * 2 - 1) * BOUNDS.z
  );
}

function pickNextAction(h) {
  const r = Math.random();
  if (r < 0.3)      { h.state = 'wheel'; h.target.copy(WHEEL_POINT); h.timer = 3 + Math.random() * 2; }
  else if (r < 0.5) { h.state = 'eat';   h.target.copy(BOWL_POINT);  h.timer = 2.5 + Math.random() * 2; }
  else              { h.state = 'wander'; h.target = randomPoint();  h.timer = 99; }
}

function updateHamster(h, dt, t) {
  h.timer -= dt;
  const pos = h.mesh.position;
  let moving = false;

  if (h.state === 'pause') {
    if (h.timer <= 0) pickNextAction(h);
  } else {
    const dir = new THREE.Vector3().subVectors(h.target, pos);
    dir.y = 0;
    const dist = dir.length();
    const arrived = dist < 0.15;

    if (!arrived) {
      dir.normalize();
      pos.addScaledVector(dir, h.speed * dt * (h.state === 'wander' ? 1 : 1.3));
      // smoothly face travel direction
      const want = Math.atan2(dir.x, dir.z);
      let d = want - h.mesh.rotation.y;
      while (d > Math.PI) d -= Math.PI * 2;
      while (d < -Math.PI) d += Math.PI * 2;
      h.mesh.rotation.y += d * Math.min(1, dt * 8);
      moving = true;
    }
    if (arrived && (h.state === 'wander' || h.timer <= 0)) {
      h.state = 'pause';
      h.timer = 0.6 + Math.random() * 2;
    }
  }

  // --- cute idle/walk animation ---
  if (moving) {
    h.phase += dt * 11;
    pos.y = HAMSTER_Y + Math.abs(Math.sin(h.phase)) * 0.045;   // waddle hop
    h.mesh.rotation.z = Math.sin(h.phase) * 0.08;              // little tilt
    h.mesh.rotation.x = 0;
  } else if (h.state === 'wheel') {
    // running in place at the wheel!
    h.phase += dt * 16;
    pos.y = HAMSTER_Y + Math.abs(Math.sin(h.phase)) * 0.03;
    h.mesh.rotation.z = 0;
    h.mesh.rotation.x = -0.25;                                 // lean into the wheel
    // face the wheel
    const want = Math.atan2(WHEEL_POINT.x - pos.x, WHEEL_POINT.z - pos.z);
    let d = want - h.mesh.rotation.y;
    while (d > Math.PI) d -= Math.PI * 2;
    while (d < -Math.PI) d += Math.PI * 2;
    h.mesh.rotation.y += d * Math.min(1, dt * 6);
  } else if (h.state === 'eat') {
    // nibbling: slow bob + face the bowl
    h.phase += dt * 5;
    pos.y = HAMSTER_Y + Math.sin(h.phase) * 0.015;
    h.mesh.rotation.z = 0;
    h.mesh.rotation.x = Math.sin(h.phase * 2) * 0.12;          // head-bob
    const want = Math.atan2(BOWL_POINT.x - pos.x, BOWL_POINT.z - pos.z);
    let d = want - h.mesh.rotation.y;
    while (d > Math.PI) d -= Math.PI * 2;
    while (d < -Math.PI) d += Math.PI * 2;
    h.mesh.rotation.y += d * Math.min(1, dt * 6);
  } else {
    // paused: settle down, tiny breathing
    pos.y += (HAMSTER_Y + Math.sin(t * 2 + h.phase) * 0.008 - pos.y) * Math.min(1, dt * 5);
    h.mesh.rotation.z *= 1 - Math.min(1, dt * 5);
    h.mesh.rotation.x *= 1 - Math.min(1, dt * 5);
  }
}

/* ============================================================
   MAIN LOOP
============================================================ */
const clock = new THREE.Clock();

function animate() {
  requestAnimationFrame(animate);
  const dt = Math.min(clock.getDelta(), 0.05);
  const t = clock.elapsedTime;

  hamsters.forEach(h => updateHamster(h, dt, t));

  // Wheel spins fast if anyone is using it (only the inner group rotates)
  const someoneRunning = hamsters.some(h => h.state === 'wheel');
  wheelSpeed += ((someoneRunning ? 7 : 0.4) - wheelSpeed) * Math.min(1, dt * 3);
  wheelSpin.rotation.z -= wheelSpeed * dt;
  if (wheelSpin.rotation.z < -Math.PI * 2) wheelSpin.rotation.z += Math.PI * 2;

  controls.update();
  renderer.render(scene, camera);
}
animate();

addEventListener('resize', () => {
  camera.aspect = innerWidth / innerHeight;
  camera.updateProjectionMatrix();
  renderer.setSize(innerWidth, innerHeight);
});
