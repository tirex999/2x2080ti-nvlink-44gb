const _desired = new THREE.Vector3();
const _diff = new THREE.Vector3();
const _to = new THREE.Vector3();
const X_AXIS = new THREE.Vector3(1,0,0);
const _q = new THREE.Quaternion();

function updateFish(dt, t){
  const n = fishArray.length;
  for (let i=0;i<n;i++){
    const f = fishArray[i];
    const p = f.mesh.position;

    // random wandering
    f.wanderTimer -= dt;
    if (f.wanderTimer <= 0){
      f.wanderTimer = 1.5 + Math.random()*3;
      f.wanderDir.set((Math.random()-0.5), (Math.random()-0.5)*0.5, (Math.random()-0.5)).normalize();
    }
    _desired.copy(f.wanderDir).multiplyScalar(f.speed);

    // food
    let best = null, bestD = 15;
    for (const fd of foods){
      const d = p.distanceTo(fd.mesh.position);
      if (d < bestD){ bestD = d; best = fd; }
    }
    if (best){
      f.targetFood = best;
      _to.copy(best.mesh.position).sub(p);
      const d = _to.length();
      if (d < 0.9 + 0.5*f.mesh.scale.x){
        eatFood(best, f);
      } else {
        _desired.copy(_to.normalize().multiplyScalar(f.speed*1.8));
      }
    } else f.targetFood = null;

    // collision avoidance
    for (let j=0;j<n;j++){
      if (j===i) continue;
      const o = fishArray[j].mesh.position;
      _diff.subVectors(p, o);
      const d = _diff.length();
      const r = f.avoidanceRadius + fishArray[j].mesh.scale.x*0.8;
      if (d < r && d > 1e-4){
        _desired.addScaledVector(_diff.normalize(), (r-d)/r * 6);
      }
    }

    // walls
    const mx = W/2 - 2, mz = D/2 - 2, myMin = 1.6, myMax = H - 2;
    if (p.x < -mx) _desired.x += ( -mx - p.x ) * 3;
    if (p.x >  mx) _desired.x -= ( p.x -  mx ) * 3;
    if (p.z < -mz) _desired.z += ( -mz - p.z ) * 3;
    if (p.z >  mz) _desired.z -= ( p.z -  mz ) * 3;
    if (p.y < myMin) _desired.y += (myMin - p.y) * 3;
    if (p.y > myMax) _desired.y -= (p.y - myMax) * 3;

    // integrate
    _desired.normalize().multiplyScalar(f.speed * (f.targetFood ? 1.8 : 1));
    const k = 1 - Math.exp(-3.5*dt);
    f.velocity.lerp(_desired, k);
    p.addScaledVector(f.velocity, dt);

    // hard clamp
    p.x = THREE.MathUtils.clamp(p.x, -W/2+1, W/2-1);
    p.y = THREE.MathUtils.clamp(p.y, 1, H-1);
    p.z = THREE.MathUtils.clamp(p.z, -D/2+1, D/2-1);

    // orientation
    _to.set(f.velocity.x, 0, f.velocity.z);
    if (_to.lengthSq() > 1e-4){
      _to.normalize();
      _q.setFromUnitVectors(X_AXIS, _to);
      f.mesh.quaternion.slerp(_q, 1 - Math.exp(-5*dt));
    }

    // animation
    const tt = t*f.tailSpeed + f.phase;
    f.tail.rotation.z = Math.sin(tt) * 0.55;
    f.dorsal.rotation.z = 0.35 + Math.sin(tt*0.7) * 0.08;
    f.leftFin.rotation.z = 0.5 + Math.sin(tt*0.9 + 1) * 0.35;
    f.rightFin.rotation.z = -0.5 - Math.sin(tt*0.9 + 1) * 0.35;
    f.bodyMesh.position.y = Math.sin(tt*0.5) * 0.06;
  }
}

function updateFish(dt, t){
  const n = fishArray.length;
  for (let i=0;i<n;i++){
    const f = fishArray[i], p = f.mesh.position;

    f.wanderTimer -= dt;
    if (f.wanderTimer <= 0){
      f.wanderTimer = rand(1.5, 4.5);
      f.wanderDir.set(rand(-1,1), rand(-0.4,0.4), rand(-1,1)).normalize();
    }
    _desired.copy(f.wanderDir);

    let best = null, bestD = 15;
    for (let k=0;k<foods.length;k++){
      const d = p.distanceTo(foods[k].mesh.position);
      if (d < bestD){ bestD = d; best = foods[k]; }
    }
    f.targetFood = best;
    if (best){
      _to.copy(best.mesh.position).sub(p);
      const d = _to.length();
      if (d < 1.1 + 0.4*f.mesh.scale.x){
        eatFood(best, f);
      } else {
        _desired.copy(_to.normalize());
      }
    }

    for (let j=0;j<n;j++){
      if (j===i) continue;
      const o = fishArray[j].mesh.position;
      _diff.subVectors(p, o);
      const d = _diff.length();
      const r = f.avoidanceRadius + fishArray[j].mesh.scale.x;
      if (d < r && d > 1e-4) _desired.addScaledVector(_diff.normalize(), (r-d)/r * 5);
    }

    const bx = W/2-2.2, bz = D/2-2.2;
    if (p.x < -bx) _desired.x += (-bx - p.x)*2.5;
    else if (p.x > bx) _desired.x -= (p.x - bx)*2.5;
    if (p.z < -bz) _desired.z += (-bz - p.z)*2.5;
    else if (p.z > bz) _desired.z -= (p.z - bz)*2.5;
    if (p.y < 2) _desired.y += (2 - p.y)*2.5;
    else if (p.y > H-2.2) _desired.y -= (p.y - (H-2.2))*2.5;

    const spd = f.speed * (best ? 1.9 : 1);
    if (_desired.lengthSq() > 1e-6) _desired.normalize();
    _desired.multiplyScalar(spd);
    f.velocity.lerp(_desired, 1 - Math.exp(-3.2*dt));
    p.addScaledVector(f.velocity, dt);

    p.x = THREE.MathUtils.clamp(p.x, -W/2+1.2, W/2-1.2);
    p.y = THREE.MathUtils.clamp(p.y, 1.2, H-1.4);
    p.z = THREE.MathUtils.clamp(p.z, -D/2+1.2, D/2-1.2);

    _to.set(f.velocity.x, f.velocity.y*0.4, f.velocity.z);
    if (_to.lengthSq() > 1e-4){
      _q.setFromUnitVectors(X_AXIS, _to.normalize());
      f.mesh.quaternion.slerp(_q, 1 - Math.exp(-4.5*dt));
    }

    const tt = t*f.tailSpeed + f.phase;
    f.tail.rotation.z = Math.sin(tt)*0.55;
    f.dorsal.rotation.z = Math.sin(tt*0.7)*0.1;
    f.leftFin.rotation.z = 0.5 + Math.sin(tt*0.9 + 1)*0.35;
    f.rightFin.rotation.z = -0.5 - Math.sin(tt*0.9 + 1)*0.35;
    f.body.position.y = Math.sin(tt*0.5)*0.05;
  }
}
