```js
const _desired = new THREE.Vector3(), _dir = new THREE.Vector3(), _q = new THREE.Quaternion();

function updateFish(dt, t){
  const n = fishArray.length;
  for (let i = 0; i < n; i++){
    const f = fishArray[i];
    const p = f.mesh.position;

    f.wanderT -= dt;
    if (f.wanderT <= 0){
      f.wanderT = rand(2, 5);
      f.wander.set(rand(-1,1), rand(-0.5,0.5), rand(-1,1)).normalize();
    }
    _desired.copy(f.wander);

    for (let j = 0; j < n; j++){
      if (j === i) continue;
      const o = fishArray[j].mesh.position;
      const dx = p.x - o.x, dy = p.y - o.y, dz = p.z - o.z;
      const d2 = dx*dx + dy*dy + dz*dz;
      const rr = f.avoidanceRadius;
      if (d2 < rr * rr && d2 > 1e-6){
        const d = Math.sqrt(d2), k = (1 - d / rr) / d;
        _desired.x += dx * k; _desired.y += dy * k; _desired.z += dz * k;
      }
    }

    const M = 2.2;
    if (p.x >  BX - M) _desired.x -= (p.x - (BX - M)) * 1.5;
    if (p.x < -BX + M) _desired.x += ((-BX + M) - p.x) * 1.5;
    if (p.z >  BZ - M) _desired.z -= (p.z - (BZ - M)) * 1.5;
    if (p.z < -BZ + M) _desired.z += ((-BZ + M) - p.z) * 1.5;
    if (p.y < BY_MIN + M) _desired.y += ((BY_MIN + M) - p.y) * 1.7;
    if (p.y > BY_MAX - M) _desired.y -= (p.y - (BY_MAX - M)) * 1.5;

    f.targetFood = null;
    let best = 225; // 15^2
    for (let k = 0; k < foods.length; k++){
      const fp = foods[k].mesh.position;
      const d2 = (p.x-fp.x)**2 + (p.y-fp.y)**2 + (p.z-fp.z)**2;
      if (d2 < best){ best = d2; f.targetFood = foods[k]; }
    }
    let spd = f.speed;
    if (f.targetFood){
      const fp = f.targetFood.mesh.position;
      _dir.subVectors(fp, p);
      const dl = _dir.length();
      _dir.divideScalar(dl || 1);
      _desired.copy(_dir);
      spd *= 1.9;
      const eatR = 0.8 + f.baseScale * f.grow * 0.45;
      if (dl < eatR) eatFood(f, f.targetFood);
    }

    if (_desired.lengthSq() < 1e-6) _desired.copy(f.velocity);
    _desired.normalize();
    f.velocity.lerp(_desired.multiplyScalar(spd), 1 - Math.exp(-3 * dt));
    p.addScaledVector(f.velocity, dt);

    p.x = THREE.MathUtils.clamp(p.x, -BX, BX);
    p.y = THREE.MathUtils.clamp(p.y, BY_MIN, BY_MAX);
    p.z = THREE.MathUtils.clamp(p.z, -BZ, BZ);

    if (f.velocity.lengthSq() > 1e-4){
      _dir.copy(f.velocity).normalize();
      _q.setFromUnitVectors(FWD, _dir);
      f.mesh.quaternion.slerp(_q, 1 - Math.exp(-4 * dt));
    }

    const act = 0.6 + 0.4 * Math.min(1, f.velocity.length() / (f.speed || 1));
    f.tail.rotation.y = Math.sin(t * f.tailSpeed + f.phase) * 0.55 * act;
    const flap = Math.sin(t * f.flapSpeed + f.phase) * 0.4;
    f.rightFin.rotation.z = f.finBaseR + flap;
```

```js
    const fd = foods[i], p = fd.mesh.position;
    if (!fd.landed){
      fd.vy -= 3.2 * dt;
      fd.vy *= Math.exp(-0.85 * dt);
      p.y += fd.vy * dt;
      p.x += Math.sin(t * 1.7 + fd.seed) * 0.25 * dt;
```

```js
const effects = [];
const burstGeo = new THREE.SphereGeometry(0.08, 8, 6);
function burst(pos){
  for (let i = 0; i < 6; i++){
    const m = new THREE.Mesh(burstGeo, new THREE.MeshBasicMaterial({ color: 0xbff3ff, transparent: true, opacity: 0.9, depthWrite: false }));
    m.position.copy(pos);
    scene.add(m);
    effects.push({
      mesh: m, life: 0.55, max: 0.55,
      vel: new THREE.Vector3(rand(-1.6,1.6), rand(0.8,2.4), rand(-1.6,1.6))
    });
  }
}
function updateEffects(dt){
  for (let i = effects.length - 1; i >= 0; i--){
    const e = effects[i];
    e.life -= dt;
    if (e.life <= 0){
      scene.remove(e.mesh); e.mesh.material.dispose();
      effects.splice(i, 1); continue;
    }
    e.vel.y += 2.2 * dt;
    e.mesh.position.addScaledVector(e.vel, dt);
    const k = e.life / e.max;
    e.mesh.scale.setScalar(k);
    e.mesh.material.opacity = k;
  }
}
```
