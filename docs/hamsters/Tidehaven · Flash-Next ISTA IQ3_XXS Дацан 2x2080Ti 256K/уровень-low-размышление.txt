For the water surface shader:

```glsl
// Vertex shader
uniform float uTime;
uniform float uWindStrength;
varying vec2 vUv;
varying vec3 vWorldPos;
varying vec3 vNormal;

void main() {
    vUv = uv;
    vec3 pos = position;
    
    // Wave displacement
    float wave1 = sin(pos.x * 0.5 + uTime * 1.2) * 0.15 * uWindStrength;
    float wave2 = sin(pos.z * 0.7 + uTime * 0.8) * 0.1 * uWindStrength;
    float wave3 = sin((pos.x + pos.z) * 0.3 + uTime * 1.5) * 0.08 * uWindStrength;
    pos.y += wave1 + wave2 + wave3;
    
    vWorldPos = pos;
    vNormal = normalize(normalMatrix * normal);
    gl_Position = projectionMatrix * modelViewMatrix * vec4(pos, 1.0);
}
```

```glsl
// Fragment shader
uniform float uTime;
uniform float uWindStrength;
uniform float uTideLevel;
uniform vec3 uWaterColor;
uniform vec3 uIslandCenter;
uniform float uIslandRadius;
varying vec2 vUv;
varying vec3 vWorldPos;
varying vec3 vNormal;

void main() {
    // Distance from island center for foam
    float dist = length(vWorldPos.xz - uIslandCenter.xz);
    float foamFactor = smoothstep(uIslandRadius, uIslandRadius - 1.5, dist);
    
    // Base water color
    vec3 color = uWaterColor;
    
    // Fresnel effect
    vec3 viewDir = normalize(cameraPosition - vWorldPos);
    float fresnel = pow(1.0 - max(dot(viewDir, vNormal), 0.0), 3.0);
    
    // Foam
    float foam = foamFactor * (0.5 + 0.5 * sin(vWorldPos.x * 3.0 + uTime * 2.0) * sin(vWorldPos.z * 2.5 + uTime * 1.5));
    color = mix(color, vec3(1.0), foam * 0.7);
    
    // Alpha: more transparent in the middle, more opaque at edges
    float alpha = 0.4 + fresnel * 0.3 + foam * 0.3;
    
    gl_FragColor = vec4(color, alpha);
}
```

For the seabed caustics:

```glsl
// Fragment shader
uniform float uTime;
uniform float uWindStrength;
varying vec2 vUv;

float caustic(vec2 uv, float time) {
    float c = 0.0;
    for (int i = 0; i < 3; i++) {
        float fi = float(i);
        vec2 p = uv * (3.0 + fi) + time * (0.3 + fi * 0.1);
        c += sin(p.x) * sin(p.y) * 0.3;
    }
    return c;
}

void main() {
    vec3 sandColor = vec3(0.76, 0.68, 0.52);
    float c = caustic(vUv, uTime) * 0.15 * uWindStrength;
    vec3 color = sandColor + vec3(c);
    gl_FragColor = vec4(color, 1.0);
}
```
