function hash2i(x,y){let n=(Math.imul(x,374761393)+Math.imul(y,668265263))|0;n=Math.imul(n^(n>>>13),1274126177);n^=n>>>16;return (n>>>0)/4294967296;}

function smooth(t){return t*t*(3-2*t);}
function noise2(x,y){
 const xi=Math.floor(x),yi=Math.floor(y);
 const xf=x-xi,yf=y-yi,u=smooth(xf),v=smooth(yf);
 const a=hash2i(xi,yi),b=hash2i(xi+1,yi),c=hash2i(xi,yi+1),d=hash2i(xi+1,yi+1);
 return a+(b-a)*u+(c-a)*v+(a-b-c+d)*u*v;
}
function noise3(x,y,z){
 const xi=Math.floor(x),yi=Math.floor(y),zi=Math.floor(z);
 const xf=x-xi,yf=y-yi,zf=z-zi,u=smooth(xf),v=smooth(yf),w=smooth(zf);
 const c000=hash3i(xi,yi,zi),c100=hash3i(xi+1,yi,zi),c010=hash3i(xi,yi+1,zi),c110=hash3i(xi+1,yi+1,zi),
       c001=hash3i(xi,yi,zi+1),c101=hash3i(xi+1,yi,zi+1),c011=hash3i(xi,yi+1,zi+1),c111=hash3i(xi+1,yi+1,zi+1);
 const x00=c000+(c100-c000)*u,x10=c010+(c110-c010)*u,x01=c001+(c101-c001)*u,x11=c011+(c111-c011)*u;
 const y0=x00+(x10-x00)*v,y1=x01+(x11-x01)*v;
 return y0+(y1-y0)*w;
}
function fractal2(x,y){let a=0,amp=0.5,f=1,tot=0;for(let i=0;i<4;i++){a+=noise2(x*f,y*f)*amp;tot+=amp;amp*=0.5;f*=2;}return a/tot;}
function fractal3(x,y,z){let a=0,amp=0.5,f=1,tot=0;for(let i=0;i<4;i++){a+=noise3(x*f,y*f,z*f)*amp;tot+=amp;amp*=0.5;f*=2;}return a/tot;}
