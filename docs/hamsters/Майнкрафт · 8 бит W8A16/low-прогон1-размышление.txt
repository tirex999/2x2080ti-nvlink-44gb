```
function noise2(x, z) {
    const xi = Math.floor(x), zi = Math.floor(z);
    const xf = x - xi, zf = z - zi;
    const u = smoothstep(xf), v = smoothstep(zf);
    const a = hash2(xi, zi), b = hash2(xi+1, zi);
    const c = hash2(xi, zi+1), d = hash2(xi+1, zi+1);
    return a + (b-a)*u + (c-a)*v + (a-b-c+d)*u*v;
}
```

```
function fractal2(x, z) {
    let val = 0, amp = 1, freq = 1, max = 0;
    for (let i = 0; i < 4; i++) {
        val += noise2(x*freq, z*freq) * amp;
        max += amp;
        amp *= 0.5;
        freq *= 2;
    }
    return val / max;
}
```

```javascript
function noise2(x, z) {
    const xi = Math.floor(x), zi = Math.floor(z);
    const xf = x - xi, zf = z - zi;
    const u = smoothstep(xf), v = smoothstep(zf);
    const a = hash2(xi, zi), b = hash2(xi + 1, zi);
    const c = hash2(xi, zi + 1), d = hash2(xi + 1, zi + 1);
    return a + (b - a) * u + (c - a) * v + (a - b - c + d) * u * v;
}
```

```javascript
function fractal2(x, z) {

    for (let i = 0; i < 4; i++) {
        val += noise2(x * freq, z * freq) * amp;
        max += amp;
        amp *= 0.5;
        freq *= 2;
    }
    return val / max;
}
```

```javascript
function noise3(x, y, z) {
    const xi = Math.floor(x), yi = Math.floor(y), zi = Math.floor(z);
    const xf = x - xi, yf = y - yi, zf = z - zi;
    const u = smoothstep(xf), v = smoothstep(yf), w = smoothstep(zf);
    
    const c000 = hash3(xi, yi, zi);
    const c100 = hash3(xi+1, yi, zi);
    const c010 = hash3(xi, yi+1, zi);
    const c110 = hash3(xi+1, yi+1, zi);
    const c001 = hash3(xi, yi, zi+1);
    const c101 = hash3(xi+1, yi, zi+1);
    const c011 = hash3(xi, yi+1, zi+1);
    const c111 = hash3(xi+1, yi+1, zi+1);
    
    const x00 = c000 + (c100 - c000) * u;
    const x10 = c010 + (c110 - c010) * u;
    const x01 = c001 + (c101 - c001) * u;
    const x11 = c011 + (c111 - c011) * u;
    
    const y0 = x00 + (x10 - x00) * v;
    const y1 = x01 + (x11 - x01) * v;
    
    return y0 + (y1 - y0) * w;
```

```javascript
function updateChunks() {
    const pcx = Math.floor(playerX / 16);
    const pcz = Math.floor(playerZ / 16);
    
    // Generate (radius 5, max 4)
    let generated = 0;
    const genQueue = [];
    for (let dx = -5; dx <= 5 && generated < 4; dx++) {
        for (let dz = -5; dz <= 5 && generated < 4; dz++) {
            const cx = pcx + dx, cz = pcz + dz;
            const key = cx + ',' + cz;
            if (!chunks.has(key)) {
                genQueue.push({ cx, cz, dist: dx*dx + dz*dz });
            }
        }
    }
    genQueue.sort((a, b) => a.dist - b.dist);
    for (let i = 0; i < Math.min(4, genQueue.length); i++) {
        const { cx, cz } = genQueue[i];
        chunks.set(cx + ',' + cz, { data: generateChunk(cx, cz), mesh: null });
        generated++;
    }
    
    // Mesh (radius 4, max 2)
    let meshed = 0;
    const meshQueue = [];
    for (let dx = -4; dx <= 4 && meshed < 2; dx++) {
        for (let dz = -4; dz <= 4 && meshed < 2; dz++) {
            const cx = pcx + dx, cz = pcz + dz;
            const key = cx + ',' + cz;
            const chunk = chunks.get(key);
            if (chunk && !chunk.mesh) {
                // Check 4 neighbors have data
                if (chunks.has((cx-1)+','+cz) && chunks.has((cx+1)+','+cz) &&
                    chunks.has(cx+','+(cz-1)) && chunks.has(cx+','+(cz+1))) {
                    meshQueue.push({ cx, cz, dist: dx*dx + dz*dz });
                }
            }
        }
    }
    meshQueue.sort((a, b) => a.dist - b.dist);
    for (let i = 0; i < Math.min(2, meshQueue.length); i++) {
        rebuildChunk(meshQueue[i].cx, meshQueue[i].cz);
        meshed++;
    }
    
    // Unload (beyond radius 7)
    for (const [key, chunk] of chunks) {
        const [cx, cz] = key.split(',').map(Number);
        const dx = cx - pcx, dz = cz - pcz;
        if (dx*dx + dz*dz > 49) { // 7^2 = 49
            if (chunk.mesh) {
                scene.remove(chunk.mesh);
                const idx = chunkMeshes.indexOf(chunk.mesh);
                if (idx !== -1) chunkMeshes.splice(idx, 1);
                chunk.mesh.geometry.dispose();
            }
            chunks.delete(key);
        }
    }
}
```
