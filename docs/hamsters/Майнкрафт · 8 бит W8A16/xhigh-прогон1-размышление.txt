```js
function smooth(t){ return t*t*(3-2*t); }
function noise2(x, y) {
  const xi = Math.floor(x), yi = Math.floor(y);
  const xf = x - xi, yf = y - yi;
  const a = hash2(xi, yi), b = hash2(xi+1, yi), c = hash2(xi, yi+1), d = hash2(xi+1, yi+1);
  const u = smooth(xf), v = smooth(yf);
  return a + (b-a)*u + (c-a)*v + (a - b - c + d)*u*v;
}
```

```js
function fbm2(x, y) {
  let sum = 0, amp = 0.5, freq = 1, total = 0;
  for (let i = 0; i < 4; i++) {
    sum += noise2(x*freq, y*freq) * amp;
    total += amp;
    amp *= 0.5; freq *= 2;
  }
  return sum / total;
}
```

```js
function buildMeshForChunk(cx, cz) {
  const c = CHUNKS.get(cx + ',' + cz);
  if (!c) return;
  const data = c.data;
  const ox = cx * 16, oz = cz * 16;
  const pos = [], norm = [], col = [], idx = [];
  for (let y = 0; y < 80; y++) {
    for (let lz = 0; lz < 16; lz++) {
      for (let lx = 0; lx < 16; lx++) {
        const id = data[lx + (lz << 4) + (y << 8)];
        if (id === 0) continue;
        const rgb = BLOCK_RGB[id];
        const wx = ox + lx, wz = oz + lz;
        for (let f = 0; f < 6; f++) {
          const F = FACES[f];
          const nx = lx + F.dir[0], ny = y + F.dir[1], nz = lz + F.dir[2];
          let nId;
          if (ny >= 0 && ny < 80 && nx >= 0 && nx < 16 && nz >= 0 && nz < 16) {
            nId = data[nx + (nz << 4) + (ny << 8)];
          } else {
            nId = getBlock(wx + F.dir[0], ny, wz + F.dir[2]);
          }
          if (nId !== 0) continue;
          const vi = pos.length / 3;
          for (let i = 0; i < 4; i++) {
            const cn = F.corners[i];
            pos.push(wx + cn[0], y + cn[1], wz + cn[2]);
            norm.push(F.dir[0], F.dir[1], F.dir[2]);
            col.push(rgb[0] * F.shade, rgb[1] * F.shade, rgb[2] * F.shade);
          }
          idx.push(vi, vi + 1, vi + 2, vi, vi + 2, vi + 3);
        }
      }
    }
  }
  const g = new THREE.BufferGeometry();
  g.setAttribute('position', new THREE.Float32BufferAttribute(pos, 3));
  g.setAttribute('normal', new THREE.Float32BufferAttribute(norm, 3));
  g.setAttribute('color', new THREE.Float32BufferAttribute(col, 3));
  g.setIndex(idx);
  g.computeBoundingSphere();
  const mesh = new THREE.Mesh(g, blockMaterial);
  scene.add(mesh);
  chunkMeshes.push(mesh);
  c.mesh = mesh;
}
```

```js
function hash2(x, y) {
  let n = (Math.imul(x, 374761393) + Math.imul(y, 668265263)) | 0;
  n = Math.imul(n ^ (n >>> 13), 1274126177);
  n ^= n >>> 16;

}
function hash3(x, y, z) {
  let n = (Math.imul(x, 374761393) + Math.imul(y, 668265263) + Math.imul(z, 1440662683)) | 0;
```

noise2:

```js
function noise2(x, y) {
  const xi = Math.floor(x), yi = Math.floor(y);
  const xf = x - xi, yf = y - yi;
  const a = hash2(xi, yi), b = hash2(xi + 1, yi);
  const c = hash2(xi, yi + 1), d = hash2(xi + 1, yi + 1);
  const u = sstep(xf), v = sstep(yf);
  return a + (b - a) * u + (c - a) * v + (a - b - c + d) * u * v;
}
```

fbm2 (4 octaves):

```js
function fbm2(x, y) {
  let sum = 0, amp = 0.5, tot = 0;
  for (let i = 0; i < 4; i++) {
    sum += noise2(x, y) * amp;
    tot += amp;
    x *= 2; y *= 2; amp *= 0.5;
  }
  return sum / tot;
}
```

noise3 (trilinear):

```js
function noise3(x, y, z) {
  const xi = Math.floor(x), yi = Math.floor(y), zi = Math.floor(z);
  const xf = x - xi, yf = y - yi, zf = z - zi;
  const u = sstep(xf), v = sstep(yf), w = sstep(zf);
  const c000 = hash3(xi, yi, zi),     c100 = hash3(xi+1, yi, zi);
  const c010 = hash3(xi, yi+1, zi),   c110 = hash3(xi+1, yi+1, zi);
  const c001 = hash3(xi, yi, zi+1),   c101 = hash3(xi+1, yi, zi+1);
  const c011 = hash3(xi, yi+1, zi+1), c111 = hash3(xi+1, yi+1, zi+1);
  const x00 = c000 + (c100 - c000) * u;
  const x10 = c010 + (c110 - c010) * u;
  const x01 = c001 + (c101 - c001) * u;
  const x11 = c011 + (c111 - c011) * u;
  const y0 = x00 + (x10 - x00) * v;
  const y1 = x01 + (x11 - x01) * v;
  return y0 + (y1 - y0) * w;
}
```

```js
function forEachRing(radius, fn) {
  for (let r = 0; r <= radius; r++) {
    for (let dx = -r; dx <= r; dx++) {
      for (let dz = -r; dz <= r; dz++) {
        if (Math.max(Math.abs(dx), Math.abs(dz)) !== r) continue;
        if (fn(dx, dz, r) === true) return;
      }
    }
  }
}
```

```js
function updateWorld() {
  const pcx = Math.floor(player.pos.x / 16);
  const pcz = Math.floor(player.pos.z / 16);

  let gen = 0;
  forEachRing(GEN_RADIUS, (dx, dz) => {
    if (gen >= 4) return true;
    const cx = pcx + dx, cz = pcz + dz;
    if (!CHUNKS.has(cx + ',' + cz)) {
      generateChunkData(cx, cz);
      if (++gen >= 4) return true;
    }
  });

  let mesh = 0;
  forEachRing(MESH_RADIUS, (dx, dz) => {
    if (mesh >= 2) return true;
    const cx = pcx + dx, cz = pcz + dz;
    const c = CHUNKS.get(cx + ',' + cz);
    if (!c || c.mesh) return;
    if (CHUNKS.has((cx - 1) + ',' + cz) && CHUNKS.has((cx + 1) + ',' + cz) &&
        CHUNKS.has(cx + ',' + (cz - 1)) && CHUNKS.has(cx + ',' + (cz + 1))) {
      buildMeshForChunk(cx, cz);
      if (++mesh >= 2) return true;
    }
  });

  for (const [key, c] of CHUNKS) {
    const parts = key.split(',');
    const cx = +parts[0], cz = +parts[1];
    if (Math.max(Math.abs(cx - pcx), Math.abs(cz - pcz)) > UNLOAD_RADIUS) {
      if (c.mesh) {
        scene.remove(c.mesh);
        const i = chunkMeshes.indexOf(c.mesh);
        if (i !== -1) chunkMeshes.splice(i, 1);
        c.mesh.geometry.dispose();
      }
      CHUNKS.delete(key);
    }
  }
}
```

```js
const GRAVITY = 25, JUMP = 8.5, SPEED = 5.5, HW = 0.3, PH = 1.8, EYE = 1.62;

function collides() {
  const p = player.pos;
  const x0 = Math.floor(p.x - HW), x1 = Math.floor(p.x + HW);
  const y0 = Math.floor(p.y), y1 = Math.floor(p.y + PH);
  const z0 = Math.floor(p.z - HW), z1 = Math.floor(p.z + HW);
  for (let x = x0; x <= x1; x++)
    for (let y = y0; y <= y1; y++)
      for (let z = z0; z <= z1; z++)
        if (getBlock(x, y, z) !== 0) return true;
  return false;
}

function overlapsPlayer(x, y, z) {
  const p = player.pos;
  return x + 1 > p.x - HW && x < p.x + HW &&
         y + 1 > p.y && y < p.y + PH &&
         z + 1 > p.z - HW && z < p.z + HW;
}

function tickPhysics(dt) {
  const f = (keys.KeyW ? 1 : 0) - (keys.KeyS ? 1 : 0);
  const r = (keys.KeyD ? 1 : 0) - (keys.KeyA ? 1 : 0);
  let vx = 0, vz = 0;
  if (f !== 0 || r !== 0) {
    const inv = 1 / Math.hypot(f, r);
    const sn = Math.sin(yaw), cs = Math.cos(yaw);
    vx = (-f * sn + r * cs) * SPEED * inv;
    vz = (-f * cs - r * sn) * SPEED * inv;
  }
  player.vy -= GRAVITY * dt;
  if (player.vy < -40) player.vy = -40;
  if (keys.Space && player.onGround) { player.vy = JUMP; player.onGround = false; }

  player.pos.x += vx * dt;
  if (collides()) player.pos.x -= vx * dt;
  player.pos.z += vz * dt;
  if (collides()) player.pos.z -= vz * dt;
  player.pos.y += player.vy * dt;
  player.onGround = false;
  if (collides()) {
    player.pos.y -= player.vy * dt;
    if (player.vy < 0) player.onGround = true;
    player.vy = 0;
  }
  if (player.pos.y < -20) { player.pos.copy(SPAWN); player.vy = 0; }
}
```

```js
function buildMeshForChunk(cx, cz) {
  const c = CHUNKS.get(cx + "," + cz);
  if (!c || c.mesh) return;
  const data = c.data;
  const ox = cx * W, oz = cz * W;
  const pos = [], nrm = [], col = [], idx = [];
  for (let y = 0; y < H_; y++) {
    for (let lz = 0; lz < W; lz++) {
      for (let lx = 0; lx < W; lx++) {
        const id = data[ci(lx, y, lz)];
        if (id === 0) continue;
        const rgb = BLOCK_RGB[id];
        const wx = ox + lx, wz = oz + lz;
        for (let f = 0; f < 6; f++) {
          const F = FACES[f];
          const nx = lx + F.dir[0], ny = y + F.dir[1], nz = lz + F.dir[2];
          let nId;
          if (nx >= 0 && nx < W && nz >= 0 && nz < W && ny >= 0 && ny < H_) {
            nId = data[ci(nx, ny, nz)];
          } else {
            nId = getBlock(wx + F.dir[0], ny, wz + F.dir[2]);
          }
          if (nId !== 0) continue;
          const vi = pos.length / 3;
          for (let i = 0; i < 4; i++) {
            const cn = F.corners[i];
            pos.push(wx + cn[0], y + cn[1], wz + cn[2]);
            nrm.push(F.dir[0], F.dir[1], F.dir[2]);
            col.push(rgb[0] * F.shade, rgb[1] * F.shade, rgb[2] * F.shade);
          }
          idx.push(vi, vi + 1, vi + 2, vi, vi + 2, vi + 3);
        }
      }
    }
  }
  const g = new THREE.BufferGeometry();
  g.setAttribute("position", new THREE.Float32BufferAttribute(pos, 3));
  g.setAttribute("normal", new THREE.Float32BufferAttribute(nrm, 3));
  g.setAttribute("color", new THREE.Float32BufferAttribute(col, 3));
  g.setIndex(idx);
  g.computeBoundingSphere();
  const mesh = new THREE.Mesh(g, blockMaterial);
  scene.add(mesh);
  chunkMeshes.push(mesh);
  c.mesh = mesh;
}
```

```js
function rebuildChunk(cx, cz) {
  const c = CHUNKS.get(cx + "," + cz);

    const i = chunkMeshes.indexOf(c.mesh);
```

```js
function forEachRing(radius, cb) {
  for (let r = 0; r <= radius; r++) {
    for (let dx = -r; dx <= r; dx++) {
      for (let dz = -r; dz <= r; dz++) {
        if (Math.max(Math.abs(dx), Math.abs(dz)) !== r) continue;
        if (cb(dx, dz) === true) return;
      }
    }
  }
}
```
