

Причина: в теле класса `Folkling` строка `const bodyOffsets=[new Tween(walkPulse,fallPrime),new *)&- null ];` содержит `new *`, что является синтаксической ошибкой JavaScript — весь `<script>` блок не парсится, сцена не создаётся, и в браузере видны лишь CSS-надписи на пустом холсте. Кроме того, даже при удачном парсинге ни один хомяк не инстанцируется, а цикл `requestAnimationFrame` отсутствует.

const sparkles = [];
function spawnSparkle(pos, em){

  const sp = new THREE.Sprite(mat); sp.renderOrder = 10;
  sp.position.copy(pos); sp.scale.setScalar(0.55);
  sp.userData = {t:0, vel:new THREE.Vector3((Math.random()-0.5)*0.6, 0.9+Math.random()*0.5, (Math.random()-0.5)*0.6)};

}
function updateSparkles(dt){
  for (let i=sparkles.length-1;i>=0;i--){
    const s = sparkles[i]; s.userData.t += dt;
    s.position.addScaledVector(s.userData.vel, dt);
    s.scale.setScalar(0.55 + Math.sin(s.userData.t*20)*0.04);
    if (s.userData.t > 0.7) s.material.opacity = Math.max(0, 1-(s.userData.t-0.7)/0.4);
    if (s.userData.t > 1.15){ scene.remove(s); s.material.dispose(); sparkles.splice(i,1); }
  }
}

/* ============================================================
   7. AUTONOMOUS BEHAVIOUR
============================================================ */
const HAM_NAMES = ['Banana','Noodle','Pebble','Muffin','Quiche'];
const FLOOR_Y = 0.56;
const CAGE_LIMIT = B_R - 0.8;

const WALK_SPEED = 1.4;
const TURN_SPEED = 2.5;

const TARGETS = {
  wheel:  new THREE.Vector3(wheelPos.x, FLOOR_Y, wheelPos.z),
  tunnel: new THREE.Vector3(tunnelPos.x, FLOOR_Y, tunnelPos.z),
  bowl:   new THREE.Vector3(bowlPos.x, FLOOR_Y, bowlPos.z)
};

class Hamster{
  constructor(idx){
    const fur = PALETTE[idx % PALETTE.length];
    this.art = makeHamster(fur);
    this.name = HAM_NAMES[idx % HAM_NAMES.length];
    scene.add(this.art.group);

    const angle = idx / HAM_NAMES.length * Math.PI * 2 + 0.5;
    const dist = 1.0 + (idx % 3) * 0.8;
    this.art.group.position.set(
      Math.cos(angle)*dist, FLOOR_Y, Math.sin(angle)*dist
    );
    this.art.group.rotateY(Math.random()*Math.PI*2);
    this.art.group.add(nameTag(this.name, 'rgba(138,96,66,0.95)'));

    this.heading = Math.random()*Math.PI*2;
    this.state = 'walk';
    this.timer = 1 + Math.random()*2;
    this.target = null;
    this.walkPhase = Math.random()*10;
    this.cheekScale = 0;
    this.bobPhase = 0;
  }

  pickNextState(){
    const r = Math.random();
    if (r < 0.25){
      this.state = 'go_wheel';
      this.target = TARGETS.wheel.clone();
      this.timer = 3 + Math.random()*2;
    } else if (r < 0.45){
      this.state = 'go_tunnel';
      this.target = TARGETS.tunnel.clone();
      this.timer = 2.5;
    } else if (r < 0.65){
      this.state = 'go_bowl';
      this.target = TARGETS.bowl.clone();
      this.timer = 3 + Math.random()*2;
    } else if (r < 0.82){
      this.state = 'pause';
      this.timer = 1.5 + Math.random()*2;
    } else {
      this.state = 'walk';
      this.timer = 2 + Math.random()*3;
      this.heading += (Math.random()-0.5)*2.5;
    }
  }

  update(dt){
    this.timer -= dt;
    this.bobPhase += dt * 8;

    // Cheek animation
    const targetCheek = (this.state==='eat') ? 1 : 0;
    this.cheekScale += (targetCheek - this.cheekScale) * dt * 4;
    this.art.cheeks.forEach(c => c.scale.setScalar(this.cheekScale * 0.8));

    switch(this.state){
      case 'walk': this.doWalk(dt); break;
      case 'pause': this.doPause(dt); break;
      case 'go_wheel': this.doGoToTarget(dt, 'on_wheel'); break;
      case 'on_wheel': this.doWheel(dt); break;
      case 'go_tunnel': this.doGoToTarget(dt, 'in_tunnel'); break;
      case 'in_tunnel': this.doTunnel(dt); break;
      case 'go_bowl': this.doGoToTarget(dt, 'eat'); break;
      case 'eat': this.doEat(dt); break;
    }

    // Clamp to cage
    const p = this.art.group.position;
    const r = Math.sqrt(p.x*p.x + p.z*p.z);
    if (r > CAGE_LIMIT){
      const s = CAGE_LIMIT / r;
      p.x *= s; p.z *= s;
    }
    p.y = FLOOR_Y;

    // Walk bob
    if (this.state === 'walk' || this.state.startsWith('go_')){
      this.art.group.position.y = FLOOR_Y + Math.abs(Math.sin(this.walkPhase))*0.04;
    } else {
      this.art.group.position.y = FLOOR_Y;
    }

    // Foot shuffle
    const shuf = (this.state==='walk'||this.state.startsWith('go_')) ? Math.sin(this.walkPhase*2)*0.06 : 0;
    this.art.feet[0].position.y = 0.1 + shuf;
    this.art.feet[1].position.y = 0.1 - shuf;
    this.art.feet[2].position.y = 0.1 - shuf;
    this.art.feet[3].position.y = 0.1 + shuf;
  }

  doWalk(dt){
    this.walkPhase += dt * 10;
    const dx = Math.sin(this.heading), dz = Math.cos(this.heading);
    this.art.group.position.x += dx * WALK_SPEED * dt;
    this.art.group.position.z += dz * WALK_SPEED * dt;
    this.art.group.rotation.y = this.heading;
    if (this.timer <= 0) this.pickNextState();
  }

  doPause(dt){
    // idle wiggle
    this.art.head.rotation.z = Math.sin(this.bobPhase*0.5)*0.08;
    if (this.timer <= 0) this.pickNextState();
  }

  doGoToTarget(dt, arrivalState){
    if (!this.target) { this.state='walk'; return; }
    const p = this.art.group.position;
    const dx = this.target.x - p.x, dz = this.target.z - p.z;
    const dist = Math.sqrt(dx*dx+dz*dz);
    if (dist < 0.5){
      this.state = arrivalState;
      this.timer = 2.5 + Math.random()*2;
      // face target
      this.art.group.rotation.y = Math.atan2(dx, dz);
      if (arrivalState==='on_wheel'){
        wheelSpinSpeed = 3;
        spawnSparkle(p.clone().add(new THREE.Vector3(0,1,0)), '🏃‍♂️');
      }
      if (arrivalState==='in_tunnel'){
        tunnelOccupants++;
        tunnelMat.color.setHex(tunnelActiveColor);
        spawnSparkle(p.clone().add(new THREE.Vector3(0,1,0)), '🕳️');
      }
      if (arrivalState==='eat'){
        spawnSparkle(p.clone().add(new THREE.Vector3(0,1,0)), '🧀');
      }
      return;
    }
    this.walkPhase += dt * 10;
    const speed = WALK_SPEED * dt;
    p.x += (dx/dist)*speed;
    p.z += (dz/dist)*speed;
    this.art.group.rotation.y = Math.atan2(dx, dz);
  }

  doWheel(dt){
    // bounce in place, simulate running
    this.art.group.position.y = FLOOR_Y + Math.abs(Math.sin(this.bobPhase*1.5))*0.08;
    this.art.head.rotation.z = Math.sin(this.bobPhase)*0.1;
    if (this.timer <= 0){
      this.state = 'walk';
      wheelSpinSpeed = 0;
      this.heading = Math.random()*Math.PI*2;
      this.timer = 2 + Math.random()*3;
      spawnSparkle(this.art.group.position.clone().add(new THREE.Vector3(0,1,0)), '😄');
    }
  }

  doTunnel(dt){
    // shrink into tunnel
    const s = Math.max(0.1, 1 - (2.5 - this.timer)/1.5);
    this.art.group.scale.setScalar(s);
    if (this.timer <= 0){
      // pop out other side
      this.art.group.scale.setScalar(1);
      this.art.group.position.x = tunnelPos.x + 1.2;
      this.art.group.position.z = tunnelPos.z + 0.8;
      tunnelOccupants--;
      if (tunnelOccupants <= 0){ tunnelOccupants=0; tunnelMat.color.setHex(tunnelInactiveColor); }
      this.state = 'walk';
      this.heading = Math.random()*Math.PI*2;
      this.timer = 2 + Math.random()*3;
      spawnSparkle(this.art.group.position.clone().add(new THREE.Vector3(0,1,0)), '✨');
    }
  }

  doEat(dt){
    // bob head up/down (gnawing)
    this.art.head.position.y = 0.52 + Math.sin(this.bobPhase*2)*0.05;
    this.art.head.rotation.x = Math.sin(this.bobPhase*2)*0.1;
    if (this.timer <= 0){
      this.art.head.position.y = 0.52;
      this.art.head.rotation.x = 0;
      this.state = 'walk';
      this.heading = Math.random()*Math.PI*2;
      this.timer = 2 + Math.random()*3;
      spawnSparkle(this.art.group.position.clone().add(new THREE.Vector3(0,1,0)), '😋');
    }
  }
}

const clock = new THREE.Clock();

function animate(){
  requestAnimationFrame(animate);
  const dt = Math.min(clock.getDelta(), 0.05);

  // Update hamsters
  for (const h of hamsters) h.update(dt);

  // Wheel spin
  if (wheelSpinSpeed > 0){
    spin.rotation.z += wheelSpinSpeed * dt;
    wheelSpinSpeed *= 0.995;
    if (wheelSpinSpeed < 0.1) wheelSpinSpeed = 0;
  }

  // Sparkles
  updateSparkles(dt);

  controls.update();
  renderer.render(scene, camera);
}
