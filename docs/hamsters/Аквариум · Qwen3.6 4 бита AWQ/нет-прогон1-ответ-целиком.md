Вот полный код проекта в одном HTML-файле. Вы можете сохранить его как `index.html` и открыть в любом современном браузере.

Проект включает в себя генерацию процедурной геометрии для рыбок, физику движения, систему частиц (пузыри), кормление и красивый UI.

    <script>
        // --- Глобальные переменные ---
        let scene, camera, renderer, controls;
        let aquariumBox, tankEdges;
        let fishArray = [];
        let foodArray = [];
        let bubbleArray = [];
        let raycaster = new THREE.Raycaster();
        let mouse = new THREE.Vector2();
        let clock = new THREE.Clock();
        let mainLight, ambientLight;
        let isLightOn = true;

        // Размеры аквариума
        const TANK_WIDTH = 36;
        const TANK_HEIGHT = 24;
        const TANK_DEPTH = 20;

        // Цветовые схемы рыбок
        const FISH_COLORS = [0xff7f00, 0x007fff, 0xff4500, 0x800080, 0xff0000, 0x00ff00, 0xff69b4, 0xffd700];

        // FPS счетчик
        let frames = 0;
        let lastTime = performance.now();

        init();
        animate();

        function init() {
            // 1. Сцена и Камера
            scene = new THREE.Scene();
            scene.background = new THREE.Color(0x001e36); // Темно-синий фон
            scene.fog = new THREE.FogExp2(0x001e36, 0.015); // Туман воды

            camera = new THREE.PerspectiveCamera(45, window.innerWidth / window.innerHeight, 0.1, 1000);
            camera.position.set(0, 10, 55);

            // 2. Рендерер
            renderer = new THREE.WebGLRenderer({ antialias: true });
            renderer.setSize(window.innerWidth, window.innerHeight);
            renderer.shadowMap.enabled = true;
            renderer.shadowMap.type = THREE.PCFSoftShadowMap;
            document.body.appendChild(renderer.domElement);

            // 3. Контроллеры
            controls = new THREE.OrbitControls(camera, renderer.domElement);
            controls.enableDamping = true;
            controls.dampingFactor = 0.05;
            controls.maxPolarAngle = Math.PI / 1.8;
            controls.minDistance = 10;
            controls.maxDistance = 60;
            controls.target.set(0, 0, 0);

            // 4. Освещение
            setupLighting();

            // 5. Аквариум (Стекло, Песок, Камни, Водоросли)
            createAquariumEnvironment();

            // 6. Начальные объекты
            for(let i=0; i<15; i++) addFish();
            createBubbles(30);

            // 7. События
            window.addEventListener('resize', onWindowResize);
            window.addEventListener('click', onMouseClick);

            document.getElementById('loading').style.opacity = 0;
        }

        function setupLighting() {
            ambientLight = new THREE.AmbientLight(0x404040, 0.4);
            scene.add(ambientLight);

            mainLight = new THREE.DirectionalLight(0xffffff, 1.2);
            mainLight.position.set(20, 40, 20);
            mainLight.castShadow = true;
            mainLight.shadow.mapSize.width = 2048;
            mainLight.shadow.mapSize.height = 2048;
            scene.add(mainLight);

            // Подводные огни
            const p1 = new THREE.PointLight(0x00ffff, 0.8, 40);
            p1.position.set(-10, 10, -5);
            scene.add(p1);

            const p2 = new THREE.PointLight(0x0000ff, 0.8, 40);
            p2.position.set(10, 10, 5);
            scene.add(p2);
        }

        function createAquariumEnvironment() {
            // --- Стекло ---
            const glassGeo = new THREE.BoxGeometry(TANK_WIDTH, TANK_HEIGHT, TANK_DEPTH);
            const glassMat = new THREE.MeshPhysicalMaterial({
                color: 0xffffff,
                metalness: 0.1,
                roughness: 0.05,
                transmission: 0.95, // Преломление
                thickness: 0.5,
                transparent: true,
                opacity: 0.3,
                side: THREE.DoubleSide
            });
            aquariumBox = new THREE.Mesh(glassGeo, glassMat);
            aquariumBox.receiveShadow = true;
            scene.add(aquariumBox);

            // Рамка (Wireframe)
            const edges = new THREE.EdgesGeometry(glassGeo);
            tankEdges = new THREE.LineSegments(edges, new THREE.LineBasicMaterial({ color: 0x88ccff, opacity: 0.5, transparent: true }));
            scene.add(tankEdges);

            // --- Песок ---
            const sandGeo = new THREE.PlaneGeometry(TANK_WIDTH, TANK_DEPTH, 32, 32);
            // Неровности песка
            const posAttribute = sandGeo.attributes.position;
            for (let i = 0; i < posAttribute.count; i++) {
                const z = posAttribute.getZ(i);
                posAttribute.setZ(i, z + (Math.random() * 0.5));
            }
            sandGeo.computeVertexNormals();

            const sandMat = new THREE.MeshStandardMaterial({ 
                color: 0xe6c288, 
                roughness: 1,
                side: THREE.DoubleSide
            });
            const sand = new THREE.Mesh(sandGeo, sandMat);
            sand.rotation.x = -Math.PI / 2;
            sand.position.y = -TANK_HEIGHT / 2;
            sand.receiveShadow = true;
            scene.add(sand);

            // --- Камни ---
            const rockGeo = new THREE.DodecahedronGeometry(1.5, 0); // Низкополигональный
            for(let i=0; i<8; i++) {
                const rockMat = new THREE.MeshStandardMaterial({ color: 0x555555 + Math.random()*0x222222, roughness: 0.8 });
                const rock = new THREE.Mesh(rockGeo, rockMat);
                
                // Деформация камня
                const scale = 0.5 + Math.random() * 1;
                rock.scale.set(scale, scale * 0.7, scale);
                
                rock.position.set(
                    (Math.random() - 0.5) * (TANK_WIDTH - 4),
                    -TANK_HEIGHT / 2 + 0.5,
                    (Math.random() - 0.5) * (TANK_DEPTH - 4)
                );
                rock.rotation.set(Math.random(), Math.random(), Math.random());
                rock.castShadow = true;
                rock.receiveShadow = true;
                scene.add(rock);
            }

            // --- Водоросли ---
            const algaeColors = [0x228b22, 0x32cd32, 0x006400];
            for(let i=0; i<12; i++) {
                const points = [];
                const height = 4 + Math.random() * 6;
                for(let j=0; j<=5; j++) {
                    const y = (j/5) * height;
                    const x = Math.sin(j * 0.5) * 0.5; // Изгиб
                    points.push(new THREE.Vector3(x, y - TANK_HEIGHT/2, 0));
                }
                
                const curve = new THREE.CatmullRomCurve3(points);
                const tubeGeo = new THREE.TubeGeometry(curve, 12, 0.15, 4, false);
                const tubeMat = new THREE.MeshLambertMaterial({ color: algaeColors[Math.floor(Math.random()*algaeColors.length)] });
                const algae = new THREE.Mesh(tubeGeo, tubeMat);
                
                algae.position.set(
                    (Math.random() - 0.5) * (TANK_WIDTH - 2),
                    0,
                    (Math.random() - 0.5) * (TANK_DEPTH - 2)
                );
                algae.rotation.z = (Math.random() - 0.5) * 0.5;
                
                // Добавляем пользовательские данные для анимации
                algae.userData = { 
                    baseRotX: algae.rotation.x, 
                    baseRotZ: algae.rotation.z,
                    speed: 0.5 + Math.random() * 0.5,
                    offset: Math.random() * 100
                };
                
                scene.add(algae);
                // Сохраняем ссылку для анимации (можно вынести в отдельный массив)
                if(!window.algaeList) window.algaeList = [];
                window.algaeList.push(algae);
            }
        }

        // --- Создание Рыбки ---
        function createFishMesh(color) {
            const fishGroup = new THREE.Group();

            // Материалы
            const bodyMat = new THREE.MeshPhongMaterial({ color: color, shininess: 100, specular: 0x222222 });
            const finMat = new THREE.MeshPhongMaterial({ color: color, transparent: true, opacity: 0.9 });
            const eyeMat = new THREE.MeshBasicMaterial({ color: 0xffffff });
            const pupilMat = new THREE.MeshBasicMaterial({ color: 0x000000 });

            // 1. Тело (Сфера, сплюснутая)
            const bodyGeo = new THREE.SphereGeometry(1, 16, 12);
            bodyGeo.scale(1.8, 1, 0.8); // Вытягиваем
            const body = new THREE.Mesh(bodyGeo, bodyMat);
            fishGroup.add(body);

            // 2. Хвост (Треугольная плоскость)
            const tailShape = new THREE.Shape();
            tailShape.moveTo(0, 0);
            tailShape.lineTo(1.5, 1);
            tailShape.lineTo(1.5, -1);
            tailShape.lineTo(0, 0);
            
            const tailGeo = new THREE.ShapeGeometry(tailShape);
            const tail = new THREE.Mesh(tailGeo, finMat);
            tail.position.set(-1.6, 0, 0); // Сзади тела
            tail.rotation.y = Math.PI / 2; // Поворот чтобы махать по Z
            fishGroup.add(tail);

            // 3. Плавники
            // Верхний
            const topFinGeo = new THREE.PlaneGeometry(0.8, 0.8);
            const topFin = new THREE.Mesh(topFinGeo, finMat);
            topFin.position.set(0, 0.8, 0);
            topFin.rotation.x = -Math.PI / 4;
            fishGroup.add(topFin);

            // Боковые
            const sideFinGeo = new THREE.PlaneGeometry(0.6, 0.4);
            const leftFin = new THREE.Mesh(sideFinGeo, finMat);
            leftFin.position.set(0.2, -0.2, 0.6);
            leftFin.rotation.y = -Math.PI / 3;
            leftFin.rotation.x = Math.PI / 4;
            fishGroup.add(leftFin);

            const rightFin = new THREE.Mesh(sideFinGeo, finMat);
            rightFin.position.set(0.2, -0.2, -0.6);
            rightFin.rotation.y = Math.PI / 3;
            rightFin.rotation.x = Math.PI / 4;
            fishGroup.add(rightFin);

            // 4. Глаза
            const eyeGeo = new THREE.SphereGeometry(0.15, 8, 8);
            const pupilGeo = new THREE.SphereGeometry(0.08, 8, 8);
            
            const leftEye = new THREE.Mesh(eyeGeo, eyeMat);
            leftEye.position.set(1.0, 0.3, 0.5);
            const leftPupil = new THREE.Mesh(pupilGeo, pupilMat);
            leftPupil.position.z = 0.1;
            leftEye.add(leftPupil);
            fishGroup.add(leftEye);

            const rightEye = new THREE.Mesh(eyeGeo, eyeMat);
            rightEye.position.set(1.0, 0.3, -0.5);
            const rightPupil = new THREE.Mesh(pupilGeo, pupilMat);
            rightPupil.position.z = 0.1;
            rightEye.add(rightPupil);
            fishGroup.add(rightEye);

            return { group: fishGroup, tail: tail, leftFin: leftFin, rightFin: rightFin, topFin: topFin };
        }

        function addFish() {
            const color = FISH_COLORS[Math.floor(Math.random() * FISH_COLORS.length)];
            const meshData = createFishMesh(color);
            const scale = 0.6 + Math.random() * 0.6;
            
            meshData.group.scale.set(scale, scale, scale);
            
            // Случайная позиция внутри
            meshData.group.position.set(
                (Math.random() - 0.5) * (TANK_WIDTH - 4),
                (Math.random() - 0.5) * (TANK_HEIGHT - 4),
                (Math.random() - 0.5) * (TANK_DEPTH - 4)
            );

            scene.add(meshData.group);

            // Данные поведения
            const fishObj = {
                mesh: meshData.group,
                tail: meshData.tail,
                leftFin: meshData.leftFin,
                rightFin: meshData.rightFin,
                topFin: meshData.topFin,
                velocity: new THREE.Vector3(Math.random()-0.5, Math.random()-0.5, Math.random()-0.5).normalize().multiplyScalar(0.1),
                speed: 0.05 + Math.random() * 0.05,
                tailSpeed: 5 + Math.random() * 5,
                phase: Math.random() * 100,
                targetFood: null,
                avoidanceRadius: 4.0,
                radius: 1.5 * scale
            };

            fishArray.push(fishObj);
            updateStats();
        }

        // --- Логика Пузырей ---
        function createBubble() {
            const geo = new THREE.SphereGeometry(0.2 + Math.random()*0.2, 8, 8);
            const mat = new THREE.MeshPhysicalMaterial({
                color: 0xffffff,
                transmission: 0.9,
                roughness: 0.1,
                metalness: 0.1,
                transparent: true,
                opacity: 0.5
            });
            const mesh = new THREE.Mesh(geo, mat);
            
            mesh.position.set(
                (Math.random() - 0.5) * (TANK_WIDTH - 2),
                -TANK_HEIGHT / 2 + 1,
                (Math.random() - 0.5) * (TANK_DEPTH - 2)
            );

            mesh.userData = {
                velocityY: 0.05 + Math.random() * 0.05,
                wobbleOffset: Math.random() * Math.PI * 2,
                wobbleSpeed: 2 + Math.random() * 3
            };

            scene.add(mesh);
            bubbleArray.push(mesh);
        }

        function createBubbles(count) {
            for(let i=0; i<count; i++) createBubble();
        }

        function addBubbles() {
            createBubbles(10);
        }

        // --- Логика Корма ---
        function spawnFood(x, y, z) {
            const geo = new THREE.SphereGeometry(0.2, 8, 8);
            const mat = new THREE.MeshStandardMaterial({ color: 0xffaa00, emissive: 0x221100 });
            const mesh = new THREE.Mesh(geo, mat);
            
            // Если координаты не переданы (клик по UI), спавним сверху по центру
            mesh.position.set(x || 0, y || TANK_HEIGHT/2, z || 0);
            
            scene.add(mesh);
            
            foodArray.push({
                mesh: mesh,
                velocity: new THREE.Vector3((Math.random()-0.5)*0.01, -0.05, (Math.random()-0.5)*0.01) // Гравитация
            });
        }

        // --- Управление ---
        function onMouseClick(event) {
            // Игнорируем клики по UI
            if(event.target.closest('.ui-panel')) return;

            mouse.x = (event.clientX / window.innerWidth) * 2 - 1;
            mouse.y = -(event.clientY / window.innerHeight) * 2 + 1;

            raycaster.setFromCamera(mouse, camera);
            
            // Проверяем пересечение со стеклянной коробкой
            const intersects = raycaster.intersectObject(aquariumBox);
            
            if (intersects.length > 0) {
                const point = intersects[0].point;
                // Ограничиваем спавн внутри коробки
                const safeX = Math.max(-TANK_WIDTH/2 + 1, Math.min(TANK_WIDTH/2 - 1, point.x));
                const safeY = Math.max(-TANK_HEIGHT/2 + 1, Math.min(TANK_HEIGHT/2 - 1, point.y));
                const safeZ = Math.max(-TANK_DEPTH/2 + 1, Math.min(TANK_DEPTH/2 - 1, point.z));
                
                spawnFood(safeX, safeY, safeZ);
            } else {
                // Если кликнули мимо, кидаем сверху случайно
                spawnFood();
            }
        }

        function toggleLight() {
            isLightOn = !isLightOn;
            mainLight.intensity = isLightOn ? 1.2 : 0.0;
            ambientLight.intensity = isLightOn ? 0.4 : 0.1;
        }

        function updateStats() {
            document.getElementById('fish-count').innerText = fishArray.length;
        }

        function onWindowResize() {
            camera.aspect = window.innerWidth / window.innerHeight;
            camera.updateProjectionMatrix();
            renderer.setSize(window.innerWidth, window.innerHeight);
        }

        // --- Главный цикл анимации ---
        function animate() {
            requestAnimationFrame(animate);

            const time = clock.getElapsedTime();
            const delta = clock.getDelta(); // Не используем везде, но полезно для физики

            // FPS Counter
            frames++;
            const currentTime = performance.now();
            if (currentTime >= lastTime + 1000) {
                document.getElementById('fps-counter').innerText = frames;
                frames = 0;
                lastTime = currentTime;
            }

            controls.update();

            // 1. Анимация Водорослей
            if(window.algaeList) {
                window.algaeList.forEach(algae => {
                    const d = algae.userData;
                    algae.rotation.x = d.baseRotX + Math.sin(time * d.speed + d.offset) * 0.1;
                    algae.rotation.z = d.baseRotZ + Math.cos(time * d.speed * 0.8 + d.offset) * 0.1;
                });
            }

            // 2. Анимация Пузырей
            for (let i = bubbleArray.length - 1; i >= 0; i--) {
                const b = bubbleArray[i];
                b.position.y += b.userData.velocityY;
                b.position.x += Math.sin(time * b.userData.wobbleSpeed + b.userData.wobbleOffset) * 0.02;
                
                // Сброс если достигли верха
                if(b.position.y > TANK_HEIGHT/2) {
                    b.position.y = -TANK_HEIGHT/2;
                    b.position.x = (Math.random() - 0.5) * (TANK_WIDTH - 2);
                }
            }

            // 3. Анимация Корма
            for (let i = foodArray.length - 1; i >= 0; i--) {
                const f = foodArray[i];
                f.mesh.position.add(f.velocity);
                
                // Удаление если достиг дна
                if(f.mesh.position.y <= -TANK_HEIGHT/2 + 0.5) {
                    scene.remove(f.mesh);
                    foodArray.splice(i, 1);
                }
            }

            // 4. Логика Рыбок
            fishArray.forEach(fish => {
                // --- Движение ---
                let moveDir = fish.velocity.clone();

                // Поиск еды
                if (!fish.targetFood) {
                    // Ищем ближайшую еду в радиусе 15
                    let closestDist = 15;
                    for(let f of foodArray) {
                        const dist = fish.mesh.position.distanceTo(f.mesh.position);
                        if(dist < closestDist) {
                            closestDist = dist;
                            fish.targetFood = f;
                        }
                    }
                } else {
                    // Если цель есть, проверяем актуальность
                    if (!foodArray.includes(fish.targetFood)) {
                        fish.targetFood = null;
                    } else {
                        // Плывем к еде
                        const dirToFood = new THREE.Vector3().subVectors(fish.targetFood.mesh.position, fish.mesh.position).normalize();
                        // Интерполяция скорости к направлению еды
                        moveDir.lerp(dirToFood, 0.05);
                        moveDir.normalize().multiplyScalar(fish.speed * 2.5); // Быстрее плывем к еде
                        
                        // Съели?
                        if (moveDir.length() < 0.01 || fish.mesh.position.distanceTo(fish.targetFood.mesh.position) < 1.0) {
                            // Удаляем еду
                            const idx = foodArray.indexOf(fish.targetFood);
                            if(idx > -1) {
                                scene.remove(fish.targetFood.mesh);
                                foodArray.splice(idx, 1);
                                
                                // Рост рыбки
                                const currentScale = fish.mesh.scale.x;
                                const growAmount = currentScale * 1.05;
                                fish.mesh.scale.set(growAmount, growAmount, growAmount);
                                fish.radius = 1.5 * growAmount;
                            }
                            fish.targetFood = null;
                        }
                    }
                }

                // --- Избегание других рыбок ---
                for (let other of fishArray) {
                    if (other === fish) continue;
                    const dist = fish.mesh.position.distanceTo(other.mesh.position);
                    if (dist < fish.avoidanceRadius) {
                        const pushDir = new THREE.Vector3().subVectors(fish.mesh.position, other.mesh.position).normalize();
                        moveDir.add(pushDir.multiplyScalar(0.05));
                    }
                }

                // --- Столкновение со стенами ---
                const padding = 1.5;
                const hw = TANK_WIDTH/2 - padding;
                const hh = TANK_HEIGHT/2 - padding;
                const hd = TANK_DEPTH/2 - padding;

                // Отражение вектора скорости
                if (fish.mesh.position.x > hw) moveDir.x -= 0.1;
                if (fish.mesh.position.x < -hw) moveDir.x += 0.1;
                if (fish.mesh.position.y > hh) moveDir.y -= 0.1;
                if (fish.mesh.position.y < -hh) moveDir.y += 0.1;
                if (fish.mesh.position.z > hd) moveDir.z -= 0.1;
                if (fish.mesh.position.z < -hd) moveDir.z += 0.1;

                // Ограничение максимальной скорости
                moveDir.clampLength(0, fish.speed * 2);

                // Применение движения
                fish.mesh.position.add(moveDir);

                // --- Поворот тела ---
                // Плавно поворачиваем рыбку в направлении движения
                if (moveDir.lengthSq() > 0.0001) {
                    const lookTarget = fish.mesh.position.clone().add(moveDir);
                    fish.mesh.lookAt(lookTarget);
                    
                    // Анимация хвоста (синус от времени)
                    const tailSwing = Math.sin(time * fish.tailSpeed + fish.phase) * 0.3;
                    fish.tail.rotation.y = Math.PI/2 + tailSwing;
                    
                    // Анимация плавников
