void main(){

    vec2 i = p + vec2(cos(tt - p.y) + sin(tt + p.x), sin(tt - p.y) + cos(tt + p.x));
    c += 1.0/length(vec2(p.x/(sin(i.x+tt)/inten), p.y/(cos(i.y+tt)/inten)));
