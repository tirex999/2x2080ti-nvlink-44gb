      mesh:g, tail, leftFin:lf, rightFin:rf, dorsal,
      baseScale:g.scale.x,
      velocity:new THREE.Vector3(Math.cos(angle),0,Math.sin(angle)),
      speed:rand(3,5.5),
      tailSpeed:rand(7,13),
      phase:rand(0,Math.PI*2),
      wanderTimer:rand(1,4),
      avoidanceRadius:rand(3.5,6),
      targetFood:null
    };
  }
  for(let i=0;i<15;i++) fishArray.push(createFish());

  // ---------- Food ----------
  const foodArray=[];
  const foodGeo=new THREE.SphereGeometry(0.28,10,10);
  const foodMat=new THREE.MeshStandardMaterial({color:0xff7a2f,roughness:0.5,emissive:0x552200,emissiveIntensity:0.4});
  function addFood(pos){
    const m=new THREE.Mesh(foodGeo,foodMat);
    m.position.copy(pos);
    m.userData={vel:new THREE.Vector3(rand(-0.4,0.4),rand(-1,0),rand(-0.4,0))};
    scene.add(m); foodArray.push(m);
  }

  // ---------- Raycasting (feed on click) ----------
  const raycaster=new THREE.Raycaster();
  const plane=new THREE.Plane(new THREE.Vector3(0,1,0),0);
  const ndc=new THREE.Vector2(); const hit=new THREE.Vector3();
  let downX=0,downY=0,downT=0;
  canvas.addEventListener('pointerdown',e=>{downX=e.clientX;downY=e.clientY;downT=performance.now();});
  canvas.addEventListener('pointerup',e=>{
    if(performance.now()-downT>260) return;
    if(Math.hypot(e.clientX-downX,e.clientY-downY)>6) return;
    ndc.x=(e.clientX/innerWidth)*2-1; ndc.y=-(e.clientY/innerHeight)*2+1;
    raycaster.setFromCamera(ndc,camera);
    if(raycaster.ray.intersectPlane(plane,hit)){
      hit.x=THREE.MathUtils.clamp(hit.x,-BOUND.x,BOUND.x);
      hit.y=THREE.MathUtils.clamp(hit.y,FLOOR_Y+0.6,BOUND.yTop);
      hit.z=THREE.MathUtils.clamp(hit.z,-BOUND.z,BOUND.z);
      addFood(hit);
      for(let i=0;i<3;i++) makeBubble();
    }
  });

  // ---------- Behavior ----------
  function updateFish(dt,time){
    for(const f of fishArray){
      f.velocity.normalize().multiplyScalar(f.speed);
      f.wanderTimer-=dt;
      if(f.wanderTimer<=0){
        f.velocity.x+=(Math.random()-0.5)*f.speed*0.9;
        f.velocity.y+=(Math.random()-0.5)*f.speed*0.5;
        f.velocity.z+=(Math.random()-0.5)*f.speed*0.9;
        f.wanderTimer=rand(2,5);
      }
      // separation
      for(const o of fishArray){
        if(o===f) continue;
        const dx=f.mesh.position.x-o.mesh.position.x,
              dy=f.mesh.position.y-o.mesh.position.y,
              dz=f.mesh.position.z-o.mesh.position.z;
        const d2=dx*dx+dy*dy+dz*dz, r=f.avoidanceRadius;
        if(d2<r*r && d2>0.0001){
          const d=Math.sqrt(d2), w=(1-d/r)*f.speed*0.9;
          f.velocity.x+=dx/d*w; f.velocity.y+=dy/d*w; f.velocity.z+=dz/d*w;
        }
      }
      // seek food
      if(!f.targetFood || !foodArray.includes(f.targetFood)) f.targetFood=null;
      if(f.targetFood){
        const d=f.mesh.position.distanceTo(f.targetFood.position);
        if(d<1.3){
          scene.remove(f.targetFood);
          foodArray.splice(foodArray.indexOf(f.targetFood),1);
          f.baseScale=Math.min(f.baseScale*1.05,2.4);
          f.targetFood=null;
        }else{
          const dirv=new THREE.Vector3().subVectors(f.targetFood.position,f.mesh.position).normalize();
          f.velocity.lerp(dirv.multiplyScalar(f.speed*1.6),0.08);
        }
      }
      // walls
      const p=f.mesh.position, turn=0.25;
      if(p.x>BOUND.x){f.velocity.x-=turn; if(p.x>BOUND.x)p.x=BOUND.x;}
      if(p.x<-BOUND.x){f.velocity.x+=turn; if(p.x<-BOUND.x)p.x=-BOUND.x;}
      if(p.z>BOUND.z){f.velocity.z-=turn; if(p.z>BOUND.z)p.z=BOUND.z;}
      if(p.z<-BOUND.z){f.velocity.z+=turn; if(p.z<-BOUND.z)p.z=-BOUND.z;}
      if(p.y>BOUND.yTop){f.velocity.y-=turn; if(p.y>BOUND.yTop)p.y=BOUND.yTop;}
      if(p.y<BOUND.yBot){f.velocity.y+=turn; if(p.y<BOUND.yBot)p.y=BOUND.yBot;}

      f.velocity.normalize().multiplyScalar(f.speed);
      p.addScaledVector(f.velocity,dt);

      // orient
      f.mesh.lookAt(p.x+f.velocity.x,p.y+f.velocity.y,p.z+f.velocity.z);

      // animate
      const tp=f.phase+time*f.tailSpeed;
      f.tail.rotation.z=Math.sin(tp)*0.6;
      f.leftFin.rotation.z=Math.sin(tp*0.7)*0.4;
      f.rightFin.rotation.z=-Math.sin(tp*0.7)*0.4;
      f.dorsal.rotation.z=Math.sin(tp*0.5)*0.15;
      f.mesh.scale.setScalar(f.baseScale);
    }
  }

  function updateFood(dt){
    for(let i=foodArray.length-1;i>=0;i--){
      const fd=foodArray[i], v=fd.userData.vel;
      v.y-=3.5*dt;
      fd.position.addScaledVector(v,dt);
      fd.rotation.y+=dt*3;
      if(fd.position.y<FLOOR_Y+0.3){
        fd.position.y=FLOOR_Y+0.3; v.y*=-0.3; v.x*=0.6; v.z*=0.6;
        if(Math.abs(v.y)<0.5){ scene.remove(fd); foodArray.splice(i,1); }
      }
    }
  }

  function updateBubbles(dt,time){
    for(const b of bubbles){
      const u=b.userData;
      b.position.y+=u.speed*dt;
      b.position.x=u.baseX+Math.sin(time*1.5+u.phase)*u.amp;
      b.position.z=u.baseZ+Math.cos(time*1.2+u.phase)*u.amp;
      if(b.position.y>BOUND.yTop){
        b.position.y=FLOOR_Y+rand(0,2);
        b.position.x=rand(-BOUND.x,BOUND.x);
        b.position.z=rand(-BOUND.z,BOUND.z);
        u.baseX=b.position.x; u.baseZ=b.position.z;
      }
    }
  }

  function updateWeeds(time){
    for(const w of weeds){
      w.rotation.x=w.userData.base.x+Math.sin(time*0.8+w.userData.ph)*w.userData.sway;
      w.rotation.z=w.userData.base.z+Math.cos(time*0.6+w.userData.ph)*w.userData.sway*0.7;
    }
  }

  // ---------- UI ----------
  const elFish=document.getElementById('fishCount');
  const elFps=document.getElementById('fps');
  const elBub=document.getElementById('bubbleCount');
  const toastEl=document.getElementById('toast');
  let toastTimer;
  function toast(msg){
    toastEl.textContent=msg; toastEl.classList.add('show');
    clearTimeout(toastTimer); toastTimer=setTimeout(()=>toastEl.classList.remove('show'),1600);
  }
  function refreshCounts(){ elFish.textContent=fishArray.length; elBub.textContent=bubbles.length; }
  refreshCounts();

  let lightOn=true;
  const btnLight=document.querySelector('.b-light');
  document.querySelectorAll('.btn').forEach(b=>{
    b.addEventListener('click',()=>{
      const a=b.dataset.act;
      if(a==='add'){
        if(fishArray.length>=40){toast('Максимум 40 рыбок');return;}
        fishArray.push(createFish()); refreshCounts(); toast('Рыбка добавлена 🐟');
      }else if(a==='bubble'){
        for(let i=0;i<10;i++) makeBubble(); refreshCounts(); toast('+10 пузырей ◦');
      }else if(a==='light'){
        lightOn=!lightOn;
        dir.intensity=lightOn?1.15:0.15;
        ambient.intensity=lightOn?0.4:0.18;
        pl1.intensity=lightOn?0.9:1.4; pl2.intensity=lightOn?0.85:1.3;
        btnLight.classList.toggle('off',!lightOn);
        btnLight.textContent=lightOn?'☀ Свет':'☾ Свет';
        toast(lightOn?'Основной свет вкл':'Основной свет выкл');
      }
    });
  });

  // ---------- Loop ----------
  const clock=new THREE.Clock();
  let frames=0, fpsAccum=0;
  function animate(){
    requestAnimationFrame(animate);
    const dt=Math.min(clock.getDelta(),0.05);
    const time=clock.elapsedTime;

    controls.update();
    updateFish(dt,time);
    updateFood(dt);

    caustic.tex.offset.x+=dt*0.02; caustic.tex.offset.y+=dt*0.015;
    caustic.mesh.material.opacity=0.4+Math.sin(time*1.2)*0.12;
    pl1.intensity=(lightOn?0.9:1.4)+Math.sin(time*2)*0.12;
    pl2.intensity=(lightOn?0.85:1.3)+Math.cos(time*1.7)*0.12;

    renderer.render(scene,camera);

    frames++; fpsAccum+=dt;
    if(fpsAccum>=0.5){ elFps.textContent=Math.round(frames/fpsAccum); frames=0; fpsAccum=0; }
  }
  animate();

  addEventListener('resize',()=>{
    camera.aspect=innerWidth/innerHeight; camera.updateProjectionMatrix();
    renderer.setSize(innerWidth,innerHeight);
  });

  // hide loader after first paint
  requestAnimationFrame(()=>requestAnimationFrame(()=>{
    document.getElementById('loader').classList.add('hide');
  }));
})();

Что внутри и как это работает:

- **Рыбки** — группа из тела/головы (масштабированные сферы), трёх полосок, глаз с зрачками, спинного/боковых плавников и анимируемого хвоста. 8 палитр, случайный масштаб 0.6–1.2, индивидуальная скорость/частота маха/фаза.
- **ИИ** — разделение (отталкивание), плавный отскок от стенок, случайное блуждание, погоня за кормом; после поедания размер растёт на 5% (до предела).
- **Окружение** — стеклянный бокс (`transmission: 0.95`) с wireframe-рамкой, процедурный песок с неровностями (принимает тени), 8 деформированных камней, 12 водорослей на `TubeGeometry`+`CatmullRomCurve3` с покачиванием, плюс «каустики» (движущийся световой узор на дне).
- **Пузыри** — 30 на старте, поднимаются с синусоидальным покачиванием и сбрасываются у поверхности; кнопка добавляет ещё 10.
- **Кормление** — клик по воде (raycast в плоскость) бросает корм с гравитацией; рыбки в радиусе 15 плывут к нему, съедают → удаляется + рост. Корм исчезает при оседании на дно.
- **Свет** — Ambient + Directional (тени PCFSoft, карта 2048) + два пульсирующих PointLight; кнопка «Свет» переключает режим (при выключенном — усиливается «подводный» свет).
- **Камера/UI** — OrbitControls (вращение/пан/зум 10–60, ограничение угла, damping), glass‑панели, счётчики рыбок/FPS/пузырей, тосты, адаптив и экран загрузки.

Кликни по воде, подбрось корм и понаблюдай, как стая стягивается и растёт.
