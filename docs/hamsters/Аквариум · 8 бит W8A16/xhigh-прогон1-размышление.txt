function updateFish(f, dt, t){
  const p = f.mesh.position;
  // wander
  f.wanderT -= dt;
  if(f.wanderT <= 0){
    f.wanderT = 2 + Math.random()*4;
    f.wanderDir.set(Math.random()-0.5, (Math.random()-0.5)*0.6, Math.random()-0.5).normalize();
  }
  f.wanderDir.x += (Math.random()-0.5)*dt*2.5;
  f.wanderDir.y += (Math.random()-0.5)*dt*1.2;
  f.wanderDir.z += (Math.random()-0.5)*dt*2.5;
  f.wanderDir.normalize();

  _desired.copy(f.wanderDir).multiplyScalar(f.speed);

  // food
  f.targetFood = null;
  let bestD = 15;
  for(const fd of foods){
    const d = p.distanceTo(fd.mesh.position);
    if(d < bestD){ bestD = d; f.targetFood = fd; }
  }
  if(f.targetFood){
    _diff.copy(f.targetFood.mesh.position).sub(p).normalize();
    _desired.copy(_diff).multiplyScalar(f.speed*2.1);
    if(bestD < 0.95) eatFood(f, f.targetFood);
  }

  // collision avoidance
  for(const o of fishArray){
    if(o === f) continue;
    _diff.copy(p).sub(o.mesh.position);
    const d = _diff.length();
    if(d < f.avoidanceRadius && d > 0.001){
      _diff.multiplyScalar((1 - d/f.avoidanceRadius) * f.speed * 2.6 / d);
      _desired.add(_diff);
    }
  }

  // walls
  const MX=14.5, MZ=6.5, MYT=9, MYB=-9.5;
  if(p.x >  MX) _desired.x -= (p.x-MX)*6;
  if(p.x < -MX) _desired.x += (-MX-p.x)*6;
  if(p.z >  MZ) _desired.z -= (p.z-MZ)*6;
  if(p.z < -MZ) _desired.z += (-MZ-p.z)*6;
  if(p.y >  MYT) _desired.y -= (p.y-MYT)*6;
  if(p.y < MYB) _desired.y += (MYB-p.y)*6;

  // steering
  f.velocity.lerp(_desired, 1 - Math.exp(-2.5*dt));
  const sp = f.velocity.length() || 0.001;
  const maxS = f.speed * (f.targetFood ? 2.1 : 1.25);
  const minS = f.speed * 0.45;
  if(sp > maxS) f.velocity.multiplyScalar(maxS/sp);
  else if(sp < minS) f.velocity.multiplyScalar(minS/sp);

  p.addScaledVector(f.velocity, dt);
  p.x = THREE.MathUtils.clamp(p.x, -16.8, 16.8);
  p.y = THREE.MathUtils.clamp(p.y, -10.8, 11);
  p.z = THREE.MathUtils.clamp(p.z, -8.8, 8.8);

  // orientation

  _orient.lookAt(p.x + f.velocity.x, p.y + f.velocity.y*0.6, p.z + f.velocity.z);
  f.mesh.quaternion.slerp(_orient.quaternion, Math.min(1, dt*5));

  // animation

  const wag2 = Math.sin(t*f.tailSpeed*0.5 + f.phase*1.7);
  f.tail.rotation.z = wag*0.55 + wag2*0.1;
