            for (let y = 0; y < 80; y++) {

                if (y === 0) block = 3; // unbreakable stone
                else if (y <= H) {

                    else if (y < H) {

                        if (H <= 16) block = 4; // sand

function buildChunkMesh(cx, cz) {
    const chunk = chunks.get(cx + ',' + cz);
    if (!chunk || !chunk.data) return;
    
    const positions = [];
    const normals = [];
    const colors = [];
    
    for (let lx = 0; lx < 16; lx++) {
        for (let y = 0; y < 80; y++) {
            for (let lz = 0; lz < 16; lz++) {
                const wx = cx * 16 + lx;
                const wz = cz * 16 + lz;
                const block = chunk.data[lx * 80 * 16 + y * 16 + lz];
                if (block === 0) continue;
                
                const color = BLOCK_COLORS[block];
                const r = ((color >> 16) & 0xff) / 255;
                const g = ((color >> 8) & 0xff) / 255;
                const b = (color & 0xff) / 255;
                
                for (let f = 0; f < 6; f++) {
                    const face = FACES[f];
                    const nx = wx + face.dir[0];
                    const ny = y + face.dir[1];
                    const nz = wz + face.dir[2];
                    
                    if (getBlock(nx, ny, nz) !== 0) continue;
                    
                    // Add face
                    const light = face.light;
                    const cr = r * light;
                    const cg = g * light;
                    const cb = b * light;
                    
                    // Two triangles from 4 corners
                    const c = face.corners;
                    // Triangle 1: 0, 1, 2
                    // Triangle 2: 0, 2, 3
                    const indices = [0, 1, 2, 0, 2, 3];
                    for (const i of indices) {
                        positions.push(wx + c[i][0], y + c[i][1], wz + c[i][2]);
                        normals.push(face.dir[0], face.dir[1], face.dir[2]);
                        colors.push(cr, cg, cb);
                    }
                }
            }
        }
    }
    
    // Remove old mesh if exists
    if (chunk.mesh) {
        scene.remove(chunk.mesh);
        chunk.mesh.geometry.dispose();
    }
    
    if (positions.length === 0) {
        chunk.mesh = null;
        return;
    }
    
    const geometry = new THREE.BufferGeometry();
    geometry.setAttribute('position', new THREE.Float32BufferAttribute(positions, 3));
    geometry.setAttribute('normal', new THREE.Float32BufferAttribute(normals, 3));
    geometry.setAttribute('color', new THREE.Float32BufferAttribute(colors, 3));
    
    const mesh = new THREE.Mesh(geometry, material);
    scene.add(mesh);
    chunk.mesh = mesh;
    
    // Update global mesh list for raycasting
    updateMeshList();
}
