function noise3(x,y,z){
  const xi=Math.floor(x),yi=Math.floor(y),zi=Math.floor(z);
  const xf=x-xi,yf=y-yi,zf=z-zi;
  const u=xf*xf*(3-2*xf),v=yf*yf*(3-2*yf),w=zf*zf*(3-2*zf);
  function h(a,b,c){return hash(xi+a,yi+b,zi+c);}
  const c000=h(0,0,0),c100=h(1,0,0),c010=h(0,1,0),c110=h(1,1,0),c001=h(0,0,1),c101=h(1,0,1),c011=h(0,1,1),c111=h(1,1,1);
  const x00=c000+(c100-c000)*u, x10=c010+(c110-c010)*u, x01=c001+(c101-c001)*u, x11=c011+(c111-c011)*u;
  const y0=x00+(x10-x00)*v, y1=x01+(x11-x01)*v;
  return y0+(y1-y0)*w;
}
