

**Причина:** игрок спавнится и начинает падать ещё до того, как данные чанка под ним сгенерированы (генерация идёт по 4 чанка/кадр начиная с дальнего угла), поэтому он проваливается сквозь ещё несуществующую землю внутрь твёрдых блоков, где все грани скрыты (между двумя solid-блоками грань не строится) — камера оказывается внутри «пустоты» без единого полигона, и виден только синий фон с облаками.

Исправление: синхронно сгенерировать чанк игрока и его соседей до первого кадра физики.

  const BLOCK_COLORS = [

  ];

    { n: [-1, 0, 0], c: [[0,0,1],[0,1,1],[0,1,0],[0,0,0]], b: 0.8 }, // -X

    { n: [ 0,-1, 0], c: [[0,0,0],[1,0,0],[1,0,1],[0,0,1]], b: 0.55}, // -Y
    { n: [ 0, 0, 1], c: [[0,0,1],[1,0,1],[1,1,1],[0,1,1]], b: 0.8 }, // +Z
    { n: [ 0, 0,-1], c: [[1,0,0],[0,0,0],[0,1,0],[1,1,0]], b: 0.8 }  // -Z

  //  Deterministic value noise (no Math.random anywhere)

  function hashInt(x, y) {
    let h = (x | 0) ^ ((y | 0) * 374761393);
    h = (h ^ (h >>> 13)) * 1274126177;
    h = h ^ (h >>> 16);
    return ((h >>> 0) % 100000) / 100000;
  }
  function hashInt3(x, y, z) {
    let h = (x | 0) ^ ((y | 0) * 374761393) ^ ((z | 0) * 668265263);
    h = (h ^ (h >>> 13)) * 1274126177;
    h = h ^ (h >>> 16);
    return ((h >>> 0) % 100000) / 100000;
  }
  function smooth(t) { return t * t * (3 - 2 * t); }

  function valueNoise2D(x, y) {
    const xi = Math.floor(x), yi = Math.floor(y);
    const xf = x - xi, yf = y - yi;
    const v00 = hashInt(xi, yi),     v10 = hashInt(xi + 1, yi);
    const v01 = hashInt(xi, yi + 1), v11 = hashInt(xi + 1, yi + 1);
    const u = smooth(xf), v = smooth(yf);
    const a = v00 + (v10 - v00) * u;
    const b = v01 + (v11 - v01) * u;
    return a + (b - a) * v;
  }
  function valueNoise3D(x, y, z) {
    const xi = Math.floor(x), yi = Math.floor(y), zi = Math.floor(z);
    const xf = x - xi, yf = y - yi, zf = z - zi;
    const c000 = hashInt3(xi, yi, zi),     c100 = hashInt3(xi + 1, yi, zi);
    const c010 = hashInt3(xi, yi + 1, zi), c110 = hashInt3(xi + 1, yi + 1, zi);
    const c001 = hashInt3(xi, yi, zi + 1), c101 = hashInt3(xi + 1, yi, zi + 1);
    const c011 = hashInt3(xi, yi + 1, zi + 1), c111 = hashInt3(xi + 1, yi + 1, zi + 1);
    const u = smooth(xf), v = smooth(yf), w = smooth(zf);
    const x00 = c000 + (c100 - c000) * u;
    const x10 = c010 + (c110 - c010) * u;
    const x01 = c001 + (c101 - c001) * u;
    const x11 = c011 + (c111 - c011) * u;
    const y0 = x00 + (x10 - x00) * v;
    const y1 = x01 + (x11 - x01) * v;
    return y0 + (y1 - y0) * w;
  }
  function fractal2D(x, y) {
    let sum = 0, amp = 0.5, freq = 1;
    for (let i = 0; i < 4; i++) {
      sum += valueNoise2D(x * freq, y * freq) * amp;
      freq *= 2; amp *= 0.5;
    }
    return sum;
  }
  function fractal3D(x, y, z) {
    let sum = 0, amp = 0.5, freq = 1;
    for (let i = 0; i < 4; i++) {
      sum += valueNoise3D(x * freq, y * freq, z * freq) * amp;
      freq *= 2; amp *= 0.5;
    }
    return sum;
  }

  // ------------------------------------------------------------------
  //  Global chunk storage + read / write helpers (negative coords safe)
  // ------------------------------------------------------------------
  const chunks = new Map();   // key "cx,cz" -> { data: Uint8Array, mesh: Mesh|null }

  function getChunkData(cx, cz) {
    let c = chunks.get(cx + ',' + cz);
    if (!c) { c = { data: new Uint8Array(CHUNK_SIZE * CHUNK_SIZE * CHUNK_HEIGHT), mesh: null }; chunks.set(cx + ',' + cz, c); }
    return c.data;
  }
  function idx(lx, ly, lz) { return (lz * CHUNK_SIZE + lx) * CHUNK_HEIGHT + ly; }

  function getBlock(wx, wy, wz) {
    if (wy < 0 || wy >= CHUNK_HEIGHT) return AIR;
    const cx = Math.floor(wx / CHUNK_SIZE), cz = Math.floor(wz / CHUNK_SIZE);
    const lx = wx - cx * CHUNK_SIZE, lz = wz - cz * CHUNK_SIZE;
    const c = chunks.get(cx + ',' + cz);
    if (!c) return AIR;
    return c.data[idx(lx, wy, lz)];
  }
  function setBlock(wx, wy, wz, id) {
    if (wy < 0 || wy >= CHUNK_HEIGHT) return;
    const cx = Math.floor(wx / CHUNK_SIZE), cz = Math.floor(wz / CHUNK_SIZE);
    const lx = wx - cx * CHUNK_SIZE, lz = wz - cz * CHUNK_SIZE;
    const c = chunks.get(cx + ',' + cz);
    if (!c) return;
    c.data[idx(lx, wy, lz)] = id;
  }

  // ------------------------------------------------------------------
  //  Terrain generation
  // ------------------------------------------------------------------
  function columnHeight(wx, wz) {
    const m = fractal2D(wx * 0.004, wz * 0.004);
    const h = fractal2D(wx * 0.02, wz * 0.02);
    return Math.floor(5 + m * m * 58 + h * 10);
  }

  function generateChunk(cx, cz) {
    const data = getChunkData(cx, cz);
    const baseX = cx * CHUNK_SIZE, baseZ = cz * CHUNK_SIZE;

    for (let lz = 0; lz < CHUNK_SIZE; lz++) {
      for (let lx = 0; lx < CHUNK_SIZE; lx++) {
        const wx = baseX + lx, wz = baseZ + lz;
        const H = columnHeight(wx, wz);
        let surfaceType;
        if (H >= 46) surfaceType = SNOW;
        else if (H >= 37) surfaceType = STONE;
        else if (H <= 16) surfaceType = SAND;
        else surfaceType = GRASS;
        let fillType;
        if (H <= 16) fillType = SAND;
        else if (H >= 37) fillType = STONE;
        else fillType = DIRT;

        for (let y = 0; y < CHUNK_HEIGHT; y++) {
          let id;
          if (y === 0) id = STONE;
          else if (y < H - 3) id = STONE;
          else if (y < H) id = fillType;
          else if (y === H) id = surfaceType;
          else id = AIR;
          data[idx(lx, y, lz)] = id;
        }

        // Caves
        for (let y = 3; y < H - 2; y++) {
          if (fractal3D(wx * 0.09, y * 0.09, wz * 0.09) > 0.67) {
            data[idx(lx, y, lz)] = AIR;
          }
        }
      }
    }

    // Trees
    for (let lz = 0; lz < CHUNK_SIZE; lz++) {
      for (let lx = 0; lx < CHUNK_SIZE; lx++) {
        const wx = baseX + lx, wz = baseZ + lz;
        const H = columnHeight(wx, wz);
        if (data[idx(lx, H, lz)] !== GRASS) continue;
        if (hashInt(wx * 73856093, wz * 19349663) >= 0.02) continue;
        if (lx < 1 || lx > 14 || lz < 1 || lz > 14) continue;
        const top = H + 4;
        if (top >= CHUNK_HEIGHT) continue;

        for (let t = 1; t <= 4; t++) {
          const ty = H + t;
          if (ty < CHUNK_HEIGHT && data[idx(lx, ty, lz)] === AIR) data[idx(lx, ty, lz)] = WOOD;
        }
        for (let dy = 0; dy < 2; dy++) {
          const ty = H + 3 + dy;
          for (let dx = -2; dx <= 2; dx++) {
            for (let dz = -2; dz <= 2; dz++) {
              const nx = lx + dx, nz = lz + dz;
              if (nx < 0 || nx >= CHUNK_SIZE || nz < 0 || nz >= CHUNK_SIZE) continue;
              if (ty < CHUNK_HEIGHT && data[idx(nx, ty, nz)] === AIR) data[idx(nx, ty, nz)] = LEAVES;
            }
          }
        }
        {
          const ty = H + 5;
          for (let dx = -1; dx <= 1; dx++) {
            for (let dz = -1; dz <= 1; dz++) {
              const nx = lx + dx, nz = lz + dz;
              if (nx < 0 || nx >= CHUNK_SIZE || nz < 0 || nz >= CHUNK_SIZE) continue;
              if (ty < CHUNK_HEIGHT && data[idx(nx, ty, nz)] === AIR) data[idx(nx, ty, nz)] = LEAVES;
            }
          }
        }
        {
          const ty = H + 6;
          if (ty < CHUNK_HEIGHT && data[idx(lx, ty, lz)] === AIR) data[idx(lx, ty, lz)] = LEAVES;
        }
      }
    }
  }

  // ------------------------------------------------------------------
  //  Meshing (one BufferGeometry per chunk, hidden faces culled)
  // ------------------------------------------------------------------
  function buildChunkGeometry(cx, cz) {
    const data = getChunkData(cx, cz);
    const positions = [], normals = [], colors = [];
    const baseX = cx * CHUNK_SIZE, baseZ = cz * CHUNK_SIZE;
    const tmp = new THREE.Color();

    for (let lz = 0; lz < CHUNK_SIZE; lz++) {
      for (let lx = 0; lx < CHUNK_SIZE; lx++) {
        for (let y = 0; y < CHUNK_HEIGHT; y++) {
          const id = data[idx(lx, y, lz)];
          if (id === AIR) continue;
          const wx = baseX + lx, wz = baseZ + lz;
          tmp.setHex(BLOCK_COLORS[id]);
          for (let f = 0; f < 6; f++) {
            const face = FACES[f];
            if (getBlock(wx + face.n[0], y + face.n[1], wz + face.n[2]) !== AIR) continue;
            const n = face.n, b = face.b;
            const cr = tmp.r * b, cg = tmp.g * b, cb = tmp.b * b;
            const corners = face.c;
            const v0 = [wx + corners[0][0], y + corners[0][1], wz + corners[0][2]];
            const v1 = [wx + corners[1][0], y + corners[1][1], wz + corners[1][2]];
            const v2 = [wx + corners[2][0], y + corners[2][1], wz + corners[2][2]];
            const v3 = [wx + corners[3][0], y + corners[3][1], wz + corners[3][2]];
            positions.push(v0[0],v0[1],v0[2], v1[0],v1[1],v1[2], v2[0],v2[1],v2[2]);
            positions.push(v0[0],v0[1],v0[2], v2[0],v2[1],v2[2], v3[0],v3[1],v3[2]);
            for (let k = 0; k < 6; k++) { normals.push(n[0], n[1], n[2]); }
            for (let k = 0; k < 6; k++) { colors.push(cr, cg, cb); }
          }
        }
      }
    }

    const geo = new THREE.BufferGeometry();
    geo.setAttribute('position', new THREE.Float32BufferAttribute(positions, 3));
    geo.setAttribute('normal', new THREE.Float32BufferAttribute(normals, 3));
    geo.setAttribute('color', new THREE.Float32BufferAttribute(colors, 3));
    return geo;
  }

  const sharedMaterial = new THREE.MeshLambertMaterial({ vertexColors: true });
  const chunkMeshes = [];

  function rebuildChunk(cx, cz) {
    const c = chunks.get(cx + ',' + cz);
    if (!c) return;
    if (c.mesh) {
      scene.remove(c.mesh);
      c.mesh.geometry.dispose();
      const mi = chunkMeshes.indexOf(c.mesh);
      if (mi >= 0) chunkMeshes.splice(mi, 1);
      c.mesh = null;
    }
    const geo = buildChunkGeometry(cx, cz);
    if (geo.attributes.position.count > 0) {
      const mesh = new THREE.Mesh(geo, sharedMaterial);
      mesh.userData.cx = cx;
      mesh.userData.cz = cz;
      scene.add(mesh);
      chunkMeshes.push(mesh);
      c.mesh = mesh;
    }
  }

    const lim = Math.PI / 2 - 0.05;
    if (pitch > lim) pitch = lim;
    if (pitch < -lim) pitch = -lim;
  });

  document.addEventListener('keydown', function (e) {
    keys[e.code] = true;
    if (e.code >= 'Digit1' && e.code <= 'Digit7') {
      selectSlot(parseInt(e.code.slice(5), 10) - 1);
    }
  });
  document.addEventListener('keyup', function (e) { keys[e.code] = false; });

  document.addEventListener('wheel', function (e) {
    if (document.pointerLockElement !== document.body) return;
    selectSlot(selected + (e.deltaY > 0 ? 1 : -1));
  });

  document.addEventListener('contextmenu', function (e) { e.preventDefault(); });

  document.addEventListener('mousedown', function (e) {
    if (document.pointerLockElement !== document.body) return;
    if (e.button === 0) doBreak();
    else if (e.button === 2) doPlace();
  });

  window.addEventListener('resize', function () {
    camera.aspect = window.innerWidth / window.innerHeight;
    camera.updateProjectionMatrix();
    renderer.setSize(window.innerWidth, window.innerHeight);
  });

  // ------------------------------------------------------------------
  //  Hotbar UI
  // ------------------------------------------------------------------
  const hotbarEl = document.getElementById('hotbar');
  const slotEls = [];
  HOTBAR_BLOCKS.forEach(function (id, i) {
    const slot = document.createElement('div');
    slot.className = 'slot';
    const sw = document.createElement('div');
    sw.className = 'swatch';
    sw.style.background = '#' + BLOCK_COLORS[id].toString(16).padStart(6, '0');
    const num = document.createElement('div');
    num.className = 'num';
    num.textContent = String(i + 1);
    slot.appendChild(sw);
    slot.appendChild(num);
    hotbarEl.appendChild(slot);
    slotEls.push(slot);
  });
  let selected = 0;
  function selectSlot(i) {
    selected = ((i % 7) + 7) % 7;
    slotEls.forEach(function (el, k) { el.classList.toggle('selected', k === selected); });
  }
  selectSlot(0);

  // ------------------------------------------------------------------
  //  Raycast targeting + break / place
  // ------------------------------------------------------------------
  const raycaster = new THREE.Raycaster();
  raycaster.far = REACH;
  const centerVec = new THREE.Vector2(0, 0);
  let targetCell = null;

  function updateTarget() {
    targetCell = null;
    outline.visible = false;
    if (document.pointerLockElement !== document.body) return;
    raycaster.setFromCamera(centerVec, camera);
    const hits = raycaster.intersectObjects(chunkMeshes, false);
    if (!hits.length) return;
    const hit = hits[0];
    const n = hit.face.normal;
    const p = hit.point;
    const bx = Math.floor(p.x - n.x * 0.5);
    const by = Math.floor(p.y - n.y * 0.5);
    const bz = Math.floor(p.z - n.z * 0.5);
    targetCell = { x: bx, y: by, z: bz, px: Math.floor(p.x + n.x * 0.5), py: Math.floor(p.y + n.y * 0.5), pz: Math.floor(p.z + n.z * 0.5) };
    outline.visible = true;
    outline.position.set(bx + 0.5, by + 0.5, bz + 0.5);
  }

  function rebuildAt(wx, wy, wz) {
    const cx = Math.floor(wx / CHUNK_SIZE), cz = Math.floor(wz / CHUNK_SIZE);
    rebuildChunk(cx, cz);
    const lx = wx - cx * CHUNK_SIZE, lz = wz - cz * CHUNK_SIZE;
    if (lx === 0) rebuildChunk(cx - 1, cz);
    if (lx === CHUNK_SIZE - 1) rebuildChunk(cx + 1, cz);
    if (lz === 0) rebuildChunk(cx, cz - 1);
    if (lz === CHUNK_SIZE - 1) rebuildChunk(cx, cz + 1);
  }

  function doBreak() {
    if (!targetCell) return;
    const t = targetCell;
    if (t.y === 0) return;
    if (getBlock(t.x, t.y, t.z) === AIR) return;
    setBlock(t.x, t.y, t.z, AIR);
    rebuildAt(t.x, t.y, t.z);
  }

  function doPlace() {
    if (!targetCell) return;
    const t = targetCell;
    if (getBlock(t.px, t.py, t.pz) !== AIR) return;
    const minX = player.x - PLAYER_HALF, maxX = player.x + PLAYER_HALF;
    const minY = player.y, maxY = player.y + PLAYER_HEIGHT;
    const minZ = player.z - PLAYER_HALF, maxZ = player.z + PLAYER_HALF;
    const bx0 = t.px, bx1 = t.px + 1, by0 = t.py, by1 = t.py + 1, bz0 = t.pz, bz1 = t.pz + 1;
    if (bx1 > minX && bx0 < maxX && by1 > minY && by0 < maxY && bz1 > minZ && bz0 < maxZ) return;
    setBlock(t.px, t.py, t.pz, HOTBAR_BLOCKS[selected]);
    rebuildAt(t.px, t.py, t.pz);
  }

  // ------------------------------------------------------------------
  //  Collision
  // ------------------------------------------------------------------
  function collides(px, py, pz) {
    const minX = px - PLAYER_HALF, maxX = px + PLAYER_HALF;
    const minY = py, maxY = py + PLAYER_HEIGHT;
    const minZ = pz - PLAYER_HALF, maxZ = pz + PLAYER_HALF;
    const x0 = Math.floor(minX), x1 = Math.floor(maxX - 1e-6);
    const y0 = Math.floor(minY), y1 = Math.floor(maxY - 1e-6);
    const z0 = Math.floor(minZ), z1 = Math.floor(maxZ - 1e-6);
    for (let x = x0; x <= x1; x++)
      for (let y = y0; y <= y1; y++)
        for (let z = z0; z <= z1; z++)
          if (getBlock(x, y, z) !== AIR) return true;
    return false;
  }

  // ------------------------------------------------------------------
  //  World streaming
  // ------------------------------------------------------------------
  function updateWorld() {
    const pcx = Math.floor(player.x / CHUNK_SIZE);
    const pcz = Math.floor(player.z / CHUNK_SIZE);

    // Generate data within 5 chunks (max 4 / frame)
    let genCount = 0;
    for (let dz = -5; dz <= 5 && genCount < 4; dz++) {
      for (let dx = -5; dx <= 5 && genCount < 4; dx++) {
        const cx = pcx + dx, cz = pcz + dz;
        const key = cx + ',' + cz;
        if (!chunks.has(key)) {
          generateChunk(cx, cz);
          genCount++;
        }
      }
    }

    // Build meshes within 4 chunks once 4 neighbours have data (max 2 / frame)
    let meshCount = 0;
    for (let dz = -4; dz <= 4 && meshCount < 2; dz++) {
      for (let dx = -4; dx <= 4 && meshCount < 2; dx++) {
        const cx = pcx + dx, cz = pcz + dz;
        const c = chunks.get(cx + ',' + cz);
        if (!c) continue;
        if (c.mesh) continue;
        if (!chunks.has((cx - 1) + ',' + cz)) continue;
        if (!chunks.has((cx + 1) + ',' + cz)) continue;
        if (!chunks.has(cx + ',' + (cz - 1))) continue;
        if (!chunks.has(cx + ',' + (cz + 1))) continue;
        rebuildChunk(cx, cz);
        meshCount++;
      }
    }

    // Dispose anything farther than 7 chunks
    for (const key of Array.from(chunks.keys())) {
      const parts = key.split(',');
      const cx = parseInt(parts[0], 10), cz = parseInt(parts[1], 10);
      if (Math.abs(cx - pcx) > 7 || Math.abs(cz - pcz) > 7) {
        const c = chunks.get(key);
        if (c.mesh) {
          scene.remove(c.mesh);
          c.mesh.geometry.dispose();
          const i = chunkMeshes.indexOf(c.mesh);
          if (i >= 0) chunkMeshes.splice(i, 1);
          c.mesh = null;
        }
        chunks.delete(key);
      }
    }
  }

  // ------------------------------------------------------------------
  //  Player physics
  // ------------------------------------------------------------------
  function updatePlayer(dt) {
    if (keys['KeyW']) { player.vx += Math.sin(yaw) * WALK_SPEED * dt; player.vz += -Math.cos(yaw) * WALK_SPEED * dt; }
    if (keys['KeyS']) { player.vx += -Math.sin(yaw) * WALK_SPEED * dt; player.vz += Math.cos(yaw) * WALK_SPEED * dt; }
    if (keys['KeyA']) { player.vx += -Math.cos(yaw) * WALK_SPEED * dt; player.vz += -Math.sin(yaw) * WALK_SPEED * dt; }
    if (keys['KeyD']) { player.vx += Math.cos(yaw) * WALK_SPEED * dt; player.vz += Math.sin(yaw) * WALK_SPEED * dt; }

    const hs = Math.hypot(player.vx, player.vz);
    if (hs > WALK_SPEED) { const k = WALK_SPEED / hs; player.vx *= k; player.vz *= k; }

    if (keys['Space'] && player.onGround) { player.vy = JUMP_VEL; player.onGround = false; }

    player.vy -= GRAVITY * dt;

    let nx = player.x + player.vx * dt;
    if (collides(nx, player.y, player.z)) { nx = player.x; player.vx = 0; }
    player.x = nx;

    let nz = player.z + player.vz * dt;
    if (collides(player.x, player.y, nz)) { nz = player.z; player.vz = 0; }
    player.z = nz;

    let ny = player.y + player.vy * dt;
    if (collides(player.x, ny, player.z)) {
      if (player.vy < 0) player.onGround = true;
      ny = player.y; player.vy = 0;
    } else {
      player.onGround = false;
    }
    player.y = ny;

    if (player.y < -20) spawn();
  }

  // ------------------------------------------------------------------
  //  Per-frame visuals
  // ------------------------------------------------------------------
  function updateVisuals(dt) {
    camera.position.set(player.x, player.y + EYE_HEIGHT, player.z);
    camera.rotation.set(pitch, yaw, 0);

    water.position.set(player.x, WATER_LEVEL, player.z);

    for (let i = 0; i < clouds.length; i++) {
      const c = clouds[i];
      c.position.x += c.userData.speed * dt;
      if (c.position.x - player.x > 160) c.position.x -= 320;
      if (c.position.x - player.x < -160) c.position.x += 320;
    }
  }

  // ------------------------------------------------------------------
  //  Main loop
  // ------------------------------------------------------------------
  let lastTime = performance.now();
  function animate() {
    requestAnimationFrame(animate);
    const now = performance.now();
    let dt = (now - lastTime) / 1000;
    lastTime = now;
    if (dt > 0.05) dt = 0.05;

    updateWorld();
    updatePlayer(dt);
    updateTarget();
    updateVisuals(dt);

    renderer.render(scene, camera);
  }

  // ------------------------------------------------------------------
  //  Boot — pre-generate player's chunk neighbourhood BEFORE first frame
  // ------------------------------------------------------------------
  spawn();
  updateWorld();
  animate();

})();
</script>
