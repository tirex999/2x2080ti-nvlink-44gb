Value noise:
```js
function hash2(x, z){
  let n = x*374761393 + z*668265263;
  n = (n ^ (n >> 13)) * 1274126177;
  n = n ^ (n >> 16);
  // map to 0..1
  return ((n >>> 0) % 100000) / 100000;
}
```
Note: JS bitwise ops are 32-bit signed; n can overflow to float. Use Math.imul for multiplication to keep int32.

```js
function ihash(x, z){
  let n = Math.imul(x, 374761393) + Math.imul(z, 668265263);
  n = Math.imul(n ^ (n >>> 13), 1274126177);
  n ^= n >>> 16;
  return (n >>> 0) / 4294967296;
}
```

Smoothstep t*t*(3-t). Noise2:
```js
function noise2(x, z){
  const xi = Math.floor(x), zi = Math.floor(z);
  const xf = x - xi, zf = z - zi;
  const u = xf*xf*(3-xf), v = zf*zf*(3-zf);
  const a = ihash(xi, zi), b = ihash(xi+1, zi), c = ihash(xi, zi+1), d = ihash(xi+1, zi+1);
  return a + (b-a)*u + (c-a)*v + (a-b-c+d)*u*v;
}
```

```js
function fbm2(x, z){
```

```js
function noise3(x,y,z){
  const xi=Math.floor(x), yi=Math.floor(y), zi=Math.floor(z);
  const xf=x-xi, yf=y-yi, zf=z-zi;
  const u=xf*xf*(3-xf), v=yf*yf*(3-yf), w=zf*zf*(3-zf);
  // trilinear
  const c000=ihash3(xi,yi,zi), c100=ihash3(xi+1,yi,zi), c010=ihash3(xi,yi+1,zi), c110=ihash3(xi+1,yi+1,zi);
  const c001=ihash3(xi,yi,zi+1), c101=ihash3(xi+1,yi,zi+1), c011=ihash3(xi,yi+1,zi+1), c111=ihash3(xi+1,yi+1,zi+1);
  const x00=c000+(c100-c000)*u, x10=c010+(c110-c010)*u;
  const x01=c001+(c101-c001)*u, x11=c011+(c111-c011)*u;
  const y0=x00+(x10-x00)*v, y1=x01+(x11-x01)*v;
  return y0+(y1-y0)*w;
}
function fbm3(x,y,z){

  for(let i=0;i<3;i++){ s+=amp*noise3(x*f,y*f,z*f); f*=2;amp*=0.5;}
```

```js
function genChunk(cx, cz){
  const data = new Uint8Array(CH*CH*CT);
  for(let lx=0;lx<16;lx++) for(let lz=0;lz<16;lz++){
    const wx=cx*16+lx, wz=cz*16+lz;
    const m=fbm2(wx*0.004, wz*0.004), h=fbm2(wx*0.02, wz*0.02);
    let H = Math.floor(5 + m*m*58 + h*10);
    H = Math.max(1, Math.min(79, H));
    for(let y=0;y<CT;y++){
      let b;
      if(y===0) b=3;
      else if(y<H-3) b=3;
      else if(y<H) b = H<=16?4 : (H>=37?3:2);
      else if(y===H) b = H>=46?7 : H>=37?3 : H<=16?4 : 1;

      data[(lx*16+lz)*CT+y]=b;
```

```js
  if(c.mesh){ scene.remove(c.mesh); c.mesh.geometry.dispose(); }
  const pos=[], norm=[], col=[];
  for(let lx=0;lx<16;lx++)for(let lz=0;lz<16;lz++)for(let y=0;y<CT;y++){
    const id=c.data[(lx*16+lz)*CT+y];
```
