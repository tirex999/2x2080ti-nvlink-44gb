function noise2D(x, z) {
    const ix = Math.floor(x), iz = Math.floor(z);
    const fx = x - ix, fz = z - iz;
    const sx = fx * fx * (3 - 2 * fx);
    const sz = fz * fz * (3 - 2 * fz);
    const a = hash2D(ix, iz);
    const b = hash2D(ix + 1, iz);
    const c = hash2D(ix, iz + 1);
    const d = hash2D(ix + 1, iz + 1);
    return a + (b - a) * sx + (c - a) * sz + (a - b - c + d) * sx * sz;
}

    const ix = Math.floor(x), iy = Math.floor(y), iz = Math.floor(z);
    const fx = x - ix, fy = y - iy, fz = z - iz;
    const sx = fx * fx * (3 - 2 * fx);

function generateChunkData(cx, cz) {
    const key = cx + ',' + cz;
    if (chunks.has(key)) return;
    
    const data = new Uint8Array(16 * 80 * 16);
    const baseX = cx * 16;
    const baseZ = cz * 16;
    
    for (let lz = 0; lz < 16; lz++) {
        for (let lx = 0; lx < 16; lx++) {
            const wx = baseX + lx;
            const wz = baseZ + lz;
            
            // Height
            const m = fractal2D(wx * 0.004, wz * 0.004);
            const h = fractal2D(wx * 0.02, wz * 0.02);
            const H = Math.floor(5 + m * m * 58 + h * 10);
            
            // Fill column
            for (let y = 0; y <= H && y < 80; y++) {
                let id;
                if (y === 0) {
                    id = 3; // stone (unbreakable)
                } else if (y < H - 3) {
                    id = 3; // stone
                } else if (y < H) {
                    // 3 layers under surface
                    if (H <= 16) id = 4; // sand
                    else if (H >= 37) id = 3; // stone
                    else id = 2; // dirt
                } else {
                    // Surface
                    if (H >= 46) id = 7; // snow
                    else if (H >= 37) id = 3; // stone
                    else if (H <= 16) id = 4; // sand
                    else id = 1; // grass
                }
                data[(y * 16 + lz) * 16 + lx] = id;
            }
            
            // Caves
            for (let y = 3; y < H - 1 && y < 80; y++) {

                    const n = noise3D(wx * 0.09, y * 0.09, wz * 0.09);
                    if (n > 0.67) {
                        data[(y * 16 + lz) * 16 + lx] = 0;
                    }
                }
            }
        }
    }
    
    // Trees
    for (let lz = 2; lz < 14; lz++) {
        for (let lx = 2; lx < 14; lx++) {

                        const tz = lz + dz;
