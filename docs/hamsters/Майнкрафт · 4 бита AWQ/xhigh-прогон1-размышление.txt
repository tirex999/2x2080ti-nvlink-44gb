function generateChunk(cx, cz) {
  const key = cx + ',' + cz;
  if (chunks.has(key)) return;
  const data = new Uint8Array(CHUNK_SIZE*CHUNK_SIZE*HEIGHT);
  const ox = cx*CHUNK_SIZE, oz = cz*CHUNK_SIZE;
  for (let lx=0; lx<CHUNK_SIZE; lx++) {
    for (let lz=0; lz<CHUNK_SIZE; lz++) {
      const wx = ox+lx, wz = oz+lz;
      const H = terrainHeight(wx,wz);
      const under = H <= 16 ? 4 : (H >= 37 ? 3 : 2);
      const surf = H >= 46 ? 7 : (H >= 37 ? 3 : (H <= 16 ? 4 : 1));
      for (let y=0; y<=H; y++) {
        let id;
        if (y === 0) id = 3;
        else if (y < H - 3) id = 3;
        else if (y < H) id = under;
        else id = surf;
        data[(y*CHUNK_SIZE+lz)*CHUNK_SIZE+lx] = id;
      }
      for (let y=3; y<=H-2; y++) {
        if (noise3(wx*0.09, y*0.09, wz*0.09) > 0.67) {
          data[(y*CHUNK_SIZE+lz)*CHUNK_SIZE+lx] = 0;
        }
      }
      if (surf === 1 && hash2i(wx, wz) < 0.02 && lx >= 2 && lx <= 13 && lz >= 2 && lz <= 13 && H + 6 < HEIGHT) {
        // trunk
        for (let ty=H+1; ty<=H+4; ty++) {
          data[(ty*16+lz)*16+lx] = 5;
        }
        // leaves 5x5 twice
        for (let ly=H+3; ly<=H+4; ly++) {
          for (let dx=-2; dx<=2; dx++) {
            for (let dz=-2; dz<=2; dz++) {
              const ix=lx+dx, iz=lz+dz;
              if (ix<0||ix>15||iz<0||iz>15||ly<0||ly>=HEIGHT) continue;
              const idx=(ly*16+iz)*16+ix;
              if (data[idx] === 0) data[idx] = 6;
            }
          }
        }
        // 3x3
        for (let dx=-1; dx<=1; dx++) for (let dz=-1; dz<=1; dz++) {
          const ix=lx+dx, iz=lz+dz;
          if (ix<0||ix>15||iz<0||iz>15) continue;
          const idx=((H+5)*16+iz)*16+ix;
          if (data[idx] === 0) data[idx] = 6;
        }
        // top
        {
          const idx=((H+6)*16+lz)*16+lx;
          if (data[idx] === 0) data[idx] = 6;
        }
      }
    }
  }
  chunks.set(key, {data: data, mesh: null, cx: cx, cz: cz});
}

function buildChunkMesh(cx,cz){

    for (let lz=0; lz<CHUNK_SIZE; lz++) {
      for (let lx=0; lx<CHUNK_SIZE; lx++) {
        const id = c.data[(y*CHUNK_SIZE+lz)*CHUNK_SIZE+lx];
        if (id === 0) continue;
        const col = BLOCK_COLORS[id];
        const wx = ox + lx, wz = oz + lz;
        for (let f=0; f<FACES.length; f++) {
          const face = FACES[f];
          if (getBlock(wx + face.dir[0], y + face.dir[1], wz + face.dir[2]) !== 0) continue;
          const shade = face.shade;
          const r = col[0]*shade, g=col[1]*shade, b=col[2]*shade;
          const c0=face.corners[0], c1=face.corners[1], c2=face.corners[2], c3=face.corners[3];
          // add triangle 0,1,2
          addVertex(positions,normals,colors, wx+c0[0], y+c0[1], wz+c0[2], face.dir, r,g,b);
          addVertex(... c1)
          addVertex(... c2)
          addVertex(... c0)
          addVertex(... c2)
          addVertex(... c3)
        }
      }
    }
  }
  if (positions.length > 0) {
    const geo = new THREE.BufferGeometry();
    geo.setAttribute('position', new THREE.Float32BufferAttribute(positions,3));
    geo.setAttribute('normal', new THREE.Float32BufferAttribute(normals,3));
    geo.setAttribute('color', new THREE.Float32BufferAttribute(colors,3));
    geo.computeBoundingSphere();
