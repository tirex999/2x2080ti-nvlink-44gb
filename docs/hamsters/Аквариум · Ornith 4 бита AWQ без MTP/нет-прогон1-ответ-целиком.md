function init() {
  clock = new THREE.Clock();

  // Сцена
  scene = new THREE.Scene();
  scene.background = new THREE.Color(0x0a2a55);
  scene.fog = new THREE.FogExp2(0x0a3a66, 0.012);

  // Камера
  camera = new THREE.PerspectiveCamera(55, window.innerWidth / window.innerHeight, 0.1, 500);
  camera.position.set(0, 8, 48);

  // Рендерер
  renderer = new THREE.WebGLRenderer({ antialias: true });
  renderer.setSize(window.innerWidth, window.innerHeight);
  renderer.setPixelRatio(Math.min(window.devicePixelRatio, 2));
  renderer.shadowMap.enabled = true;
  renderer.shadowMap.type = THREE.PCFSoftShadowMap;
  renderer.outputFormat = THREE.RGBAFormat;
  document.body.appendChild(renderer.domElement);

  // Управление камерой
  controls = new THREE.OrbitControls(camera, renderer.domElement);
  controls.enableDamping = true;
  controls.dampingFactor = 0.06;
  controls.rotateOnHorizontal = false;
  controls.mouse = { left: 'rotate', right: 'pan', wheel: 'zoom' };

function buildTank() {
  const w = container.x, h = container.y, dep = container.z;
  const geo = new THREE.BoxGeometry(w, h, dep);
  const glassMat = new THREE.MeshPhysicalMaterial({
    color: 0xaaddff, transparent: true, opacity: 0.12,
    roughness: 0.1, metalness: 0, transmission: 0.95,
    side: THREE.DoubleSide, depthWrite: false,
    clearcoat: 1.0, clearcoatRoughness: 0.1
  });
  const tank = new THREE.Mesh(geo, glassMat);
  tank.renderOrder = -1;
  scene.add(tank);

  // Рамка (edges)
  const edges = new THREE.EdgesGeometry(geo);
  const lineMat = new THREE.LineBasicMaterial({ color: 0x7fd8ff, transparent: true, opacity: 0.55 });
  const wire = new THREE.LineSegments(edges, lineMat);
  scene.add(wire);
}

// ---------- ПЕСЧАНОЕ ДНО ----------
function buildSand() {
  const seg = 64;
  const geo = new THREE.PlaneGeometry(container.x * 2, container.z * 2, seg, seg);
  const pos = geo.attributes.position.array;
  for (let i = 0; i < pos.length; i += 3) {
    pos[i] += (Math.random() - 0.5) * 1.2;
    pos[i + 1] += (Math.random() - 0.5) * 1.2;
  }
  geo.computeVertexNormals();
  const mat = new THREE.MeshStandardMaterial({ color: 0xd9c080, roughness: 1, flatShading: true });
  sandMesh = new THREE.Mesh(geo, mat);
  sandMesh.rotation.x = -Math.PI / 2;
  sandMesh.position.y = -container.y / 2 + 0.15;
  sandMesh.receiveShadow = true;
  scene.add(sandMesh);
  groundY = sandMesh.position.y;
}

// ---------- КАМНИ ----------
function buildRocks() {
  const rockMat = new THREE.MeshStandardMaterial({ color: 0x7a7a7a, roughness: 0.95, flatShading: true });
  for (let i = 0; i < 8; i++) {
    const detail = 0;
    const base = new THREE.DodecahedronGeometry(1, detail);
    const p = base.attributes.position.array;
    for (let j = 0; j < p.length; j += 3) {
      const s = 0.75 + Math.random() * 0.5;
      p[j] *= s; p[j + 1] *= s * 0.7; p[j + 2] *= s;
    }
    base.computeVertexNormals();
    const rock = new THREE.Mesh(base, rockMat.clone());
    rock.material.color.setHSL(0, 0, 0.28 + Math.random() * 0.12);
    const sc = 1.2 + Math.random() * 2.2;
    rock.scale.set(sc, sc * 0.6, sc);
    rock.position.set((Math.random() - 0.5) * (container.x - 6), groundY + 0.3, (Math.random() - 0.5) * (container.z - 6));
    rock.rotation.set(Math.random() * Math.PI, Math.random() * Math.PI, Math.random() * Math.PI);
    rock.castShadow = true;
    rock.receiveShadow = true;
    scene.add(rock);
  }
}

function buildSeaweed() {
  const positions = [];
  for (let i = 0; i < 12; i++) {
    positions.push(new THREE.Vector3((Math.random() - 0.5) * (container.x - 8), groundY, (Math.random() - 0.5) * (container.z - 8)));
  }
  positions.forEach((pos, idx) => {
    const pts = [];
    const stalks = 2 + Math.floor(Math.random() * 3);
    for (let s = 0; s < stalks; s++) {
      const ox = (Math.random() - 0.5) * 0.6;
      const oz = (Math.random() - 0.5) * 0.6;
      const n = 6;
      const heights = [];
      for (let k = 0; k <= n; k++) heights.push((k / n) * (3 + Math.random() * 3));
      for (let k = 0; k < n; k++) {
        const y0 = heights[k], y1 = heights[k + 1];
        const sway = Math.sin(k * 0.8 + idx) * 0.4;
        pts.push(new THREE.Vector3(ox + sway * (k / n), y0, oz + Math.cos(k) * 0.2 * (k / n)));
        pts.push(new THREE.Vector3(ox + sway * ((k + 1) / n), y1, oz + Math.cos(k + 1) * 0.2 * ((k + 1) / n)));
      }
    }
    const curve = new THREE.CatmullRomCurve3(pts);
    const tubeGeo = new THREE.TubeGeometry(curve, 8, 0.12 + Math.random() * 0.12, 5, false);
    const hue = 0.28 + Math.random() * 0.12;
    const tubeMat = new THREE.MeshStandardMaterial({ color: new THREE.Color().setHSL(hue, 0.6, 0.35), roughness: 0.9, emissive: new THREE.Color().setHSL(hue, 0.5, 0.12), side: THREE.DoubleSide });
    const seaweed = new THREE.Mesh(tubeGeo, tubeMat);
    seaweed.position.copy(pos);
    seaweed.userData.phase = Math.random() * Math.PI * 2;
    seaweed.userData.speed = 0.6 + Math.random() * 0.6;
    seaweed.userData.baseRotZ = (Math.random() - 0.5) * 0.3;
    seaweed.userData.baseRotX = (Math.random() - 0.5) * 0.2;
    scene.add(seaweed);
    seaweeds.push(seaweed);
  });
}

// ---------- СОЗДАНИЕ РЫБКИ ----------
function createFish() {
  const scheme = colorSchemes[Math.floor(Math.random() * colorSchemes.length)];
  const group = new THREE.Group();
  const bodyColor = scheme.body;
  const finColor = scheme.fin;

  const bodyMat = new THREE.MeshStandardMaterial({ color: bodyColor, roughness: 0.4, metalness: 0.05, flatShading: true });
  const finMat = new THREE.MeshStandardMaterial({ color: finColor, roughness: 0.5, transparent: true, opacity: 0.9, flatShading: true, side: THREE.DoubleSide });
  const eyeWhiteMat = new THREE.MeshStandardMaterial({ color: 0xffffff, roughness: 0.2 });
  const pupilMat = new THREE.MeshStandardMaterial({ color: 0x111111, roughness: 0.1 });

  // Тело (растянутая сфера)
  const bodyGeo = new THREE.SphereGeometry(0.5, 16, 12);
  bodyGeo.scale(1.7, 1.0, 0.85);
  const body = new THREE.Mesh(bodyGeo, bodyMat);
  body.castShadow = true;
  group.add(body);

  // Глаза
  const eyeGeo = new THREE.SphereGeometry(0.16, 10, 10);
  const pupilGeo = new THREE.SphereGeometry(0.09, 8, 8);
  [-1, 1].forEach(side => {
    const eye = new THREE.Mesh(eyeGeo, eyeWhiteMat);
    eye.position.set(1.0, 0.12, side * 0.32);
    group.add(eye);
    const pupil = new THREE.Mesh(pupilGeo, pupilMat);
    pupil.position.set(1.14, 0.12, side * 0.32);
    group.add(pupil);
  });

  // Хвост (наклонённый квадрат, вращается по Z)
  const tailShape = new THREE.Shape();
  tailShape.moveTo(0, 0);
  tailShape.lineTo(-0.5, 0.4);
  tailShape.lineTo(-0.7, 0.15);
  tailShape.lineTo(-0.5, -0.4);
  tailShape.lineTo(0, -0.1);
  tailShape.lineTo(-0.5, -0.4);
  const tailGeo = new THREE.ShapeGeometry(tailShape);
  const tail = new THREE.Mesh(tailGeo, finMat);
  tail.rotation.y = Math.PI;
  tail.position.set(-1.5, 0, 0);
  tail.castShadow = true;
  group.add(tail);

  // Верхний плавник
  const dorsalShape = new THREE.Shape();
  dorsalShape.moveTo(0, 0);
  dorsalShape.quadraticCurveTo(0.3, 0.6, 0.7, 0.15);
  const dorsalGeo = new THREE.ShapeGeometry(dorsalShape);
  const dorsalFin = new THREE.Mesh(dorsalGeo, finMat);
  dorsalFin.position.set(-0.2, 0.5, 0);
  dorsalFin.rotation.z = 0.15;
  group.add(dorsalFin);

  // Боковые плавники (левый / правый)
  const sideShape = new THREE.Shape();
  sideShape.moveTo(0, 0);
  sideShape.lineTo(0.5, -0.15);
  sideShape.lineTo(0.3, 0.15);
  const sideGeo = new THREE.ShapeGeometry(sideShape);
  const leftFin = new THREE.Mesh(sideGeo, finMat);
  leftFin.position.set(-0.3, -0.2, 0.45);
  leftFin.rotation.z = -0.3;
  group.add(leftFin);
  const rightFin = new THREE.Mesh(sideGeo, finMat);
  rightFin.position.set(-0.3, -0.2, -0.45);
  rightFin.rotation.z = 0.3;
  rightFin.rotation.y = Math.PI;
  group.add(rightFin);

  const scale = 0.6 + Math.random() * 0.6;
  group.scale.setScalar(scale);

  // Индивидуальные параметры
  const speed = 2.5 + Math.random() * 3.0;
  const obj = {
    mesh: group,
    tail: tail,
    leftFin: leftFin,
    rightFin: rightFin,
    dorsalFin: dorsalFin,
    velocity: new THREE.Vector3(),
    speed: speed,
    tailSpeed: 4 + Math.random() * 5,
    phase: Math.random() * Math.PI * 2,
    targetFood: null,
    avoidanceRadius: 3 + Math.random() * 2,
    size: scale,
    wanderPhase: Math.random() * Math.PI * 2
  };
  // Начальная позиция внутри аквариума
  group.position.set((Math.random() - 0.5) * (container.x - 6), (Math.random() - 0.5) * (container.y - 4), (Math.random() - 0.5) * (container.z - 6));
  scene.add(group);
  fishArray.push(obj);
  return obj;
}

// ---------- КОРМ ----------
function spawnFood(x, y, z) {
  const geo = new THREE.SphereGeometry(0.15, 6, 6);
  const mat = new THREE.MeshStandardMaterial({ color: 0x8B4513, roughness: 0.8 });
  const food = new THREE.Mesh(geo, mat);
  food.position.set(x, y, z);
  food.userData = { velocity: new THREE.Vector3(0, 0, 0) };
  scene.add(food);
  foodArray.push(food);
}

// ---------- ОБНОВЛЕНИЕ РЫБ ----------
function updateFish(dt) {
  const halfW = container.x / 2 - 1;
  const halfH = container.y / 2 - 1;
  const halfD = container.z / 2 - 1;

  for (let i = 0; i < fishArray.length; i++) {
    const f = fishArray[i];
    const pos = f.mesh.position;

    // --- Столкновение с едой (цель) ---
    let nearestFood = null, nearestDist = 15;
    for (let j = 0; j < foodArray.length; j++) {
      const fp = foodArray[j].position;
      const d = pos.distanceTo(fp);
      if (d < nearestDist) { nearestDist = d; nearestFood = foodArray[j]; }
    }

    let ax = new THREE.Vector3();
    if (nearestFood) {
      f.targetFood = nearestFood;
      ax.copy(nearestFood.position).sub(pos).multiplyScalar(1.0);
    } else {
      // --- Избегание других рыбок ---
      for (let j = 0; j < fishArray.length; j++) {
        if (i === j) continue;
        const of = fishArray[j].mesh.position;
        const d = of.distanceTo(pos);
        if (d < f.avoidanceRadius && d > 0.001) {
          const away = new THREE.Vector3().subVectors(of, pos).normalize().multiplyScalar((f.avoidanceRadius - d) / f.avoidanceRadius);
          ax.add(away);
        }
      }
      // --- Случайное блуждание ---
      f.wanderPhase += dt * 0.5;
      ax.x += Math.sin(f.wanderPhase) * 0.4;
      ax.z += Math.cos(f.wanderPhase * 1.3) * 0.4;
    }

    // --- Отражение от стенок ---
    ['x', 'y', 'z'].forEach(axis => {
      const lim = axis === 'y' ? halfH : (axis === 'x' ? halfW : halfD);
      if (pos[axis] > lim) ax[axis] -= (pos[axis] - lim) * 3;
      if (pos[axis] < -lim) ax[axis] -= (-lim - pos[axis]) * 3;
    });

    // --- Интеграция скорости ---
    f.velocity.addScaledVector(ax, dt);
    const sp = f.velocity.length();
    const maxSp = f.speed;
    if (sp > maxSp) f.velocity.multiplyScalar(maxSp / sp);
    f.velocity.multiplyScalar(0.96); // трение

    // --- Обновление позиции ---
    pos.addScaledVector(f.velocity, dt);
    // ограничение
    pos.x = Math.max(-halfW, Math.min(halfW, pos.x));
    pos.y = Math.max(-halfH, Math.min(halfH, pos.y));
    pos.z = Math.max(-halfD, Math.min(halfD, pos.z));

    // --- Поворот в направлении движения ---
    const velLen = f.velocity.length();
    if (velLen > 0.2) {
      const dir = f.velocity.clone().normalize();
      const angle = Math.atan2(dir.z, dir.x);
      f.mesh.rotation.y = angle;
    }

    // --- Анимация хвоста (вращение по Z) и плавников ---
    f.tail.rotation.z = Math.sin(clock.elapsedTime * f.tailSpeed + f.phase) * 0.4;
    const finWave = Math.sin(clock.elapsedTime * f.tailSpeed * 1.5 + f.phase + 1) * 0.35;
    f.leftFin.rotation.z = -0.3 + finWave;
    f.rightFin.rotation.z = 0.3 - finWave;
    f.dorsalFin.rotation.z = 0.15 + Math.sin(clock.elapsedTime * f.tailSpeed * 0.7 + f.phase) * 0.12;

    // --- Поедание еды ---
    if (f.targetFood) {
      const fd = pos.distanceTo(f.targetFood.position);
      if (fd < 1.0) {
        // съедаем
        scene.remove(f.targetFood);
        foodArray = foodArray.filter(x => x !== f.targetFood);
        // рост на 5%
        f.size *= 1.05;
        f.mesh.scale.setScalar(f.size);
        f.targetFood = null;
      }
    } else if (foodArray.length === 0) {
      f.targetFood = null;
    }
  }
}

// ---------- ОБНОВЛЕНИЕ КОРМА ----------
function updateFood(dt) {
  const gravity = -9.8;
  const floorY = groundY + 0.2;
  for (let i = 0; i < foodArray.length; i++) {
    const food = foodArray[i];
    food.userData.velocity.y += gravity * dt;
    food.position.y += food.userData.velocity.y * dt;
    if (food.position.y <= floorY) {
      scene.remove(food);
      foodArray.splice(i, 1);
      i--;
    }
  }
}

function updateBubbles(dt) {
  const topY = container.y / 2 - 0.5;
  const bottomY = -container.y / 2 + 0.5;
  for (let i = 0; i < bubbleArray.length; i++) {
    const b = bubbleArray[i];
    b.position.y += b.userData.speed * dt;
    b.position.x += Math.sin(clock.elapsedTime * b.userData.swaySpeed + b.userData.swayPhase) * 0.3 * dt;
    b.position.z += Math.cos(clock.elapsedTime * b.userData.swaySpeed + b.userData.swayPhase) * 0.3 * dt;
    if (b.position.y > topY) {
      b.position.y = bottomY;
      b.position.x = (Math.random() - 0.5) * (container.x - 4);
      b.position.z = (Math.random() - 0.5) * (container.z - 4);
    }
  }
}

// ---------- ОБНОВЛЕНИЕ ВОДОРОСЛЕЙ ----------
function updateSeaweed(dt) {
  for (let i = 0; i < seaweeds.length; i++) {
    const sw = seaweeds[i];
    const t = clock.elapsedTime;
    sw.rotation.z = sw.userData.baseRotZ + Math.sin(t * sw.userData.speed + sw.userData.phase) * 0.12;
    sw.rotation.x = sw.userData.baseRotX + Math.cos(t * sw.userData.speed * 0.8 + sw.userData.phase) * 0.08;
  }
}

// ---------- КЛИК ДЛЯ КОРМЛЕНИЯ ----------
function onCanvasClick(event) {
  // Не обрабатываем, если перетащили камеру
  if (controls._moved) { controls._moved = false; return; }
  mouse.x = (event.clientX / window.innerWidth) * 2 - 1;
  mouse.y = -(event.clientY / window.innerHeight) * 2 + 1;
  raycaster.setFromCamera(mouse, camera);
  const intersects = raycaster.intersectObject(waterPlane, true);
  if (intersects.length > 0) {
    const p = intersects[0].point;
    // ограничим внутри аквариума
    p.x = Math.max(-container.x / 2 + 1, Math.min(container.x / 2 - 1, p.x));
    p.z = Math.max(-container.z / 2 + 1, Math.min(container.z / 2 - 1, p.z));
    spawnFood(p.x, Math.max(container.y / 2 - 1, p.y), p.z);
    updateStats();
  }
}

let prevTime = Date.now();
function animate() {
  requestAnimationFrame(animate);
  const dt = Math.min((Date.now() - prevTime) / 1000, 0.05);
  prevTime = Date.now();

  updateFish(dt);
  updateFood(dt);
  updateBubbles(dt);
  updateSeaweed(dt);

  controls.update();
  renderer.render(scene, camera);

  // FPS
  frameCount++;
  fpsTimer += Date.now() - (prevTime - dt * 1000);
  const now = Date.now();
  if (now - (animate._last || (animate._last = now)) >= 500) {
    currentFps = Math.round(frameCount * 1000 / (now - animate._last));
    animate._last = now;
    frameCount = 0;
