function buildFish(scale){
  const g=new THREE.Group();
  const bodyColor=colorSchemes[Math.floor(Math.random()*colorSchemes.length)];
  const bodyMat=new THREE.MeshStandardMaterial({color:bodyColor,roughness:0.5,metalness:0.05});

  tail.rotation.z=-Math.PI/2;tail.position.set(-1.35,0,0);g.add(tail);

  g.traverse(o=>{if(o.isMesh){o.castShadow=true;o.receiveShadow=true}});
  g.scale.setScalar(scale);
  return {mesh:g,tail,leftFin,rightFin};
}

function animate(){
  requestAnimationFrame(animate);
  const t=performance.now()*0.001,delta=Math.min(0.05,performance.now()-lastT)/1000;lastT=performance.now();

  /* рыба → еда */

    const f=fishArray[i];
    const fp=f.mesh.position;
    let tx=0,ty=0,tz=0;
    /* поиск еды */
    let best=1e9,bf=null;
    for(let j=0;j<foods.length;j++){
      const d=fp.distanceTo(foods[j].position);
      if(d<best){best=d;bf=foods[j];}
    }
    if(bf&&best<FOOD_RADIUS){
      f.targetFood=bf;
      const dx=bf.position.x-fp.x,dy=bf.position.y-fp.y,dz=bf.position.z-fp.z;
      const dl=Math.hypot(dx,dy,dz)||1;
      tx=dx/dl;ty=dy/dl;tz=dz/dl;
      if(best<1.4){removeFood(foods.indexOf(bf));f.age++;f.maxScale*=1.05;f.mesh.scale.setScalar(Math.min(f.scale*f.maxScale/f.scale,1.4)*f.scale);f.mesh.scale.setScalar(f.scale*(f.age>0?Math.min(f.maxScale,1.35):1));}
    }else f.targetFood=null;

    /* избегание других рыб */
    for(let j=0;j<fishArray.length;j++){
      if(i===j)continue;
      const o=fishArray[j].mesh.position;
      const dx=fp.x-o.x,dz=fp.z-o.z,dd=Math.sqrt(dx*dx+dz*dz);
      if(dd<f.avoidanceRadius&&dd>0.001){
        const push=(f.avoidanceRadius-dd)/f.avoidanceRadius;
        tx+=dx/dd*push*2;tz+=dz/dd*push*2;
      }
    }

    /* блуждание */
    tx+=(Math.random()-0.5)*1.5;
    ty+=(Math.random()-0.5)*0.8;
    tz+=(Math.random()-0.5)*1.5;

    /* стены */
    const margin=3;
    if(fp.x> W/2-margin)tx-=3; if(fp.x<-W/2+margin)tx+=3;
    if(fp.y> H/2-margin)ty-=3; if(fp.y<-H/2+margin)ty+=3;
    if(fp.z> D/2-margin)tz-=3; if(fp.z<-D/2+margin)tz+=3;

    const v=f.velocity;
    v.x+=(tx)*delta*2;v.y+=(ty)*delta*2;v.z+=(tz)*delta*2;
    const sp=Math.hypot(v.x,v.y,v.z);
    if(sp>f.speed)v.multiplyScalar(f.speed/sp);
    v.multiplyScalar(0.94);
    fp.addScaledVector(v,delta*6);

    /* ориентация */
    if(Math.hypot(v.x,v.z)>0.1)f.mesh.rotation.y=Math.atan2(v.v,v.z);
    f.mesh.rotation.z=Math.atan2(v.y,Math.hypot(v.x,v.z))*0.4;

    /* анимация */
    f.phase+=f.tailSpeed*delta;
    f.tail.rotation.y=Math.sin(f.phase)*0.5;
    f.leftFin.rotation.z=Math.sin(f.phase*0.8)*0.4;
    f.rightFin.rotation.z=-Math.sin(f.phase*0.8)*0.4;

    f.scale+= (f.maxScale-f.scale)*0.02;
    f.mesh.scale.setScalar(f.scale);
  }

  /* пузыри */
  for(let i=0;i<bubbles.length;i++){

}

## Что реализовано

**🐟 Рыбки (15 шт.):** анатомически детализированные модели — вытянутое тело (скалированная сфера), глаза с зрачками, анимируемый хвост, спинной и боковые плавники. 8 цветовых схем, размер 0.6–1.2, плавное махание хвостом и плавниками, поворот по направлению движения.

**🧠 ИИ-поведение:** система избегания столкновений (отталкивание), отражение от стен, случайное блуждание, преследование корма в радиусе 15 единиц, рост на 5% после поедания, индивидуальные параметры скорости/радиуса/фазы.

**🌊 Окружение:** стеклянный контейнер 36×24×20 с `transmission: 0.95`, видимая рамка (edges), песчаное дно с procedural неровностями, 8 деформированных дедекаэдров-камней, 12 водорослей из `TubeGeometry` + `CatmullRomCurve3` с качающейся анимацией.

**🫧 Пузыри (30 шт.):** прозрачные сферы со стеклом, движение вверх с sinus/cos wave, сброс у поверхности.

**🍽️ Кормление:** клик по воде создаёт падающий корм (гравитация), рыбки его находят, съедают (рост 5%), удаление у дна.

**💡 Освещение:** Ambient, Directional с тенями 2048² PCFSoftShadowMap, 2 PointLight (голубой/синий), переключение кнопкой.

**🎮 OrbitControls:** вращение/панорама/зум с damping и ограничением угла.

**🖥️ Glass UI:** градиентный заголовок, полупрозрачные панели с `backdrop-filter`, счётчики FPS/рыбок/пузырей/корма.
