function makeHamster(bodyColor, bellyColor) {
  const g = new THREE.Group();
  const bodyMat = mat(bodyColor);
  const bellyMat = mat(bellyColor);

  // Body (squashed sphere)
  const body = new THREE.Mesh(new THREE.SphereGeometry(0.5, 9, 7), bodyMat);
  body.scale.set(1, 0.85, 1.2);
  body.position.y = 0.45;
  body.castShadow = true;
  g.add(body);

  // Belly patch
  const belly = new THREE.Mesh(new THREE.SphereGeometry(0.42, 8, 6), bellyMat);
  belly.scale.set(0.9, 0.7, 0.9);
  belly.position.set(0, 0.36, 0.18);
  g.add(belly);

  // Head
  const head = new THREE.Mesh(new THREE.SphereGeometry(0.32, 8, 6), bodyMat);
  head.position.set(0, 0.58, 0.48);
  head.castShadow = true;
  g.add(head);

  // Ears
  const earMat = mat(bellyColor);
  [-1, 1].forEach(s => {
    const ear = new THREE.Mesh(new THREE.SphereGeometry(0.12, 6, 5), bodyMat);
    ear.position.set(0.19 * s, 0.85, 0.42);
    const inner = new THREE.Mesh(new THREE.SphereGeometry(0.07, 5, 4), earMat);
    inner.position.set(0.19 * s, 0.85, 0.47);
    g.add(ear, inner);
  });

  // Eyes (kept refs for blinking)
  const eyeMat = mat(0x2b2b2b, { roughness: 0.3 });
  const eyes = [];
  [-1, 1].forEach(s => {
    const e = new THREE.Mesh(new THREE.SphereGeometry(0.055, 6, 5), eyeMat);
    e.position.set(0.15 * s, 0.62, 0.75);
    g.add(e);
    eyes.push(e);
  });

  // Nose + cheeks
  const nose = new THREE.Mesh(new THREE.SphereGeometry(0.05, 5, 4), mat(0xf28ba8));
  nose.position.set(0, 0.55, 0.8);
  const cheekMat = mat(0xf7a8b8);
  [-1, 1].forEach(s => {
    const c = new THREE.Mesh(new THREE.SphereGeometry(0.07, 5, 4), cheekMat);
    c.position.set(0.2 * s, 0.5, 0.72);
    g.add(c);
  });
  g.add(nose);

  // Feet + stubby tail
  const footMat = mat(bellyColor);
  [[-0.25, 0.32], [0.25, 0.32], [-0.25, -0.32], [0.25, -0.32]].forEach(([x, z]) => {
    const f = new THREE.Mesh(new THREE.SphereGeometry(0.09, 5, 4), footMat);
    f.scale.set(1, 0.6, 1.4);
    f.position.set(x, 0.07, z);
    g.add(f);
  });
  const tail = new THREE.Mesh(new THREE.SphereGeometry(0.08, 5, 4), bodyMat);
  tail.position.set(0, 0.42, -0.58);
  g.add(tail);

  return { group: g, body, head, eyes };
}

/* ============================================================
   SIMPLE AUTONOMOUS BEHAVIOR (state machine)
============================================================ */
const tmpVec = new THREE.Vector3();

function pickRandomPoint(v) {
  v.set(
    (Math.random() * 2 - 1) * (BOUNDS.x - 0.5),
    0.17,
    (Math.random() * 2 - 1) * (BOUNDS.z - 0.5)
  );
  return v;
}

function chooseNextState(h) {
  const r = Math.random();
  if (r < 0.55) {
    h.state = 'wander';
    pickRandomPoint(h.target);
  } else if (r < 0.8) {
    h.state = 'pause';
    h.timer = 0.6 + Math.random() * 1.8;
  } else if (r < 0.92) {
    h.state = 'eat';
    // stand just in front of the bowl
    h.target.copy(BOWL_POS).add(tmpVec.set(0, 0.17, 0.95));
  } else {
    h.state = 'wheel';
    // stand right in front of the wheel
    h.target.copy(WHEEL_POS).add(tmpVec.set(0, 0.17, 0.2));
    h.target.z += 1.0;
  }
}

function steerToward(h, dt) {
  tmpVec.subVectors(h.target, h.group.position);
  tmpVec.y = 0;
  const dist = tmpVec.length();
  if (dist > 0.001) {
    const desired = Math.atan2(tmpVec.x, tmpVec.z);
    // shortest-angle turn
    let diff = desired - h.group.rotation.y;
    diff = Math.atan2(Math.sin(diff), Math.cos(diff));
    h.group.rotation.y += diff * Math.min(1, dt * 6);
    return dist;
  }
  return 0;
}

function updateHamster(h, dt, t) {
  let moving = false;

  switch (h.state) {
    case 'wander': {
      const d = steerToward(h, dt);
      if (d < 0.25) {
        h.state = 'pause';
        h.timer = 0.5 + Math.random() * 1.5;
      } else {
        h.group.position.addScaledVector(tmpVec.normalize(), h.speed * dt);
        moving = true;
      }
      break;
    }
    case 'pause': {
      h.timer -= dt;
      if (h.timer <= 0) chooseNextState(h);
      break;
    }
    case 'eat': {
      const d = steerToward(h, dt);
      if (d < 0.3) {
        // face the bowl and chew
        h.group.rotation.y = Math.atan2(BOWL_POS.x - h.group.position.x,
                                        BOWL_POS.z - h.group.position.z);
        h.head.scale.setScalar(1 + Math.sin(t * 14) * 0.08);
        h.timer -= dt;
        if (h.timer <= 0) { h.timer = 2.5 + Math.random() * 2; h.state = 'wander'; pickRandomPoint(h.target); }
      } else {
        h.group.position.addScaledVector(tmpVec.normalize(), h.speed * dt);
        moving = true;
      }
      break;
    }
    case 'wheel': {
      const d = steerToward(h, dt);
      if (d < 0.35) {
        // face the wheel and scurry!
        h.group.rotation.y = Math.atan2(WHEEL_POS.x - h.group.position.x,
                                        WHEEL_POS.z - h.group.position.z);
        wheelSpin = Math.max(wheelSpin, 7);
        h.head.scale.setScalar(1 + Math.sin(t * 20) * 0.1);
        h.timer -= dt;
        if (h.timer <= 0) { h.timer = 2 + Math.random() * 2; h.state = 'wander'; pickRandomPoint(h.target); }
      } else {
        h.group.position.addScaledVector(tmpVec.normalize(), h.speed * dt);
        moving = true;
      }
      break;
    }
  }

  // First-time state setup guard (eat/wheel timers start on arrival)
  if ((h.state === 'eat' || h.state === 'wheel') && h.timer === 0) h.timer = 0.01;

  // Keep inside the cage
  h.group.position.x = THREE.MathUtils.clamp(h.group.position.x, -BOUNDS.x, BOUNDS.x);
  h.group.position.z = THREE.MathUtils.clamp(h.group.position.z, -BOUNDS.z, BOUNDS.z);

  // Squishy bob while moving, gentle breathing while still
  const bob = moving ? Math.sin(t * 12 + h.phase) * 0.035 : Math.sin(t * 3 + h.phase) * 0.012;
  h.group.position.y = 0.17 + Math.abs(bob);
  h.body.scale.y = 0.85 * (1 - bob * 1.5);

  // Tilt head forward when running
  h.head.rotation.x = moving ? 0.12 : Math.sin(t * 2 + h.phase) * 0.05;

  // Blinking
  h.blinkTimer -= dt;
  if (h.blinkTimer < 0) h.blinkTimer = 2 + Math.random() * 4;
  const blink = h.blinkTimer < 0.12 ? 0.1 : 1;
  h.eyes.forEach(e => e.scale.y = blink);
}

/* ============================================================
   MAIN LOOP
============================================================ */
const clock = new THREE.Clock();

function animate() {
  requestAnimationFrame(animate);
  const dt = Math.min(clock.getDelta(), 0.05);
  const t = clock.elapsedTime;

  hamsters.forEach(h => updateHamster(h, dt, t));

  // Wheel spins fast, then coasts to a stop
  wheelSpin *= Math.pow(0.4, dt);
  if (wheelSpin < 0.05) wheelSpin = 0;
  wheel.rotation.x -= wheelSpin * dt;

  controls.update();
  renderer.render(scene, camera);
}
animate();

addEventListener('resize', () => {
  camera.aspect = innerWidth / innerHeight;
  camera.updateProjectionMatrix();
  renderer.setSize(innerWidth, innerHeight);
});

## What's in the scene

| Element | Details |
|---|---|
| **Cage** | Wooden base, sand bedding, corner posts, top rails, and light semi-transparent bars so the hamsters stay visible from every angle |
| **4 hamsters** | Orange, golden, grey-blue, and white — built from flat-shaded spheres with ears, blushing cheeks, feet, and stubby tails |
| **Interactive wheel** | Hamsters approach it, face it, and scurry — the wheel spins and *coasts to a stop* when they leave |
| **Food bowl** | Hamsters walk over and chew (their heads bob) for a couple of seconds |
| **Decor** | Green tunnel, little red-roofed house, scattered hay tufts |

## Behavior (simple state machine)

Each hamster cycles through four states with randomized timing:
- **Wander** → picks a random spot, turns smoothly (shortest arc), walks with a squishy bob
- **Pause** → idles, breathes, looks around
- **Eat** → heads to the bowl, faces it, chews
- **Wheel** → heads to the wheel, faces it, spins it

All positions are clamped to the cage bounds, and they blink on their own schedule. The camera uses damped OrbitControls with limits so you can never get lost under the floor.
