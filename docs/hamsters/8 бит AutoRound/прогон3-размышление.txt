Movement code:

```js
function updateHamster(h, dt){
  h.timer -= dt;
  switch(h.state){
    case 'wander':
      moveToward(h, h.target, dt);
      if(dist < 0.15 || h.timer<=0) setState(h,'pause', rand(0.5,2));
      break;
    case 'pause':
      if(h.timer<=0) pickAction(h);
      break;
    case 'toWheel':
      if(wheelBusy && !h.isAtWheel) { setState(h,'wander'); }
      else { moveToward(h, wheelSpot, dt); if close -> 'atWheel' timer rand(3,6)}
      break;
    case 'atWheel':
      faceWheel; bob; wheelSpeed target high; if timer<=0 -> wander
      break;
    case 'toFood': similar
    case 'eat': bob head; timer -> wander
  }
}
```
