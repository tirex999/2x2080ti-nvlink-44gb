2D noise:
```js
function smooth(t){ return t*t*(3-2*t); }
function noise2(x, z){
  const xi = Math.floor(x), zi = Math.floor(z);
  const xf = x - xi, zf = z - zi;
  const a = hash2(xi, zi), b = hash2(xi+1, zi), c = hash2(xi, zi+1), d = hash2(xi+1, zi+1);
  const u = smooth(xf), v = smooth(zf);
  return a + (b-a)*u + (c-a)*v + (a - b - c + d)*u*v;
}
```

fbm 4 octaves:
```js
function fbm2(x, z){
  let sum = 0, amp = 1, freq = 1, norm = 0;
  for(let i=0;i<4;i++){
    sum += noise2(x*freq, z*freq) * amp;
    norm += amp;
    amp *= 0.5; freq *= 2;
  }
  return sum / norm;
}
```

3D noise: trilinear interp of hash3 with smoothstep.
```js
function noise3(x,y,z){
  const xi=Math.floor(x), yi=Math.floor(y), zi=Math.floor(z);
  const xf=x-xi, yf=y-yi, zf=z-zi;
  const u=smooth(xf), v=smooth(yf), w=smooth(zf);
  // 8 corners
  const c000=hash3(xi,yi,zi), c100=hash3(xi+1,yi,zi), c010=hash3(xi,yi+1,zi), c110=hash3(xi+1,yi+1,zi);
  const c001=hash3(xi,yi,zi+1), c101=hash3(xi+1,yi,zi+1), c011=hash3(xi,yi+1,zi+1), c111=hash3(xi+1,yi+1,zi+1);
  const x00=c000+(c100-c000)*u, x10=c010+(c110-c010)*u;
  const x01=c001+(c101-c001)*u, x11=c011+(c111-c011)*u;
  const y0=x00+(x10-x00)*v, y1=x01+(x11-x01)*v;
  return y0+(y1-y0)*w;
}
```

buildMesh:
```js
function buildMesh(cx, cz){
  const c = chunks.get(cx+','+cz);
  if(!c) return;
  if(c.mesh){ removeMesh(c); }
  const pos=[], norm=[], col=[], idx=[];
  const baseX = cx*CHUNK, baseZ = cz*CHUNK;
  const data = c.data;
  const tmpColor = new THREE.Color();
  for(let y=0;y<HEIGHT;y++){
    for(let lz=0;lz<CHUNK;lz++){
      for(let lx=0;lx<CHUNK;lx++){
        const id = data[(y*CHUNK+lz)*CHUNK+lx];
        if(id===0) continue;
        const wx = baseX+lx, wz = baseZ+lz;
        tmpColor.setHex(BLOCKS[id].color);
        for(let f=0;f<6;f++){
          const face = FACES[f];
          if(getBlock(wx+face.dir[0], y+face.dir[1], wz+face.dir[2]) !== 0) continue;
          const base = pos.length/3;
          for(let k=0;k<4;k++){
            pos.push(wx+face.corners[k][0], y+face.corners[k][1], wz+face.corners[k][2]);
            norm.push(face.dir[0], face.dir[1], face.dir[2]);
            col.push(tmpColor.r*face.shade, tmpColor.g*face.shade, tmpColor.b*face.shade);
          }
          idx.push(base, base+1, base+2, base+2, base+1, base+3);
        }
      }
    }
  }
  const geo = new THREE.BufferGeometry();
  geo.setAttribute('position', new THREE.Float32BufferAttribute(pos,3));
  geo.setAttribute('normal', new THREE.Float32BufferAttribute(norm,3));
  geo.setAttribute('color', new THREE.Float32BufferAttribute(col,3));
  geo.setIndex(idx);
  const mesh = new THREE.Mesh(geo, chunkMaterial);
  scene.add(mesh);
  chunkMeshes.push(mesh);
  c.mesh = mesh;
}
```

```js
function generateChunk(cx, cz){
  const key = cx+','+cz;
  if(chunks.has(key)) return;
  const data = new Uint8Array(HEIGHT*CHUNK*CHUNK);
  for(let lx=0;lx<CHUNK;lx++){
    for(let lz=0;lz<CHUNK;lz++){
      const wx = cx*CHUNK+lx, wz = cz*CHUNK+lz;
      const m = fbm2(wx*0.004, wz*0.004);
      const h = fbm2(wx*0.02, wz*0.02);
      const H = Math.floor(5 + m*m*58 + h*10);
      const surf = H>=46?7 : H>=37?3 : H<=16?4 : 1;
      const sub  = H<=16?4 : H>=37?3 : 2;
      for(let y=0;y<=H && y<HEIGHT;y++){
        let id;
        if(y===0) id=3;
        else if(y<H-3) id=3;
        else if(y<H) id=sub;
        else id=surf;
        data[(y*CHUNK+lz)*CHUNK+lx]=id;
      }
      for(let y=3;y<=H-2;y++){
        if(fbm3(wx*0.09, y*0.09, wz*0.09) > 0.67) data[(y*CHUNK+lz)*CHUNK+lx]=0;
      }
    }
  }
  // trees
  for(let lx=2;lx<14;lx++){
    for(let lz=2;lz<14;lz++){
      const wx = cx*CHUNK+lx, wz = cz*CHUNK+lz;
      let sy=-1;
      for(let y=HEIGHT-1;y>=0;y--){ if(data[(y*CHUNK+lz)*CHUNK+lx]!==0){ sy=y; break; } }
      if(sy<1 || sy>68) continue;
      if(data[(sy*CHUNK+lz)*CHUNK+lx]!==1) continue;
      if(treeHash(wx,wz) >= 0.02) continue;
      for(let t=1;t<=4;t++){
        const y=sy+t;
        if(y>=HEIGHT) break;
        data[(y*CHUNK+lz)*CHUNK+lx]=5;
      }
      const leaf=(x,y,z)=>{
        if(y<0||y>=HEIGHT) return;
        const i2=(y*CHUNK+z)*CHUNK+x;
        if(data[i2]===0) data[i2]=6;
      };
      for(const ty of [sy+4, sy+5]){
        for(let dx2=-2;dx2<=2;dx2++)
          for(let dz2=-2;dz2<=2;dz2++)
            leaf(lx+dx2, ty, lz+dz2);
      }
      for(let dx2=-1;dx2<=1;dx2++)
        for(let dz2=-1;dz2<=1;dz2++)
          leaf(lx+dx2, sy+6, lz+dz2);
      leaf(lx, sy+7, lz);
    }
  }
  chunks.set(key, {data, mesh:null});
}
```

```js
function byDistance(pcx, pcz, radius, fn){
  const list = [];
  for(let dx=-radius; dx<=radius; dx++)
    for(let dz=-radius; dz<=radius; dz++)
      list.push([dx*dx+dz*dz, pcx+dx, pcz+dz]);
  list.sort((a,b)=>a[0]-b[0]);
  for(let i=0;i<list.length;i++) fn(list[i][1], list[i][2]);
}
```

```js
function updateWorld(){
  const pcx = Math.floor(player.pos.x / CHUNK);
  const pcz = Math.floor(player.pos.z / CHUNK);
  let gen = 0;
  byDistance(pcx, pcz, 5, (cx, cz) => {
    if(gen >= 4) return;
    if(!chunks.has(cx + ',' + cz)){ generateChunk(cx, cz); gen++; }
  });
  let built = 0;
  byDistance(pcx, pcz, 4, (cx, cz) => {
    if(built >= 2) return;
    const c = chunks.get(cx + ',' + cz);
    if(!c || c.mesh) return;
    if(!chunks.has((cx-1)+','+cz) || !chunks.has((cx+1)+','+cz) ||
       !chunks.has(cx+','+(cz-1)) || !chunks.has(cx+','+(cz+1))) return;
    buildMesh(cx, cz); built++;
  });
  const dead = [];
  chunks.forEach((c, key) => {
    const p = key.split(',');
    const dx = Math.abs(+p[0] - pcx), dz = Math.abs(+p[1] - pcz);
    if(dx > 7 || dz > 7) dead.push(key);
  });
  for(const key of dead){
    const c = chunks.get(key);
    if(c.mesh) removeMesh(c);
    chunks.delete(key);
  }
}
```
