Smoothstep: `t * t * (3 - 2 * t)`

2D noise:
```javascript
function noise2(x, z) {
    const xi = Math.floor(x), zi = Math.floor(z);
    const xf = x - xi, zf = z - zi;
    const u = smoothstep(xf), v = smoothstep(zf);
    const a = hash2(xi, zi);
    const b = hash2(xi + 1, zi);
    const c = hash2(xi, zi + 1);
    const d = hash2(xi + 1, zi + 1);
    return a + (b - a) * u + (c - a) * v + (a - b - c + d) * u * v;
}
```

Fractal (4 octaves):
```javascript
function fractal2(x, z) {
    let val = 0, amp = 1, freq = 1, max = 0;
    for (let i = 0; i < 4; i++) {
        val += noise2(x * freq, z * freq) * amp;
        max += amp;
        amp *= 0.5;
        freq *= 2;
    }
    return val / max;
}
```

Now for the chunk system:

```javascript
const CHUNK_SIZE = 16;
const CHUNK_HEIGHT = 80;
const chunks = new Map();
const chunkMeshes = [];

function chunkKey(cx, cz) { return cx + ',' + cz; }

function getBlock(wx, wy, wz) {
    if (wy < 0 || wy >= CHUNK_HEIGHT) return 0;
    const cx = Math.floor(wx / CHUNK_SIZE);
    const cz = Math.floor(wz / CHUNK_SIZE);
    const chunk = chunks.get(chunkKey(cx, cz));
    if (!chunk) return 0;
    const lx = wx - cx * CHUNK_SIZE;
    const lz = wz - cz * CHUNK_SIZE;
    return chunk.data[wy * CHUNK_SIZE * CHUNK_SIZE + lz * CHUNK_SIZE + lx];
}

function setBlock(wx, wy, wz, id) {
    if (wy < 0 || wy >= CHUNK_HEIGHT) return;
    const cx = Math.floor(wx / CHUNK_SIZE);
    const cz = Math.floor(wz / CHUNK_SIZE);
    const chunk = chunks.get(chunkKey(cx, cz));
    if (!chunk) return;
    const lx = wx - cx * CHUNK_SIZE;
    const lz = wz - cz * CHUNK_SIZE;
    chunk.data[wy * CHUNK_SIZE * CHUNK_SIZE + lz * CHUNK_SIZE + lx] = id;
}
```
