Movement helper:
```js
function walkTo(h, tx, tz, dt, speed){
  const dx = tx - h.pos.x, dz = tz - h.pos.z;
  const d = Math.hypot(dx,dz);
  if (d < 0.02) return 0;
  let ux = dx/d, uz = dz/d;
  // obstacle avoidance
  for (const o of obstacles){
    const ox = h.pos.x - o.x, oz = h.pos.z - o.z;
    const od = Math.hypot(ox,oz);
    if (od < o.r + 0.4 && od > 0.0001){
      const push = (o.r + 0.4 - od);
      ux += (ox/od) * push * 2.2; uz += (oz/od) * push * 2.2;
      const l = Math.hypot(ux,uz); ux/=l; uz/=l;
    }
  }
  const v = speed;
  h.pos.x = clamp(h.pos.x + ux*v*dt, -BOUND.x, BOUND.x);
  h.pos.z = clamp(h.pos.z + uz*v*dt, -BOUND.z, BOUND.z);
  const want = Math.atan2(-uz, ux);
  h.heading = angleLerp(h.heading, want, 1 - Math.pow(0.0015, dt));
  h.speed = v;
  return d;
}
```
